    77	/* ── Animated Counter ──────────────────────────────────────────────── */
    78	const AnimatedCounter = ({
    79	  value,
    80	  duration = 2,
    81	}: {
    82	  value: number;
    83	  duration?: number;
    84	}) => {
    85	  const [display, setDisplay] = useState(0);
    86	  const springValue = useSpring(0, { stiffness: 40, damping: 25 });
    87	
    88	  useEffect(() => {
    89	    springValue.set(value);
    90	  }, [value, springValue]);
    91	
    92	  useEffect(() => {
    93	    const unsubscribe = springValue.on("change", (v) =>
    94	      setDisplay(Math.round(v))
    95	    );
    96	    return unsubscribe;
    97	  }, [springValue]);
    98	
    99	  return <>{display.toLocaleString("pt-MZ")}</>;
   100	};
