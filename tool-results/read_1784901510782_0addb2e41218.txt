   770→   770→interface LanguageContextValue {
   771→   771→  lang: Lang;
   772→   772→  setLang: (l: Lang, options?: { explicit?: boolean }) => void;
   773→   773→  t: (key: string, vars?: Record<string, string>) => string;
   774→   774→}
   775→   775→
   776→   776→const LanguageContext = createContext<LanguageContextValue | null>(null);
   777→   777→
   778→   778→function format(template: string, vars?: Record<string, string>) {
   779→   779→  if (!vars) return template;
   780→   780→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
   781→   781→}
   782→   782→
   783→   783→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
   784→   784→
   785→   785→export function LanguageProvider({ children }: { children: ReactNode }) {
   786→   786→  // English is the universal default. Only honor an explicit non-English pick
   787→   787→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
   788→   788→  // PT defaults from previous releases are reset to English automatically.
   789→   789→  const [lang, setLangState] = useState<Lang>(() => {
   790→   790→    if (typeof window === "undefined") return "en";
   791→   791→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
   792→   792→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
   793→   793→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
   794→   794→    return resolveLangFromCountry(getStoredCountry());
   795→   795→  });
   796→   796→
   797→   797→  useEffect(() => {
   798→   798→    localStorage.setItem("bateu_lang", lang);
   799→   799→    document.documentElement.lang = lang;
   800→   800→  }, [lang]);
   801→   801→
   802→   802→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
   803→   803→    if (!SUPPORTED.includes(l)) return;
   804→   804→    setLangState(l);
   805→   805→    if (typeof window !== "undefined" && options?.explicit !== false) {
   806→   806→      localStorage.setItem("bateu_lang_explicit", "1");
   807→   807→    }
   808→   808→  };
   809→   809→
   810→   810→  const t = (key: string, vars?: Record<string, string>) => {
   811→   811→    // Prioritize the currently SELECTED LANGUAGE first, then English as fallback, then PT
   812→   812→    const en = translations.en!;
   813→   813→    const dict = translations[lang] ?? en;
   814→   814→    const value = dict[key] ?? en[key] ?? translations.pt?.[key] ?? key;
   815→   815→    return format(value, vars);
   816→   816→  };
   817→   817→
   818→   818→  return (
   819→   819→    <LanguageContext.Provider value={{ lang, setLang, t }}>
   820→   820→      {children}
   821→   821→    </LanguageContext.Provider>
   822→   822→  );
   823→   823→}
   824→   824→
   825→   825→export function useLanguage() {
   826→   826→  const ctx = useContext(LanguageContext);
   827→   827→  if (!ctx) {
   828→   828→    return {
   829→   829→      lang: "en" as Lang,
   830→   830→      setLang: () => {},
   831→   831→      t: (k: string, vars?: Record<string, string>) =>
   832→   832→        format(translations.en?.[k] ?? k, vars),
   833→   833→    };
   834→   834→  }
   835→   835→  return ctx;
   836→   836→}
