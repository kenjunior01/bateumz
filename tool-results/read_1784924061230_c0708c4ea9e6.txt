  2375→interface LanguageContextValue {
  2376→  lang: Lang;
  2377→  setLang: (l: Lang, options?: { explicit?: boolean }) => void;
  2378→  t: (key: string, vars?: Record<string, string>) => string;
  2379→}
  2380→
  2381→const LanguageContext = createContext<LanguageContextValue | null>(null);
  2382→
  2383→function format(template: string, vars?: Record<string, string>) {
  2384→  if (!vars) return template;
  2385→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
  2386→}
  2387→
  2388→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
  2389→
  2390→export function LanguageProvider({ children }: { children: ReactNode }) {
  2391→  // English is the universal default. Only honor an explicit non-English pick
  2392→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
  2393→  // PT defaults from previous releases are reset to English automatically.
  2394→  const [lang, setLangState] = useState<Lang>(() => {
  2395→    if (typeof window === "undefined") return "en";
  2396→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
  2397→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  2398→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  2399→    return resolveLangFromCountry(getStoredCountry());
  2400→  });
  2401→
  2402→  useEffect(() => {
  2403→    localStorage.setItem("bateu_lang", lang);
  2404→    document.documentElement.lang = lang;
  2405→  }, [lang]);
  2406→
  2407→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
  2408→    if (!SUPPORTED.includes(l)) return;
  2409→    setLangState(l);
  2410→    if (typeof window !== "undefined" && options?.explicit !== false) {
  2411→      localStorage.setItem("bateu_lang_explicit", "1");
  2412→    }
  2413→  };
  2414→
  2415→  const t = (key: string, vars?: Record<string, string>) => {
  2416→    // Prioritize the currently SELECTED LANGUAGE first, then English as fallback, then PT
  2417→    const en = translations.en!;
  2418→    const dict = translations[lang] ?? en;
  2419→    const value = dict[key] ?? en[key] ?? translations.pt?.[key] ?? key;
  2420→    return format(value, vars);
  2421→  };
  2422→
  2423→  return (
  2424→    <LanguageContext.Provider value={{ lang, setLang, t }}>
  2425→      {children}
  2426→    </LanguageContext.Provider>
  2427→  );
  2428→}
  2429→
  2430→export function useLanguage() {
  2431→  const ctx = useContext(LanguageContext);
  2432→  if (!ctx) {
  2433→    return {
  2434→      lang: "en" as Lang,
  2435→      setLang: () => {},
  2436→      t: (k: string, vars?: Record<string, string>) =>
  2437→        format(translations.en?.[k] ?? k, vars),
  2438→    };
  2439→  }
  2440→  return ctx;
  2441→}
