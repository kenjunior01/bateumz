   104→function shuffle<T>(arr: T[]): T[] {
   105→  const a = [...arr];
   106→  for (let i = a.length - 1; i > 0; i--) {
   107→    const j = Math.floor(Math.random() * (i + 1));
   108→    [a[i], a[j]] = [a[j], a[i]];
   109→  }
   110→  return a;
   111→}
