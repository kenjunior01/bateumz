    65→function pickPattern(total: number, count: number): number[] {
    66→  const pool = Array.from({ length: total }, (_, i) => i);
    67→  for (let i = pool.length - 1; i > 0; i--) {
    68→    const j = Math.floor(Math.random() * (i + 1));
    69→    [pool[i], pool[j]] = [pool[j], pool[i]];
    70→  }
    71→  return pool.slice(0, Math.min(count, total)).sort((a, b) => a - b);
    72→}
    73→
    74→function evalSelection(pattern: Set<number>, sel: Set<number>): RoundResult {
    75→  let correct = 0;
    76→  let wrong = 0;
    77→  sel.forEach((i) => (pattern.has(i) ? correct++ : wrong++));
    78→  const missed = [...pattern].filter((i) => !sel.has(i)).length;
    79→  const points = correct * 10 - wrong * 5;
    80→  return { correct, wrong, missed, points };
    81→}

   191→export default function PatternMemory({ onScore, liveCode }: Props) {
   192→  /* ── state ── */
   193→  const [phase, setPhase] = useState<Phase>('intro');
   194→  const [round, setRound] = useState(0);
   195→  const [pattern, setPattern] = useState<Set<number>>(new Set());
   196→  const [p1Sel, setP1Sel] = useState<Set<number>>(new Set());
   197→  const [p2Sel, setP2Sel] = useState<Set<number>>(new Set());
   198→  const [p1Total, setP1Total] = useState(0);
   199→  const [p2Total, setP2Total] = useState(0);
   200→  const [lastResult, setLastResult] = useState<{ p1: RoundResult; p2: RoundResult } | null>(null);
   201→  const [memProgress, setMemProgress] = useState(100);
   202→  const [countdownNum, setCountdownNum] = useState(3);
   203→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   204→  const scoresRef = useRef({ p1: 0, p2: 0 });
   205→  const onScoreCalled = useRef(false);
   206→
   207→  const cfg = ROUNDS[round] ?? ROUNDS[ROUNDS.length - 1];
   208→  const memMs = MEM_DURATION_MS[round] ?? 3000;
   209→  const patternArr = [...pattern];
   210→  const cs = cellSize(cfg.gridSize);
   211→
   212→  /* ── cleanup helper ── */
   213→  const clearTimer = useCallback(() => {
   214→    if (timerRef.current) {
   215→      clearInterval(timerRef.current);
   216→      timerRef.current = null;
   217→    }
   218→  }, []);
   219→
   220→  /* ── round bootstrap ── */
   221→  const initRound = useCallback((r: number) => {
   222→    const c = ROUNDS[r];
   223→    const pat = pickPattern(c.gridSize * c.gridSize, c.patternSize);
   224→    setPattern(new Set(pat));
   225→    setP1Sel(new Set());
   226→    setP2Sel(new Set());
   227→    setMemProgress(100);
   228→    setLastResult(null);
   229→    setCountdownNum(3);
   230→    setPhase('countdown');
   231→  }, []);
   232→
   233→  /* ── countdown timer ── */
   234→  useEffect(() => {
   235→    if (phase !== 'countdown') return;
   236→    clearTimer();
   237→    timerRef.current = setInterval(() => {
   238→      setCountdownNum((prev) => {
   239→        const next = prev - 1;
   240→        if (next <= 0) {
   241→          clearTimer();
   242→          setPhase('memorize');
   243→          return 0;
   244→        }
   245→        return next;
   246→      });
   247→    }, 700);
   248→    return clearTimer;
   249→  }, [phase, clearTimer]);
   250→
   251→  /* ── memorize timer ── */
   252→  useEffect(() => {
   253→    if (phase !== 'memorize') return;
   254→    clearTimer();
   255→    const start = Date.now();
   256→    timerRef.current = setInterval(() => {
   257→      const elapsed = Date.now() - start;
   258→      const pct = Math.max(0, 100 - (elapsed / memMs) * 100);
   259→      setMemProgress(pct);
   260→      if (elapsed >= memMs) {
   261→        clearTimer();
   262→        setPhase('p1_turn');
   263→      }
   264→    }, 50);
   265→    return clearTimer;
   266→  }, [phase, memMs, clearTimer]);
   267→
   268→  /* ── handlers ── */
   269→  const handleStart = useCallback(() => {
   270→    scoresRef.current = { p1: 0, p2: 0 };
   271→    onScoreCalled.current = false;
   272→    setRound(0);
   273→    setP1Total(0);
   274→    setP2Total(0);
   275→    setLastResult(null);
   276→    initRound(0);
   277→  }, [initRound]);
   278→
   279→  const handleTap = useCallback(
   280→    (idx: number) => {
   281→      if (phase !== 'p1_turn' && phase !== 'p2_turn') return;
   282→      const setter = phase === 'p1_turn' ? setP1Sel : setP2Sel;
   283→      setter((prev) => {
   284→        const next = new Set(prev);
   285→        if (next.has(idx)) next.delete(idx);
   286→        else next.add(idx);
   287→        return next;
   288→      });
   289→    },
   290→    [phase],
   291→  );
   292→
   293→  const handleConfirm = useCallback(
   294→    (player: 1 | 2) => {
   295→      if (player === 1 && phase === 'p1_turn') {
   296→        setPhase('p2_turn');
   297→        return;
   298→      }
   299→      if (player === 2 && phase === 'p2_turn') {
   300→        const p1r = evalSelection(pattern, p1Sel);
   301→        const p2r = evalSelection(pattern, p2Sel);
   302→        setLastResult({ p1: p1r, p2: p2r });
   303→        scoresRef.current.p1 += p1r.points;
   304→        scoresRef.current.p2 += p2r.points;
   305→        setP1Total(scoresRef.current.p1);
   306→        setP2Total(scoresRef.current.p2);
   307→        setPhase('round_result');
   308→      }
   309→    },
   310→    [phase, pattern, p1Sel, p2Sel],
   311→  );
   312→
   313→  const handleNextRound = useCallback(() => {
   314→    const nr = round + 1;
   315→    if (nr >= TOTAL_ROUNDS) {
   316→      setPhase('game_over');
   317→      if (onScore && !onScoreCalled.current) {
   318→        onScoreCalled.current = true;
   319→        const { p1, p2 } = scoresRef.current;
   320→        if (p1 > p2) {
   321→          onScore('Jogador 1', p1);
   322→        } else if (p2 > p1) {
   323→          onScore('Jogador 2', p2);
   324→        } else {
   325→          onScore('Jogador 1', p1);
   326→          onScore('Jogador 2', p2);
   327→        }
   328→      }
   329→    } else {
   330→      setRound(nr);
   331→      initRound(nr);
   332→    }
   333→  }, [round, initRound, onScore]);
   334→
   335→  const handleRestart = useCallback(() => {
   336→    clearTimer();
   337→    handleStart();
   338→  }, [clearTimer, handleStart]);
   339→
   340→  /* ── derived ── */
   341→  const currentSel = phase === 'p1_turn' ? p1Sel : phase === 'p2_turn' ? p2Sel : new Set<number>();
   342→  const activePlayer: 1 | 2 = phase === 'p1_turn' ? 1 : 2;
   343→
   344→  /* ================================================================ */
   345→/*  RENDER                                                           */
   346→/* ================================================================ */
   347→
   348→  return (
   349→    <div className="min-h-[480px] bg-gradient-to-b from-zinc-900 to-zinc-950 text-white flex flex-col items-center justify-center p-4 select-none overflow-hidden rounded-2xl border border-zinc-800/60 shadow-2xl">
   350→      {/* ── Score bar ── */}
   351→      <AnimatePresence>
   352→        {phase !== 'intro' && (
   353→          <motion.div
   354→            initial={{ opacity: 0, y: -12 }}
   355→            animate={{ opacity: 1, y: 0 }}
   356→            exit={{ opacity: 0 }}
   357→            className="w-full flex justify-center"
   358→          >
   359→            <ScoreBar
   360→              p1Score={p1Total}
   361→              p2Score={p2Total}
   362→              round={round}
   363→              highlight={phase === 'p1_turn' ? 1 : phase === 'p2_turn' ? 2 : undefined}
   364→            />
   365→          </motion.div>
   366→        )}
   367→      </AnimatePresence>
   368→
   369→      {/* ── Main content ── */}
   370→      <AnimatePresence mode="wait">
   371→        {/* ==================== INTRO ==================== */}
   372→        {phase === 'intro' && (
   373→          <motion.div
   374→            key="intro"
   375→            {...fadeSlide}
   376→            className="flex flex-col items-center gap-5 text-center max-w-sm"
   377→          >
   378→            <motion.div
   379→              initial={{ scale: 0, rotate: -20 }}
   380→              animate={{ scale: 1, rotate: 0 }}
   381→              transition={{ delay: 0.1, type: 'spring', stiffness: 200, damping: 15 }}
   382→              className="w-20 h-20 rounded-2xl bg-gradient-to-br from-cyan-500/20 to-pink-500/20 border border-zinc-700/60 flex items-center justify-center mb-1"
   383→            >
   384→              <Brain className="w-10 h-10 text-cyan-400" />
   385→            </motion.div>
   386→
   387→            <h1 className="text-3xl sm:text-4xl font-extrabold tracking-tight">
   388→              <span className="bg-gradient-to-r from-cyan-400 to-cyan-300 bg-clip-text text-transparent">
   389→                Memória
   390→              </span>{' '}
   391→              <span className="bg-gradient-to-r from-pink-400 to-pink-300 bg-clip-text text-transparent">
   392→                VS
   393→              </span>
   394→            </h1>
   395→
   396→            <p className="text-zinc-400 text-sm leading-relaxed">
   397→              Um padrão acende brevemente na grade.
   398→              <br />
   399→              Cada jogador recria de memória.
   400→              <br />
   401→              <span className="text-cyan-400/80">Acerto = +10 pts</span>{' '}
   402→              <span className="text-zinc-600">•</span>{' '}
   403→              <span className="text-red-400/80">Erro = −5 pts</span>
   404→            </p>
   405→
   406→            <div className="flex items-center gap-3 text-xs">
   407→              <Badge className="bg-cyan-500/20 text-cyan-300 border border-cyan-500/30 px-3 py-0.5">
   408→                Jogador 1
   409→              </Badge>
   410→              <span className="text-zinc-500 font-extrabold text-sm">VS</span>
   411→              <Badge className="bg-pink-500/20 text-pink-300 border border-pink-500/30 px-3 py-0.5">
   412→                Jogador 2
   413→              </Badge>
   414→            </div>
   415→
   416→            <div className="flex flex-col gap-0.5 text-[11px] text-zinc-500 mt-1">

   421→            <Button

   442→                key={countdownNum}
   443→                initial={{ scale: 2.5, opacity: 0 }}

   448→                  'text-8xl font-black tabular-nums',
   449→                  countdownNum === 3 && 'text-cyan-400',

   454→                {countdownNum}

   456→            </AnimatePresence>
   457→            <p className="text-zinc-600 text-xs">

   462→

   468→            className="flex flex-col items-center gap-4"

   485→                  className="h-full bg-gradient-to-r from-amber-500 to-orange-500 rounded-full"
   486→                  initial={{ width: '100%' }}
   487→                  animate={{ width: `${memProgress}%` }}
   488→                  transition={{ duration: 0.05 }}

   491→              <span className="text-xs text-amber-300 tabular-nums w-8 text-right">
   492→                {((memProgress / 100) * (memMs / 1000)).toFixed(1)}s
   493→              </span>
   494→            </div>
   495→
   496→            {/* Grid with staggered lit cells */}
   497→            <div
   498→              className="grid gap-2"
   499→              style={{
   500→                gridTemplateColumns: `repeat(${cfg.gridSize}, ${cs}px)`,
   501→              }}
   502→            >
   503→              {Array.from({ length: cfg.gridSize * cfg.gridSize }, (_, i) => {
   504→                const isLit = pattern.has(i);
   505→                const litIdx = patternArr.indexOf(i);
   506→                const delay = isLit ? litIdx * 0.08 : 0;
   507→                return (
   508→                  <motion.div
   509→                    key={i}
   510→                    initial={{
   511→                      backgroundColor: 'rgba(39,39,42,0.6)',
   512→                      scale: 0.7,
   513→                      boxShadow: '0 0 0px 0px transparent',

   526→                      'rounded-xl border-2',
   527→                      isLit && 'animate-pulse',

   531→                );

   537→        {/* ==================== P1 / P2 TURN ==================== */}

   540→            key={phase}
   541→            {...popIn}

   546→              selected={currentSel.size}
