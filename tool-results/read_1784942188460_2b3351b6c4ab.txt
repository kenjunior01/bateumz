    82→
    83→let _id = 0;
    84→const uid = (p: string) => `\${p}-\${++_id}`;
    85→
    86→const shuffle = <T,>(a: T[]): T[] => {
    87→  const b = [...a];
    88→  for (let i = b.length - 1; i > 0; i--) {
    89→    const j = Math.floor(Math.random() * (i + 1));
    90→    [b[i], b[j]] = [b[j], b[i]];
    91→  }
    92→  return b;
    93→};
    94→
    95→const pick = <T,>(a: T[]): T => a[Math.floor(Math.random() * a.length)];
    96→
    97→const otherColor = (c: string) => {
    98→  const pool = COLORS.filter((v) => v !== c);
    99→  return pick(pool);
   100→};
   101→
   102→const calcSpeedBonus = (timeLeft: number): number => {
   103→  if (timeLeft >= 50) return 5;
   104→  if (timeLeft >= 40) return 3;
   105→  if (timeLeft >= 25) return 1;
   106→  return 0;
   107→};

   113→function makeShape(id: string): Shape {
   114→  const type = pick<ShapeType>(['circle', 'rectangle', 'triangle']);
   115→  const w = 28 + Math.floor(Math.random() * 52);
   116→  return {
   117→    id,
   118→    type,
   119→    x: 3 + Math.random() * 78,
   120→    y: 3 + Math.random() * 72,
   121→    width: w,
   122→    height: type === 'circle' ? w : 28 + Math.floor(Math.random() * 52),
   123→    color: pick(COLORS),
   124→    rotation: Math.floor(Math.random() * 360),
   125→  };
   126→}
   127→
   128→function generateBaseScene(): Shape[] {
   129→  const shapes: Shape[] = [];
   130→  for (let i = 0; i < SHAPES_COUNT; i++) shapes.push(makeShape(uid('s')));
   131→  return shapes;
   132→}
   133→
   134→function createRound(): {
   135→  base: Shape[];
   136→  modified: Shape[];
   137→  diffs: Difference[];
   138→} {
   139→  const base = generateBaseScene();
   140→  const modified = base.map((s) => ({ ...s }));
   141→  const diffs: Difference[] = [];
   142→  const candidates = shuffle(base);
   143→  const types = shuffle(ALL_DIFF_TYPES);
   144→
   145→  for (let i = 0; i < 5; i++) {
   146→    const t = types[i];
   147→    const did = uid('d');
   148→
   149→    if (t === 'extra') {
   150→      const ns = makeShape(uid('ex'));
   151→      ns.x = 5 + Math.random() * 78;
   152→      ns.y = 5 + Math.random() * 70;
   153→      modified.push(ns);
   154→      diffs.push({
   155→        id: did, modShapeId: ns.id, type: t,
   156→        found: false, foundBy: null,
   157→        baseX: ns.x, baseY: ns.y, modX: ns.x, modY: ns.y,
   158→      });
   159→      continue;
   160→    }
   161→
   162→    const sh = candidates[i];
   163→
   164→    if (t === 'missing') {
   165→      const idx = modified.findIndex((s) => s.id === sh.id);
   166→      if (idx !== -1) modified.splice(idx, 1);
   167→      diffs.push({
   168→        id: did, baseShapeId: sh.id, type: t,
   169→        found: false, foundBy: null,
   170→        baseX: sh.x, baseY: sh.y, modX: sh.x, modY: sh.y,
   171→      });
   172→    } else if (t === 'color_change') {
   173→      const m = modified.find((s) => s.id === sh.id)!;
   174→      m.color = otherColor(sh.color);
   175→      diffs.push({
   176→        id: did, baseShapeId: sh.id, modShapeId: sh.id, type: t,
   177→        found: false, foundBy: null,
   178→        baseX: sh.x, baseY: sh.y, modX: m.x, modY: m.y,
   179→      });
   180→    } else if (t === 'position_shift') {
   181→      const m = modified.find((s) => s.id === sh.id)!;
   182→      const dx = (Math.random() > 0.5 ? 1 : -1) * (10 + Math.random() * 12);
   183→      const dy = (Math.random() > 0.5 ? 1 : -1) * (10 + Math.random() * 12);
   184→      m.x = Math.max(3, Math.min(82, m.x + dx));
   185→      m.y = Math.max(3, Math.min(75, m.y + dy));
   186→      diffs.push({
   187→        id: did, baseShapeId: sh.id, modShapeId: sh.id, type: t,
   188→        found: false, foundBy: null,
   189→        baseX: sh.x, baseY: sh.y, modX: m.x, modY: m.y,
   190→      });
   191→    } else if (t === 'size_change') {
   192→      const m = modified.find((s) => s.id === sh.id)!;
   193→      const bigger = Math.random() > 0.5;
   194→      const f = bigger ? 1.4 + Math.random() * 0.5 : 0.4 + Math.random() * 0.35;
   195→      m.width = Math.max(14, Math.min(100, Math.floor(m.width * f)));
   196→      m.height = m.type === 'circle'
   197→        ? m.width
   198→        : Math.max(14, Math.min(100, Math.floor(m.height * f)));
   199→      diffs.push({
   200→        id: did, baseShapeId: sh.id, modShapeId: sh.id, type: t,
   201→        found: false, foundBy: null,
   202→        baseX: sh.x, baseY: sh.y, modX: m.x, modY: m.y,
   203→      });
   204→    }
   205→  }
   206→
   207→  return { base, modified, diffs };
   208→}

   330→
   331→function ScenePanel({
   332→  shapes, diffs, floats, onShapeClick, onFloatDone, side, playing,
   333→}: {
   334→  shapes: Shape[];
   335→  diffs: Difference[];
   336→  floats: FloatFeedback[];
   337→  onShapeClick: (shapeId: string, px: number, py: number) => void;
   338→  onFloatDone: (id: string) => void;
   339→  side: 'base' | 'mod';
   340→  playing: boolean;
   341→}) {
   342→  const foundIds = new Set(
   343→    diffs
   344→      .filter((d) => d.found)
   345→      .flatMap((d) => [d.baseShapeId, d.modShapeId].filter(Boolean) as string[]),
   346→  );
   347→
   348→  const sideFloats = floats.filter((f) => {
   349→    return side === 'base'
   350→      ? diffs.some((d) => Math.abs(d.baseX - f.x) < 10 && Math.abs(d.baseY - f.y) < 10)
   351→      : diffs.some((d) => Math.abs(d.modX - f.x) < 10 && Math.abs(d.modY - f.y) < 10);
   352→  });
   353→
   354→  return (
   355→    <div className="relative aspect-[4/3] w-full overflow-hidden rounded-lg border border-white/10 bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900">
   356→      {/* subtle grid */}
   357→      <div
   358→        className="pointer-events-none absolute inset-0 opacity-[0.04]"
   359→        style={{
   360→          backgroundImage:
   361→            'linear-gradient(rgba(255,255,255,.15) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,.15) 1px, transparent 1px)',
   362→          backgroundSize: '40px 40px',
   363→        }}
   364→      />
   365→
   366→      {/* label */}
   367→      <div className="pointer-events-none absolute left-2 top-2 z-10">
   368→        <Badge variant="secondary" className="text-[10px] font-medium uppercase tracking-wider opacity-60">
   369→          {side === 'base' ? 'Cena A' : 'Cena B'}
   370→        </Badge>
   371→      </div>
   372→
   373→      {/* shapes */}
   374→      {shapes.map((s) => (
   375→        <ShapeEl
   376→          key={s.id}
   377→          shape={s}
   378→          disabled={!playing}
   379→          isFound={foundIds.has(s.id)}
   380→          onClick={() => onShapeClick(s.id, s.x, s.y)}
   381→        />
   382→      ))}
   383→
   384→      {/* found markers */}
   385→      {diffs
   386→        .filter((d) => d.found)
   387→        .map((d) => (
   388→          <FoundMarker
   389→            key={\`mk-\${d.id}-\${side}\`}
   390→            x={side === 'base' ? d.baseX : d.modX}
   391→            y={side === 'base' ? d.baseY : d.modY}
   392→            player={d.foundBy!}
   393→          />
   394→        ))}
   395→
   396→      {/* floating feedbacks */}
   397→      {sideFloats.map((f) => (
   398→        <FloatingText key={f.id} fb={f} onDone={() => onFloatDone(f.id)} />
   399→      ))}
   400→    </div>
   401→  );
   402→}

   408→export default function SpotDifference({ onScore, liveCode }: Props) {

   412→  const [p1Score, setP1Score] = useState(0);
   413→  const [p2Score, setP2Score] = useState(0);
   414→  const [activePlayer, setActivePlayer] = useState<Player>(1);
   415→  const [baseShapes, setBaseShapes] = useState<Shape[]>([]);
   416→  const [modShapes, setModShapes] = useState<Shape[]>([]);
   417→  const [diffs, setDiffs] = useState<Difference[]>([]);
   418→  const [floats, setFloats] = useState<FloatFeedback[]>([]);
   419→  const [roundP1, setRoundP1] = useState(0);
   420→  const [roundP2, setRoundP2] = useState(0);
   421→  const [sceneKey, setSceneKey] = useState(0);
   422→
   423→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   424→
   425→  const foundCount = diffs.filter((d) => d.found).length;
   426→  const p1Found = diffs.filter((d) => d.found && d.foundBy === 1).length;
   427→  const p2Found = diffs.filter((d) => d.found && d.foundBy === 2).length;
   428→
   429→  /* ---- start round ---- */
   430→  const startRound = useCallback(() => {
   431→    _id = 0;
   432→    const { base, modified, diffs: d } = createRound();
   433→    setBaseShapes(base);
   434→    setModShapes(modified);
   435→    setDiffs(d);
   436→    setTimeLeft(ROUND_TIME);
   437→    setFloats([]);
   438→    setRoundP1(0);

   447→    if (round >= TOTAL_ROUNDS) {
   448→      setStatus('gameOver');
   449→      onScore?.('Jogador 1', p1Score);
   450→      onScore?.('Jogador 2', p2Score);
   451→    } else {
   452→      setStatus('roundEnd');
   453→    }
   454→  }, [round, p1Score, p2Score, onScore]);

   457→  useEffect(() => {
   458→    if (status !== 'playing') return;
   459→    timerRef.current = setInterval(() => {
   460→      setTimeLeft((prev) => {
   461→        if (prev <= 1) {
   462→          clearInterval(timerRef.current!);
   463→          timerRef.current = null;
   464→          setTimeout(() => {
   465→            setDiffs((d) => d.map((dd) => ({ ...dd, found: true })));
   466→          }, 300);
   467→          setTimeout(() => endRound(), 2200);
   468→          return 0;
   469→        }
   470→        return prev - 1;
   471→      });
   472→    }, 1000);
   473→    return () => { if (timerRef.current) clearInterval(timerRef.current); };
   474→  }, [status, endRound]);
   475→
   476→  /* ---- early end when all found ---- */

   479→    const t = setTimeout(() => endRound(), 1200);

   570→    <div className="relative flex min-h-[520px] w-full flex-col items-center gap-4 rounded-2xl border border-white/10 bg-gradient-to-br from-slate-950 via-slate-900 to-slate-950 p-4">
   571→
   572→      {/* active-player bar */}

   575→          className="absolute inset-x-0 top-0 h-1 rounded-t-2xl"
   576→          style={{ backgroundColor: activePlayer === 1 ? P1_COLOR : P2_COLOR }}
