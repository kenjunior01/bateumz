    10→interface Props {

    15→type GameState = "waiting" | "playing" | "gameOver";

    17→type Direction = "up" | "down" | "left" | "right";
    18→
    19→interface Point {
    20→  x: number;
    21→  y: number;
    22→}
    23→
    24→interface Snake {
    25→  body: Point[];
    26→  direction: Direction;
    27→  nextDirection: Direction;
    28→  growing: boolean;
    29→}
    30→
    31→const GRID_SIZE = 20;
    32→
    33→const SPEED_MAP: Record<SpeedLevel, number> = {
    34→  slow: 200,
    35→  normal: 130,
    36→  fast: 75,
    37→};
    38→
    39→const INITIAL_SNAKE_1: Snake = {
    40→  body: [
    41→    { x: 3, y: 10 },
    42→    { x: 2, y: 10 },
    43→    { x: 1, y: 10 },
    44→  ],
    45→  direction: "right",
    46→  nextDirection: "right",
    47→  growing: false,
    48→};
    49→
    50→const INITIAL_SNAKE_2: Snake = {
    51→  body: [
    52→    { x: 16, y: 10 },
    53→    { x: 17, y: 10 },
    54→    { x: 18, y: 10 },
    55→  ],
    56→  direction: "left",
    57→  nextDirection: "left",
    58→  growing: false,
    59→};
    60→
    61→const OPPOSITE: Record<Direction, Direction> = {
    62→  up: "down",
    63→  down: "up",
    64→  left: "right",
    65→  right: "left",
    66→};
    67→
    68→const DIRECTION_DELTA: Record<Direction, Point> = {
    69→  up: { x: 0, y: -1 },
    70→  down: { x: 0, y: 1 },
    71→  left: { x: -1, y: 0 },
    72→  right: { x: 1, y: 0 },
    73→};
    74→
    75→function randomFood(snake1: Snake, snake2: Snake): Point {
    76→  const occupied = new Set<string>();
    77→  for (const p of snake1.body) occupied.add(`${p.x},${p.y}`);
    78→  for (const p of snake2.body) occupied.add(`${p.x},${p.y}`);
    79→  let pt: Point;
    80→  do {
    81→    pt = {
    82→      x: Math.floor(Math.random() * GRID_SIZE),
    83→      y: Math.floor(Math.random() * GRID_SIZE),
    84→    };
    85→  } while (occupied.has(`${pt.x},${pt.y}`));
    86→  return pt;
    87→}
    88→
    89→function cloneSnake(s: Snake): Snake {
    90→  return {
    91→    body: s.body.map((p) => ({ ...p })),
    92→    direction: s.direction,
    93→    nextDirection: s.nextDirection,
    94→    growing: s.growing,
    95→  };
    96→}
    97→
    98→function pointsEqual(a: Point, b: Point): boolean {
    99→  return a.x === b.x && a.y === b.y;
   100→}
   101→
   102→export default function SnakeBattle({ onScore, liveCode }: Props) {

   104→  const containerRef = useRef<HTMLDivElement>(null);
   105→  const gameLoopRef = useRef<ReturnType<typeof setInterval> | null>(null);
   106→  const lastTickRef = useRef<number>(0);
   107→  const rafRef = useRef<number>(0);
   108→
   109→  const [gameState, setGameState] = useState<GameState>("waiting");
   110→  const [speed, setSpeed] = useState<SpeedLevel>("normal");
   111→  const [round, setRound] = useState(1);
   112→  const [score1, setScore1] = useState(0);
   113→  const [score2, setScore2] = useState(0);
   114→  const [length1, setLength1] = useState(3);
   115→  const [length2, setLength2] = useState(3);

   126→  useEffect(() => {
   127→    gameStateRef.current = gameState;

   129→
   130→  useEffect(() => {
   131→    speedRef.current = speed;

   164→      gameLoopRef.current = null;

   166→    if (rafRef.current) {
   167→      cancelAnimationFrame(rafRef.current);
   168→      rafRef.current = 0;
   169→    }
   170→    setRound(1);
   171→    setScore1(0);
   172→    setScore2(0);
   173→    resetRound();
   174→  }, [resetRound]);
   175→
   176→  const endRound = useCallback(
   177→    (w: "Jogador 1" | "Jogador 2" | "empate") => {
   178→      if (gameLoopRef.current) {
   179→        clearInterval(gameLoopRef.current);
   180→        gameLoopRef.current = null;
   181→      }
   182→      if (rafRef.current) {
   183→        cancelAnimationFrame(rafRef.current);
   184→        rafRef.current = 0;
   185→      }
   186→      setWinner(w);
   187→      setGameState("gameOver");

   189→      if (w === "Jogador 1") {

   191→        setScore1(newScore);

   194→        const newScore = score2 + 10 + length2;
   195→        setScore2(newScore);
   196→        onScore?.("Jogador 2", newScore);
   197→      }
   198→    },
   199→    [onScore, score1, score2, length1, length2]
   200→  );
   201→
   202→  const tick = useCallback(() => {
   203→    const s1 = snake1Ref.current;
   204→    const s2 = snake2Ref.current;
   205→    const food = foodRef.current;
   206→
   207→    s1.direction = s1.nextDirection;
   208→    s2.direction = s2.nextDirection;
   209→
   210→    const d1 = DIRECTION_DELTA[s1.direction];
   211→    const d2 = DIRECTION_DELTA[s2.direction];
   212→
   213→    const newHead1: Point = {
   214→      x: s1.body[0].x + d1.x,
   215→      y: s1.body[0].y + d1.y,
   216→    };
   217→    const newHead2: Point = {
   218→      x: s2.body[0].x + d2.x,
   219→      y: s2.body[0].y + d2.y,
   220→    };
   221→
   222→    const hitWall1 =
   223→      newHead1.x < 0 || newHead1.x >= GRID_SIZE || newHead1.y < 0 || newHead1.y >= GRID_SIZE;
   224→    const hitWall2 =
   225→      newHead2.x < 0 || newHead2.x >= GRID_SIZE || newHead2.y < 0 || newHead2.y >= GRID_SIZE;
   226→
   227→    const hitSelf1 = s1.body.some((p) => pointsEqual(p, newHead1));
   228→    const hitSelf2 = s2.body.some((p) => pointsEqual(p, newHead2));
   229→
   230→    const hitOther1 = s2.body.some((p) => pointsEqual(p, newHead1));
   231→    const hitOther2 = s1.body.some((p) => pointsEqual(p, newHead2));
   232→
   233→    const headCollision = pointsEqual(newHead1, newHead2);
   234→
   235→    const dead1 = hitWall1 || hitSelf1 || hitOther1;
   236→    const dead2 = hitWall2 || hitSelf2 || hitOther2;
   237→
   238→    if (dead1 && dead2) {
   239→      endRound("empate");
   240→      return;
   241→    }
   242→    if (dead1) {
   243→      endRound("Jogador 2");
   244→      return;
   245→    }
   246→    if (dead2) {
   247→      endRound("Jogador 1");
   248→      return;
   249→    }
   250→    if (headCollision) {
   251→      endRound("empate");
   252→      return;
   253→    }
   254→
   255→    s1.body.unshift(newHead1);
   256→    s2.body.unshift(newHead2);
   257→
   258→    const ate1 = pointsEqual(newHead1, food);
   259→    const ate2 = pointsEqual(newHead2, food);
   260→
   261→    if (ate1) {
   262→      setLength1((l) => l + 1);
   263→      setFlashScore(1);
   264→      setTimeout(() => setFlashScore(null), 400);
   265→      foodRef.current = randomFood(s1, s2);
   266→    } else {
   267→      s1.body.pop();
   268→    }
   269→
   270→    if (ate2) {
   271→      setLength2((l) => l + 1);
   272→      setFlashScore(2);
   273→      setTimeout(() => setFlashScore(null), 400);
   274→      foodRef.current = randomFood(s1, s2);
   275→    } else {
   276→      s2.body.pop();
   277→    }
   278→  }, [endRound]);
   279→
   280→  const draw = useCallback(() => {
   281→    const canvas = canvasRef.current;
   282→    if (!canvas) return;
   283→    const ctx = canvas.getContext("2d");
   284→    if (!ctx) return;
   285→
   286→    const w = canvas.width;
   287→    const h = canvas.height;
   288→    const cell = w / GRID_SIZE;
   289→
   290→    ctx.clearRect(0, 0, w, h);
   291→
   292→    ctx.fillStyle = "#0f172a";
   293→    ctx.fillRect(0, 0, w, h);
   294→
   295→    ctx.strokeStyle = "rgba(148, 163, 184, 0.06)";
   296→    ctx.lineWidth = 0.5;
   297→    for (let i = 0; i <= GRID_SIZE; i++) {
   298→      const pos = i * cell;
   299→      ctx.beginPath();
   300→      ctx.moveTo(pos, 0);
   301→      ctx.lineTo(pos, h);
   302→      ctx.stroke();
   303→      ctx.beginPath();
   304→      ctx.moveTo(0, pos);
   305→      ctx.lineTo(w, pos);
   306→      ctx.stroke();
   307→    }
   308→
   309→    const food = foodRef.current;
   310→    const fx = food.x * cell + cell / 2;
   311→    const fy = food.y * cell + cell / 2;
   312→    const foodRadius = cell * 0.35;
   313→
   314→    const glowRadius = cell * 0.9;
   315→    const glow = ctx.createRadialGradient(fx, fy, 0, fx, fy, glowRadius);
   316→    glow.addColorStop(0, "rgba(251, 191, 36, 0.35)");
   317→    glow.addColorStop(1, "rgba(251, 191, 36, 0)");
   318→    ctx.fillStyle = glow;
   319→    ctx.beginPath();
   320→    ctx.arc(fx, fy, glowRadius, 0, Math.PI * 2);
   321→    ctx.fill();
   322→
   323→    const foodGrad = ctx.createRadialGradient(
   324→      fx - foodRadius * 0.3,
   325→      fy - foodRadius * 0.3,
   326→      0,
   327→      fx,
   328→      fy,
   329→      foodRadius
   330→    );
   331→    foodGrad.addColorStop(0, "#fde68a");
   332→    foodGrad.addColorStop(0.6, "#fbbf24");
   333→    foodGrad.addColorStop(1, "#d97706");
   334→    ctx.fillStyle = foodGrad;
   335→    ctx.beginPath();
   336→    ctx.arc(fx, fy, foodRadius, 0, Math.PI * 2);
   337→    ctx.fill();
   338→
   339→    const drawSnake = (
   340→      snake: Snake,
   341→      color1: string,
   342→      color2: string,
   343→      glowColor: string
   344→    ) => {
   345→      const body = snake.body;
   346→      for (let i = body.length - 1; i >= 0; i--) {
   347→        const seg = body[i];
   348→        const t = i / Math.max(body.length - 1, 1);
   349→        const cx = seg.x * cell + cell / 2;
   350→        const cy = seg.y * cell + cell / 2;
   351→
   352→        const isHead = i === 0;
   353→        const segSize = isHead ? cell * 0.45 : cell * 0.38 - t * cell * 0.08;
   354→
   355→        if (isHead) {
   356→          const headGlow = ctx.createRadialGradient(cx, cy, 0, cx, cy, cell * 0.7);
   357→          headGlow.addColorStop(0, glowColor);
   358→          headGlow.addColorStop(1, "rgba(0,0,0,0)");
   359→          ctx.fillStyle = headGlow;
   360→          ctx.beginPath();
   361→          ctx.arc(cx, cy, cell * 0.7, 0, Math.PI * 2);
   362→          ctx.fill();
   363→        }
   364→
   365→        const segGrad = ctx.createRadialGradient(
   366→          cx - segSize * 0.3,
   367→          cy - segSize * 0.3,
   368→          0,
   369→          cx,
   370→          cy,
   371→          segSize
   372→        );
   373→        segGrad.addColorStop(0, color1);
   374→        segGrad.addColorStop(1, color2);
   375→        ctx.fillStyle = segGrad;
   376→
   377→        ctx.beginPath();
   378→        ctx.roundRect(
   379→          seg.x * cell + (cell - segSize * 2) / 2,
   380→          seg.y * cell + (cell - segSize * 2) / 2,
   381→          segSize * 2,
   382→          segSize * 2,
   383→          isHead ? segSize * 0.4 : segSize * 0.35
   384→        );
   385→        ctx.fill();
   386→
   387→        if (isHead) {
   388→          const dir = snake.direction;
   389→          const eyeOffset = cell * 0.14;
   390→          const eyeRadius = cell * 0.07;
   391→          const pupilRadius = cell * 0.04;
   392→
   393→          let e1: Point, e2: Point;
   394→          if (dir === "right") {
   395→            e1 = { x: cx + eyeOffset * 0.5, y: cy - eyeOffset };
   396→            e2 = { x: cx + eyeOffset * 0.5, y: cy + eyeOffset };
   397→          } else if (dir === "left") {
   398→            e1 = { x: cx - eyeOffset * 0.5, y: cy - eyeOffset };
   399→            e2 = { x: cx - eyeOffset * 0.5, y: cy + eyeOffset };
   400→          } else if (dir === "up") {
   401→            e1 = { x: cx - eyeOffset, y: cy - eyeOffset * 0.5 };
   402→            e2 = { x: cx + eyeOffset, y: cy - eyeOffset * 0.5 };
   403→          } else {
   404→            e1 = { x: cx - eyeOffset, y: cy + eyeOffset * 0.5 };
   405→            e2 = { x: cx + eyeOffset, y: cy + eyeOffset * 0.5 };
   406→          }
   407→
   408→          ctx.fillStyle = "#ffffff";
   409→          ctx.beginPath();
   410→          ctx.arc(e1.x, e1.y, eyeRadius, 0, Math.PI * 2);
   411→          ctx.fill();
   412→          ctx.beginPath();
   413→          ctx.arc(e2.x, e2.y, eyeRadius, 0, Math.PI * 2);
   414→          ctx.fill();
   415→
   416→          const pupilShift = cell * 0.03;
   417→          const pd = DIRECTION_DELTA[dir];
   418→          ctx.fillStyle = "#0f172a";
   419→          ctx.beginPath();
   420→          ctx.arc(e1.x + pd.x * pupilShift, e1.y + pd.y * pupilShift, pupilRadius, 0, Math.PI * 2);
   421→          ctx.fill();
   422→          ctx.beginPath();
   423→          ctx.arc(e2.x + pd.x * pupilShift, e2.y + pd.y * pupilShift, pupilRadius, 0, Math.PI * 2);
   424→          ctx.fill();
   425→        }
   426→      }
   427→    };
   428→
   429→    drawSnake(
   430→      snake1Ref.current,
   431→      "#22d3ee",
   432→      "#0891b2",
   433→      "rgba(34, 211, 238, 0.2)"
   434→    );
   435→    drawSnake(
   436→      snake2Ref.current,
   437→      "#f472b6",
   438→      "#c026d3",
   439→      "rgba(244, 114, 182, 0.2)"

   441→  }, []);
   442→

   461→  }, [gameLoop]);
   462→
   463→  useEffect(() => {
   464→    const handleKey = (e: KeyboardEvent) => {

   467→      const s1 = snake1Ref.current;
   468→      const s2 = snake2Ref.current;
   469→
   470→      switch (e.key.toLowerCase()) {

   472→          if (s1.direction !== "down") s1.nextDirection = "up";
   473→          e.preventDefault();
   474→          break;
   475→        case "s":
   476→          if (s1.direction !== "up") s1.nextDirection = "down";
   477→          e.preventDefault();
   478→          break;
   479→        case "a":
   480→          if (s1.direction !== "right") s1.nextDirection = "left";
   481→          e.preventDefault();
   482→          break;
   483→        case "d":
   484→          if (s1.direction !== "left") s1.nextDirection = "right";
   485→          e.preventDefault();
   486→          break;
   487→        case "arrowup":
   488→          if (s2.direction !== "down") s2.nextDirection = "up";
   489→          e.preventDefault();
   490→          break;
   491→        case "arrowdown":
   492→          if (s2.direction !== "up") s2.nextDirection = "down";
   493→          e.preventDefault();
   494→          break;
   495→        case "arrowleft":
   496→          if (s2.direction !== "right") s2.nextDirection = "left";
   497→          e.preventDefault();
   498→          break;
   499→        case "arrowright":
   500→          if (s2.direction !== "left") s2.nextDirection = "right";
   501→          e.preventDefault();
   502→          break;
   503→      }
   504→    };
   505→    window.addEventListener("keydown", handleKey);
   506→    return () => window.removeEventListener("keydown", handleKey);
   507→  }, []);
   508→
   509→  const startGame = useCallback(() => {
   510→    resetRound();
   511→    setGameState("playing");
   512→    lastTickRef.current = performance.now();
   513→  }, [resetRound]);
   514→
   515→  const nextRound = useCallback(() => {
   516→    const nextR = round + 1;
   517→    setRound(nextR);
   518→    resetRound();
   519→    setGameState("playing");
   520→    lastTickRef.current = performance.now();
   521→  }, [round, resetRound]);
   522→
   523→  const setDirection = useCallback((player: 1 | 2, dir: Direction) => {
   524→    if (gameStateRef.current !== "playing") return;
   525→    const snake = player === 1 ? snake1Ref.current : snake2Ref.current;
   526→    if (OPPOSITE[dir] !== snake.direction) {
   527→      snake.nextDirection = dir;
   528→    }
   529→  }, []);
   530→
   531→  const speedLabel: Record<SpeedLevel, string> = {
   532→    slow: "Lento",
   533→    normal: "Normal",
   534→    fast: "Rápido",
   535→  };
   536→
   537→  return (
   538→    <div className="flex flex-col items-center gap-3 w-full max-w-[520px] mx-auto select-none">
   539→      <div className="bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 rounded-xl p-3 w-full">
   540→        <div className="flex items-center justify-between">
   541→          <div className="flex flex-col items-center gap-1 min-w-[80px]">
   542→            <span className="text-xs text-cyan-300 font-semibold tracking-wide uppercase">
   543→              Jogador 1
   544→            </span>
   545→            <motion.div
   546→              key={length1}
   547→              initial={{ scale: 1.4, color: "#22d3ee" }}
   548→              animate={{ scale: 1, color: "#94a3b8" }}
   549→              transition={{ duration: 0.3 }}
   550→              className="text-xl font-bold"
   551→            >
   552→              {length1}
   553→            </motion.div>
   554→            <motion.span
   555→              key={`cum1-${score1}`}
   556→              initial={{ scale: flashScore === 1 ? 1.5 : 1 }}
   557→              animate={{ scale: 1 }}
   558→              className="text-[10px] text-slate-400"
   559→            >
   560→              Total: {score1}
   561→            </motion.span>
   562→          </div>
   563→
   564→          <div className="flex flex-col items-center gap-1">
   565→            <Badge
   566→              variant="outline"
   567→              className="border-amber-500/30 text-amber-400 text-[10px] bg-amber-500/5"
   568→            >
   569→              <Coins className="w-3 h-3 mr-1" />
   570→              Round {round}
   571→            </Badge>
   572→            <div className="flex items-center gap-1.5 mt-1">
   573→              <span className="text-[10px] text-slate-500">Velocidade:</span>
   574→              <div className="flex gap-1">
   575→                {(Object.keys(SPEED_MAP) as SpeedLevel[]).map((s) => (
   576→                  <button
   577→                    key={s}
   578→                    onClick={() => gameState === "waiting" && setSpeed(s)}
