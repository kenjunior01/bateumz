    75→    75→function randomFood(snake1: Snake, snake2: Snake): Point {
    76→    76→  const occupied = new Set<string>();
    77→    77→  for (const p of snake1.body) occupied.add(`${p.x},${p.y}`);
    78→    78→  for (const p of snake2.body) occupied.add(`${p.x},${p.y}`);
    79→    79→  let pt: Point;
    80→    80→  do {
    81→    81→    pt = {
    82→    82→      x: Math.floor(Math.random() * GRID_SIZE),
    83→    83→      y: Math.floor(Math.random() * GRID_SIZE),
    84→    84→    };
    85→    85→  } while (occupied.has(`${pt.x},${pt.y}`));
    86→    86→  return pt;
    87→    87→}
