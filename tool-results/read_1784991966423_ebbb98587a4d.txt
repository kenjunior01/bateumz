     1→import { useEffect, useState, useCallback, useRef, useMemo } from "react";
     2→import { motion, AnimatePresence } from "framer-motion";
     3→import { RotateCcw, Timer, Palette } from "lucide-react";
     4→import { Button } from "@/components/ui/button";
     5→import { Badge } from "@/components/ui/badge";
     6→import { cn } from "@/lib/utils";
     7→
     8→/* ------------------------------------------------------------------ */
     9→/*  Types                                                              */
    10→/* ------------------------------------------------------------------ */
    11→
    12→interface Props {
    13→  onScore?: (name: string, score: number) => void;

    16→
    17→interface ColorDef {
    18→  id: string;
    19→  name: string;
    20→  hex: string;
    21→  tailwind: string;
    22→}
    23→
    24→type GamePhase =
    25→  | "idle"
    26→  | "countdown"
    27→  | "playing"
    28→  | "roundResult"
    29→  | "gameOver";
    30→

    32→  word: ColorDef;
    33→  inkColor: ColorDef;
    34→  choices: ColorDef[];

    36→
    37→/* ------------------------------------------------------------------ */

    39→/* ------------------------------------------------------------------ */
    40→
    41→const ALL_COLORS: ColorDef[] = [
    42→  { id: "vermelho", name: "Vermelho", hex: "#EF4444", tailwind: "bg-red-500" },
    43→  { id: "azul",     name: "Azul",     hex: "#3B82F6", tailwind: "bg-blue-500" },
    44→  { id: "verde",    name: "Verde",    hex: "#22C55E", tailwind: "bg-green-500" },
    45→  { id: "amarelo",  name: "Amarelo",  hex: "#EAB308", tailwind: "bg-yellow-500" },
    46→  { id: "roxo",     name: "Roxo",     hex: "#A855F7", tailwind: "bg-purple-500" },
    47→  { id: "laranja",  name: "Laranja",  hex: "#F97316", tailwind: "bg-orange-500" },
    48→];
    49→
    50→const TOTAL_ROUNDS = 15;
    51→const INITIAL_TIME_MS = 5000;
    52→const MIN_TIME_MS = 2000;
    53→
    54→/* ------------------------------------------------------------------ */
    55→/*  Helpers                                                            */
    56→/* ------------------------------------------------------------------ */
    57→
    58→function shuffle<T>(arr: T[]): T[] {
    59→  const a = [...arr];
    60→  for (let i = a.length - 1; i > 0; i--) {
    61→    const j = Math.floor(Math.random() * (i + 1));
    62→    [a[i], a[j]] = [a[j], a[i]];
    63→  }
    64→  return a;
    65→}
    66→
    67→function makeRound(): RoundData {
    68→  const word = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    69→  let ink: ColorDef;
    70→  do {
    71→    ink = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    72→  } while (ink.id === word.id);
    73→  const others = shuffle(ALL_COLORS.filter((c) => c.id !== ink.id)).slice(0, 3);
    74→  return { word, inkColor: ink, choices: shuffle([ink, ...others]) };
    75→}
    76→
    77→function roundDuration(roundNum: number): number {
    78→  const t = (roundNum - 1) / (TOTAL_ROUNDS - 1);
    79→  return Math.round(INITIAL_TIME_MS - t * (INITIAL_TIME_MS - MIN_TIME_MS));
    80→}
    81→
    82→/* ------------------------------------------------------------------ */
    83→/*  Animation variants                                                 */
    84→/* ------------------------------------------------------------------ */
    85→
    86→const wordIn = {
    87→  hidden:  { scale: 0, opacity: 0, rotate: -12 },
    88→  visible: { scale: 1, opacity: 1, rotate: 0, transition: { type: "spring", stiffness: 260, damping: 20 } },
    89→  exit:    { scale: 0, opacity: 0, rotate: 12 },
    90→};
    91→
    92→const countPop = {
    93→  initial: { scale: 0, opacity: 0 },
    94→  animate: { scale: 1, opacity: 1, transition: { type: "spring", stiffness: 300, damping: 15 } },
    95→  exit:    { scale: 2.5, opacity: 0, transition: { duration: 0.22 } },
    96→};
    97→
    98→/* ------------------------------------------------------------------ */
    99→/*  Component                                                          */
   100→/* ------------------------------------------------------------------ */
   101→
   102→export default function ColorMatch({ onScore, liveCode }: Props) {
   103→  /* state */
   104→  const [phase, setPhase]               = useState<GamePhase>("idle");
   105→  const [round, setRound]               = useState(1);
   106→  const [p1Score, setP1Score]           = useState(0);
   107→  const [p2Score, setP2Score]           = useState(0);
   108→  const [p1Streak, setP1Streak]         = useState(0);
   109→  const [p2Streak, setP2Streak]         = useState(0);
   110→  const [p1MaxStreak, setP1MaxStreak]   = useState(0);
   111→  const [p2MaxStreak, setP2MaxStreak]   = useState(0);
   112→  const [roundData, setRoundData]       = useState<RoundData | null>(null);
   113→  const [timerProgress, setTimerProgress] = useState(1);
   114→  const [countdownVal, setCountdownVal] = useState(3);
   115→  const [lastWinner, setLastWinner]     = useState<1 | 2 | "timeout" | null>(null);
   116→  const [lastCorrect, setLastCorrect]    = useState<string | null>(null);
   117→
   118→  /* per-button feedback – tracked via unique keys to force re-mount for animation */
   119→  const [p1Feedback, setP1Feedback] = useState<{ colorId: string; type: "correct" | "wrong" } | null>(null);
   120→  const [p2Feedback, setP2Feedback] = useState<{ colorId: string; type: "correct" | "wrong" } | null>(null);
   121→  const [feedbackTick, setFeedbackTick] = useState(0);
   122→
   123→  /* score bounce trigger */
   124→  const [p1Bounce, setP1Bounce] = useState(0);
   125→  const [p2Bounce, setP2Bounce] = useState(0);
   126→
   127→  /* refs */
   128→  const timerRef  = useRef<ReturnType<typeof setTimeout> | null>(null);
   129→  const tickRef   = useRef<ReturnType<typeof setInterval> | null>(null);
   130→  const lockedRef = useRef(false);
   131→
   132→  /* cleanup */
   133→  const clearTimers = useCallback(() => {
   134→    if (timerRef.current) { clearTimeout(timerRef.current); timerRef.current = null; }
   135→    if (tickRef.current)  { clearInterval(tickRef.current);  tickRef.current = null; }
   136→    lockedRef.current = false;
   137→  }, []);
   138→
   139→  useEffect(() => () => clearTimers(), [clearTimers]);
   140→
   141→  /* ---- start / restart game ---- */
   142→  const startGame = useCallback(() => {
   143→    clearTimers();
   144→    setP1Score(0); setP2Score(0);
   145→    setP1Streak(0); setP2Streak(0);
   146→    setP1MaxStreak(0); setP2MaxStreak(0);
   147→    setRound(1); setCountdownVal(3);
   148→    setLastWinner(null); setLastCorrect(null);
   149→    setP1Feedback(null); setP2Feedback(null);
   150→    setFeedbackTick(0);
   151→    setP1Bounce(0); setP2Bounce(0);
   152→    setPhase("countdown");
   153→  }, [clearTimers]);
   154→
   155→  /* ---- countdown tick ---- */
   156→  useEffect(() => {
   157→    if (phase !== "countdown") return;
   158→    if (countdownVal <= 0) {
   159→      const rd = makeRound();
   160→      setRoundData(rd);
   161→      setTimerProgress(1);
   162→      setLastWinner(null);
   163→      setLastCorrect(null);
   164→      setP1Feedback(null); setP2Feedback(null);
   165→      setFeedbackTick((t) => t + 1);
   166→      lockedRef.current = false;
   167→      setPhase("playing");
   168→      return;
   169→    }
   170→    const id = setTimeout(() => setCountdownVal((v) => v - 1), 800);
   171→    return () => clearTimeout(id);
   172→  }, [phase, countdownVal]);
   173→
   174→  /* ---- round timer (starts when phase goes to "playing") ---- */
   175→  useEffect(() => {
   176→    if (phase !== "playing") return;
   177→    const dur = roundDuration(round);
   178→    const t0 = performance.now();
   179→
   180→    tickRef.current = setInterval(() => {
   181→      const elapsed = performance.now() - t0;
   182→      setTimerProgress(Math.max(0, 1 - elapsed / dur));
   183→    }, 30);
   184→
   185→    timerRef.current = setTimeout(() => {
   186→      clearTimers();
   187→      setLastWinner("timeout");
   188→      setP1Streak(0); setP2Streak(0);
   189→      setPhase("roundResult");
   190→    }, dur);
   191→
   192→    return () => clearTimers();
   193→    // eslint-disable-next-line react-hooks/exhaustive-deps
   194→  }, [phase, round]);
   195→
   196→  /* ---- handle answer ---- */
   197→  const handleAnswer = useCallback(
   198→    (player: 1 | 2, colorId: string) => {
   199→      if (lockedRef.current || phase !== "playing" || !roundData) return;
   200→
   201→      const isCorrect = colorId === roundData.inkColor.id;
   202→
   203→      if (isCorrect) {
   204→        lockedRef.current = true;
   205→        clearTimers();
   206→
   207→        const isP1 = player === 1;
   208→        const prevStreak = isP1 ? p1Streak : p2Streak;
   209→        const newStreak  = prevStreak + 1;
   210→        const bonus       = Math.min(newStreak, 5); // up to +5 streak bonus
   211→
   212→        if (isP1) {
   213→          setP1Score((s) => s + 1 + bonus);
   214→          setP1Streak(newStreak);
   215→          setP1MaxStreak((m) => Math.max(m, newStreak));
   216→          setP2Streak(0);
   217→          setP1Bounce((b) => b + 1);
   218→          setP1Feedback({ colorId, type: "correct" });
   219→        } else {
   220→          setP2Score((s) => s + 1 + bonus);
   221→          setP2Streak(newStreak);
   222→          setP2MaxStreak((m) => Math.max(m, newStreak));
   223→          setP1Streak(0);
   224→          setP2Bounce((b) => b + 1);
   225→          setP2Feedback({ colorId, type: "correct" });
   226→        }
   227→
   228→        setFeedbackTick((t) => t + 1);
   229→        setLastWinner(player);
   230→        setLastCorrect(roundData.inkColor.name);
   231→        setPhase("roundResult");
   232→      } else {
   233→        const isP1 = player === 1;
   234→        if (isP1) {
   235→          setP1Score((s) => Math.max(0, s - 1));
   236→          setP1Streak(0);
   237→          setP1Bounce((b) => b + 1);
   238→          setP1Feedback({ colorId, type: "wrong" });
   239→        } else {
   240→          setP2Score((s) => Math.max(0, s - 1));
   241→          setP2Streak(0);
   242→          setP2Bounce((b) => b + 1);
   243→          setP2Feedback({ colorId, type: "wrong" });
   244→        }
   245→        setFeedbackTick((t) => t + 1);
   246→      }
   247→    },
   248→    [phase, roundData, p1Streak, p2Streak, clearTimers],
   249→  );
   250→
   251→  /* ---- advance / finish ---- */
   252→  const advanceRound = useCallback(() => {
   253→    if (round >= TOTAL_ROUNDS) {
   254→      setPhase("gameOver");
   255→      if (p1Score !== p2Score) {
   256→        const winner = p1Score > p2Score ? "Jogador 1" : "Jogador 2";
   257→        onScore?.(winner, Math.max(p1Score, p2Score));
   258→      }
   259→    } else {
   260→      setRound((r) => r + 1);
   261→      setCountdownVal(3);
   262→      setP1Feedback(null); setP2Feedback(null);
   263→      setFeedbackTick((t) => t + 1);
   264→      setPhase("countdown");
   265→    }
   266→  }, [round, p1Score, p2Score, onScore]);
   267→
   268→  /* ---- back to menu ---- */
   269→  const goToMenu = useCallback(() => {
   270→    clearTimers();
   271→    setPhase("idle");
   272→  }, [clearTimers]);
   273→
   274→  /* ---- derived ---- */
   275→  const speedLevel = useMemo(() => Math.ceil((round / TOTAL_ROUNDS) * 10), [round]);
   276→  const timerGradient = useMemo(() => {
   277→    if (timerProgress > 0.5)  return "from-green-500 to-green-400";
   278→    if (timerProgress > 0.25) return "from-yellow-500 to-orange-400";
   279→    return "from-red-500 to-red-400";
   280→  }, [timerProgress]);
   281→
   282→  /* ================================================================ */
   283→  /*  RENDER                                                          */
   284→  /* ================================================================ */
   285→
   286→  return (
   287→    <div className="relative flex min-h-[600px] w-full max-w-2xl flex-col items-center justify-center rounded-2xl border border-white/10 bg-gray-950 px-4 py-6 sm:px-6">
   288→      {/* ==================== IDLE ==================== */}
   289→      <AnimatePresence mode="wait">
   290→        {phase === "idle" && (
   291→          <motion.div
   292→            key="idle"
   293→            initial={{ opacity: 0, y: 30 }}
   294→            animate={{ opacity: 1, y: 0 }}
   295→            exit={{ opacity: 0, y: -20 }}
   296→            className="flex flex-col items-center gap-6 text-center"
   297→          >
   298→            <div className="flex items-center gap-3">
   299→              <Palette className="h-10 w-10 text-purple-400" />
   300→              <h1 className="text-3xl font-bold bg-gradient-to-r from-cyan-400 to-pink-400 bg-clip-text text-transparent">
   301→                Cor versus Palavra
   302→              </h1>
   303→            </div>
   304→
   305→            <p className="max-w-md text-gray-400 text-sm leading-relaxed">
   306→              A palavra mostrada estará escrita numa{" "}
   307→              <span className="text-white font-semibold">COR diferente</span> do que diz.
   308→              Você deve identificar a{" "}
   309→              <span className="text-cyan-400 font-semibold">COR DA TINTA</span>, não a palavra!
   310→              Dois jogadores competem — o mais rápido ganha o ponto.
   311→            </p>
   312→
   313→            <div className="flex flex-wrap justify-center gap-2">
   314→              {ALL_COLORS.map((c) => (
   315→                <span
   316→                  key={c.id}
   317→                  className="inline-flex items-center rounded-full px-3 py-1 text-xs font-medium text-white"
   318→                  style={{ backgroundColor: c.hex }}
   319→                >
   320→                  {c.name}
   321→                </span>
   322→              ))}
   323→            </div>
   324→
   325→            <Button
   326→              size="lg"
   327→              onClick={startGame}
   328→              className="mt-2 bg-gradient-to-r from-cyan-600 to-pink-600 hover:from-cyan-500 hover:to-pink-500 text-white font-bold px-8 text-lg"
   329→            >
   330→              Jogar
   331→            </Button>
   332→          </motion.div>
   333→        )}
   334→
   335→        {/* ==================== COUNTDOWN ==================== */}
   336→        {phase === "countdown" && countdownVal > 0 && (
   337→          <motion.div
   338→            key={`cd-${countdownVal}`}
   339→            variants={countPop}
   340→            initial="initial"
   341→            animate="animate"
   342→            exit="exit"
   343→            className="flex flex-col items-center gap-4"
   344→          >
   345→            <span className="text-8xl font-black text-white drop-shadow-[0_0_30px_rgba(168,85,247,0.6)]">
   346→              {countdownVal}
   347→            </span>
   348→            <span className="text-gray-400 text-sm">Preparar...</span>
   349→          </motion.div>
   350→        )}
   351→
   352→        {/* ==================== PLAYING / ROUND RESULT ==================== */}
   353→        {(phase === "playing" || phase === "roundResult") && roundData && (
   354→          <motion.div
   355→            key={`round-${round}`}
   356→            initial={{ opacity: 0 }}
   357→            animate={{ opacity: 1 }}
   358→            exit={{ opacity: 0 }}
   359→            className="flex w-full flex-col items-center gap-4"
   360→          >
   361→            {/* ---- Scoreboard ---- */}
   362→            <div className="flex w-full items-center justify-between rounded-xl bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 px-3 py-2 sm:px-5 sm:py-3">
   363→              <div className="flex items-center gap-2">
   364→                <Badge className="bg-cyan-600/80 text-white text-xs">P1</Badge>
   365→                <motion.span
   366→                  key={`p1s-${p1Bounce}`}
   367→                  animate={{ y: [0, -14, 0] }}
   368→                  transition={{ type: "spring", stiffness: 400, damping: 12 }}
   369→                  className="text-xl font-bold text-cyan-300"
   370→                >
   371→                  {p1Score}
   372→                </motion.span>
   373→                {p1Streak >= 2 && (
   374→                  <motion.span
   375→                    key={`p1st-${p1Streak}`}
   376→                    initial={{ scale: 0.5 }}
   377→                    animate={{ scale: 1 }}
   378→                    className="text-sm"
   379→                  >
   380→                    🔥{p1Streak}
   381→                  </motion.span>
   382→                )}
   383→              </div>
   384→
   385→              <span className="text-xs font-semibold tracking-widest text-gray-400 uppercase sm:text-sm">
   386→                Cor vs Palavra
   387→              </span>
   388→
   389→              <div className="flex items-center gap-2">
   390→                {p2Streak >= 2 && (
   391→                  <motion.span
   392→                    key={`p2st-${p2Streak}`}
   393→                    initial={{ scale: 0.5 }}
   394→                    animate={{ scale: 1 }}
   395→                    className="text-sm"
   396→                  >
   397→                    🔥{p2Streak}
   398→                  </motion.span>
   399→                )}
   400→                <motion.span
   401→                  key={`p2s-${p2Bounce}`}
   402→                  animate={{ y: [0, -14, 0] }}
   403→                  transition={{ type: "spring", stiffness: 400, damping: 12 }}
   404→                  className="text-xl font-bold text-pink-300"
   405→                >
   406→                  {p2Score}
   407→                </motion.span>
   408→                <Badge className="bg-pink-600/80 text-white text-xs">P2</Badge>
   409→              </div>
   410→            </div>
   411→
   412→            {/* ---- Round / Timer / Speed ---- */}
   413→            <div className="flex w-full items-center gap-3 text-xs text-gray-400 sm:text-sm">
   414→              <span className="font-medium whitespace-nowrap">
   415→                Round {round}/{TOTAL_ROUNDS}
   416→              </span>
   417→
   418→              <div className="relative h-3 flex-1 overflow-hidden rounded-full bg-gray-800">
   419→                <motion.div
   420→                  className={cn("absolute inset-y-0 left-0 rounded-full bg-gradient-to-r", timerGradient)}
   421→                  animate={{ width: `${timerProgress * 100}%` }}
   422→                  transition={{ ease: "linear", duration: 0.05 }}
   423→                />
   424→              </div>
   425→
   426→              <div className="flex items-center gap-1 whitespace-nowrap">
   427→                <Timer className="h-3.5 w-3.5" />
   428→                <span>Vel. {speedLevel}</span>
   429→              </div>
   430→            </div>
   431→
   432→            {/* ---- Question ---- */}
   433→            <p className="text-sm text-gray-400 mt-1 sm:text-base">
   434→              Qual a COR do texto?
   435→            </p>
   436→
   437→            {/* ---- Word display ---- */}
   438→            <div className="flex min-h-[120px] w-full items-center justify-center rounded-2xl border border-white/5 bg-gray-900/60 px-8 py-6 sm:min-h-[140px]">
   439→              <AnimatePresence mode="wait">
   440→                <motion.span
   441→                  key={`${roundData.word.id}-${roundData.inkColor.id}`}
   442→                  variants={wordIn}
   443→                  initial="hidden"
   444→                  animate="visible"
   445→                  exit="exit"
   446→                  className="select-none text-5xl font-black tracking-wider sm:text-7xl"
   447→                  style={{ color: roundData.inkColor.hex }}
   448→                >
   449→                  {roundData.word.name}
   450→                </motion.span>
   451→              </AnimatePresence>
   452→            </div>
   453→
   454→            {/* ---- Round result message ---- */}
   455→            <AnimatePresence>
   456→              {phase === "roundResult" && lastWinner && (
   457→                <motion.div
   458→                  initial={{ opacity: 0, y: -10 }}
   459→                  animate={{ opacity: 1, y: 0 }}
   460→                  exit={{ opacity: 0 }}
   461→                  className={cn(
   462→                    "rounded-lg px-4 py-2 text-sm font-bold",
   463→                    lastWinner === "timeout" && "bg-yellow-500/20 text-yellow-300",
   464→                    lastWinner === 1 && "bg-cyan-500/20 text-cyan-300",
   465→                    lastWinner === 2 && "bg-pink-500/20 text-pink-300",
   466→                  )}
   467→                >
   468→                  {lastWinner === "timeout" && "Tempo esgotado!"}
   469→                  {lastWinner === 1 && "Jogador 1 — Correto! ✅"}
   470→                  {lastWinner === 2 && "Jogador 2 — Correto! ✅"}
   471→                  {lastCorrect && (
   472→                    <span className="ml-1 text-gray-400">(era {lastCorrect})</span>
   473→                  )}
   474→                </motion.div>
   475→              )}
   476→            </AnimatePresence>
   477→
   478→            {/* ---- Player 1 & 2 buttons ---- */}
   479→            <div className="flex w-full flex-col items-center gap-6 sm:flex-row sm:justify-center sm:gap-10">
   480→              {/* Player 1 */}
   481→              <div className="flex flex-col items-center gap-1">
   482→                <span className="mb-1 text-xs font-bold text-cyan-400 tracking-widest">
   483→                  JOGADOR 1
   484→                </span>
   485→                <div className="grid grid-cols-2 gap-2">
   486→                  {roundData.choices.map((c) => {
   487→                    const fb = p1Feedback;
   488→                    const isHit = fb?.colorId === c.id;
   489→                    return (
   490→                      <motion.button
   491→                        key={`p1-${c.id}-${feedbackTick}`}
   492→                        whileTap={{ scale: 0.92 }}
   493→                        initial={
   494→                          isHit && fb?.type === "correct"
   495→                            ? { scale: 1 }
   496→                            : isHit && fb?.type === "wrong"
   497→                              ? { x: 0 }
   498→                              : {}
   499→                        }
   500→                        animate={
   501→                          isHit && fb?.type === "correct"
   502→                            ? {
   503→                                scale: [1, 1.18, 1],
   504→                                boxShadow: [
   505→                                  "0 0 0 0 rgba(34,197,94,0)",
   506→                                  "0 0 0 12px rgba(34,197,94,0.6)",
   507→                                  "0 0 0 0 rgba(34,197,94,0)",
   508→                                ],
   509→                                transition: { duration: 0.45 },
   510→                              }
   511→                            : isHit && fb?.type === "wrong"
   512→                              ? {
   513→                                  x: [0, -10, 10, -10, 10, 0],
   514→                                  transition: { duration: 0.4 },
   515→                                }
   516→                            : {}
   517→                        }
   518→                        onClick={() => handleAnswer(1, c.id)}
