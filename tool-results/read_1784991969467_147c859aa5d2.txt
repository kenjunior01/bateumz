    58→    58→function shuffle<T>(arr: T[]): T[] {
    59→    59→  const a = [...arr];
    60→    60→  for (let i = a.length - 1; i > 0; i--) {
    61→    61→    const j = Math.floor(Math.random() * (i + 1));
    62→    62→    [a[i], a[j]] = [a[j], a[i]];
    63→    63→  }
    64→    64→  return a;
    65→    65→}
    66→    66→
    67→    67→function makeRound(): RoundData {
    68→    68→  const word = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    69→    69→  let ink: ColorDef;
    70→    70→  do {
    71→    71→    ink = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    72→    72→  } while (ink.id === word.id);
    73→    73→  const others = shuffle(ALL_COLORS.filter((c) => c.id !== ink.id)).slice(0, 3);
    74→    74→  return { word, inkColor: ink, choices: shuffle([ink, ...others]) };
    75→    75→}
    76→    76→
    77→    77→function roundDuration(roundNum: number): number {
    78→    78→  const t = (roundNum - 1) / (TOTAL_ROUNDS - 1);
    79→    79→  return Math.round(INITIAL_TIME_MS - t * (INITIAL_TIME_MS - MIN_TIME_MS));
    80→    80→}

   101→   101→
   102→   102→export default function ColorMatch({ onScore, liveCode }: Props) {
   103→   103→  /* state */
   104→   104→  const [phase, setPhase]               = useState<GamePhase>("idle");
   105→   105→  const [round, setRound]               = useState(1);
   106→   106→  const [p1Score, setP1Score]           = useState(0);
   107→   107→  const [p2Score, setP2Score]           = useState(0);
   108→   108→  const [p1Streak, setP1Streak]         = useState(0);
   109→   109→  const [p2Streak, setP2Streak]         = useState(0);
   110→   110→  const [p1MaxStreak, setP1MaxStreak]   = useState(0);
   111→   111→  const [p2MaxStreak, setP2MaxStreak]   = useState(0);
   112→   112→  const [roundData, setRoundData]       = useState<RoundData | null>(null);
   113→   113→  const [timerProgress, setTimerProgress] = useState(1);
   114→   114→  const [countdownVal, setCountdownVal] = useState(3);
   115→   115→  const [lastWinner, setLastWinner]     = useState<1 | 2 | "timeout" | null>(null);
   116→   116→  const [lastCorrect, setLastCorrect]    = useState<string | null>(null);
   117→   117→
   118→   118→  /* per-button feedback – tracked via unique keys to force re-mount for animation */
   119→   119→  const [p1Feedback, setP1Feedback] = useState<{ colorId: string; type: "correct" | "wrong" } | null>(null);
   120→   120→  const [p2Feedback, setP2Feedback] = useState<{ colorId: string; type: "correct" | "wrong" } | null>(null);
   121→   121→  const [feedbackTick, setFeedbackTick] = useState(0);
   122→   122→
   123→   123→  /* score bounce trigger */
   124→   124→  const [p1Bounce, setP1Bounce] = useState(0);
   125→   125→  const [p2Bounce, setP2Bounce] = useState(0);
   126→   126→
   127→   127→  /* refs */
   128→   128→  const timerRef  = useRef<ReturnType<typeof setTimeout> | null>(null);
   129→   129→  const tickRef   = useRef<ReturnType<typeof setInterval> | null>(null);
   130→   130→  const lockedRef = useRef(false);
   131→   131→
   132→   132→  /* cleanup */
   133→   133→  const clearTimers = useCallback(() => {
   134→   134→    if (timerRef.current) { clearTimeout(timerRef.current); timerRef.current = null; }
   135→   135→    if (tickRef.current)  { clearInterval(tickRef.current);  tickRef.current = null; }
   136→   136→    lockedRef.current = false;
   137→   137→  }, []);
   138→   138→
   139→   139→  useEffect(() => () => clearTimers(), [clearTimers]);
   140→   140→
   141→   141→  /* ---- start / restart game ---- */
   142→   142→  const startGame = useCallback(() => {
   143→   143→    clearTimers();
   144→   144→    setP1Score(0); setP2Score(0);
   145→   145→    setP1Streak(0); setP2Streak(0);
   146→   146→    setP1MaxStreak(0); setP2MaxStreak(0);
   147→   147→    setRound(1); setCountdownVal(3);
   148→   148→    setLastWinner(null); setLastCorrect(null);
   149→   149→    setP1Feedback(null); setP2Feedback(null);
   150→   150→    setFeedbackTick(0);
   151→   151→    setP1Bounce(0); setP2Bounce(0);
   152→   152→    setPhase("countdown");
   153→   153→  }, [clearTimers]);
   154→   154→
   155→   155→  /* ---- countdown tick ---- */
   156→   156→  useEffect(() => {
   157→   157→    if (phase !== "countdown") return;
   158→   158→    if (countdownVal <= 0) {
   159→   159→      const rd = makeRound();
   160→   160→      setRoundData(rd);
   161→   161→      setTimerProgress(1);
   162→   162→      setLastWinner(null);
   163→   163→      setLastCorrect(null);
   164→   164→      setP1Feedback(null); setP2Feedback(null);
   165→   165→      setFeedbackTick((t) => t + 1);
   166→   166→      lockedRef.current = false;
   167→   167→      setPhase("playing");
   168→   168→      return;
   169→   169→    }
   170→   170→    const id = setTimeout(() => setCountdownVal((v) => v - 1), 800);
   171→   171→    return () => clearTimeout(id);
   172→   172→  }, [phase, countdownVal]);

   176→   176→    if (phase !== "playing") return;
   177→   177→    const dur = roundDuration(round);
   178→   178→    const t0 = performance.now();
   179→   179→
   180→   180→    tickRef.current = setInterval(() => {
   181→   181→      const elapsed = performance.now() - t0;
   182→   182→      setTimerProgress(Math.max(0, 1 - elapsed / dur));
   183→   183→    }, 30);
   184→   184→
   185→   185→    timerRef.current = setTimeout(() => {
   186→   186→      clearTimers();
   187→   187→      setLastWinner("timeout");
   188→   188→      setP1Streak(0); setP2Streak(0);
   189→   189→      setPhase("roundResult");
   190→   190→    }, dur);
   191→   191→
   192→   192→    return () => clearTimers();
   193→   193→    // eslint-disable-next-line react-hooks/exhaustive-deps
   194→   194→  }, [phase, round]);
   195→   195→
   196→   196→  /* ---- handle answer ---- */
   197→   197→  const handleAnswer = useCallback(
   198→   198→    (player: 1 | 2, colorId: string) => {
   199→   199→      if (lockedRef.current || phase !== "playing" || !roundData) return;
   200→   200→
   201→   201→      const isCorrect = colorId === roundData.inkColor.id;
   202→   202→
   203→   203→      if (isCorrect) {
   204→   204→        lockedRef.current = true;
   205→   205→        clearTimers();
   206→   206→
   207→   207→        const isP1 = player === 1;
   208→   208→        const prevStreak = isP1 ? p1Streak : p2Streak;
   209→   209→        const newStreak  = prevStreak + 1;
   210→   210→        const bonus       = Math.min(newStreak, 5); // up to +5 streak bonus
   211→   211→
   212→   212→        if (isP1) {
   213→   213→          setP1Score((s) => s + 1 + bonus);
   214→   214→          setP1Streak(newStreak);
   215→   215→          setP1MaxStreak((m) => Math.max(m, newStreak));
   216→   216→          setP2Streak(0);
   217→   217→          setP1Bounce((b) => b + 1);
   218→   218→          setP1Feedback({ colorId, type: "correct" });
   219→   219→        } else {
   220→   220→          setP2Score((s) => s + 1 + bonus);
   221→   221→          setP2Streak(newStreak);
   222→   222→          setP2MaxStreak((m) => Math.max(m, newStreak));
   223→   223→          setP1Streak(0);
   224→   224→          setP2Bounce((b) => b + 1);
   225→   225→          setP2Feedback({ colorId, type: "correct" });
   226→   226→        }
   227→   227→
   228→   228→        setFeedbackTick((t) => t + 1);
   229→   229→        setLastWinner(player);
   230→   230→        setLastCorrect(roundData.inkColor.name);
   231→   231→        setPhase("roundResult");
   232→   232→      } else {
   233→   233→        const isP1 = player === 1;
   234→   234→        if (isP1) {
   235→   235→          setP1Score((s) => Math.max(0, s - 1));
   236→   236→          setP1Streak(0);
   237→   237→          setP1Bounce((b) => b + 1);
   238→   238→          setP1Feedback({ colorId, type: "wrong" });
   239→   239→        } else {
   240→   240→          setP2Score((s) => Math.max(0, s - 1));
   241→   241→          setP2Streak(0);
   242→   242→          setP2Bounce((b) => b + 1);
   243→   243→          setP2Feedback({ colorId, type: "wrong" });
   244→   244→        }
   245→   245→        setFeedbackTick((t) => t + 1);
   246→   246→      }
   247→   247→    },
   248→   248→    [phase, roundData, p1Streak, p2Streak, clearTimers],
   249→   249→  );
   250→   250→
   251→   251→  /* ---- advance / finish ---- */
   252→   252→  const advanceRound = useCallback(() => {
   253→   253→    if (round >= TOTAL_ROUNDS) {
   254→   254→      setPhase("gameOver");
   255→   255→      if (p1Score !== p2Score) {
   256→   256→        const winner = p1Score > p2Score ? "Jogador 1" : "Jogador 2";
   257→   257→        onScore?.(winner, Math.max(p1Score, p2Score));
   258→   258→      }
   259→   259→    } else {
   260→   260→      setRound((r) => r + 1);
   261→   261→      setCountdownVal(3);
   262→   262→      setP1Feedback(null); setP2Feedback(null);
   263→   263→      setFeedbackTick((t) => t + 1);
   264→   264→      setPhase("countdown");
   265→   265→    }
   266→   266→  }, [round, p1Score, p2Score, onScore]);
   267→   267→
   268→   268→  /* ---- back to menu ---- */
   269→   269→  const goToMenu = useCallback(() => {
   270→   270→    clearTimers();
   271→   271→    setPhase("idle");
   272→   272→  }, [clearTimers]);
   273→   273→
   274→   274→  /* ---- derived ---- */
   275→   275→  const speedLevel = useMemo(() => Math.ceil((round / TOTAL_ROUNDS) * 10), [round]);
   276→   276→  const timerGradient = useMemo(() => {
   277→   277→    if (timerProgress > 0.5)  return "from-green-500 to-green-400";
   278→   278→    if (timerProgress > 0.25) return "from-yellow-500 to-orange-400";
   279→   279→    return "from-red-500 to-red-400";
   280→   280→  }, [timerProgress]);
   281→   281→
   282→   282→  /* ================================================================ */
   283→   283→  /*  RENDER                                                          */
   284→   284→  /* ================================================================ */
   285→   285→
   286→   286→  return (
   287→   287→    <div className="relative flex min-h-[600px] w-full max-w-2xl flex-col items-center justify-center rounded-2xl border border-white/10 bg-gray-950 px-4 py-6 sm:px-6">
   288→   288→      {/* ==================== IDLE ==================== */}
   289→   289→      <AnimatePresence mode="wait">
   290→   290→        {phase === "idle" && (
   291→   291→          <motion.div
   292→   292→            key="idle"
   293→   293→            initial={{ opacity: 0, y: 30 }}
   294→   294→            animate={{ opacity: 1, y: 0 }}
   295→   295→            exit={{ opacity: 0, y: -20 }}
   296→   296→            className="flex flex-col items-center gap-6 text-center"
   297→   297→          >
   298→   298→            <div className="flex items-center gap-3">
   299→   299→              <Palette className="h-10 w-10 text-purple-400" />
   300→   300→              <h1 className="text-3xl font-bold bg-gradient-to-r from-cyan-400 to-pink-400 bg-clip-text text-transparent">
   301→   301→                Cor versus Palavra
   302→   302→              </h1>
   303→   303→            </div>
   304→   304→
   305→   305→            <p className="max-w-md text-gray-400 text-sm leading-relaxed">
   306→   306→              A palavra mostrada estará escrita numa{" "}
   307→   307→              <span className="text-white font-semibold">COR diferente</span> do que diz.
   308→   308→              Você deve identificar a{" "}
   309→   309→              <span className="text-cyan-400 font-semibold">COR DA TINTA</span>, não a palavra!
   310→   310→              Dois jogadores competem — o mais rápido ganha o ponto.
   311→   311→            </p>
   312→   312→
   313→   313→            <div className="flex flex-wrap justify-center gap-2">

   316→   316→                  key={c.id}

   323→   323→            </div>
   324→   324→
   325→   325→            <Button
   326→   326→              size="lg"
   327→   327→              onClick={startGame}
   328→   328→              className="mt-2 bg-gradient-to-r from-cyan-600 to-pink-600 hover:from-cyan-500 hover:to-pink-500 text-white font-bold px-8 text-lg"
   329→   329→            >
   330→   330→              Jogar
   331→   331→            </Button>

   333→   333→        )}
   334→   334→
   335→   335→        {/* ==================== COUNTDOWN ==================== */}

   346→   346→              {countdownVal}

   351→   351→
   352→   352→        {/* ==================== PLAYING / ROUND RESULT ==================== */}
   353→   353→        {(phase === "playing" || phase === "roundResult") && roundData && (
   354→   354→          <motion.div
   355→   355→            key={`round-${round}`}
   356→   356→            initial={{ opacity: 0 }}

   363→   363→              <div className="flex items-center gap-2">
   364→   364→                <Badge className="bg-cyan-600/80 text-white text-xs">P1</Badge>
   365→   365→                <motion.span
   366→   366→                  key={`p1s-${p1Bounce}`}
   367→   367→                  animate={{ y: [0, -14, 0] }}
   368→   368→                  transition={{ type: "spring", stiffness: 400, damping: 12 }}
   369→   369→                  className="text-xl font-bold text-cyan-300"
   370→   370→                >
   371→   371→                  {p1Score}
   372→   372→                </motion.span>
   373→   373→                {p1Streak >= 2 && (

   383→   383→              </div>
   384→   384→
   385→   385→              <span className="text-xs font-semibold tracking-widest text-gray-400 uppercase sm:text-sm">
   386→   386→                Cor vs Palavra
   387→   387→              </span>

   389→   389→              <div className="flex items-center gap-2">
   390→   390→                {p2Streak >= 2 && (
   391→   391→                  <motion.span
   392→   392→                    key={`p2st-${p2Streak}`}
   393→   393→                    initial={{ scale: 0.5 }}

   396→   396→                  >
   397→   397→                    🔥{p2Streak}
   398→   398→                  </motion.span>

   400→   400→                <motion.span
   401→   401→                  key={`p2s-${p2Bounce}`}
   402→   402→                  animate={{ y: [0, -14, 0] }}
   403→   403→                  transition={{ type: "spring", stiffness: 400, damping: 12 }}
   404→   404→                  className="text-xl font-bold text-pink-300"
   405→   405→                >
   406→   406→                  {p2Score}

   411→   411→
   412→   412→            {/* ---- Round / Timer / Speed ---- */}
   413→   413→            <div className="flex w-full items-center gap-3 text-xs text-gray-400 sm:text-sm">
