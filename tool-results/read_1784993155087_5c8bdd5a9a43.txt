    42→function generateProblem(round: number): MathProblem {
    43→  const roundIndex = round - 1;
    44→  let difficulty: Difficulty;
    45→  let expression: string;
    46→  let answer: number;
    47→
    48→  if (roundIndex < 3) {
    49→    difficulty = "Fácil";
    50→    const ops = ["+", "-"];
    51→    const op = ops[Math.floor(Math.random() * ops.length)];
    52→    const a = Math.floor(Math.random() * 20) + 1;
    53→    const b = Math.floor(Math.random() * 20) + 1;
    54→    if (op === "+") {
    55→      expression = `${a} + ${b}`;
    56→      answer = a + b;
    57→    } else {
    58→      const big = Math.max(a, b);
    59→      const small = Math.min(a, b);
    60→      expression = `${big} - ${small}`;
    61→      answer = big - small;
    62→    }
    63→  } else if (roundIndex < 6) {
    64→    difficulty = "Médio";
    65→    if (Math.random() < 0.5) {
    66→      const a = Math.floor(Math.random() * 12) + 1;
    67→      const b = Math.floor(Math.random() * 12) + 1;
    68→      expression = `${a} × ${b}`;
    69→      answer = a * b;
    70→    } else {
    71→      const a = Math.floor(Math.random() * 41) + 10;
    72→      const b = Math.floor(Math.random() * 41) + 10;
    73→      expression = `${a} + ${b}`;
    74→      answer = a + b;
    75→    }
    76→  } else {
    77→    difficulty = "Difícil";
    78→    const twoStepType = Math.floor(Math.random() * 4);
    79→    if (twoStepType === 0) {
    80→      const a = Math.floor(Math.random() * 8) + 2;
    81→      const b = Math.floor(Math.random() * 8) + 2;
    82→      const c = Math.floor(Math.random() * 15) + 1;
    83→      expression = `${a} × ${b} + ${c}`;
    84→      answer = a * b + c;
    85→    } else if (twoStepType === 1) {
    86→      const a = Math.floor(Math.random() * 8) + 2;
    87→      const b = Math.floor(Math.random() * 8) + 2;
    88→      const c = Math.floor(Math.random() * (a * b - 1)) + 1;
    89→      expression = `${a * b} - ${c}`;
    90→      answer = a * b - c;
    91→    } else if (twoStepType === 2) {
    92→      const a = Math.floor(Math.random() * 10) + 2;
    93→      const b = Math.floor(Math.random() * 10) + 2;
    94→      const c = Math.floor(Math.random() * 5) + 1;
    95→      const sum = a + b;
    96→      expression = `(${a} + ${b}) × ${c}`;
    97→      answer = sum * c;
    98→    } else {
    99→      const a = Math.floor(Math.random() * 50) + 20;
   100→      const b = Math.floor(Math.random() * 30) + 10;
   101→      const c = Math.floor(Math.random() * 8) + 2;
   102→      expression = `${a} + ${b} × ${c}`;
   103→      answer = a + b * c;
   104→    }
   105→  }
   106→
   107→  return { expression, answer, difficulty };
   108→}
   109→
   110→function getDifficultyColor(d: Difficulty): string {
   111→  if (d === "Fácil") return "bg-green-500/20 text-green-400 border-green-500/30";
   112→  if (d === "Médio") return "bg-yellow-500/20 text-yellow-400 border-yellow-500/30";
   113→  return "bg-red-500/20 text-red-400 border-red-500/30";
   114→}

   116→function ConfettiParticles({ color }: { color: string }) {
   117→  const particles = Array.from({ length: 20 }, (_, i) => ({
   118→    id: i,
   119→    x: (Math.random() - 0.5) * 300,
   120→    y: -(Math.random() * 250 + 50),
   121→    rotate: Math.random() * 720 - 360,
   122→    scale: Math.random() * 0.8 + 0.3,
   123→    hue: Math.random() * 60 - 30,
   124→  }));
   125→
   126→  return (
   127→    <div className="absolute inset-0 pointer-events-none overflow-hidden z-50">
   128→      {particles.map((p) => (
   129→        <motion.div
   130→          key={p.id}
   131→          className={cn("absolute w-2 h-2 rounded-sm", color)}
   132→          style={{
   133→            left: "50%",
   134→            top: "50%",
   135→            backgroundColor: color.includes("cyan")
   136→              ? `hsl(${180 + p.hue}, 90%, 60%)`
   137→              : `hsl(${330 + p.hue}, 90%, 65%)`,
   138→          }}
   139→          initial={{ x: 0, y: 0, rotate: 0, opacity: 1 }}
   140→          animate={{
   141→            x: p.x,
   142→            y: p.y,
   143→            rotate: p.rotate,
   144→            opacity: 0,
   145→            scale: p.scale,
   146→          }}
   147→          transition={{ duration: 1.2, ease: "easeOut" }}
   148→        />
   149→      ))}
   150→    </div>
   151→  );
   152→}

   154→export default function QuickMath({ onScore, liveCode }: Props) {
   155→  const [gameState, setGameState] = useState<GameState>("idle");
   156→  const [currentRound, setCurrentRound] = useState(1);
   157→  const [problem, setProblem] = useState<MathProblem | null>(null);
   158→  const [countdown, setCountdown] = useState(COUNTDOWN_SECONDS);
   159→  const [timeLeft, setTimeLeft] = useState(15);
   160→  const [timerDuration, setTimerDuration] = useState(15);
   161→  const [wrongMode, setWrongMode] = useState<WrongMode>("retry");
   162→  const [p1Input, setP1Input] = useState("");
   163→  const [p2Input, setP2Input] = useState("");
   164→  const [p1Stats, setP1Stats] = useState<PlayerStats>({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   165→  const [p2Stats, setP2Stats] = useState<PlayerStats>({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   166→  const [p1Locked, setP1Locked] = useState(false);
   167→  const [p2Locked, setP2Locked] = useState(false);
   168→  const [p1Wrong, setP1Wrong] = useState(false);
   169→  const [p2Wrong, setP2Wrong] = useState(false);
   170→  const [roundResult, setRoundResult] = useState<RoundResult | null>(null);
   171→  const [roundStartTime, setRoundStartTime] = useState(0);
   172→  const [showConfetti, setShowConfetti] = useState(false);
   173→  const [confettiColor, setConfettiColor] = useState("cyan");
   174→  const p1Ref = useRef<HTMLInputElement>(null);
   175→  const p2Ref = useRef<HTMLInputElement>(null);
   176→  const timerInterval = useRef<ReturnType<typeof setInterval> | null>(null);
   177→  const roundEnded = useRef(false);
   178→
   179→  const clearTimer = useCallback(() => {
   180→    if (timerInterval.current) {
   181→      clearInterval(timerInterval.current);
   182→      timerInterval.current = null;
   183→    }
   184→  }, []);
   185→
   186→  const startTimer = useCallback(
   187→    (duration: number) => {
   188→      clearTimer();
   189→      setTimeLeft(duration);
   190→      const start = Date.now();
   191→      timerInterval.current = setInterval(() => {
   192→        const elapsed = (Date.now() - start) / 1000;
   193→        const remaining = Math.max(0, duration - elapsed);
   194→        setTimeLeft(remaining);
   195→        if (remaining <= 0) {
   196→          clearTimer();
   197→          if (!roundEnded.current) {
   198→            roundEnded.current = true;
   199→            setRoundResult({ winner: null, winnerName: null, p1Time: null, p2Time: null, timedOut: true });
   200→            setGameState("roundResult");
   201→            toast.error("Tempo!");
   202→          }
   203→        }
   204→      }, 50);
   205→    },
   206→    [clearTimer]
   207→  );
   208→
   209→  const startGame = useCallback(() => {
   210→    const firstProblem = generateProblem(1);
   211→    setP1Stats({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   212→    setP2Stats({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   213→    setCurrentRound(1);
   214→    setProblem(firstProblem);
   215→    setP1Input("");
   216→    setP2Input("");
   217→    setP1Locked(false);
   218→    setP2Locked(false);
   219→    setP1Wrong(false);
   220→    setP2Wrong(false);
   221→    setRoundResult(null);
   222→    setCountdown(COUNTDOWN_SECONDS);
   223→    roundEnded.current = false;
   224→    setGameState("countdown");
   225→  }, []);
   226→
   227→  useEffect(() => {
   228→    if (gameState !== "countdown") return;
   229→    if (countdown <= 0) {
   230→      setGameState("playing");
   231→      setRoundStartTime(Date.now());
   232→      startTimer(timerDuration);
   233→      return;
   234→    }
   235→    const t = setTimeout(() => setCountdown((c) => c - 1), 800);
   236→    return () => clearTimeout(t);
   237→  }, [gameState, countdown, timerDuration, startTimer]);
   238→
   239→  useEffect(() => {
   240→    if (gameState !== "playing" && gameState !== "countdown") {
   241→      clearTimer();
   242→    }
   243→    return () => clearTimer();
   244→  }, [gameState, clearTimer]);
   245→
   246→  const handleWin = useCallback(
   247→    (player: 1 | 2) => {
   248→      if (roundEnded.current) return;
   249→      roundEnded.current = true;
   250→      clearTimer();
   251→      const elapsed = (Date.now() - roundStartTime) / 1000;
   252→      const name = player === 1 ? "Jogador 1" : "Jogador 2";
   253→      const setStats = player === 1 ? setP1Stats : setP2Stats;
   254→      const otherSetStats = player === 1 ? setP2Stats : setP1Stats;
   255→
   256→      setStats((prev) => ({
   257→        correct: prev.correct + 1,
   258→        totalTime: prev.totalTime + elapsed,
   259→        streak: prev.streak + 1,
   260→        bestStreak: Math.max(prev.bestStreak, prev.streak + 1),
   261→      }));
   262→
   263→      otherSetStats((prev) => ({
   264→        ...prev,
   265→        streak: 0,
   266→      }));
   267→
   268→      setRoundResult({
   269→        winner: player,
   270→        winnerName: name,
   271→        p1Time: player === 1 ? elapsed : null,
   272→        p2Time: player === 2 ? elapsed : null,
   273→        timedOut: false,
   274→      });
   275→
   276→      setConfettiColor(player === 1 ? "cyan" : "pink");
   277→      setShowConfetti(true);
   278→      setTimeout(() => setShowConfetti(false), 1500);
   279→
   280→      setGameState("roundResult");
   281→      toast.success(`${name} — Correto! (${elapsed.toFixed(1)}s)`);
   282→    },
   283→    [clearTimer, roundStartTime]
   284→  );
   285→
   286→  const handleSubmit = useCallback(
   287→    (player: 1 | 2) => {
   288→      if (gameState !== "playing" || !problem || roundEnded.current) return;
   289→      if (player === 1 && p1Locked) return;
   290→      if (player === 2 && p2Locked) return;
   291→
   292→      const input = player === 1 ? p1Input : p2Input;
   293→      const parsed = parseFloat(input);
   294→      if (isNaN(parsed)) return;
   295→
   296→      if (parsed === problem.answer) {
   297→        if (player === 1) setP1Locked(true);
   298→        else setP2Locked(true);
   299→        handleWin(player);
   300→      } else {
   301→        if (wrongMode === "retry") {
   302→          if (player === 1) {
   303→            setP1Wrong(true);
   304→            setTimeout(() => {
   305→              setP1Wrong(false);
   306→              setP1Input("");
   307→              p1Ref.current?.focus();
   308→            }, 500);
   309→          } else {
   310→            setP2Wrong(true);
   311→            setTimeout(() => {
   312→              setP2Wrong(false);
   313→              setP2Input("");
   314→              p2Ref.current?.focus();
   315→            }, 500);
   316→          }
   317→        } else {
   318→          if (player === 1) {
   319→            setP1Locked(true);
   320→            setP1Wrong(true);
   321→          } else {
   322→            setP2Locked(true);
   323→            setP2Wrong(true);
   324→          }
   325→          const otherLocked = player === 1 ? p2Locked : p1Locked;
   326→          if (otherLocked) {
   327→            if (!roundEnded.current) {
   328→              roundEnded.current = true;
   329→              clearTimer();
   330→              const otherPlayer = player === 1 ? 2 : 1;
   331→              const otherName = otherPlayer === 1 ? "Jogador 1" : "Jogador 2";
   332→              const elapsed = (Date.now() - roundStartTime) / 1000;
   333→              const otherSetStats = otherPlayer === 1 ? setP1Stats : setP2Stats;
   334→              const loserSetStats = player === 1 ? setP2Stats : setP1Stats;
   335→
   336→              otherSetStats((prev) => ({
   337→                correct: prev.correct + 1,
   338→                totalTime: prev.totalTime + elapsed,
   339→                streak: prev.streak + 1,
   340→                bestStreak: Math.max(prev.bestStreak, prev.streak + 1),
   341→              }));
   342→              loserSetStats((prev) => ({ ...prev, streak: 0 }));
   343→
   344→              setRoundResult({
   345→                winner: otherPlayer,
   346→                winnerName: otherName,
   347→                p1Time: otherPlayer === 1 ? elapsed : null,
   348→                p2Time: otherPlayer === 2 ? elapsed : null,
   349→                timedOut: false,
   350→              });
   351→
   352→              setConfettiColor(otherPlayer === 1 ? "cyan" : "pink");
   353→              setShowConfetti(true);
   354→              setTimeout(() => setShowConfetti(false), 1500);
   355→
   356→              setGameState("roundResult");
   357→              toast.success(`${otherName} — Correto! (${elapsed.toFixed(1)}s)`);
   358→            }
   359→          }
   360→        }
   361→      }
   362→    },
   363→    [gameState, problem, p1Input, p2Input, p1Locked, p2Locked, wrongMode, handleWin, clearTimer, roundStartTime]
   364→  );
   365→
   366→  const nextRound = useCallback(() => {
   367→    if (currentRound >= TOTAL_ROUNDS) {
   368→      const p1Score = p1Stats.correct * 10 + (p1Stats.correct > 0 ? Math.max(0, Math.round((timerDuration - p1Stats.totalTime / p1Stats.correct) * 2)) : 0);
   369→      const p2Score = p2Stats.correct * 10 + (p2Stats.correct > 0 ? Math.max(0, Math.round((timerDuration - p2Stats.totalTime / p2Stats.correct) * 2)) : 0);
   370→      onScore?.("Jogador 1", p1Score);
   371→      onScore?.("Jogador 2", p2Score);
   372→      setGameState("gameOver");
   373→      return;
   374→    }
   375→
   376→    const nextProblem = generateProblem(currentRound + 1);
   377→    setCurrentRound((r) => r + 1);
   378→    setProblem(nextProblem);
   379→    setP1Input("");
   380→    setP2Input("");
   381→    setP1Locked(false);
   382→    setP2Locked(false);
   383→    setP1Wrong(false);
   384→    setP2Wrong(false);
   385→    setRoundResult(null);
   386→    roundEnded.current = false;
   387→    setGameState("playing");
   388→    setRoundStartTime(Date.now());
   389→    startTimer(timerDuration);
   390→  }, [currentRound, p1Stats, p2Stats, timerDuration, onScore, startTimer]);
   391→
   392→  const resetAll = useCallback(() => {
   393→    clearTimer();
   394→    roundEnded.current = false;
   395→    setGameState("idle");
   396→    setCurrentRound(1);
   397→    setProblem(null);
   398→    setP1Input("");
   399→    setP2Input("");
   400→    setP1Stats({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   401→    setP2Stats({ correct: 0, totalTime: 0, streak: 0, bestStreak: 0 });
   402→    setP1Locked(false);
   403→    setP2Locked(false);
   404→    setP1Wrong(false);
   405→    setP2Wrong(false);
   406→    setRoundResult(null);
   407→    setShowConfetti(false);
   408→  }, [clearTimer]);
   409→
   410→  const timerPercent = gameState === "playing" ? (timeLeft / timerDuration) * 100 : 100;
   411→  const timerBarColor = timeLeft > timerDuration * 0.5
   412→    ? "from-green-500 to-green-400"
   413→    : timeLeft > timerDuration * 0.25
   414→    ? "from-yellow-500 to-orange-400"
   415→    : "from-red-500 to-red-400";
   416→
   417→  const p1AvgTime = p1Stats.correct > 0 ? (p1Stats.totalTime / p1Stats.correct).toFixed(1) : "—";
   418→  const p2AvgTime = p2Stats.correct > 0 ? (p2Stats.totalTime / p2Stats.correct).toFixed(1) : "—";
   419→
   420→  return (
   421→    <div className="w-full max-w-3xl mx-auto flex flex-col gap-4 relative">
   422→      {showConfetti && <ConfettiParticles color={confettiColor} />}
   423→
   424→      {/* Scoreboard */}
   425→      <div className="bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 rounded-xl px-4 py-3 flex items-center justify-between gap-2">
   426→        <motion.div
   427→          className="flex items-center gap-2"
   428→          key={p1Stats.correct}
   429→          animate={{ scale: [1, 1.15, 1] }}
   430→          transition={{ duration: 0.3 }}
   431→        >
   432→          <span className="text-cyan-400 font-bold text-sm sm:text-base">Jogador 1</span>
   433→          <span className="text-cyan-300 text-lg sm:text-xl font-black">{p1Stats.correct}</span>
   434→          <span className="text-cyan-500/60 text-xs">✓</span>
   435→        </motion.div>
   436→
   437→        <div className="flex flex-col items-center">
   438→          <h2 className="text-white font-black text-sm sm:text-base tracking-wider">
   439→            DUELO DE MATEMÁTICA
   440→          </h2>
   441→          {liveCode && (
   442→            <span className="text-white/30 text-[10px] font-mono">{liveCode}</span>
   443→          )}
   444→        </div>
   445→
   446→        <motion.div
   447→          className="flex items-center gap-2"
   448→          key={p2Stats.correct}
   449→          animate={{ scale: [1, 1.15, 1] }}
   450→          transition={{ duration: 0.3 }}
   451→        >
   452→          <span className="text-pink-500/60 text-xs">✓</span>
   453→          <span className="text-pink-400 text-lg sm:text-xl font-black">{p2Stats.correct}</span>
   454→          <span className="text-pink-400 font-bold text-sm sm:text-base">Jogador 2</span>
   455→        </motion.div>
   456→      </div>
   457→
   458→      {/* Round / Timer / Difficulty */}
   459→      {gameState !== "idle" && (
   460→        <div className="flex items-center justify-between gap-2 text-xs sm:text-sm">
   461→          <span className="text-white/60 font-medium">
   462→            Round {currentRound}/{TOTAL_ROUNDS}
   463→          </span>
   464→          <div className="flex items-center gap-2">
   465→            <Badge variant="outline" className={cn("text-[10px] sm:text-xs", getDifficultyColor(problem?.difficulty ?? "Fácil"))}>
   466→              {problem?.difficulty ?? "Fácil"}
   467→            </Badge>
   468→            <span className="flex items-center gap-1 text-white/60">
   469→              <Timer className="w-3 h-3" />
   470→              {Math.ceil(timeLeft)}s
   471→            </span>
   472→          </div>
   473→        </div>
   474→      )}
   475→
   476→      {/* Timer bar */}
   477→      {gameState === "playing" && (
   478→        <div className="w-full h-2 bg-white/5 rounded-full overflow-hidden">
   479→          <motion.div
   480→            className={cn("h-full rounded-full bg-gradient-to-r", timerBarColor)}
   481→            initial={{ width: "100%" }}
   482→            animate={{ width: `${timerPercent}%` }}
   483→            transition={{ duration: 0.1, ease: "linear" }}
   484→          />
   485→        </div>
   486→      )}
   487→
   488→      {/* IDLE STATE */}
   489→      <AnimatePresence mode="wait">
   490→        {gameState === "idle" && (
   491→          <motion.div
   492→            key="idle"
   493→            initial={{ opacity: 0, y: 20 }}
   494→            animate={{ opacity: 1, y: 0 }}
   495→            exit={{ opacity: 0, y: -20 }}
   496→            className="flex flex-col items-center gap-6 py-10"
   497→          >
   498→            <div className="flex items-center gap-3 text-white/80">
   499→              <Calculator className="w-10 h-10 text-cyan-400" />
   500→              <h1 className="text-3xl sm:text-4xl font-black bg-gradient-to-r from-cyan-400 to-pink-400 bg-clip-text text-transparent">
   501→                Duelo de Matemática
   502→              </h1>
   503→              <Calculator className="w-10 h-10 text-pink-400" />
   504→            </div>
   505→
   506→            <p className="text-white/50 text-sm text-center max-w-md">
   507→              10 rodadas de problemas matemáticos. O primeiro a responder corretamente ganha o ponto!
   508→              A dificuldade aumenta a cada rodada.
   509→            </p>
   510→
   511→            <div className="flex flex-wrap items-center justify-center gap-3">
   512→              <div className="flex items-center gap-2">
   513→                <span className="text-white/40 text-xs">Timer:</span>
   514→                {[10, 15, 20, 30].map((t) => (
   515→                  <button
   516→                    key={t}
   517→                    onClick={() => setTimerDuration(t)}
   518→                    className={cn(
   519→                      "px-3 py-1 rounded-md text-xs font-semibold transition-all",
   520→                      timerDuration === t
   521→                        ? "bg-cyan-500/30 text-cyan-300 border border-cyan-500/50"
   522→                        : "bg-white/5 text-white/40 border border-white/10 hover:bg-white/10"
   523→                    )}
