    42→    42→function generateProblem(round: number): MathProblem {
    43→    43→  const roundIndex = round - 1;
    44→    44→  let difficulty: Difficulty;
    45→    45→  let expression: string;
    46→    46→  let answer: number;
    47→    47→
    48→    48→  if (roundIndex < 3) {
    49→    49→    difficulty = "Fácil";
    50→    50→    const ops = ["+", "-"];
    51→    51→    const op = ops[Math.floor(Math.random() * ops.length)];
    52→    52→    const a = Math.floor(Math.random() * 20) + 1;
    53→    53→    const b = Math.floor(Math.random() * 20) + 1;
    54→    54→    if (op === "+") {
    55→    55→      expression = `${a} + ${b}`;
    56→    56→      answer = a + b;
    57→    57→    } else {
    58→    58→      const big = Math.max(a, b);
    59→    59→      const small = Math.min(a, b);
    60→    60→      expression = `${big} - ${small}`;
    61→    61→      answer = big - small;
    62→    62→    }
    63→    63→  } else if (roundIndex < 6) {
    64→    64→    difficulty = "Médio";
    65→    65→    if (Math.random() < 0.5) {
    66→    66→      const a = Math.floor(Math.random() * 12) + 1;
    67→    67→      const b = Math.floor(Math.random() * 12) + 1;
    68→    68→      expression = `${a} × ${b}`;
    69→    69→      answer = a * b;
    70→    70→    } else {
    71→    71→      const a = Math.floor(Math.random() * 41) + 10;
    72→    72→      const b = Math.floor(Math.random() * 41) + 10;
    73→    73→      expression = `${a} + ${b}`;
    74→    74→      answer = a + b;
    75→    75→    }
    76→    76→  } else {
    77→    77→    difficulty = "Difícil";
    78→    78→    const twoStepType = Math.floor(Math.random() * 4);
    79→    79→    if (twoStepType === 0) {
    80→    80→      const a = Math.floor(Math.random() * 8) + 2;
    81→    81→      const b = Math.floor(Math.random() * 8) + 2;
    82→    82→      const c = Math.floor(Math.random() * 15) + 1;
    83→    83→      expression = `${a} × ${b} + ${c}`;
    84→    84→      answer = a * b + c;
    85→    85→    } else if (twoStepType === 1) {
    86→    86→      const a = Math.floor(Math.random() * 8) + 2;
    87→    87→      const b = Math.floor(Math.random() * 8) + 2;
    88→    88→      const c = Math.floor(Math.random() * (a * b - 1)) + 1;
    89→    89→      expression = `${a * b} - ${c}`;
    90→    90→      answer = a * b - c;
    91→    91→    } else if (twoStepType === 2) {
    92→    92→      const a = Math.floor(Math.random() * 10) + 2;
    93→    93→      const b = Math.floor(Math.random() * 10) + 2;
    94→    94→      const c = Math.floor(Math.random() * 5) + 1;
    95→    95→      const sum = a + b;
    96→    96→      expression = `(${a} + ${b}) × ${c}`;
    97→    97→      answer = sum * c;
    98→    98→    } else {
    99→    99→      const a = Math.floor(Math.random() * 50) + 20;
   100→   100→      const b = Math.floor(Math.random() * 30) + 10;
   101→   101→      const c = Math.floor(Math.random() * 8) + 2;
   102→   102→      expression = `${a} + ${b} × ${c}`;
   103→   103→      answer = a + b * c;
   104→   104→    }
   105→   105→  }
   106→   106→
   107→   107→  return { expression, answer, difficulty };
   108→   108→}

   122→   122→    scale: Math.random() * 0.8 + 0.3,
   123→   123→    hue: Math.random() * 60 - 30,

   125→   125→
   126→   126→  return (
   127→   127→    <div className="absolute inset-0 pointer-events-none overflow-hidden z-50">

   129→   129→        <motion.div
   130→   130→          key={p.id}
   131→   131→          className={cn("absolute w-2 h-2 rounded-sm", color)}
   132→   132→          style={{
   133→   133→            left: "50%",
   134→   134→            top: "50%",
   135→   135→            backgroundColor: color.includes("cyan")
   136→   136→              ? `hsl(${180 + p.hue}, 90%, 60%)`
   137→   137→              : `hsl(${330 + p.hue}, 90%, 65%)`,
   138→   138→          }}
   139→   139→          initial={{ x: 0, y: 0, rotate: 0, opacity: 1 }}
   140→   140→          animate={{
   141→   141→            x: p.x,
   142→   142→            y: p.y,
   143→   143→            rotate: p.rotate,
   144→   144→            opacity: 0,
   145→   145→            scale: p.scale,
   146→   146→          }}
   147→   147→          transition={{ duration: 1.2, ease: "easeOut" }}
   148→   148→        />
   149→   149→      ))}
   150→   150→    </div>
   151→   151→  );
   152→   152→}

   156→   156→  const [currentRound, setCurrentRound] = useState(1);

   171→   171→  const [roundStartTime, setRoundStartTime] = useState(0);

   200→   200→            setGameState("roundResult");

   204→   204→      }, 50);
   205→   205→    },
   206→   206→    [clearTimer]
   207→   207→  );

   289→   289→      if (player === 1 && p1Locked) return;

   298→   298→        else setP2Locked(true);

   302→   302→          if (player === 1) {

   304→   304→            setTimeout(() => {

   312→   312→              setP2Wrong(false);
   313→   313→              setP2Input("");
   314→   314→              p2Ref.current?.focus();
   315→   315→            }, 500);
   316→   316→          }

   319→   319→            setP1Locked(true);
