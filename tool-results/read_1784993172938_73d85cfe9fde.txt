   155→function shuffle<T>(arr: T[]): T[] {
   156→  const a = [...arr];
   157→  for (let i = a.length - 1; i > 0; i--) {
   158→    const j = Math.floor(Math.random() * (i + 1));
   159→    [a[i], a[j]] = [a[j], a[i]];
   160→  }
   161→  return a;
   162→}

   222→function GameCard({ card, index, disabled, onClick }: CardProps) {
   223→  const isFlipped = card.status === 'flipping' || card.status === 'matched';
   224→  const isShaking = card.status === 'shaking';
   225→  const isMatched = card.status === 'matched';
   226→  const shapeDef = SHAPE_DEFS[card.shape];
   227→
   228→  return (
   229→    <motion.div
   230→      className="relative"
   231→      style={{ perspective: 600 }}
   232→      layout
   233→    >
   234→      <AnimatePresence>
   235→        {isMatched && (
   236→          <motion.div
   237→            key={`glow-${card.id}`}
   238→            initial={{ opacity: 0, scale: 0.8 }}
   239→            animate={{ opacity: [0, 1, 0.6], scale: [0.8, 1.15, 1] }}
   240→            transition={{ duration: 0.6, ease: 'easeOut' }}
   241→            className="absolute inset-0 rounded-xl z-10 pointer-events-none"
   242→            style={{
   243→              boxShadow: `0 0 20px 4px ${shapeDef.color}66, 0 0 40px 8px ${shapeDef.color}33`,
   244→            }}
   245→          />
   246→        )}
   247→      </AnimatePresence>
   248→
   249→      <motion.button
   250→        type="button"
   251→        disabled={disabled || isFlipped || isMatched}
   252→        onClick={() => onClick(index)}
   253→        animate={
   254→          isShaking
   255→            ? { x: [0, -6, 6, -6, 6, -3, 3, 0] }
   256→            : isMatched
   257→              ? { scale: [1, 1.12, 1] }
   258→              : {}
   259→        }
   260→        transition={
   261→          isShaking
   262→            ? { duration: 0.5, ease: 'easeInOut' }
   263→            : isMatched
   264→              ? { duration: 0.4, ease: 'easeOut' }
   265→              : {}
   266→        }
   267→        className={cn(
   268→          'relative w-16 h-20 sm:w-20 sm:h-24 rounded-xl cursor-pointer focus:outline-none transition-shadow',
   269→          isMatched && 'cursor-default',
   270→          disabled && !isFlipped && !isMatched && 'cursor-not-allowed opacity-60',
   271→        )}
   272→        style={{
   273→          transformStyle: 'preserve-3d',
   274→        }}
   275→        aria-label={`Carta ${index + 1}`}
   276→      >
   277→        {/* ---- Back face (face-down) ---- */}
   278→        <motion.div
   279→          className={cn(
   280→            'absolute inset-0 rounded-xl border bg-slate-800 border-slate-700 flex items-center justify-center',
   281→          )}
   282→          style={{
   283→            backfaceVisibility: 'hidden',
   284→            WebkitBackfaceVisibility: 'hidden',

   287→          transition={{ duration: 0.45, ease: [0.4, 0, 0.2, 1] }}

   289→          <div className="text-slate-500 flex flex-col items-center justify-center">

   292→          </div>
   293→          <span className="absolute bottom-0.5 right-1 text-[9px] text-slate-600 font-mono">
   294→            {index + 1 <= 9 ? index + 1 : ''}
   295→          </span>
   296→        </motion.div>
   297→
   298→        {/* ---- Front face (face-up) ---- */}
   299→        <motion.div
   300→          className={cn(
   301→            'absolute inset-0 rounded-xl border border-slate-300 dark:border-slate-400 bg-white dark:bg-slate-100 flex items-center justify-center',
   302→            isMatched && 'border-green-400',
   303→          )}
   304→          style={{
   305→            backfaceVisibility: 'hidden',
   306→            WebkitBackfaceVisibility: 'hidden',
   307→          }}
   308→          animate={{ rotateY: isFlipped ? 0 : -180 }}
   309→          transition={{ duration: 0.45, ease: [0.4, 0, 0.2, 1] }}
   310→        >
   311→          <ShapeSVG shape={card.shape} color={shapeDef.color} />
   312→          {isMatched && (
   313→            <motion.span
   314→              initial={{ opacity: 0, scale: 0 }}
   315→              animate={{ opacity: 1, scale: 1 }}
   316→              transition={{ delay: 0.3, duration: 0.25 }}
   317→              className="absolute top-0.5 right-0.5 text-green-500 text-xs"
   318→            >
   319→              ✓
   320→            </motion.span>
   321→          )}
   322→        </motion.div>
   323→      </motion.button>
   324→    </motion.div>
   325→  );
   326→}

   329→/*  Main component                                                     */
   330→/* ================================================================== */
   331→
   332→export default function MemoryCardsVS({ onScore, liveCode: _liveCode }: Props) {
   333→  /* ---- State ---- */
   334→  const [gameState, setGameState] = useState<GameState>('idle');
   335→  const [gridSize, setGridSize] = useState<GridSize>('4x4');
   336→  const [timerOption, setTimerOption] = useState<TimerOption>(90);
   337→  const [cards, setCards] = useState<CardData[]>([]);
   338→  const [flippedIndices, setFlippedIndices] = useState<number[]>([]);
   339→  const [currentPlayer, setCurrentPlayer] = useState<1 | 2>(1);
   340→  const [scores, setScores] = useState<[number, number]>([0, 0]);
   341→  const [timeRemaining, setTimeRemaining] = useState<number>(0);
   342→  const [canFlip, setCanFlip] = useState(false);
   343→  const [feedback, setFeedback] = useState<{ message: string; color: string; id: number } | null>(null);
   344→  const [winner, setWinner] = useState<1 | 2 | 'draw' | null>(null);
   345→
   346→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   347→  const feedbackIdRef = useRef(0);
   348→
   349→  const config = GRID_CONFIG[gridSize];
   350→  const totalPairs = config.pairs;
   351→  const matchedCount = cards.filter(c => c.status === 'matched').length / 2;
   352→  const pairsLeft = totalPairs - matchedCount;
   353→
   354→  /* ---- Derived ---- */
   355→  const activeGlow =
   356→    currentPlayer === 1
   357→      ? 'shadow-[0_0_20px_4px_rgba(6,182,212,0.35)] border-cyan-500/50'
   358→      : 'shadow-[0_0_20px_4px_rgba(236,72,153,0.35)] border-pink-500/50';
   359→
   360→  /* ---- Init game ---- */
   361→  const initGame = useCallback(() => {
   362→    const shapes = shuffle(ALL_SHAPES).slice(0, totalPairs);
   363→    const deck: CardData[] = [];
   364→    shapes.forEach((shape, pairIdx) => {
   365→      const color = SHAPE_DEFS[shape].color;
   366→      deck.push({ id: deck.length, pairId: pairIdx, shape, color, status: 'hidden' });
   367→      deck.push({ id: deck.length, pairId: pairIdx, shape, color, status: 'hidden' });
   368→    });
   369→    const shuffled = shuffle(deck);
   370→    setCards(shuffled);
   371→    setFlippedIndices([]);
   372→    setCurrentPlayer(1);
   373→    setScores([0, 0]);
   374→    setCanFlip(true);
   375→    setFeedback(null);
   376→    setWinner(null);
   377→    setTimeRemaining(timerOption);
   378→    setGameState('playing');
   379→  }, [totalPairs, timerOption]);
   380→
   381→  /* ---- Timer effect ---- */
   382→  useEffect(() => {
   383→    if (gameState !== 'playing' || timerOption === 0) return;
   384→    timerRef.current = setInterval(() => {
   385→      setTimeRemaining(prev => {
   386→        if (prev <= 1) {
   387→          clearInterval(timerRef.current!);
   388→          return 0;
   389→        }
   390→        return prev - 1;
   391→      });
   392→    }, 1000);
   393→    return () => {
   394→      if (timerRef.current) clearInterval(timerRef.current);
   395→    };
   396→  }, [gameState, timerOption]);
   397→
   398→  /* ---- End game when timer hits 0 ---- */
   399→  const endGame = useCallback(() => {
   400→    setGameState('ended');
   401→    setCanFlip(false);
   402→    if (timerRef.current) clearInterval(timerRef.current);
   403→
   404→    const timeBonus = timerOption > 0 ? timeRemaining : 0;
   405→
   406→    if (scores[0] > scores[1]) {
   407→      setWinner(1);
   408→      onScore?.('Jogador 1', scores[0] * 10 + timeBonus);
   409→    } else if (scores[1] > scores[0]) {
   410→      setWinner(2);
   411→      onScore?.('Jogador 2', scores[1] * 10 + timeBonus);
   412→    } else {
   413→      setWinner('draw');
   414→    }
   415→  }, [scores, timerOption, timeRemaining, onScore]);
   416→
   417→  useEffect(() => {
   418→    if (gameState === 'playing' && timerOption > 0 && timeRemaining === 0) {
   419→      endGame();
   420→    }
   421→  }, [timeRemaining, gameState, timerOption, endGame]);
   422→
   423→  /* ---- Keyboard support (keys 1-9 and 0 for 10) ---- */
   424→  const handleCardClick = useCallback(
   425→    (index: number) => {
   426→      if (!canFlip || gameState !== 'playing') return;
   427→      const card = cards[index];
   428→      if (card.status !== 'hidden') return;
   429→      if (flippedIndices.includes(index)) return;
   430→
   431→      const newFlipped = [...flippedIndices, index];
   432→      const updated = cards.map((c, i) => (i === index ? { ...c, status: 'flipping' as CardStatus } : c));
   433→      setCards(updated);
   434→      setFlippedIndices(newFlipped);
   435→
   436→      if (newFlipped.length === 2) {
   437→        setCanFlip(false);
   438→        const [a, b] = newFlipped;
   439→        if (updated[a].pairId === updated[b].pairId) {
   440→          setTimeout(() => {
   441→            setCards(prev =>
   442→              prev.map((c, i) => (i === a || i === b ? { ...c, status: 'matched' as CardStatus } : c)),
   443→            );
   444→            setScores(prev => {
   445→              const next = [...prev] as [number, number];
   446→              next[currentPlayer - 1] += 1;
   447→              return next;
   448→            });
   449→            showFeedback('Par!', 'bg-green-600 text-white');
   450→            setFlippedIndices([]);
   451→            setCanFlip(true);
   452→          }, 500);
   453→        } else {
   454→          showFeedback('N\u00e3o \u00e9 par', 'bg-red-600 text-white');
   455→          setTimeout(() => {
   456→            setCards(prev =>
   457→              prev.map((c, i) =>
   458→                i === a || i === b ? { ...c, status: 'shaking' as CardStatus } : c,
   459→              ),
   460→            );
   461→          }, 400);
   462→          setTimeout(() => {
   463→            setCards(prev =>
   464→              prev.map((c, i) =>
   465→                i === a || i === b ? { ...c, status: 'hidden' as CardStatus } : c,
   466→              ),
   467→            );
   468→            setFlippedIndices([]);
   469→            setCurrentPlayer(prev => (prev === 1 ? 2 : 1));
   470→            setCanFlip(true);
   471→          }, 1000);
   472→        }
   473→      }
   474→    },
   475→    [canFlip, cards, flippedIndices, currentPlayer, gameState],
   476→  );
   477→
   478→  useEffect(() => {
   479→    if (gameState !== 'playing' || !canFlip) return;
   480→    function handleKey(e: KeyboardEvent) {
   481→      let idx = -1;
   482→      if (e.key >= '1' && e.key <= '9') idx = parseInt(e.key) - 1;
   483→      else if (e.key === '0' && cards.length >= 10) idx = 9;
   484→      else return;
   485→      if (idx >= 0 && idx < cards.length) {
   486→        handleCardClick(idx);
   487→      }
   488→    }
   489→    window.addEventListener('keydown', handleKey);
   490→    return () => window.removeEventListener('keydown', handleKey);
   491→  }, [gameState, canFlip, cards, handleCardClick]);
   492→
   493→  /* ---- Check if all matched ---- */
   494→  useEffect(() => {
   495→    if (gameState !== 'playing') return;
   496→    if (matchedCount === totalPairs) {
   497→      endGame();
   498→    }
   499→  }, [matchedCount, totalPairs, gameState, endGame]);
   500→
   501→  /* ---- Show feedback toast ---- */
   502→  const showFeedback = useCallback((message: string, color: string) => {
   503→    const id = ++feedbackIdRef.current;
   504→    setFeedback({ message, color, id });
   505→    setTimeout(() => setFeedback(prev => (prev && prev.id === id ? null : prev)), 900);
   506→  }, []);
   507→
   508→  /* ---- Reset to idle ---- */
   509→  const resetAll = () => {
   510→    if (timerRef.current) clearInterval(timerRef.current);
   511→    setGameState('idle');
   512→    setCards([]);
   513→    setFlippedIndices([]);
   514→    setScores([0, 0]);
   515→    setCurrentPlayer(1);
   516→    setTimeRemaining(0);
   517→    setCanFlip(false);
   518→    setFeedback(null);
   519→    setWinner(null);
   520→  };
   521→
   522→  /* ---- Format timer ---- */
   523→  const fmt = (s: number) => {
   524→    const m = Math.floor(s / 60);
   525→    const sec = s % 60;
   526→    return `${m}:${sec.toString().padStart(2, '0')}`;
   527→  };
   528→
   529→  /* ================================================================== */
   530→  /*  Render                                                            */
   531→  /* ================================================================== */
   532→
   533→  return (
   534→    <div className="flex flex-col items-center gap-4 w-full max-w-lg mx-auto select-none">
   535→      {/* ---- Title ---- */}
   536→      <h2 className="text-xl sm:text-2xl font-bold bg-gradient-to-r from-cyan-400 to-pink-400 bg-clip-text text-transparent">
   537→        MEM\u00d3RIA VS CARTAS
   538→      </h2>
   539→
   540→      {/* ---- Scoreboard ---- */}
   541→      <div className="w-full rounded-xl bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 p-3 flex items-center justify-between">
   542→        {/* P1 score */}

   611→        <div
   612→          className={cn(
   613→            'flex items-center gap-2 px-3 py-1.5 rounded-lg transition-all',
   614→            currentPlayer === 2 && gameState === 'playing'
   615→              ? 'bg-pink-500/20 ring-1 ring-pink-400'

   617→          )}
   618→        >
   619→          <span className="text-pink-400/60 text-xs">pares</span>

   621→            key={`s2-${scores[1]}`}

   681→          initial={{ opacity: 0, y: -5 }}
   682→          animate={{ opacity: 1, y: 0 }}

   687→            className={cn(
   688→              'font-mono text-sm px-3 py-0.5',
   689→              timeRemaining <= 10

   692→                  ? 'border-yellow-500 text-yellow-400 bg-yellow-500/10'
   693→                  : 'border-slate-500 text-slate-300',

   695→          >

   700→

   710→            className="bg-gradient-to-r from-cyan-600 to-pink-600 hover:from-cyan-500 hover:to-pink-500 text-white px-6"
   711→          >
   712→            Iniciar Jogo
   713→          </Button>
   714→        </div>
   715→      )}
   716→
   717→      {(gameState === 'playing' || gameState === 'ended') && (
   718→        <div
   719→          className={cn(
   720→            'relative rounded-xl p-3 transition-all duration-500 border-2',
   721→            gameState === 'playing' && activeGlow,
   722→            gameState === 'ended' && 'border-slate-700',
   723→          )}
   724→        >
   725→          {/* Feedback toast */}
   726→          <AnimatePresence>
   727→            {feedback && (

   729→                key={feedback.id}
   730→                message={feedback.message}
   731→                color={feedback.color}
   732→              />
   733→            )}
   734→          </AnimatePresence>
   735→
   736→          <div
   737→            className="grid gap-2"
   738→            style={{
   739→              gridTemplateColumns: `repeat(${config.cols}, minmax(0, 1fr))`,
   740→            }}
   741→          >
   742→            {cards.map((card, idx) => (
   743→              <GameCard
   744→                key={card.id}
   745→                card={card}
   746→                index={idx}
   747→                disabled={!canFlip}
   748→                onClick={handleCardClick}
   749→              />

   751→          </div>

   754→
   755→      {/* ---- Footer: restart + pairs left ---- */}

   765→            Reiniciar Tudo
