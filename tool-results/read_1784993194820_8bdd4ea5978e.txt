     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import { Button } from "@/components/ui/button";
     6→import { Badge } from "@/components/ui/badge";
     7→import { cn } from "@/lib/utils";
     8→import { RotateCcw, ArrowUp, ArrowDown, Check } from "lucide-react";
     9→
    10→interface GuessEntry {
    11→  player: number;
    12→  value: number;
    13→  hint: "higher" | "lower" | "correct";
    14→  round: number;
    15→}
    16→
    17→type GameState = "new-round" | "guessing" | "round-result" | "game-over";
    18→
    19→interface Props {
    20→  onScore?: (name: string, score: number) => void;
    21→  liveCode?: string;
    22→}
    23→
    24→const TOTAL_ROUNDS = 5;
    25→const MIN_NUMBER = 1;
    26→const MAX_NUMBER = 100;
    27→
    28→function generateSecretNumber(): number {
    29→  return Math.floor(Math.random() * (MAX_NUMBER - MIN_NUMBER + 1)) + MIN_NUMBER;
    30→}
    31→
    32→function calculatePoints(attempts: number): number {
    33→  return Math.max(10, 50 - attempts * 5);
    34→}
    35→
    36→function NumberLine({
    37→  rangeMin,
    38→  rangeMax,
    39→  secretNumber,
    40→  revealSecret,
    41→}: {
    42→  rangeMin: number;
    43→  rangeMax: number;
    44→  secretNumber: number;
    45→  revealSecret: boolean;
    46→}) {
    47→  const leftPct = ((rangeMin - MIN_NUMBER) / (MAX_NUMBER - MIN_NUMBER)) * 100;
    48→  const rightPct = ((rangeMax - MIN_NUMBER) / (MAX_NUMBER - MIN_NUMBER)) * 100;
    49→  const widthPct = rightPct - leftPct;
    50→  const secretPct = ((secretNumber - MIN_NUMBER) / (MAX_NUMBER - MIN_NUMBER)) * 100;
    51→
    52→  return (
    53→    <div className="w-full px-2">
    54→      <div className="relative w-full h-10 rounded-full bg-slate-700/50 overflow-hidden border border-slate-600/30">
    55→        {/* Dimmed left zone */}
    56→        <div
    57→          className="absolute top-0 left-0 h-full bg-slate-700/70"
    58→          style={{ width: `${leftPct}%` }}
    59→        />
    60→        {/* Active range */}
    61→        <motion.div
    62→          className="absolute top-0 h-full"
    63→          style={{
    64→            background:
    65→              "linear-gradient(90deg, rgba(16,185,129,0.3), rgba(16,185,129,0.6), rgba(16,185,129,0.3))",
    66→            borderLeft: "2px solid rgba(16,185,129,0.8)",
    67→            borderRight: "2px solid rgba(16,185,129,0.8)",
    68→          }}
    69→          animate={{ left: `${leftPct}%`, width: `${widthPct}%` }}
    70→          transition={{ type: "spring", stiffness: 300, damping: 30 }}
    71→        >
    72→          {/* Glow effect */}
    73→          <div className="absolute inset-0 bg-emerald-400/10 blur-sm" />
    74→        </motion.div>
    75→        {/* Dimmed right zone */}
    76→        <div
    77→          className="absolute top-0 right-0 h-full bg-slate-700/70"
    78→          style={{ width: `${100 - rightPct}%` }}
    79→        />
    80→        {/* Secret number marker (only on reveal) */}
    81→        <AnimatePresence>
    82→          {revealSecret && (
    83→            <motion.div
    84→              initial={{ scale: 0, opacity: 0 }}
    85→              animate={{ scale: 1, opacity: 1 }}
    86→              exit={{ scale: 0, opacity: 0 }}
    87→              className="absolute top-1/2 -translate-y-1/2 z-10"
    88→              style={{ left: `${secretPct}%` }}
    89→            >
    90→              <div className="w-4 h-4 rounded-full bg-amber-400 border-2 border-white shadow-lg shadow-amber-400/50 -ml-2" />
    91→            </motion.div>
    92→          )}
    93→        </AnimatePresence>
    94→        {/* Tick marks */}
    95→        {[1, 25, 50, 75, 100].map((tick) => {
    96→          const pct = ((tick - MIN_NUMBER) / (MAX_NUMBER - MIN_NUMBER)) * 100;
    97→          return (
    98→            <div
    99→              key={tick}
   100→              className="absolute top-0 h-full w-px bg-slate-500/30"
   101→              style={{ left: `${pct}%` }}
   102→            />
   103→          );
   104→        })}
   105→      </div>
   106→      {/* Labels */}
   107→      <div className="flex justify-between mt-1 px-1">
   108→        <span className="text-xs text-slate-500">{MIN_NUMBER}</span>
   109→        <motion.span
   110→          className="text-xs font-bold text-emerald-400"
   111→          animate={{ left: `${leftPct}%` }}
   112→          transition={{ type: "spring", stiffness: 300, damping: 30 }}
   113→          style={{ position: "relative" }}
   114→        >
   115→          {rangeMin}
   116→        </motion.span>
   117→        <motion.span
   118→          className="text-xs font-bold text-emerald-400"
   119→          animate={{ right: `${100 - rightPct}%` }}
   120→          transition={{ type: "spring", stiffness: 300, damping: 30 }}
   121→          style={{ position: "relative" }}
   122→        >
   123→          {rangeMax}
   124→        </motion.span>
   125→        <span className="text-xs text-slate-500">{MAX_NUMBER}</span>
   126→      </div>
   127→    </div>
   128→  );
   129→}
   130→
   131→export default function GuessNumber100({ onScore, liveCode }: Props) {
   132→  const [gameState, setGameState] = useState<GameState>("new-round");
   133→  const [currentRound, setCurrentRound] = useState(1);
   134→  const [currentPlayer, setCurrentPlayer] = useState(1);
   135→  const [secretNumber, setSecretNumber] = useState<number>(0);
   136→  const [rangeMin, setRangeMin] = useState(MIN_NUMBER);
   137→  const [rangeMax, setRangeMax] = useState(MAX_NUMBER);
   138→  const [guessInput, setGuessInput] = useState("");
   139→  const [guesses, setGuesses] = useState<GuessEntry[]>([]);
   140→  const [scores, setScores] = useState([0, 0]);
   141→  const [roundWinner, setRoundWinner] = useState<number | null>(null);
   142→  const [lastHint, setLastHint] = useState<"higher" | "lower" | "correct" | null>(null);
   143→  const [roundAttempts, setRoundAttempts] = useState(0);
   144→  const inputRef = useRef<HTMLInputElement>(null);
   145→  const roundRef = useRef(currentRound);
   146→  roundRef.current = currentRound;
   147→  const [triggerScoreAnim, setTriggerScoreAnim] = useState<number | null>(null);
   148→
   149→  const startNewRound = useCallback(() => {
   150→    const newSecret = generateSecretNumber();
   151→    setSecretNumber(newSecret);
   152→    setRangeMin(MIN_NUMBER);
   153→    setRangeMax(MAX_NUMBER);
   154→    setGuessInput("");
   155→    setGuesses([]);
   156→    setRoundWinner(null);
   157→    setLastHint(null);
   158→    setRoundAttempts(0);
   159→    setCurrentPlayer(roundRef.current % 2 === 0 ? 2 : 1);
   160→    setGameState("guessing");
   161→    setTimeout(() => inputRef.current?.focus(), 100);
   162→  }, []);
   163→
   164→  const startGame = useCallback(() => {
   165→    setCurrentRound(1);
   166→    setScores([0, 0]);
   167→    setGameState("new-round");
   168→    setTimeout(() => startNewRound(), 800);
   169→  }, [startNewRound]);
   170→
   171→  useEffect(() => {
   172→    startGame();
   173→    // eslint-disable-next-line react-hooks/exhaustive-deps
   174→  }, []);
   175→
   176→  const handleGuess = () => {
   177→    const guess = parseInt(guessInput, 10);
   178→    if (isNaN(guess) || guess < MIN_NUMBER || guess > MAX_NUMBER) return;
   179→
   180→    const attempts = roundAttempts + 1;
   181→    setRoundAttempts(attempts);
   182→
   183→    let hint: "higher" | "lower" | "correct";
   184→    if (guess === secretNumber) {
   185→      hint = "correct";
   186→    } else if (guess < secretNumber) {
   187→      hint = "higher";
   188→    } else {
   189→      hint = "lower";
   190→    }
   191→
   192→    const entry: GuessEntry = {
   193→      player: currentPlayer,
   194→      value: guess,
   195→      hint,
   196→      round: currentRound,
   197→    };
   198→
   199→    setGuesses((prev) => [...prev, entry]);
   200→    setGuessInput("");
   201→    setLastHint(hint);
   202→
   203→    if (hint === "correct") {
   204→      const points = calculatePoints(attempts);
   205→      setRoundWinner(currentPlayer);
   206→      setScores((prev) => {
   207→        const newScores = [...prev] as [number, number];
   208→        newScores[currentPlayer - 1] += points;
   209→        return newScores;
   210→      });
   211→      setTriggerScoreAnim(currentPlayer);
   212→      setGameState("round-result");
   213→    } else {
   214→      // Update range
   215→      if (hint === "higher" && guess >= rangeMin) {
   216→        setRangeMin(guess + 1);
   217→      } else if (hint === "lower" && guess <= rangeMax) {
   218→        setRangeMax(guess - 1);
   219→      }
   220→      // Switch player
   221→      setCurrentPlayer(currentPlayer === 1 ? 2 : 1);
   222→      setTimeout(() => inputRef.current?.focus(), 50);
   223→    }
   224→  };
   225→
   226→  const handleNextRound = () => {
   227→    if (currentRound >= TOTAL_ROUNDS) {
   228→      setGameState("game-over");
   229→      const winnerName = scores[0] > scores[1] ? "Jogador 1" : scores[1] > scores[0] ? "Jogador 2" : "Empate";
   230→      const winnerScore = Math.max(scores[0], scores[1]);
   231→      if (onScore) {
   232→        if (scores[0] > scores[1]) {
   233→          onScore("Jogador 1", scores[0]);
   234→        } else if (scores[1] > scores[0]) {
   235→          onScore("Jogador 2", scores[1]);
   236→        }
   237→      }
   238→    } else {
   239→      setCurrentRound((r) => r + 1);
   240→      setGameState("new-round");
   241→      setTimeout(() => startNewRound(), 800);
   242→    }
   243→  };
   244→
   245→  const handleRestart = () => {
   246→    startGame();
   247→  };
   248→
   249→  const handleKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   250→    if (e.key === "Enter") {
   251→      handleGuess();
   252→    }
   253→  };
   254→
   255→  const p1Color = "text-cyan-400";
   256→  const p2Color = "text-pink-400";
   257→  const p1Bg = "bg-cyan-400/10 border-cyan-400/30";
   258→  const p2Bg = "bg-pink-400/10 border-pink-400/30";
   259→
   260→  const getPlayerColor = (player: number) => (player === 1 ? p1Color : p2Color);
   261→  const getPlayerBg = (player: number) => (player === 1 ? p1Bg : p2Bg);
   262→  const getPlayerName = (player: number) =>
   263→    player === 1 ? "Jogador 1" : "Jogador 2";
   264→
   265→  const roundGuesses = guesses.filter((g) => g.round === currentRound);
   266→
   267→  return (
   268→    <div className="w-full max-w-lg mx-auto p-4 space-y-4">
   269→      {/* Scoreboard */}
   270→      <div className="bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 rounded-xl p-4">
   271→        <div className="flex items-center justify-between">
   272→          {/* Player 1 Score */}
   273→          <div className="flex flex-col items-center gap-1 min-w-[80px]">
   274→            <span className={cn("text-xs font-medium uppercase tracking-wider", p1Color)}>
   275→              Jogador 1
   276→            </span>
   277→            <motion.div
   278→              key={`score-p1-${scores[0]}`}
   279→              initial={{ scale: 1.4, color: "#22d3ee" }}
   280→              animate={{ scale: 1, color: "#f8fafc" }}
   281→              transition={{ type: "spring", stiffness: 400, damping: 15 }}
   282→              className="text-2xl font-bold text-white"
   283→            >
   284→              {scores[0]}
   285→            </motion.div>
   286→          </div>
   287→
   288→          {/* Title */}
   289→          <div className="text-center">
   290→            <h2 className="text-lg font-bold text-white tracking-tight">
   291→              🎯 Adivinha o Número
   292→            </h2>
   293→            <Badge
   294→              variant="outline"
   295→              className="text-xs text-slate-300 border-slate-600 mt-1"
   296→            >
   297→              Round {currentRound}/{TOTAL_ROUNDS}
   298→            </Badge>
   299→          </div>
   300→
   301→          {/* Player 2 Score */}
   302→          <div className="flex flex-col items-center gap-1 min-w-[80px]">
   303→            <span className={cn("text-xs font-medium uppercase tracking-wider", p2Color)}>
   304→              Jogador 2
   305→            </span>
   306→            <motion.div
   307→              key={`score-p2-${scores[1]}`}
   308→              initial={{ scale: 1.4, color: "#f472b6" }}
   309→              animate={{ scale: 1, color: "#f8fafc" }}
   310→              transition={{ type: "spring", stiffness: 400, damping: 15 }}
   311→              className="text-2xl font-bold text-white"
   312→            >
   313→              {scores[1]}
   314→            </motion.div>
   315→          </div>
   316→        </div>
   317→      </div>
   318→
   319→      {/* Game Info Bar */}
   320→      <div className="flex flex-wrap items-center justify-center gap-2">
   321→        <Badge
   322→          variant="outline"
   323→          className={cn(
   324→            "text-xs border",
   325→            currentPlayer === 1
   326→              ? "border-cyan-400/50 text-cyan-400"
   327→              : "border-pink-400/50 text-pink-400"
   328→          )}
   329→        >
   330→          Vez de: <span className="font-bold ml-1">{getPlayerName(currentPlayer)}</span>
   331→        </Badge>
   332→        <Badge variant="outline" className="text-xs border-emerald-400/50 text-emerald-400">
   333→          Intervalo: {rangeMin} - {rangeMax}
   334→        </Badge>
   335→        <Badge variant="outline" className="text-xs border-slate-500 text-slate-300">
   336→          Tentativas: {roundAttempts}
   337→        </Badge>
   338→      </div>
   339→
   340→      {/* Visual Number Line */}
   341→      <div className="py-2">
   342→        <NumberLine
   343→          rangeMin={rangeMin}
   344→          rangeMax={rangeMax}
   345→          secretNumber={secretNumber}
   346→          revealSecret={gameState === "round-result" || gameState === "game-over"}
   347→        />
   348→      </div>
   349→
   350→      {/* New Round Overlay */}
   351→      <AnimatePresence>
   352→        {gameState === "new-round" && (
   353→          <motion.div
   354→            initial={{ opacity: 0, scale: 0.8 }}
   355→            animate={{ opacity: 1, scale: 1 }}
   356→            exit={{ opacity: 0, scale: 1.1 }}
   357→            className="flex flex-col items-center justify-center py-8 gap-3"
   358→          >
   359→            <motion.div
   360→              animate={{ rotate: [0, 10, -10, 0] }}
   361→              transition={{ repeat: Infinity, duration: 2, ease: "easeInOut" }}
   362→              className="text-5xl"
   363→            >
   364→              🎲
   365→            </motion.div>
   366→            <p className="text-white text-lg font-semibold">Novo número sorteado!</p>
   367→            <p className="text-slate-400 text-sm">Round {currentRound} começando...</p>
   368→            <motion.div
   369→              animate={{ opacity: [0.4, 1, 0.4] }}
   370→              transition={{ repeat: Infinity, duration: 1.5 }}
   371→              className="text-slate-500 text-xs"
   372→            >
   373→              Preparando...
   374→            </motion.div>
   375→          </motion.div>
   376→        )}
   377→      </AnimatePresence>
   378→
   379→      {/* Guessing State */}
   380→      <AnimatePresence>
   381→        {gameState === "guessing" && (
   382→          <motion.div
   383→            initial={{ opacity: 0, y: 20 }}
   384→            animate={{ opacity: 1, y: 0 }}
   385→            exit={{ opacity: 0, y: -20 }}
   386→            className="space-y-4"
   387→          >
   388→            {/* Current Player Indicator */}
   389→            <motion.div
   390→              key={`player-indicator-${currentPlayer}`}
   391→              initial={{ x: currentPlayer === 1 ? -50 : 50, opacity: 0 }}
   392→              animate={{ x: 0, opacity: 1 }}
   393→              transition={{ type: "spring", stiffness: 300, damping: 25 }}
   394→              className={cn(
   395→                "flex items-center gap-2 p-3 rounded-lg border text-center justify-center",
   396→                currentPlayer === 1 ? p1Bg : p2Bg
   397→              )}
   398→            >
   399→              <div
   400→                className={cn(
   401→                  "w-3 h-3 rounded-full",
   402→                  currentPlayer === 1 ? "bg-cyan-400" : "bg-pink-400"
   403→                )}
   404→              />
   405→              <span className={cn("font-semibold", getPlayerColor(currentPlayer))}>
   406→                {getPlayerName(currentPlayer)}, é a sua vez!
   407→              </span>
   408→            </motion.div>
   409→
   410→            {/* Input + Button */}
   411→            <div className="flex gap-2">
   412→              <input
   413→                ref={inputRef}
   414→                type="number"
   415→                min={MIN_NUMBER}
   416→                max={MAX_NUMBER}
   417→                value={guessInput}
   418→                onChange={(e) => setGuessInput(e.target.value)}
   419→                onKeyDown={handleKeyDown}
   420→                placeholder="Digite um número (1-100)"
   421→                className={cn(
   422→                  "flex-1 bg-slate-800/80 border rounded-lg px-4 py-3 text-white text-lg text-center",
   423→                  "placeholder:text-slate-500 placeholder:text-sm focus:outline-none focus:ring-2 transition-all",
   424→                  currentPlayer === 1
   425→                    ? "border-cyan-500/30 focus:ring-cyan-500/50"
   426→                    : "border-pink-500/30 focus:ring-pink-500/50"
   427→                )}
   428→              />
   429→              <Button
   430→                onClick={handleGuess}
   431→                disabled={
   432→                  !guessInput ||
   433→                  parseInt(guessInput, 10) < MIN_NUMBER ||
   434→                  parseInt(guessInput, 10) > MAX_NUMBER
   435→                }
   436→                className={cn(
   437→                  "px-6 py-3 font-semibold text-base transition-all",
   438→                  currentPlayer === 1
   439→                    ? "bg-cyan-600 hover:bg-cyan-500 text-white"
   440→                    : "bg-pink-600 hover:bg-pink-500 text-white"
   441→                )}
   442→              >
   443→                Chutar
   444→              </Button>
   445→            </div>
   446→
   447→            {/* Last Hint Animation */}
   448→            <AnimatePresence>
   449→              {lastHint && lastHint !== "correct" && (
   450→                <motion.div
   451→                  key={`hint-${guesses.length}`}
   452→                  initial={{ opacity: 0, y: 30 }}
   453→                  animate={{ opacity: 1, y: 0 }}
   454→                  className="flex flex-col items-center gap-1 py-2"
   455→                >
   456→                  {lastHint === "higher" ? (
   457→                    <motion.div
   458→                      initial={{ y: 20, opacity: 0 }}
   459→                      animate={{ y: [20, -10, 0], opacity: 1 }}
   460→                      transition={{ duration: 0.6 }}
   461→                      className="flex items-center gap-2"
   462→                    >
   463→                      <ArrowUp className="w-8 h-8 text-amber-400" />
   464→                      <span className="text-amber-400 font-bold text-lg">
   465→                        É MAIOR 🔼
   466→                      </span>
   467→                      <ArrowUp className="w-8 h-8 text-amber-400" />
   468→                    </motion.div>
   469→                  ) : (
   470→                    <motion.div
   471→                      initial={{ y: -20, opacity: 0 }}
   472→                      animate={{ y: [-20, 10, 0], opacity: 1 }}
   473→                      transition={{ duration: 0.6 }}
   474→                      className="flex items-center gap-2"
   475→                    >
   476→                      <ArrowDown className="w-8 h-8 text-blue-400" />
   477→                      <span className="text-blue-400 font-bold text-lg">
   478→                        É MENOR 🔽
   479→                      </span>
   480→                      <ArrowDown className="w-8 h-8 text-blue-400" />
   481→                    </motion.div>
   482→                  )}
   483→                </motion.div>
   484→              )}
   485→            </AnimatePresence>
   486→          </motion.div>
   487→        )}
   488→      </AnimatePresence>
   489→
   490→      {/* Round Result */}
   491→      <AnimatePresence>
   492→        {gameState === "round-result" && roundWinner !== null && (
   493→          <motion.div
   494→            initial={{ opacity: 0, scale: 0.8 }}
   495→            animate={{ opacity: 1, scale: 1 }}
   496→            exit={{ opacity: 0, scale: 0.8 }}
   497→            className="flex flex-col items-center gap-4 py-6"
   498→          >
   499→            {/* Celebration */}
   500→            <motion.div
   501→              animate={{
   502→                scale: [1, 1.2, 1],
   503→                rotate: [0, 5, -5, 0],
   504→              }}
   505→              transition={{ repeat: Infinity, duration: 1.5 }}
   506→              className="text-6xl"
   507→            >
   508→              🎉
   509→            </motion.div>
   510→
   511→            <motion.div
   512→              initial={{ y: 20, opacity: 0 }}
   513→              animate={{ y: 0, opacity: 1 }}
   514→              transition={{ delay: 0.2 }}
   515→              className="text-center space-y-2"
   516→            >
   517→              <div className="flex items-center justify-center gap-2">
   518→                <Check className="w-6 h-6 text-emerald-400" />
   519→                <span className="text-emerald-400 font-bold text-xl">Correto!</span>
   520→                <Check className="w-6 h-6 text-emerald-400" />
   521→              </div>
   522→
   523→              <p className={cn("text-2xl font-bold", getPlayerColor(roundWinner))}>
   524→                {getPlayerName(roundWinner)} Venceu!
   525→              </p>
   526→
   527→              <div className="space-y-1">
   528→                <p className="text-slate-300 text-sm">
   529→                  O número era{" "}
   530→                  <span className="text-amber-400 font-bold text-lg">{secretNumber}</span>
   531→                </p>
   532→                <p className="text-slate-400 text-sm">
   533→                  {roundAttempts} tentativa{roundAttempts !== 1 ? "s" : ""}
   534→                </p>
   535→                <p className="text-emerald-400 font-semibold text-sm">
   536→                  +{calculatePoints(roundAttempts)} pontos
   537→                </p>
   538→              </div>
   539→            </motion.div>
   540→
   541→            <Button
   542→              onClick={handleNextRound}
   543→              className={cn(
   544→                "px-8 py-3 font-semibold text-base transition-all",
   545→                currentPlayer === 1
   546→                  ? "bg-cyan-600 hover:bg-cyan-500 text-white"
   547→                  : "bg-pink-600 hover:bg-pink-500 text-white"
   548→              )}
   549→            >
   550→              {currentRound >= TOTAL_ROUNDS ? "Ver Resultado Final" : "Próximo Round →"}
   551→            </Button>
   552→          </motion.div>
   553→        )}
