    31→/* -------------------------------------------------------------- Helpers -- */

    33→/** Animated count-up number using requestAnimationFrame + easeOutCubic. */
    34→const AnimatedNumber = ({ value, duration = 600 }: { value: number; duration?: number }) => {
    35→  const [display, setDisplay] = useState(value);
    36→  const fromRef = useRef(value);
    37→  const rafRef = useRef<number | null>(null);
    38→
    39→  useEffect(() => {
    40→    const from = fromRef.current;
    41→    const to = value;
    42→    if (from === to) { setDisplay(to); return; }
    43→    const start = performance.now();
    44→    const tick = (now: number) => {
    45→      const t = Math.min(1, (now - start) / duration);
    46→      const eased = 1 - Math.pow(1 - t, 3);
    47→      setDisplay(Math.round(from + (to - from) * eased));
    48→      if (t < 1) rafRef.current = requestAnimationFrame(tick);
    49→      else fromRef.current = to;
    50→    };
    51→    rafRef.current = requestAnimationFrame(tick);
    52→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
    53→  }, [value, duration]);
    54→
    55→  return <>{display.toLocaleString("pt-PT")}</>;
    56→};
    57→
    58→/** 60-second engagement pulse bucketed in 5s windows. */
    59→const useEngagementPulse = (entries: LeaderEntry[]) =>
    60→  useMemo(() => {
    61→    const now = Date.now();
    62→    const BUCKETS = 12;
    63→    const WINDOW_MS = 60_000;
    64→    const STEP = WINDOW_MS / BUCKETS;
    65→    const buckets = Array.from({ length: BUCKETS }, (_, i) => {
    66→      const end = now - (BUCKETS - 1 - i) * STEP;
    67→      const start = end - STEP;
    68→      return entries.filter((e) => e.at >= start && e.at < end).length;
    69→    });
    70→    const max = Math.max(1, ...buckets);
    71→    const total = buckets.reduce((a, b) => a + b, 0);
    72→    const recent = buckets.slice(-3).reduce((a, b) => a + b, 0);
    73→    return { buckets, max, total, recent };
    74→  }, [entries]);
    75→
    76→/** Top-N games by play count. */
    77→const useGameBreakdown = (entries: LeaderEntry[], n = 3) =>
    78→  useMemo(() => {
    79→    const map = new Map<string, number>();
    80→    entries.forEach((e) => map.set(e.game, (map.get(e.game) || 0) + 1));
    81→    return Array.from(map.entries())
    82→      .map(([game, plays]) => ({ game, plays }))
    83→      .sort((a, b) => b.plays - a.plays)
    84→      .slice(0, n);
    85→  }, [entries]);
    86→
    87→/** Tiny SVG sparkline with gradient fill + live cursor dot. */
    88→const Sparkline = ({ buckets, max }: { buckets: number[]; max: number }) => {
    89→  const W = 100, H = 28;
    90→  const pts = buckets.map((b, i) => {
    91→    const x = (i / Math.max(1, buckets.length - 1)) * W;
    92→    const y = H - (b / max) * (H - 4) - 2;
    93→    return [x, y] as const;
    94→  });
    95→  const line = pts.map((p, i) => `${i === 0 ? "M" : "L"}${p[0].toFixed(1)},${p[1].toFixed(1)}`).join(" ");
    96→  const area = `${line} L${W},${H} L0,${H} Z`;
    97→  return (
    98→    <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="w-full h-7">
    99→      <defs>
   100→        <linearGradient id="lcp-spark-fill" x1="0" y1="0" x2="0" y2="1">
   101→          <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity="0.35" />
   102→          <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity="0" />
   103→        </linearGradient>
   104→        <linearGradient id="lcp-spark-stroke" x1="0" y1="0" x2="1" y2="0">
   105→          <stop offset="0%" stopColor="var(--region-primary, hsl(var(--primary)))" />
   106→          <stop offset="100%" stopColor="var(--region-secondary, hsl(var(--accent)))" />
   107→        </linearGradient>
   108→      </defs>
   109→      <path d={area} fill="url(#lcp-spark-fill)" />
   110→      <path
   111→        d={line}
   112→        fill="none"
   113→        stroke="url(#lcp-spark-stroke)"
   114→        strokeWidth="1.5"
   115→        strokeLinecap="round"
   116→        strokeLinejoin="round"
   117→      />
   118→      {pts.length > 0 && (
   119→        <circle
   120→          cx={pts[pts.length - 1][0]}
   121→          cy={pts[pts.length - 1][1]}
   122→          r="1.8"
   123→          fill="var(--region-secondary, hsl(var(--accent)))"
   124→        />
   125→      )}
   126→    </svg>
   127→  );
   128→};
