    35→/** Animated count-up number using requestAnimationFrame + easeOutCubic. */
    36→const AnimatedNumber = ({ value, duration = 600 }: { value: number; duration?: number }) => {
    37→  const [display, setDisplay] = useState(value);
    38→  const fromRef = useRef(value);
    39→  const rafRef = useRef<number | null>(null);
    40→
    41→  useEffect(() => {
    42→    const from = fromRef.current;
    43→    const to = value;
    44→    if (from === to) { setDisplay(to); return; }
    45→    const start = performance.now();
    46→    const tick = (now: number) => {
    47→      const t = Math.min(1, (now - start) / duration);
    48→      const eased = 1 - Math.pow(1 - t, 3);
    49→      setDisplay(Math.round(from + (to - from) * eased));
    50→      if (t < 1) rafRef.current = requestAnimationFrame(tick);
    51→      else fromRef.current = to;
    52→    };
    53→    rafRef.current = requestAnimationFrame(tick);
    54→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); };
    55→  }, [value, duration]);
    56→
    57→  return <>{display.toLocaleString("pt-PT")}</>;
    58→};
    59→
    60→/** 60-second engagement pulse bucketed in 5s windows. */
    61→const useEngagementPulse = (entries: LeaderEntry[]) =>
    62→  useMemo(() => {
    63→    const now = Date.now();
    64→    const BUCKETS = 12;
    65→    const WINDOW_MS = 60_000;
    66→    const STEP = WINDOW_MS / BUCKETS;
    67→    const buckets = Array.from({ length: BUCKETS }, (_, i) => {
    68→      const end = now - (BUCKETS - 1 - i) * STEP;
    69→      const start = end - STEP;
    70→      return entries.filter((e) => e.at >= start && e.at < end).length;
    71→    });
    72→    const max = Math.max(1, ...buckets);
    73→    const total = buckets.reduce((a, b) => a + b, 0);
    74→    const recent = buckets.slice(-3).reduce((a, b) => a + b, 0);
    75→    return { buckets, max, total, recent };
    76→  }, [entries]);
    77→
    78→/** Top-N games by play count. */
    79→const useGameBreakdown = (entries: LeaderEntry[], n = 3) =>
    80→  useMemo(() => {
    81→    const map = new Map<string, number>();
    82→    entries.forEach((e) => map.set(e.game, (map.get(e.game) || 0) + 1));
    83→    return Array.from(map.entries())
    84→      .map(([game, plays]) => ({ game, plays }))
    85→      .sort((a, b) => b.plays - a.plays)
    86→      .slice(0, n);
    87→  }, [entries]);
    88→
    89→/** Tiny SVG sparkline with gradient fill + live cursor dot. */
    90→const Sparkline = ({ buckets, max }: { buckets: number[]; max: number }) => {
    91→  const W = 100, H = 28;
    92→  const pts = buckets.map((b, i) => {
    93→    const x = (i / Math.max(1, buckets.length - 1)) * W;
    94→    const y = H - (b / max) * (H - 4) - 2;
    95→    return [x, y] as const;
    96→  });
    97→  const line = pts.map((p, i) => `${i === 0 ? "M" : "L"}${p[0].toFixed(1)},${p[1].toFixed(1)}`).join(" ");
    98→  const area = `${line} L${W},${H} L0,${H} Z`;
    99→  return (
   100→    <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="w-full h-7">
   101→      <defs>
   102→        <linearGradient id="lcp-spark-fill" x1="0" y1="0" x2="0" y2="1">
   103→          <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity="0.35" />
   104→          <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity="0" />
   105→        </linearGradient>
   106→        <linearGradient id="lcp-spark-stroke" x1="0" y1="0" x2="1" y2="0">
   107→          <stop offset="0%" stopColor="var(--region-primary, hsl(var(--primary)))" />
   108→          <stop offset="100%" stopColor="var(--region-secondary, hsl(var(--accent)))" />
   109→        </linearGradient>
   110→      </defs>
   111→      <path d={area} fill="url(#lcp-spark-fill)" />
   112→      <path
   113→        d={line}
   114→        fill="none"
   115→        stroke="url(#lcp-spark-stroke)"
   116→        strokeWidth="1.5"
   117→        strokeLinecap="round"
   118→        strokeLinejoin="round"
   119→      />
   120→      {pts.length > 0 && (
   121→        <circle
   122→          cx={pts[pts.length - 1][0]}
   123→          cy={pts[pts.length - 1][1]}
   124→          r="1.8"
   125→          fill="var(--region-secondary, hsl(var(--accent)))"
   126→        />
   127→      )}
   128→    </svg>
   129→  );
   130→};
