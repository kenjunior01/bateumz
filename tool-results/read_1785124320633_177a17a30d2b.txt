    87→/** Relative time formatter in Portuguese (Mozambique). */
    88→const relTime = (ts: number): string => {
    89→  const sec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
    90→  if (sec < 5) return "agora";
    91→  if (sec < 60) return `há ${sec}s`;
    92→  const min = Math.floor(sec / 60);
    93→  if (min < 60) return `há ${min}min`;
    94→  const h = Math.floor(min / 60);
    95→  return `há ${h}h`;
    96→};
