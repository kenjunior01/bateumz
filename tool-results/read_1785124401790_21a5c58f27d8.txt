    69→    69→
    70→    70→const fmtTime = (s: number) => {
    71→    71→  const safe = Math.max(0, Math.floor(s));
    72→    72→  const m = Math.floor(safe / 60).toString().padStart(2, "0");
    73→    73→  const sec = (safe % 60).toString().padStart(2, "0");
    74→    74→  return `${m}:${sec}`;
    75→    75→};
    76→    76→
    77→    77→const fmtNumber = (n: number) => n.toLocaleString("pt-PT");
    78→    78→
    79→    79→/** Animated count-up number using framer-motion's spring driver. */
    80→    80→const AnimatedCounter = ({ value, className }: { value: number; className?: string }) => {
    81→    81→  const count = useMotionValue(0);
    82→    82→  const rounded = useTransform(count, (v) => fmtNumber(Math.max(0, Math.round(v))));
    83→    83→
    84→    84→  useEffect(() => {
    85→    85→    const controls = animate(count, value, {
    86→    86→      duration: 0.6,
    87→    87→      ease: [0.16, 1, 0.3, 1],
    88→    88→    });
    89→    89→    return () => controls.stop();
    90→    90→  }, [value, count]);
    91→    91→
    92→    92→  return <motion.span className={className}>{rounded}</motion.span>;
    93→    93→};
