    68→/* ---------- helpers ---------- */
    69→
    70→const fmtTime = (s: number) => {
    71→  const safe = Math.max(0, Math.floor(s));
    72→  const m = Math.floor(safe / 60).toString().padStart(2, "0");
    73→  const sec = (safe % 60).toString().padStart(2, "0");
    74→  return `${m}:${sec}`;
    75→};
    76→
    77→const fmtNumber = (n: number) => n.toLocaleString("pt-PT");
    78→
    79→/** Animated count-up number using framer-motion's spring driver. */
    80→const AnimatedCounter = ({ value, className }: { value: number; className?: string }) => {
    81→  const count = useMotionValue(0);
    82→  const rounded = useTransform(count, (v) => fmtNumber(Math.max(0, Math.round(v))));
    83→
    84→  useEffect(() => {
    85→    const controls = animate(count, value, {
    86→      duration: 0.6,
    87→      ease: [0.16, 1, 0.3, 1],
    88→    });
    89→    return () => controls.stop();
    90→  }, [value, count]);
    91→
    92→  return <motion.span className={className}>{rounded}</motion.span>;
    93→};
