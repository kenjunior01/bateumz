    87→const fmtTime = (s: number) => {
    88→  const safe = Math.max(0, Math.floor(s));
    89→  const m = Math.floor(safe / 60).toString().padStart(2, "0");
    90→  const sec = (safe % 60).toString().padStart(2, "0");
    91→  return `${m}:${sec}`;
    92→};
    93→
    94→const fmtNumber = (n: number) => n.toLocaleString("pt-PT");
    95→
    96→/** Relative time formatter in Portuguese (Mozambique). */
    97→const relTime = (ts: number, now: number): string => {
    98→  const sec = Math.max(0, Math.floor((now - ts) / 1000));
    99→  if (sec < 5) return "agora";
   100→  if (sec < 60) return `há ${sec}s`;
   101→  const min = Math.floor(sec / 60);
   102→  if (min < 60) return `há ${min}min`;
   103→  const h = Math.floor(min / 60);
   104→  return `há ${h}h`;
   105→};

   115→/** Animated count-up number using framer-motion's spring driver. */
   116→const AnimatedCounter = ({ value, className }: { value: number; className?: string }) => {
   117→  const count = useMotionValue(0);
   118→  const rounded = useTransform(count, (v) => fmtNumber(Math.max(0, Math.round(v))));
   119→
   120→  useEffect(() => {
   121→    const controls = animate(count, value, {
   122→      duration: 0.6,
   123→      ease: [0.16, 1, 0.3, 1],
   124→    });
   125→    return () => controls.stop();
   126→  }, [value, count]);
   127→
   128→  return <motion.span className={className}>{rounded}</motion.span>;
   129→};
