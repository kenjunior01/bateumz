   470→                  </div>
   471→                  <span className="navbar-points flex items-center gap-1 rounded-full px-2.5 py-1 text-[11px] font-bold">
   472→                    <Star className="h-3 w-3 fill-current" />

   500→                          <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-primary/10 text-primary transition-all group-hover:bg-primary group-hover:text-primary-foreground group-hover:shadow-glow">

   520→              {/* Sticky CTA */}
   521→              <div className="mt-6 safe-area-bottom">

   535→                    <Link
   536→                      to="/login"

   543→                      to="/register"

   546→                    >

   548→                      {t("nav.signup", "Participar")}

   552→              </div>
   553→            </div>
   554→          </motion.div>
   555→        )}
   556→      </AnimatePresence>
   557→    </motion.nav>
