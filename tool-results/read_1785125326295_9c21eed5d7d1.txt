    73→/* ─── Animated counter hook ─── */
    74→function useAnimatedValue(target: number, duration = 600) {
    75→  const [display, setDisplay] = useState(target);
    76→  const raf = useRef<number>(0);
    77→  const start = useRef(target);
    78→
    79→  useEffect(() => {
    80→    start.current = display;
    81→    const from = start.current;
    82→    if (from === target) return;
    83→    const startTime = performance.now();
    84→    const tick = (now: number) => {
    85→      const progress = clamp((now - startTime) / duration, 0, 1);
    86→      const eased = 1 - Math.pow(1 - progress, 3);
    87→      setDisplay(Math.round(from + (target - from) * eased));
    88→      if (progress < 1) raf.current = requestAnimationFrame(tick);
    89→    };
    90→    raf.current = requestAnimationFrame(tick);
    91→    return () => cancelAnimationFrame(raf.current);
    92→  }, [target, duration]);
    93→
    94→  return display;
    95→}

    97→/* ─── Circular gauge component ─── */
    98→const RingGauge = ({
    99→  value,
   100→  max,
   101→  size = 52,
   102→  stroke = 4,
   103→  color = "var(--primary)",
   104→  children,
   105→}: {
   106→  value: number;
   107→  max: number;
   108→  size?: number;
   109→  stroke?: number;
   110→  color?: string;
   111→  children: React.ReactNode;
   112→}) => {
   113→  const r = (size - stroke) / 2;
   114→  const circ = 2 * Math.PI * r;
   115→  const pct = max > 0 ? clamp(value / max, 0, 1) : 0;
   116→
   117→  return (
   118→    <div className="relative" style={{ width: size, height: size }}>
   119→      <svg width={size} height={size} className="-rotate-90">
   120→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="currentColor" strokeWidth={stroke} className="text-muted/30" />
   121→        <motion.circle
   122→          cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={stroke} strokeLinecap="round"
   123→          strokeDasharray={circ} initial={{ strokeDashoffset: circ }}
   124→          animate={{ strokeDashoffset: circ * (1 - pct) }}
   125→          transition={{ duration: 0.8, ease: "easeOut" }}
   126→        />
   127→      </svg>
   128→      <div className="absolute inset-0 flex items-center justify-center">{children}</div>
   129→    </div>
   130→  );
   131→};
