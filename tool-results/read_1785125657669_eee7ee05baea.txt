    45→    45→  useEffect(() => {
    46→    46→    if (!inView) return;
    47→    47→    const controls = animate(count, to, {
    48→    48→      duration,
    49→    49→      ease: [0.16, 1, 0.3, 1],

    52→    52→  }, [inView, to, count, duration]);
    53→    53→
    54→    54→  return <motion.span ref={ref}>{rounded}</motion.span>;
