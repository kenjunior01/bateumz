   231→const computeQuality = (

   233→  plays: number,
   234→  games: number,
   235→  heat: number,

   237→  if (plays === 0) return 0;

   344→ * ★ REVOLUTION: Hype Meter — detects surges in plays over the last 20s vs the
   345→ * previous 20s window. Returns 0-100 representing "is the chat exploding right now".
   346→ */
   347→const computeHype = (entries: LeaderEntry[], now: number): number => {
   348→  if (!entries.length) return 0;
   349→  const recent = entries.filter((e) => e.at >= now - 20_000).length;
   350→  const prior = entries.filter((e) => e.at >= now - 40_000 && e.at < now - 20_000).length;
   351→  if (prior === 0) return recent > 0 ? Math.min(100, recent * 25) : 0;
   352→  const ratio = recent / Math.max(1, prior);
   353→  return Math.max(0, Math.min(100, Math.round((ratio - 1) * 80 + 20)));

   387→/**
   388→ * ★ REVOLUTION: Trajectory forecast.
   389→ * Predicts projected total plays at the current pace, plus ETA in seconds to reach the goal.
   390→ */
   391→const computeTrajectory = (
   392→  plays: number,
   393→  elapsedSec: number,
   394→  goal: number,
   395→): { projected: number; etaSec: number | null; pacePerMin: number } => {
   396→  if (elapsedSec < 5 || plays === 0) return { projected: plays, etaSec: null, pacePerMin: 0 };
   397→  const pacePerMin = (plays / elapsedSec) * 60;
   398→  const projected = Math.round(plays + (pacePerMin / 60) * Math.max(0, (1800 - elapsedSec))); // project to 30min mark
   399→  if (plays >= goal || pacePerMin <= 0) return { projected, etaSec: null, pacePerMin };
   400→  const remaining = goal - plays;
   401→  const etaSec = Math.round((remaining / pacePerMin) * 60);
   402→  return { projected, etaSec, pacePerMin };
   403→};

   889→const CommandPalette = ({
   890→  open,
   891→  onClose,
   892→  items,
   893→}: {
   894→  open: boolean;
   895→  onClose: () => void;
   896→  items: CommandItem[];
   897→}) => {
   898→  const [query, setQuery] = useState("");
   899→  const [active, setActive] = useState(0);
   900→  const inputRef = useRef<HTMLInputElement>(null);
   901→
   902→  useEffect(() => {
   903→    if (open) {
   904→      setQuery("");
   905→      setActive(0);
   906→      setTimeout(() => inputRef.current?.focus(), 50);
   907→    }
   908→  }, [open]);
   909→
   910→  const filtered = useMemo(() => {
   911→    if (!query.trim()) return items;
   912→    const q = query.toLowerCase();
   913→    return items.filter(
   914→      (it) =>
   915→        it.label.toLowerCase().includes(q) ||
   916→        it.group.toLowerCase().includes(q) ||
   917→        it.hint?.toLowerCase().includes(q),
   918→    );
   919→  }, [items, query]);
   920→
   921→  useEffect(() => {
   922→    setActive(0);
   923→  }, [query]);
   924→
   925→  const groups = useMemo(() => {
   926→    const map = new Map<string, CommandItem[]>();
   927→    filtered.forEach((it) => {
   928→      if (!map.has(it.group)) map.set(it.group, []);
   929→      map.get(it.group)!.push(it);
   930→    });
   931→    return [...map.entries()];
   932→  }, [filtered]);
   933→
   934→  const onKeyDown = (e: React.KeyboardEvent) => {
   935→    if (e.key === "ArrowDown") {
   936→      e.preventDefault();
   937→      setActive((a) => Math.min(a + 1, filtered.length - 1));
   938→    } else if (e.key === "ArrowUp") {
   939→      e.preventDefault();
   940→      setActive((a) => Math.max(a - 1, 0));
   941→    } else if (e.key === "Enter") {
   942→      e.preventDefault();
   943→      const it = filtered[active];
   944→      if (it && !it.disabled) {
   945→        it.run();
   946→        onClose();
   947→      }
   948→    } else if (e.key === "Escape") {
   949→      e.preventDefault();
   950→      onClose();
   951→    }
   952→  };
   953→
   954→  return (
   955→    <AnimatePresence>
   956→      {open && (
   957→        <motion.div
   958→          initial={{ opacity: 0 }}
   959→          animate={{ opacity: 1 }}
   960→          exit={{ opacity: 0 }}
   961→          transition={{ duration: 0.15 }}
   962→          className="fixed inset-0 z-[100] flex items-start justify-center pt-[15vh] px-4 bg-background/70 backdrop-blur-sm"
   963→          onClick={onClose}
   964→        >
   965→          <motion.div
   966→            initial={{ opacity: 0, y: -16, scale: 0.96 }}
   967→            animate={{ opacity: 1, y: 0, scale: 1 }}
   968→            exit={{ opacity: 0, y: -16, scale: 0.96 }}
   969→            transition={{ type: "spring", stiffness: 400, damping: 28 }}
   970→            onClick={(e) => e.stopPropagation()}
   971→            className="w-full max-w-xl rounded-2xl border border-border bg-card shadow-elegant overflow-hidden"
   972→          >
   973→            <div className="flex items-center gap-2 px-3 py-2.5 border-b border-border/60">
   974→              <Search className="h-4 w-4 text-muted-foreground shrink-0" />
   975→              <input
   976→                ref={inputRef}
   977→                value={query}
   978→                onChange={(e) => setQuery(e.target.value)}
   979→                onKeyDown={onKeyDown}
   980→                placeholder="Pesquisar ação... (ex: coroar, exportar, copiar)"
   981→                className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   982→              />
   983→              <kbd className="text-[9px] font-mono px-1.5 py-0.5 rounded border border-border bg-muted/50 text-muted-foreground">
   984→                ESC
   985→              </kbd>
