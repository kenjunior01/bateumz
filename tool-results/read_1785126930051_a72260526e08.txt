    32→/** Animated count-up number that runs once when scrolled into view. */
    33→const CountUp = ({
    34→  to,
    35→  duration = 1.4,
    36→  format,
    37→}: {
    38→  to: number;
    39→  duration?: number;
    40→  format: (n: number) => string;
    41→}) => {
    42→  const ref = useRef<HTMLSpanElement>(null);
    43→  const inView = useInView(ref, { once: true, margin: "-40px" });
    44→  const count = useMotionValue(0);
    45→  const rounded = useTransform(count, (v) => format(Math.max(0, Math.round(v))));
    46→
    47→  useEffect(() => {
    48→    if (!inView) return;
    49→    const controls = animate(count, to, { duration, ease: "easeOut" });
    50→    return () => controls.stop();
    51→  }, [inView, to, count, duration]);
    52→
    53→  return <motion.span ref={ref}>{rounded}</motion.span>;
