    69→function formatDate(iso: string) {
    70→  return new Date(iso).toLocaleDateString('pt-BR', {
    71→    day: '2-digit',
    72→    month: 'short',
    73→    year: 'numeric',
    74→  });
    75→}
    76→
    77→function formatCount(n: number): string {
    78→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
    79→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
    80→  return String(n);
    81→}

   252→export default function Blog() {
   253→  const { region } = useRegionalTheme();
   254→
   255→  /* ---------- state ---------- */
   256→  const [posts, setPosts] = useState<BlogPost[]>([]);
   257→  const [categories, setCategories] = useState<BlogCategory[]>([]);

   259→  const [featuredPost, setFeaturedPost] = useState<BlogPost | null>(null);
   260→
   261→  const [loading, setLoading] = useState(true);
   262→  const [loadingMore, setLoadingMore] = useState(false);
   263→  const [error, setError] = useState<string | null>(null);
   264→
   265→  const [searchTerm, setSearchTerm] = useState('');
   266→  const [activeCategory, setActiveCategory] = useState<string | null>(null);
   267→
   268→  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
   269→  const [hasMore, setHasMore] = useState(false);
   270→
   271→  /* ---------- data fetching ---------- */
   272→  const fetchData = useCallback(async () => {
   273→    setLoading(true);
   274→    setError(null);
   275→    try {
   276→      // Categorias
   277→      const { data: catData } = await (supabase as any)
   278→        .from('blog_categories')
   279→        .select('*')
   280→        .order('name');
   281→      setCategories((catData as BlogCategory[]) || []);
   282→
   283→      // Posts principais
   284→      let postQuery = (supabase as any)
   285→        .from('blog_posts')
   286→        .select(
   287→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   288→        )
   289→        .eq('published', true);
   290→
   291→      if (region) {
   292→        postQuery = postQuery.or(
   293→          `region_id.eq.${region.id},region_id.is.null`
   294→        );
   295→      }
   296→
   297→      const { data: postData, error: postErr } = await postQuery.order(
   298→        'published_at',
   299→        { ascending: false }
   300→      );
   301→      if (postErr) throw postErr;
   302→      setPosts((postData as BlogPost[]) || []);
   303→
   304→      // Posts em alta
   305→      let trendQuery = (supabase as any)
   306→        .from('blog_posts')
   307→        .select(
   308→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   309→        )
   310→        .eq('published', true)
   311→        .eq('is_trending', true);
   312→
   313→      if (region) {
   314→        trendQuery = trendQuery.or(
   315→          `region_id.eq.${region.id},region_id.is.null`
   316→        );
   317→      }
   318→
   319→      const { data: trendData } = await trendQuery
   320→        .order('trending_score', { ascending: false })
   321→        .limit(5);
   322→      setTrendingPosts((trendData as BlogPost[]) || []);
   323→
   324→      // Post destaque
   325→      let featQuery = (supabase as any)
   326→        .from('blog_posts')
   327→        .select(
   328→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   329→        )
   330→        .eq('published', true)
   331→        .eq('is_featured', true);
   332→
   333→      if (region) {
   334→        featQuery = featQuery.or(
   335→          `region_id.eq.${region.id},region_id.is.null`
   336→        );
   337→      }
   338→
   339→      const { data: featData } = await featQuery
   340→        .order('published_at', { ascending: false })
   341→        .limit(1)
   342→        .single();
   343→      setFeaturedPost((featData as BlogPost) || null);
   344→    } catch (err: any) {
   345→      console.error('Erro ao carregar blog:', err);
   346→      setError('Não foi possível carregar os posts. Tente novamente mais tarde.');
   347→    } finally {
   348→      setLoading(false);
   349→    }
   350→  }, [region]);
   351→
   352→  useEffect(() => {
   353→    fetchData();
   354→  }, [fetchData]);
   355→
   356→  /* ---------- filtering ---------- */
   357→  const filteredPosts = useMemo(() => {
   358→    let result = posts;
   359→
   360→    // Remover o post destaque da listagem principal para evitar duplicidade
   361→    if (featuredPost) {
   362→      result = result.filter((p) => p.id !== featuredPost.id);
   363→    }
   364→
   365→    // Filtro por categoria
   366→    if (activeCategory) {
   367→      result = result.filter((p) => p.category?.slug === activeCategory);
   368→    }
   369→
   370→    // Filtro por busca
   371→    if (searchTerm.trim()) {
   372→      const term = searchTerm.toLowerCase();
   373→      result = result.filter(
   374→        (p) =>
   375→          p.title.toLowerCase().includes(term) ||
   376→          p.summary.toLowerCase().includes(term) ||
   377→          p.seo_keywords?.some((k: string) => k.toLowerCase().includes(term))
   378→      );
   379→    }
   380→
   381→    return result;
   382→  }, [posts, activeCategory, searchTerm, featuredPost]);
   383→
   384→  const visiblePosts = useMemo(
   385→    () => filteredPosts.slice(0, visibleCount),
   386→    [filteredPosts, visibleCount]
   387→  );
   388→
   389→  useEffect(() => {
   390→    setHasMore(visibleCount < filteredPosts.length);
   391→  }, [visibleCount, filteredPosts.length]);
   392→
   393→  // Resetar contagem ao mudar filtros
   394→  useEffect(() => {
   395→    setVisibleCount(PAGE_SIZE);
   396→  }, [activeCategory, searchTerm]);
   397→
   398→  const handleLoadMore = () => {
   399→    setLoadingMore(true);
   400→    setTimeout(() => {
   401→      setVisibleCount((prev) => prev + PAGE_SIZE);
   402→      setLoadingMore(false);
   403→    }, 400);
   404→  };

   410→      <Helmet>
   411→        <title>Blog – Bateu Online | Notícias, Dicas e Sorteios</title>
   412→        <meta
   413→          name="description"
   414→          content="Fique por dentro das últimas novidades sobre sorteios online, dicas de participação, resultados e muito mais no blog do Bateu Online."
   415→        />
