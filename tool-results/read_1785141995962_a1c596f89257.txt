    85→function formatDate(iso: string): string {
    86→  return new Date(iso).toLocaleDateString("pt-BR", {
    87→    day: "numeric",
    88→    month: "long",
    89→    year: "numeric",
    90→  });
    91→}

    93→function formatNumber(n: number): string {
    94→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1).replace(".0", "")}M`;
    95→  if (n >= 1_000) return `${(n / 1_000).toFixed(1).replace(".0", "")}K`;
    96→  return String(n);
    97→}

    99→function parseTOC(html: string): TOCItem[] {
   100→  const parser = new DOMParser();
   101→  const doc = parser.parseFromString(html, "text/html");
   102→  const headings = doc.querySelectorAll("h2, h3");
   103→  const items: TOCItem[] = [];
   104→  headings.forEach((h, i) => {
   105→    const id = `section-${i}`;
   106→    h.id = id;
   107→    items.push({ id, text: h.textContent || `Seção ${i + 1}`, level: h.tagName === "H2" ? 2 : 3 });
   108→  });
   109→  return items;
   110→}
   111→
   112→function injectAnchorIds(html: string, toc: TOCItem[]): string {
   113→  const parser = new DOMParser();
   114→  const doc = parser.parseFromString(html, "text/html");
   115→  const headings = doc.querySelectorAll("h2, h3");
   116→  headings.forEach((h, i) => {
   117→    if (toc[i]) h.id = toc[i].id;
   118→  });
   119→  return doc.body.innerHTML;
   120→}
