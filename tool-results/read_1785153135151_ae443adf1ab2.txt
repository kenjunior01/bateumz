    55→const sparklineFromEntries = (entries: LeaderEntry[]): number[] => {
    56→  const now = Date.now();
    57→  const buckets = new Array(24).fill(0);
    58→  for (const e of entries) {
    59→    const age = now - e.at;
    60→    if (age > 7200000) continue;
    61→    const idx = Math.min(23, 23 - Math.floor(age / 300000));
    62→    buckets[idx] += 1;
    63→  }
    64→  return buckets;
    65→};
    66→
    67→const computeQualityScore = (entries: LeaderEntry[]): number => {
    68→  if (entries.length === 0) return 0;
    69→  const players = new Set(entries.map((e) => e.name)).size;
    70→  const games = new Set(entries.map((e) => e.game)).size;
    71→  const engagement = Math.min(100, Math.round((entries.length / Math.max(1, players)) * 25));
    72→  const diversity = Math.min(100, Math.round(games * 20));
    73→  const activity = Math.min(100, Math.round(players * 10));
    74→  return Math.round((engagement * 0.4 + diversity * 0.3 + activity * 0.3));
    75→};

    77→const QualityRing = ({ score }: { score: number }) => {

    81→  const circumference = 2 * Math.PI * radius;
    82→  const offset = circumference - (normalized / 100) * circumference;
    83→
    84→  const colorClass =
    85→    normalized >= 75 ? "text-emerald-500" : normalized >= 50 ? "text-yellow-500" : "text-orange-500";
    86→
    87→  return (
    88→    <div className="relative w-24 h-24 flex items-center justify-center">
    89→      <svg className="absolute inset-0 w-full h-full -rotate-90" viewBox="0 0 100 100">
    90→        <circle cx="50" cy="50" r={radius} fill="none" className="stroke-border/40" strokeWidth={stroke} />
    91→        <motion.circle
    92→          cx="50"
    93→          cy="50"
    94→          r={radius}
    95→          fill="none"
    96→          className={colorClass}
    97→          strokeWidth={stroke}
    98→          strokeLinecap="round"
    99→          strokeDasharray={circumference}
   100→          initial={{ strokeDashoffset: circumference }}
   101→          animate={{ strokeDashoffset: offset }}
   102→          transition={{ duration: 1.2, ease: "easeOut" }}

   120→const SparklineChart = ({ data }: { data: number[] }) => {
   121→  const maxVal = Math.max(1, ...data);
   122→  const width = 200;
   123→  const height = 48;
   124→  const points = data.map((v, i) => {
   125→    const x = (i / (data.length - 1)) * width;
   126→    const y = height - (v / maxVal) * height;
   127→    return `${x},${y}`;
   128→  });
   129→  const pathD = `M ${points.join(" L ")}`;
   130→
   131→  const areaD = `M 0,${height} L ${points.join(" L ")} L ${width},${height} Z`;
   132→
   133→  return (
   134→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-12" preserveAspectRatio="none">
   135→      <defs>
   136→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   137→          <stop offset="0%" stopColor="currentColor" stopOpacity="0.3" />
   138→          <stop offset="100%" stopColor="currentColor" stopOpacity="0" />
   139→        </linearGradient>
   140→      </defs>
   141→      <motion.path
   142→        d={areaD}
   143→        fill="url(#sparkGrad)"
   144→        className="text-primary"
   145→        initial={{ opacity: 0 }}
   146→        animate={{ opacity: 1 }}
   147→        transition={{ duration: 0.8 }}
