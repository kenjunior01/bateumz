    55→    55→const sparklineFromEntries = (entries: LeaderEntry[]): number[] => {
    56→    56→  const now = Date.now();

    58→    58→  for (const e of entries) {
    59→    59→    const age = now - e.at;
    60→    60→    if (age > 7200000) continue;
    61→    61→    const idx = Math.min(23, 23 - Math.floor(age / 300000));

    67→    67→const computeQualityScore = (entries: LeaderEntry[]): number => {
    68→    68→  if (entries.length === 0) return 0;

    70→    70→  const games = new Set(entries.map((e) => e.game)).size;
    71→    71→  const engagement = Math.min(100, Math.round((entries.length / Math.max(1, players)) * 25));

    73→    73→  const activity = Math.min(100, Math.round(players * 10));
