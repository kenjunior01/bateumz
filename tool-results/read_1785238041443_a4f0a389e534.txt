    53→const fireConfetti = (type: string = "confetti") => {
    54→  const duration = 3 * 1000;
    55→  const animationEnd = Date.now() + duration;
    56→  const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };
    57→
    58→  const randomInRange = (min: number, max: number) => Math.random() * (max - min) + min;
    59→
    60→  const interval = setInterval(() => {
    61→    const timeLeft = animationEnd - Date.now();
    62→    if (timeLeft <= 0) {
    63→      return clearInterval(interval);
    64→    }
    65→
    66→    const particleCount = 50 * (timeLeft / duration);
    67→
    68→    switch (type) {
    69→      case "fireworks":
    70→        confetti({
    71→          ...defaults,
    72→          particleCount,
    73→          spread: 70,
    74→          origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
    75→          colors: ["#ff0000", "#00ff00", "#0000ff", "#ffff00"],
    76→        });
    77→        confetti({
    78→          ...defaults,
    79→          particleCount,
    80→          spread: 70,
    81→          origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
    82→          colors: ["#ff00ff", "#00ffff", "#ff8800", "#88ff00"],
    83→        });
    84→        break;
    85→      case "stars":
    86→        confetti({
    87→          ...defaults,
    88→          particleCount: particleCount * 0.5,
    89→          shapes: ["star"],
    90→          gravity: 0.8,
    91→          scalar: 1.2,
    92→          origin: { y: 0.4 },
    93→          colors: ["#ffd700", "#ffed4e", "#ffffff"],
    94→        });
    95→        break;
    96→      case "poppers":
    97→        for (let i = 0; i < 5; i++) {
    98→          setTimeout(() => {
    99→            confetti({
   100→              ...defaults,
   101→              particleCount: 20,
   102→              origin: { x: 0.2 + i * 0.15, y: 0.6 },
   103→              spread: 60,
   104→              colors: ["#ff4444", "#44ff44", "#4444ff", "#ffff44"],
   105→            });
   106→          }, i * 200);
   107→        }
   108→        break;
   109→      case "zap":
   110→        confetti({
   111→          ...defaults,
   112→          particleCount: 100,
   113→          shapes: ["circle", "square"],
   114→          gravity: 1.2,
   115→          scalar: 0.9,
   116→          origin: { y: 0.2 },
   117→          colors: ["#ff00ff", "#00ffff", "#ff0000", "#00ff00"],
   118→        });
   119→        break;
   120→      case "confetti":
   121→      default:
   122→        confetti({
   123→          ...defaults,
   124→          particleCount,
   125→          origin: { x: randomInRange(0.1, 0.9), y: Math.random() - 0.2 },
   126→        });
   127→        confetti({
   128→          ...defaults,
   129→          particleCount,
   130→          origin: { x: randomInRange(0.2, 0.8), y: Math.random() - 0.1 },
   131→          spread: 180,
   132→        });
   133→        break;
   134→    }
   135→  }, 250);

   153→const pickWeighted = (prizes: WheelPrize[]) => {
   154→  const total = prizes.reduce((s, p) => s + Math.max(0, p.weight), 0) || 1;
   155→  let r = Math.random() * total;
   156→  for (let i = 0; i < prizes.length; i++) {
   157→    r -= Math.max(0, prizes[i].weight);
   158→    if (r <= 0) return i;
   159→  }
   160→  return prizes.length - 1;
   161→};

   525→
   526→        {/* Top: Webcam Placeholder (Vertical Mode) */}
   527→        {mode === "vertical" && (

   535→                  <img src={branding.companyLogoUrl} alt="Logo" className="h-12 object-contain" />

   537→                {branding?.companySlogan && (
   538→                  <p className="text-xs font-black uppercase tracking-[0.2em]" style={{ color: `${currentTheme.primaryColor}80` }}>{branding.companySlogan}</p>
   539→                )}
   540→              </div>
   541→            )}
   542→          </div>
   543→        )}
   544→
   545→        {/* Center: The Wheel */}
   546→        <div className="relative group">
   547→          {/* Modern Enhanced Pointer - More Precise */}
   548→          <div className="absolute -top-8 left-1/2 -translate-x-1/2 z-20 drop-shadow-2xl">
   549→            <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center border-4 border-primary shadow-lg relative" style={{ borderColor: currentTheme.primaryColor }}>

   557→          {/* Wheel Container */}

   559→            <canvas

   561→              width="600"

   563→              className="max-w-[300px] sm:max-w-[400px] md:max-w-[600px] h-auto rounded-full"

   596→          <AnimatePresence>

   650→                      <img 
   651→                        src={result.rewardImageUrl} 
   652→                        alt="Prémio" 
   653→                        className="w-32 h-32 object-cover rounded-2xl shadow-lg"
   654→                      />
   655→                    )}
   656→                  </div>
   657→                ) : (
   658→                  <p className="font-bold text-3xl flex items-center gap-3 justify-center">
   659→                    😅 {t("tryAgain")}
   660→                  </p>
   661→                )}
   662→              </motion.div>
   663→            )}
   664→          </AnimatePresence>
   665→
   666→          {/* Minimum Segments Warning */}
   667→          {prizes.length < 4 && (

   674→          <div className="rounded-2xl bg-black/20 border border-white/10 backdrop-blur-xl text-white overflow-hidden">
