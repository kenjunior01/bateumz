    53→const fireConfetti = (type: string = "confetti") => {

    59→
    60→  const interval = setInterval(() => {
    61→    const timeLeft = animationEnd - Date.now();
    62→    if (timeLeft <= 0) {
    63→      return clearInterval(interval);
    64→    }
    65→
    66→    const particleCount = 50 * (timeLeft / duration);
    67→
    68→    switch (type) {
    69→      case "fireworks":
    70→        confetti({
    71→          ...defaults,
    72→          particleCount,
    73→          spread: 70,
    74→          origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
    75→          colors: ["#ff0000", "#00ff00", "#0000ff", "#ffff00"],
    76→        });
    77→        confetti({
    78→          ...defaults,
    79→          particleCount,
    80→          spread: 70,
    81→          origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
    82→          colors: ["#ff00ff", "#00ffff", "#ff8800", "#88ff00"],
    83→        });
    84→        break;
    85→      case "stars":
    86→        confetti({
    87→          ...defaults,
    88→          particleCount: particleCount * 0.5,
    89→          shapes: ["star"],
    90→          gravity: 0.8,
    91→          scalar: 1.2,
    92→          origin: { y: 0.4 },
    93→          colors: ["#ffd700", "#ffed4e", "#ffffff"],
    94→        });
    95→        break;
    96→      case "poppers":
    97→        for (let i = 0; i < 5; i++) {
    98→          setTimeout(() => {
    99→            confetti({
   100→              ...defaults,
   101→              particleCount: 20,

   103→              spread: 60,
   104→              colors: ["#ff4444", "#44ff44", "#4444ff", "#ffff44"],
   105→            });
   106→          }, i * 200);
   107→        }
   108→        break;
   109→      case "zap":
   110→        confetti({
   111→          ...defaults,
   112→          particleCount: 100,
   113→          shapes: ["circle", "square"],
   114→          gravity: 1.2,
   115→          scalar: 0.9,
   116→          origin: { y: 0.2 },
   117→          colors: ["#ff00ff", "#00ffff", "#ff0000", "#00ff00"],

   121→      default:

   153→const pickWeighted = (prizes: WheelPrize[]) => {
   154→  const total = prizes.reduce((s, p) => s + Math.max(0, p.weight), 0) || 1;
   155→  let r = Math.random() * total;
   156→  for (let i = 0; i < prizes.length; i++) {
   157→    r -= Math.max(0, prizes[i].weight);
   158→    if (r <= 0) return i;
   159→  }
   160→  return prizes.length - 1;
   161→};

   520→        )}
   521→
   522→        {/* Top: Webcam Placeholder (Vertical Mode) */}
   523→        {mode === "vertical" && (
   524→          <div className="w-full max-w-md space-y-4 text-center">
   525→            <div className="w-full h-40 rounded-2xl bg-black/30 border-2 border-white/20 flex items-center justify-center">
   526→              <p className="text-white/50 text-sm">{t("wheel.webcam")}</p>
   527→            </div>
   528→            {(branding?.companyLogoUrl || branding?.companySlogan) && (
   529→              <div className="flex flex-col items-center gap-2">
   530→                {branding?.companyLogoUrl && (
   531→                  <img src={branding.companyLogoUrl} alt="Logo" className="h-12 object-contain" />
   532→                )}
   533→                {branding?.companySlogan && (
   534→                  <p className="text-xs font-black uppercase tracking-[0.2em]" style={{ color: `${currentTheme.primaryColor}80` }}>{branding.companySlogan}</p>
   535→                )}
   536→              </div>
   537→            )}
   538→          </div>
   539→        )}
   540→
   541→        {/* Center: The Wheel */}
   542→        <div className="relative group">
   543→          {/* Modern Enhanced Pointer - More Precise */}
   544→          <div className="absolute -top-8 left-1/2 -translate-x-1/2 z-20 drop-shadow-2xl">
   545→            <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center border-4 border-primary shadow-lg relative" style={{ borderColor: currentTheme.primaryColor }}>
   546→              <div className="absolute -bottom-1 left-1/2 -translate-x-1/2">
   547→                <div className="w-0 h-0 border-l-[16px] border-r-[16px] border-t-[28px] border-l-transparent border-r-transparent" style={{ borderTopColor: currentTheme.primaryColor }} />
   548→              </div>
   549→              <Sparkles className="w-6 h-6" style={{ color: currentTheme.primaryColor }} />
   550→            </div>
   551→          </div>
   552→
   553→          {/* Wheel Container */}
   554→          <div className="relative p-6 rounded-full bg-white/5 border-2 border-white/10 backdrop-blur-sm shadow-[0_0_80px_rgba(0,0,0,0.6)]" style={{ borderColor: `${currentTheme.primaryColor}40` }}>
   555→            <canvas
   556→              ref={canvasRef}
   557→              width="600"
   558→              height="600"
   559→              className="max-w-[300px] sm:max-w-[400px] md:max-w-[600px] h-auto rounded-full"
   560→              style={{
   561→                transform: `rotate(${rotation}deg)`,
   562→                transition: spinning ? 'none' : 'transform 0.5s cubic-bezier(0.1, 0, 0.3, 1)'
   563→              }}
   564→            />
   565→            
   566→            {/* Center Button */}
   567→            <button
   568→                onClick={spin}
   569→                disabled={spinning || loading}
   570→                className={`absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-24 h-24 rounded-full 
   571→                flex flex-col items-center justify-center font-black text-sm transition-all duration-300 z-20
   572→                ${spinning || loading ? 'bg-gray-500 scale-95 opacity-50 cursor-not-allowed' : 'bg-white hover:scale-110 shadow-2xl active:scale-90 cursor-pointer'}`}
   573→                style={{ color: wheelConfig?.wheel_background_color || currentTheme.primaryColor }}
   574→              >
   575→                {loading || spinning ? (
   576→                  <span className="flex items-center gap-2">
   577→                    <Sparkles className="w-6 h-6 animate-spin" />
   578→                    {t("wheel.spinning")}
   579→                  </span>
   580→                ) : (
   581→                  <span className="flex items-center gap-2">
   582→                    <PartyPopper className="w-6 h-6" />
   583→                    {t("wheel.spin")}
   584→                  </span>
   585→                )}
   586→              </button>
   587→          </div>
   588→        </div>
   589→
   590→        {/* Right Side: Last Winner & Rewards (Horizontal Mode) or Bottom (Vertical Mode) */}
   591→        <div className={`${mode === "horizontal" ? "w-full md:w-96" : "w-full max-w-md"} space-y-6`}>
   592→          <AnimatePresence>
   593→            {result && (
   594→              <motion.div
   595→                initial={{ opacity: 0, scale: 0.3, y: 100 }}
   596→                animate={{ opacity: 1, scale: 1, y: 0 }}
   597→                exit={{ opacity: 0, scale: 0.3, y: -100 }}
   598→                transition={{ type: "spring", bounce: 0.6, duration: 0.8 }}
   599→                className={`rounded-3xl p-8 text-center space-y-4 shadow-2xl border-2 ${
   600→                  result.rewardType === "none" || isNoWinLabel(result.label)
   601→                    ? "bg-secondary/90 text-muted-foreground border-white/20"
   602→                    : "border-white/20"
   603→                }`}
   604→                style={
   605→                  result.rewardType !== "none" && !isNoWinLabel(result.label)
   606→                    ? {
   607→                        background: `linear-gradient(135deg, ${currentTheme.primaryColor}66, ${currentTheme.primaryColor}22)`,
   608→                        borderColor: `${currentTheme.primaryColor}66`,
   609→                        color: currentTheme.textColor,
   610→                      }
   611→                    : undefined
   612→                }
   613→              >
   614→                {result.rewardType !== "none" && !isNoWinLabel(result.label) ? (
   615→                  <div className="flex flex-col items-center gap-4">
   616→                    <motion.div
   617→                      animate={{
   618→                        rotate: [0, 15, -15, 0],
   619→                        scale: [1, 1.4, 1],
   620→                      }}
   621→                      transition={{ duration: 0.6, repeat: Infinity, repeatDelay: 2 }}
   622→                      className="text-yellow-400"
   623→                    >
   624→                      {getResultIcon(result)}
   625→                    </motion.div>
   626→                    <motion.p
   627→                      initial={{ y: 40, opacity: 0 }}
   628→                      animate={{ y: 0, opacity: 1 }}
   629→                      transition={{ delay: 0.2 }}
   630→                      className="font-black text-3xl"
   631→                    >
   632→                      🎉 {t("wheel.win")}! 🎉
   633→                    </motion.p>
   634→                    <motion.p
   635→                      initial={{ scale: 0 }}
   636→                      animate={{ scale: 1 }}
   637→                      transition={{ delay: 0.4, type: "spring", stiffness: 200 }}
   638→                      className="text-6xl font-black tracking-tighter"
   639→                    >
   640→                      {result.rewardValue || result.label}
   641→                    </motion.p>
   642→                    {result.description && (
   643→                      <p className="opacity-90 text-lg">{result.description}</p>
   644→                    )}
   645→                    {result.rewardImageUrl && (
   646→                      <img 
   647→                        src={result.rewardImageUrl} 
   648→                        alt="Prémio" 
   649→                        className="w-32 h-32 object-cover rounded-2xl shadow-lg"
   650→                      />
   651→                    )}
   652→                  </div>
   653→                ) : (
   654→                  <p className="font-bold text-3xl flex items-center gap-3 justify-center">
   655→                    😅 {t("tryAgain")}
   656→                  </p>
   657→                )}
   658→              </motion.div>
   659→            )}
   660→          </AnimatePresence>
   661→
   662→          {/* Minimum Segments Warning */}
   663→          {prizes.length < 4 && (
   664→            <div className="flex items-start gap-3 p-3 rounded-xl bg-amber-500/10 border border-amber-500/20 text-amber-200">
   665→              <AlertTriangle className="h-5 w-5 flex-shrink-0 text-amber-400" />
   666→              <p className="text-sm">{t("wheel.minSegmentsWarning")}</p>
   667→            </div>
   668→          )}
   669→
   670→          <div className="rounded-2xl bg-black/20 border border-white/10 backdrop-blur-xl text-white overflow-hidden">
   671→            <div className="bg-white/5 p-4 border-b border-white/10 flex items-center justify-between">
   672→              <span className="text-xs uppercase tracking-wider opacity-70 font-bold">{t("wheel.prizes")}</span>
   673→              {editable && onChange && (
   674→                <Sheet>
   675→                  <SheetTrigger asChild>
   676→                    <button className="inline-flex items-center gap-1 px-3 py-1 rounded-full bg-card border border-border text-[11px] font-medium hover:bg-secondary">
   677→                      <Settings2 className="h-3.5 w-3.5" /> {t("wheel.config")}
   678→                    </button>
   679→                  </SheetTrigger>
   680→                  <SheetContent side="bottom" className="rounded-t-3xl max-h-[85vh] overflow-y-auto">
   681→                    <SheetHeader className="mb-4"><SheetTitle>{t("wheel.configureTitle")}</SheetTitle></SheetHeader>
   682→                    <div className="space-y-4 pb-6">
   683→                      {prizes.map((p) => {
   684→                        const pct = ((Math.max(0, p.weight) / totalWeight) * 100).toFixed(1);
   685→                        return (
   686→                          <div key={p.id} className="rounded-2xl border border-border p-3 space-y-2">
   687→                            <div className="flex items-center gap-2">
   688→                              <input
   689→                                type="color"
   690→                                value={p.color}
   691→                                onChange={(e) => patchPrize(p.id, { color: e.target.value })}
   692→                                className="h-9 w-9 rounded-md border border-border bg-transparent cursor-pointer"
   693→                              />
   694→                              <Input
   695→                                value={p.label}
   696→                                onChange={(e) => patchPrize(p.id, { label: e.target.value })}
   697→                                placeholder={t("wheel.prizeName")}
   698→                                className="flex-1"
   699→                              />
   700→                              <button
   701→                                onClick={() => removePrize(p.id)}
   702→                                className="p-2 rounded-md hover:bg-destructive/10 text-destructive"

   704→                              >
   705→                                <Trash2 className="h-4 w-4" />
   706→                              </button>
   707→                            </div>
   708→                            <div className="grid grid-cols-2 gap-2">
   709→                              <div className="space-y-1">
   710→                                <Label className="text-[11px] text-muted-foreground">{t("wheel.prizeValue")}</Label>
   711→                                <Input
   712→                                  value={p.rewardValue || ""}
   713→                                  onChange={(e) => patchPrize(p.id, { rewardValue: e.target.value })}
   714→                                  placeholder="Ex: 100 MZN"
   715→                                  className="text-xs"
   716→                                />
   717→                              </div>
   718→                              <div className="space-y-1">
   719→                                <Label className="text-[11px] text-muted-foreground">{t("wheel.weight")} ({pct}%)</Label>
   720→                                <Input
   721→                                  type="number"
   722→                                  min={0}
   723→                                  max={1000}
   724→                                  value={p.weight}
   725→                                  onChange={(e) => patchPrize(p.id, { weight: Math.max(0, Number(e.target.value) || 0) })}
   726→                                  className="text-xs"
   727→                                />
   728→                              </div>
   729→                            </div>
   730→                            <div className="space-y-1">
   731→                              <Label className="text-[11px] text-muted-foreground">{t("wheel.effectType")}</Label>
   732→                              <select
   733→                                value={p.effectType || "confetti"}
   734→                                onChange={(e) => patchPrize(p.id, { effectType: e.target.value as any })}
   735→                                className="w-full px-3 py-2 rounded-xl bg-card border border-border text-xs"
   736→                              >
   737→                                <option value="confetti">Confetti</option>
   738→                                <option value="fireworks">Fireworks</option>
   739→                                <option value="stars">Stars</option>
   740→                                <option value="poppers">Poppers</option>
   741→                                <option value="zap">Zap</option>
   742→                              </select>
   743→                            </div>
   744→                          </div>
   745→                        );
   746→                      })}
   747→                      <button
   748→                        onClick={addPrize}
   749→                        className="w-full flex items-center justify-center gap-2 py-3 rounded-2xl border-2 border-dashed border-border text-sm text-muted-foreground hover:bg-secondary"
   750→                      >
   751→                        <Plus className="h-4 w-4" /> {t("wheel.addPrize")}
   752→                      </button>
   753→                      <p className="text-[11px] text-muted-foreground text-center">
   754→                        {t("wheel.weightHint")}
   755→                      </p>
   756→                    </div>
   757→                  </SheetContent>
   758→                </Sheet>
   759→              )}
   760→            </div>
   761→            <div className="p-0 max-h-[300px] overflow-y-auto custom-scrollbar">
   762→              {prizes.map((p, i) => (
   763→                <div key={i} className="flex items-center gap-4 p-4 hover:bg-white/5 transition-colors border-b border-white/5 last:border-0">
   764→                  <div className="w-3 h-3 rounded-full" style={{ backgroundColor: p.color }} />
   765→                  <div className="flex-1">
   766→                    <p className="font-bold text-sm">{p.label}</p>
   767→                    {p.rewardValue && <p className="text-xs opacity-50">{p.rewardValue}</p>}
   768→                  </div>
   769→                  {p.rewardImageUrl && (
   770→                    <img src={p.rewardImageUrl} alt="" className="w-8 h-8 rounded object-cover" />
   771→                  )}
   772→                </div>
   773→              ))}
   774→            </div>
   775→          </div>
   776→        </div>
   777→      </div>
   778→
   779→      {/* Floating Controls */}
   780→      <div className="fixed bottom-8 right-8 flex gap-3 z-50">
   781→        <button 
   782→          onClick={() => setSoundEnabled(!soundEnabled)}
   783→          className="p-3 rounded-full bg-white/10 border border-white/20 text-white hover:bg-white/20 transition-all"
   784→        >
   785→          {soundEnabled ? <Volume2 className="h-5 w-5" /> : <VolumeX className="h-5 w-5" />}
   786→        </button>
   787→        <button className="p-3 rounded-full bg-white/10 border border-white/20 text-white hover:bg-white/20 transition-all">
   788→          <Share2 className="h-5 w-5" />
   789→        </button>
   790→      </div>
   791→    </div>
   792→  );
   793→};
