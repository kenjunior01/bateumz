  1147→  1147→

  1150→  1150→
  1151→  1151→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1152→  1152→  items,
  1153→  1153→  statuses,
  1154→  1154→  sorts,
  1155→  1155→  renderItem,
  1156→  1156→  getStatus,
  1157→  1157→  getSortValue,
  1158→  1158→  formatScore,
  1159→  1159→  emptyMessage,
  1160→  1160→  extraAction,
  1161→  1161→  pageSize = 9,
  1162→  1162→}: {
  1163→  1163→  items: T[];
  1164→  1164→  statuses: StatusOption[];
  1165→  1165→  sorts: SortOption[];
  1166→  1166→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1167→  1167→  getStatus: (item: T) => string;
  1168→  1168→  getSortValue: (item: T, sort: string) => number;
  1169→  1169→  formatScore?: (item: T, sort: string) => string | undefined;
  1170→  1170→  emptyMessage: string;
  1171→  1171→  extraAction?: React.ReactNode;
  1172→  1172→  pageSize?: number;
  1173→  1173→}) {
  1174→  1174→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1175→  1175→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1176→  1176→  const [page, setPage] = useState(1);
  1177→  1177→  const [transitioning, setTransitioning] = useState(false);
  1178→  1178→
  1179→  1179→  // Reset page + show brief skeleton when filter/sort changes
  1180→  1180→  useEffect(() => {
  1181→  1181→    setPage(1);
  1182→  1182→    setTransitioning(true);
  1183→  1183→    const t = setTimeout(() => setTransitioning(false), 280);
  1184→  1184→    return () => clearTimeout(t);
  1185→  1185→  }, [status, sort]);
  1186→  1186→
  1187→  1187→  const filtered = useMemo(() => {
  1188→  1188→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1189→  1189→    if (sort === "recent") {
  1190→  1190→      list = [...list].sort((a, b) => {
  1191→  1191→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1192→  1192→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1193→  1193→        return tb - ta;
  1194→  1194→      });
  1195→  1195→    } else {
  1196→  1196→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1197→  1197→    }

  1199→  1199→  }, [items, status, sort, getStatus, getSortValue]);

  1209→  1209→      map[s] = (map[s] ?? 0) + 1;
  1210→  1210→    }
  1211→  1211→    return map;
  1212→  1212→  }, [items, getStatus]);
  1213→  1213→
  1214→  1214→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1215→  1215→

  1217→  1217→    return <EmptyState message={emptyMessage} />;
  1218→  1218→  }
  1219→  1219→
  1220→  1220→  return (
  1221→  1221→    <div>
  1222→  1222→      {/* Controls */}
  1223→  1223→      <div className="mb-4 flex flex-col gap-3">
  1224→  1224→        <div className="flex flex-wrap gap-1.5">
  1225→  1225→          {statuses.map((s) => {
  1226→  1226→            const active = status === s.value;
  1227→  1227→            const count = counts[s.value] ?? 0;
  1228→  1228→            return (
  1229→  1229→              <button
  1230→  1230→                key={s.value}
  1231→  1231→                onClick={() => setStatus(s.value)}
  1232→  1232→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1233→  1233→                  active
  1234→  1234→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1235→  1235→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1236→  1236→                }`}
  1237→  1237→              >
  1238→  1238→                {s.label}
  1239→  1239→                <span className="ml-1 opacity-70">({count})</span>
  1240→  1240→              </button>
  1241→  1241→            );
  1242→  1242→          })}
  1243→  1243→        </div>
  1244→  1244→
  1245→  1245→        <div className="flex items-center justify-between gap-2">
  1246→  1246→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1247→  1247→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1248→  1248→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1249→  1249→          </p>

  1259→  1259→                    {s.label}
  1260→  1260→                  </SelectItem>
  1261→  1261→                ))}
  1262→  1262→              </SelectContent>

  1264→  1264→          </div>
