  1146→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */
  1147→
  1148→interface StatusOption { value: string; label: string }
  1149→interface SortOption { value: string; label: string }
  1150→
  1151→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1152→  items,
  1153→  statuses,
  1154→  sorts,
  1155→  renderItem,
  1156→  getStatus,
  1157→  getSortValue,
  1158→  formatScore,
  1159→  emptyMessage,
  1160→  extraAction,
  1161→  pageSize = 9,
  1162→}: {
  1163→  items: T[];
  1164→  statuses: StatusOption[];
  1165→  sorts: SortOption[];
  1166→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1167→  getStatus: (item: T) => string;
  1168→  getSortValue: (item: T, sort: string) => number;
  1169→  formatScore?: (item: T, sort: string) => string | undefined;
  1170→  emptyMessage: string;
  1171→  extraAction?: React.ReactNode;
  1172→  pageSize?: number;
  1173→}) {
  1174→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1175→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1176→  const [page, setPage] = useState(1);
  1177→  const [transitioning, setTransitioning] = useState(false);
  1178→
  1179→  // Reset page + show brief skeleton when filter/sort changes
  1180→  useEffect(() => {
  1181→    setPage(1);
  1182→    setTransitioning(true);
  1183→    const t = setTimeout(() => setTransitioning(false), 280);
  1184→    return () => clearTimeout(t);
  1185→  }, [status, sort]);
  1186→
  1187→  const filtered = useMemo(() => {
  1188→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1189→    if (sort === "recent") {
  1190→      list = [...list].sort((a, b) => {
  1191→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1192→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1193→        return tb - ta;
  1194→      });
  1195→    } else {
  1196→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
