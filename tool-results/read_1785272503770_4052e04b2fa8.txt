  1295→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */
  1296→
  1297→interface StatusOption { value: string; label: string }
  1298→interface SortOption { value: string; label: string }
  1299→
  1300→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1301→  items,
  1302→  statuses,
  1303→  sorts,
  1304→  renderItem,
  1305→  getStatus,
  1306→  getSortValue,
  1307→  formatScore,
  1308→  emptyMessage,
  1309→  extraAction,
  1310→  pageSize = 9,
  1311→}: {
  1312→  items: T[];
  1313→  statuses: StatusOption[];
  1314→  sorts: SortOption[];
  1315→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1316→  getStatus: (item: T) => string;
  1317→  getSortValue: (item: T, sort: string) => number;
  1318→  formatScore?: (item: T, sort: string) => string | undefined;
  1319→  emptyMessage: string;
  1320→  extraAction?: React.ReactNode;
  1321→  pageSize?: number;
  1322→}) {
  1323→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1324→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1325→  const [page, setPage] = useState(1);
  1326→  const [transitioning, setTransitioning] = useState(false);
  1327→
  1328→  // Reset page + show brief skeleton when filter/sort changes
  1329→  useEffect(() => {
  1330→    setPage(1);
  1331→    setTransitioning(true);
  1332→    const t = setTimeout(() => setTransitioning(false), 280);
  1333→    return () => clearTimeout(t);
  1334→  }, [status, sort]);
  1335→
  1336→  const filtered = useMemo(() => {
  1337→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1338→    if (sort === "recent") {
  1339→      list = [...list].sort((a, b) => {
  1340→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1341→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1342→        return tb - ta;
  1343→      });
  1344→    } else {
  1345→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1346→    }
  1347→    return list;
  1348→  }, [items, status, sort, getStatus, getSortValue]);
  1349→
  1350→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  1351→  const visible = filtered.slice(0, page * pageSize);
  1352→  const canLoadMore = page < totalPages;
  1353→
  1354→  const counts = useMemo(() => {
  1355→    const map: Record<string, number> = { all: items.length };
  1356→    for (const i of items) {
  1357→      const s = getStatus(i);
  1358→      map[s] = (map[s] ?? 0) + 1;
  1359→    }
  1360→    return map;
  1361→  }, [items, getStatus]);
  1362→
  1363→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1364→
  1365→  if (items.length === 0) {
  1366→    return <EmptyState message={emptyMessage} />;
  1367→  }
  1368→
  1369→  return (
  1370→    <div>
  1371→      {/* Controls */}
  1372→      <div className="mb-4 flex flex-col gap-3">
  1373→        <div className="flex flex-wrap gap-1.5">
  1374→          {statuses.map((s) => {
  1375→            const active = status === s.value;
  1376→            const count = counts[s.value] ?? 0;
  1377→            return (
  1378→              <button
  1379→                key={s.value}
  1380→                onClick={() => setStatus(s.value)}
  1381→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1382→                  active
  1383→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1384→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1385→                }`}
  1386→              >
  1387→                {s.label}
  1388→                <span className="ml-1 opacity-70">({count})</span>
  1389→              </button>
  1390→            );
  1391→          })}
  1392→        </div>
  1393→
  1394→        <div className="flex items-center justify-between gap-2">
  1395→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1396→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1397→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1398→          </p>
  1399→          <div className="flex items-center gap-2">
  1400→            {extraAction}
  1401→            <Select value={sort} onValueChange={setSort}>
  1402→              <SelectTrigger className="h-8 text-xs w-[140px] sm:w-[170px]">
  1403→                <SelectValue />
  1404→              </SelectTrigger>
  1405→              <SelectContent>
  1406→                {sorts.map((s) => (
  1407→                  <SelectItem key={s.value} value={s.value} className="text-xs">
  1408→                    {s.label}
  1409→                  </SelectItem>
  1410→                ))}
  1411→              </SelectContent>
  1412→            </Select>
  1413→          </div>
  1414→        </div>
  1415→      </div>
  1416→
  1417→      {transitioning ? (
  1418→        <SkeletonCardGrid count={Math.min(pageSize, 6)} />
  1419→      ) : filtered.length === 0 ? (
  1420→        <EmptyState message="Nenhum resultado para os filtros selecionados." />
  1421→      ) : (
  1422→        <>
  1423→          <Grid>
  1424→            {visible.map((item, idx) =>
  1425→              renderItem(item, {
  1426→                rank: idx + 1,
  1427→                total: filtered.length,
  1428→                criterion: sortLabel,
  1429→                scoreLabel: formatScore?.(item, sort),
  1430→              }),
  1431→            )}
  1432→          </Grid>
  1433→
  1434→          {canLoadMore && (
  1435→            <div className="mt-6 flex justify-center">
  1436→              <Button variant="outline" size="sm" onClick={() => setPage((p) => p + 1)} className="gap-2">
  1437→                Carregar mais
  1438→                <span className="text-xs text-muted-foreground">
  1439→                  ({visible.length}/{filtered.length})
  1440→                </span>
  1441→              </Button>
  1442→            </div>
  1443→          )}
  1444→          {!canLoadMore && filtered.length > pageSize && (
  1445→            <p className="mt-6 text-center text-[11px] text-muted-foreground">
  1446→              Mostrando todos os {filtered.length} itens
  1447→            </p>
  1448→          )}
  1449→        </>
  1450→      )}
  1451→    </div>
  1452→  );
  1453→}
