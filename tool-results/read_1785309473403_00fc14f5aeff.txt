    23→// ─── Particle burst on winner ─────────────────────────────────────────
    24→function ConfettiBurst({ active }: { active: boolean }) {
    25→  if (!active) return null;
    26→  const particles = Array.from({ length: 60 }, (_, i) => {
    27→    const angle = (i / 60) * Math.PI * 2;
    28→    const distance = 120 + Math.random() * 200;
    29→    const size = 4 + Math.random() * 8;
    30→    const colors = [
    31→      "hsl(var(--primary))",
    32→      "hsl(var(--accent))",
    33→      "hsl(45, 100%, 60%)",
    34→      "hsl(280, 80%, 60%)",
    35→      "hsl(0, 90%, 60%)",
    36→      "hsl(180, 80%, 50%)",
    37→    ];
    38→    return { angle, distance, size, color: colors[i % colors.length], delay: Math.random() * 0.4 };
    39→  });
    40→
    41→  return (
    42→    <div className="absolute inset-0 pointer-events-none overflow-hidden">
    43→      {particles.map((p, i) => (
    44→        <motion.div
    45→          key={i}
    46→          className="absolute rounded-full"
    47→          style={{
    48→            width: p.size,
    49→            height: p.size,
    50→            backgroundColor: p.color,
    51→            left: "50%",
    52→            top: "50%",
    53→          }}
    54→          initial={{ x: 0, y: 0, opacity: 1, scale: 0 }}
    55→          animate={{
    56→            x: Math.cos(p.angle) * p.distance,
    57→            y: Math.sin(p.angle) * p.distance - 50,
    58→            opacity: [0, 1, 1, 0],
    59→            scale: [0, 1.5, 1, 0.5],
    60→            rotate: Math.random() * 720,
    61→          }}
    62→          transition={{
    63→            duration: 2 + Math.random(),
    64→            delay: p.delay,
    65→            ease: "easeOut",
    66→          }}
    67→        />
    68→      ))}
    69→    </div>
    70→  );
    71→}

   353→// ─── Main component ───────────────────────────────────────────────────
   354→const LiveDraw = () => {
   355→  const { slug: routeSlug } = useParams<{ slug: string }>();
   356→  const navigate = useNavigate();
   357→  const [raffle, setRaffle] = useState<any>(null);
   358→  const [participants, setParticipants] = useState<Participant[]>([]);
   359→  const [loading, setLoading] = useState(true);
   360→  const [phase, setPhase] = useState<"idle" | "countdown" | "drawing" | "winner">("idle");
   361→  const [currentNumber, setCurrentNumber] = useState<number | null>(null);
   362→  const [winner, setWinner] = useState<Participant | null>(null);
   363→  const [globeNumbers, setGlobeNumbers] = useState<number[]>([]);
   364→  const [heartbeat, setHeartbeat] = useState(false);
   365→
   366→  useEffect(() => {
   367→    if (!routeSlug) return;
   368→    const fetchData = async () => {
   369→      const isUUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(routeSlug);
   370→      const raffleQuery = isUUID
   371→        ? supabase.from("raffles").select("*").eq("id", routeSlug).single()
   372→        : supabase.from("raffles").select("*").eq("slug", routeSlug).single();
   373→
   374→      const r = await raffleQuery;
   375→      if (r.data) {
   376→        setRaffle(r.data);
   377→        const raffleId = r.data.id;
   378→        const [p, w] = await Promise.all([
   379→          supabase.from("participants").select("*").eq("raffle_id", raffleId).eq("status", "active"),
   380→          supabase.from("participants").select("*").eq("raffle_id", raffleId).eq("status", "winner").maybeSingle(),
   381→        ]);
   382→        if (p.data) {
   383→          setParticipants(p.data);
   384→          setGlobeNumbers(p.data.map((x) => x.ticket_number));
   385→        }
   386→        if (w.data) {
   387→          setWinner(w.data);
   388→          setPhase("winner");
   389→        }
   390→      }
   391→      setLoading(false);
   392→    };
   393→    fetchData();
   394→  }, [routeSlug]);
   395→
   396→  const startDraw = useCallback(() => {
   397→    if (participants.length === 0 || !raffle) return;
   398→    setPhase("countdown");
   399→  }, [participants, raffle]);
   400→
   401→  const onCountdownComplete = useCallback(async () => {
   402→    setPhase("drawing");
   403→    setHeartbeat(true);
   404→    playDrumRoll();
   405→
   406→    // Dramatic number cycling — accelerate then slow down
   407→    const totalCycles = 40;
   408→    for (let i = 0; i < totalCycles; i++) {
   409→      const randomIdx = Math.floor(Math.random() * participants.length);
   410→      setCurrentNumber(participants[randomIdx].ticket_number);
   411→      if (i % 3 === 0) playTickSound();
   412→      // Speed up first, then dramatically slow down
   413→      const speed = i < 20 ? 50 + i * 2 : 50 + i * 8;
   414→      await new Promise((r) => setTimeout(r, speed));
   415→    }
   416→
   417→    // Final dramatic pause
   418→    await new Promise(r => setTimeout(r, 800));
   419→
   420→    // Pick winner
   421→    const winnerIdx = Math.floor(Math.random() * participants.length);
   422→    const selectedWinner = participants[winnerIdx];
   423→    setCurrentNumber(selectedWinner.ticket_number);
   424→
   425→    // Save to DB
   426→    await supabase.from("participants").update({ status: "winner" }).eq("id", selectedWinner.id);
   427→    await supabase.from("raffles").update({ status: "completed" }).eq("id", raffle.id);
   428→    await supabase.from("luck_points").insert({
   429→      user_id: selectedWinner.user_id,
   430→      points: 500,
   431→      action: "bonus",
   432→      description: `Vencedor do sorteio ao vivo: ${raffle.title}`,
   433→      raffle_id: raffle.id,
   434→    });
   435→
   436→    // Dramatic pause before reveal
   437→    await new Promise(r => setTimeout(r, 1500));
   438→    playWinSound();
   439→    setWinner(selectedWinner);
   440→    setPhase("winner");
   441→    setHeartbeat(false);
   442→  }, [participants, raffle]);
   443→
   444→  if (loading) {
   445→    return (
   446→      <div className="min-h-screen bg-background flex items-center justify-center">
   447→        <div className="h-8 w-8 animate-spin rounded-full border-2 border-primary border-t-transparent" />
   448→      </div>
   449→    );
   450→  }
   451→
   452→  if (!raffle) {
   453→    return (
   454→      <div className="min-h-screen bg-background flex items-center justify-center">
   455→        <p className="text-muted-foreground">Sorteio não encontrado</p>
   456→      </div>
   457→    );
   458→  }
   459→
   460→  return (

   462→      {/* Ambient background effects */}

   465→        <div className="absolute bottom-1/4 right-1/4 w-[400px] h-[400px] rounded-full bg-accent/5 blur-[120px]" />
   466→        {phase === "drawing" && (
   467→          <motion.div
   468→            className="absolute inset-0"
   469→            animate={{ opacity: [0.05, 0.15, 0.05] }}
   470→            transition={{ duration: 1, repeat: Infinity }}
   471→            style={{ background: "radial-gradient(circle at 50% 50%, hsl(var(--primary) / 0.1), transparent 60%)" }}
   472→          />
   473→        )}
   474→      </div>
   475→
   476→      <Navbar />
   477→      <div className="container mx-auto px-4 pt-28 pb-20 relative z-10">
   478→        <button onClick={() => navigate(-1)} className="flex items-center gap-2 text-muted-foreground hover:text-foreground transition-colors mb-8">
   479→          <ArrowLeft className="h-4 w-4" /> Voltar
   480→        </button>
   481→
   482→        <div className="max-w-3xl mx-auto text-center">

   515→              <motion.div
   516→                animate={phase === "drawing" ? { rotate: -360 } : {}}
   517→                transition={phase === "drawing" ? { repeat: Infinity, duration: 3, ease: "linear" } : {}}
   518→                className="absolute inset-3 rounded-full border-2 border-dashed border-accent/20"
   519→              />
   520→              <motion.div
   521→                animate={phase === "drawing" ? { rotate: 360 } : {}}
   522→                transition={phase === "drawing" ? { repeat: Infinity, duration: 5, ease: "linear" } : {}}
   523→                className="absolute inset-6 rounded-full border border-primary/10"
   524→              />
   525→
   526→              {/* Globe core */}
   527→              <div className="absolute inset-8 rounded-full bg-gradient-to-br from-secondary via-card to-secondary border border-border overflow-hidden">
   528→                <div className="absolute inset-0 flex items-center justify-center">
   529→                  {phase === "drawing" || currentNumber ? (
   530→                    <motion.div
   531→                      key={currentNumber}
   532→                      initial={{ scale: 0.5, opacity: 0, rotateY: -90 }}
   533→                      animate={{ scale: 1, opacity: 1, rotateY: 0 }}
   534→                      transition={{ duration: 0.1 }}
   535→                      className="text-center"
   536→                    >
   537→                      <p className="font-display text-7xl font-black text-primary"
   538→                        style={{ textShadow: "0 0 40px hsl(var(--primary) / 0.4)" }}>
   539→                        {currentNumber}
   540→                      </p>
   541→                      <p className="text-xs text-muted-foreground mt-2 font-medium">
   542→                        {phase === "drawing" ? "Sorteando..." : winner ? "VENCEDOR!" : ""}
   543→                      </p>
   544→                    </motion.div>
   545→                  ) : (
   546→                    <div className="text-center">
   547→                      <motion.div
   548→                        animate={{ rotate: [0, 5, -5, 0] }}
   549→                        transition={{ duration: 3, repeat: Infinity }}
   550→                      >
   551→                        <Trophy className="h-14 w-14 text-muted-foreground/30 mx-auto mb-2" />
   552→                      </motion.div>
   553→                      <p className="text-sm text-muted-foreground">Pronto para sortear</p>
   554→                    </div>
   555→                  )}
   556→                </div>
   557→
   558→                {/* Orbiting numbers */}
   559→                {phase === "drawing" &&
   560→                  globeNumbers.slice(0, 16).map((num, i) => (
   561→                    <motion.div
   562→                      key={`orbit-${num}`}
   563→                      animate={{
   564→                        x: [0, Math.cos((i / 16) * Math.PI * 2) * 90, 0],
   565→                        y: [0, Math.sin((i / 16) * Math.PI * 2) * 90, 0],
   566→                        opacity: [0.2, 0.6, 0.2],
   567→                        scale: [0.8, 1.2, 0.8],
   568→                      }}
   569→                      transition={{ repeat: Infinity, duration: 1.5 + i * 0.15, ease: "linear" }}
   570→                      className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 text-xs font-mono font-bold text-primary/30"
   571→                    >
   572→                      {num}
   573→                    </motion.div>
   574→                  ))}
   575→
   576→                {/* Heartbeat glow */}
   577→                {heartbeat && (
   578→                  <motion.div
   579→                    className="absolute inset-0 rounded-full"
   580→                    animate={{
   581→                      boxShadow: [
   582→                        "inset 0 0 20px hsl(var(--primary) / 0.1)",
   583→                        "inset 0 0 60px hsl(var(--primary) / 0.3)",
   584→                        "inset 0 0 20px hsl(var(--primary) / 0.1)",
   585→                      ],
   586→                    }}
   587→                    transition={{ duration: 0.8, repeat: Infinity }}
   588→                  />
   589→                )}
   590→              </div>
   591→
   592→              {/* Glow effect */}
   593→              {phase === "drawing" && (
   594→                <motion.div
   595→                  animate={{
   596→                    opacity: [0.2, 0.5, 0.2],
   597→                    scale: [1, 1.05, 1],
   598→                  }}
   599→                  transition={{ repeat: Infinity, duration: 1.2 }}
   600→                  className="absolute inset-4 rounded-full"
   601→                  style={{ boxShadow: "0 0 80px 30px hsl(var(--primary) / 0.15)" }}
   602→                />
   603→              )}
   604→            </div>
   605→          </motion.div>
   606→
   607→          {/* Number slot display during draw */}
   608→          <AnimatePresence>
   609→            {(phase === "drawing" || (phase === "winner" && !winner)) && currentNumber && (
   610→              <motion.div
   611→                initial={{ opacity: 0, y: 20 }}
   612→                animate={{ opacity: 1, y: 0 }}
   613→                exit={{ opacity: 0 }}
   614→                className="mb-8"
   615→              >
   616→                <NumberSlot number={currentNumber} isRevealing={phase === "drawing"} />
   617→              </motion.div>
   618→            )}
   619→          </AnimatePresence>
   620→
   621→          {/* Stats */}
   622→          <div className="grid grid-cols-3 gap-4 mb-8 max-w-md mx-auto">
   623→            <Card className="glass">
   624→              <CardContent className="p-4 text-center">
   625→                <Users className="h-5 w-5 text-primary mx-auto mb-1" />
   626→                <p className="font-display text-xl font-bold text-foreground">{participants.length}</p>
   627→                <p className="text-xs text-muted-foreground">Participantes</p>
   628→              </CardContent>
   629→            </Card>
   630→            <Card className="glass">
   631→              <CardContent className="p-4 text-center">
   632→                <Ticket className="h-5 w-5 text-accent mx-auto mb-1" />
   633→                <p className="font-display text-xl font-bold text-foreground">{raffle.sold_tickets}</p>
   634→                <p className="text-xs text-muted-foreground">Bilhetes</p>
   635→              </CardContent>
   636→            </Card>
   637→            <Card className="glass">
   638→              <CardContent className="p-4 text-center">
   639→                <Trophy className="h-5 w-5 text-primary mx-auto mb-1" />
   640→                <p className="font-display text-xl font-bold text-foreground">{formatMZN(raffle.prize_value)}</p>
   641→                <p className="text-xs text-muted-foreground">Prémio</p>
   642→              </CardContent>
   643→            </Card>
   644→          </div>
   645→
   646→          {/* Draw button */}
   647→          {phase === "idle" && !winner && (
   648→            <motion.div
   649→              initial={{ opacity: 0, y: 20 }}
   650→              animate={{ opacity: 1, y: 0 }}
   651→              transition={{ delay: 0.3 }}
   652→            >
   653→              <motion.div
   654→                animate={{ scale: [1, 1.03, 1] }}
   655→                transition={{ duration: 2, repeat: Infinity }}
   656→              >
   657→                <Button
   658→                  onClick={startDraw}
   659→                  disabled={participants.length === 0}
   660→                  size="lg"
   661→                  className="gap-3 h-16 px-12 text-xl font-black glow-primary rounded-2xl"
   662→                >
   663→                  <Sparkles className="h-6 w-6" />
   664→                  INICIAR SORTEIO
   665→                </Button>
   666→              </motion.div>
   667→              {participants.length === 0 && (
   668→                <p className="text-sm text-muted-foreground mt-4">Nenhum participante ativo neste sorteio.</p>
   669→              )}
   670→            </motion.div>
   671→          )}
   672→
   673→          {/* Drawing state text */}
   674→          {phase === "drawing" && (
   675→            <motion.p
   676→              className="text-lg font-bold text-primary"
   677→              animate={{ opacity: [0.5, 1, 0.5] }}
   678→              transition={{ duration: 1, repeat: Infinity }}
   679→            >
   680→              ⚡ Escolhendo o vencedor...
   681→            </motion.p>
   682→          )}
   683→        </div>
   684→      </div>
   685→
   686→      {/* Countdown overlay */}
   687→      <AnimatePresence>
   688→        {phase === "countdown" && (
   689→          <PreDrawCountdown seconds={3} onComplete={onCountdownComplete} />
   690→        )}
   691→      </AnimatePresence>
   692→
   693→      {/* Winner reveal overlay */}
   694→      <AnimatePresence>
   695→        {phase === "winner" && winner && (
   696→          <WinnerReveal
   697→            winner={winner}
   698→            raffle={raffle}
   699→            onClose={() => setPhase("idle")}
   700→          />
   701→        )}
   702→      </AnimatePresence>
   703→
   704→      <Footer />
   705→    </div>
   706→  );
   707→};
   708→
   709→export default LiveDraw;
