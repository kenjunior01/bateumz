    59→const getBranding = async (code: string): Promise<OverlayBranding> => {
    60→  try {
    61→    const { data: session } = await supabase
    62→      .from('scheduled_lives')
    63→      .select('business_user_id')
    64→      .eq('live_code', code)
    65→      .single();

    67→      const { data: brand } = await supabase
    68→        .from('company_branding')
    69→        .select('*')
    70→        .eq('user_id', session.business_user_id)
    71→        .single();

    74→          companyName: brand.company_name || undefined,
    75→          companySlogan: brand.company_slogan || undefined,
    76→          companyLogoUrl: brand.company_logo_url || undefined,

    93→const AnimatedNumber = ({ value, color, size = 'text-2xl' }: { value: number; color?: string; size?: string }) => {
    94→  const display = useRef(value);
    95→  const [animating, setAnimating] = useState(false);
    96→  const [displayVal, setDisplayVal] = useState(value);
    97→
    98→  useEffect(() => {
    99→    if (value !== display.current) {
   100→      const diff = value - display.current;
   101→      display.current = value;
   102→      setAnimating(true);
   103→      let current = displayVal;
   104→      const steps = 12;
   105→      let step = 0;
   106→      const inc = diff / steps;
   107→      const iv = setInterval(() => {
   108→        step++;
   109→        current += inc;
   110→        setDisplayVal(Math.round(current));
   111→        if (step >= steps) {
   112→          clearInterval(iv);
   113→          setDisplayVal(value);
   114→          setTimeout(() => setAnimating(false), 200);
   115→        }
   116→      }, 30);
   117→      return () => clearInterval(iv);
   118→    }
   119→  }, [value]);
   120→
   121→  return (
   122→    <motion.span
   123→      className={`${size} font-black tabular-nums leading-none inline-block`}
   124→      style={{ color }}
   125→      animate={animating ? { scale: [1, 1.15, 1], y: [0, -3, 0] } : { scale: 1, y: 0 }}
   126→      transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   127→    >
   128→      {displayVal.toLocaleString()}
   129→    </motion.span>
   130→  );
   131→};

   158→  const particles = useMemo(() =>
   159→    Array.from({ length: 8 }, (_, i) => ({

   162→      distance: 30 + Math.random() * 40,

   164→      duration: 0.4 + Math.random() * 0.3,
   165→    })), []);

   167→  return (
   168→    <div className="absolute pointer-events-none" style={{ left: x, top: y }}>
   169→      {particles.map(p => (
   170→        <motion.div
   171→          key={p.id}
   172→          className="absolute rounded-full"
   173→          style={{
   174→            width: p.size,
   175→            height: p.size,
   176→            backgroundColor: color,
   177→            left: -p.size / 2,
   178→            top: -p.size / 2,
   179→          }}
   180→          initial={{ x: 0, y: 0, opacity: 1, scale: 1 }}
   181→          animate={{
   182→            x: Math.cos(p.angle * Math.PI / 180) * p.distance,
   183→            y: Math.sin(p.angle * Math.PI / 180) * p.distance,
   184→            opacity: 0,
   185→            scale: 0,
   186→          }}
   187→          transition={{ duration: p.duration, ease: 'easeOut' }}
   188→        />
   189→      ))}
