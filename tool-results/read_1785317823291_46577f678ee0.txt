    64→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    65→  const [count, setCount] = useState(0);
    66→  const rafRef = useRef<number>(0);
    67→
    68→  useEffect(() => {
    69→    if (!enabled || target === 0) {
    70→      setCount(target);
    71→      return;
    72→    }
    73→    const start = performance.now();
    74→    const step = (now: number) => {
    75→      const elapsed = now - start;
    76→      const progress = Math.min(elapsed / duration, 1);
    77→      const eased = 1 - Math.pow(1 - progress, 3);
    78→      setCount(Math.round(eased * target));
    79→      if (progress < 1) {
    80→        rafRef.current = requestAnimationFrame(step);
    81→      }
    82→    };
    83→    rafRef.current = requestAnimationFrame(step);
    84→    return () => cancelAnimationFrame(rafRef.current);
    85→  }, [target, duration, enabled]);
    86→
    87→  return count;
    88→}

    90→function formatDuration(seconds: number): string {
    91→  if (seconds < 60) return `${seconds}s`;
    92→  const m = Math.floor(seconds / 60);
    93→  const s = seconds % 60;
    94→  return s > 0 ? `${m}m ${s}s` : `${m}m`;
    95→}

    97→function getStartDate(range: DateRange): Date | null {
    98→  if (range === "all") return null;
    99→  const now = new Date();
   100→  const days = range === "7d" ? 7 : range === "30d" ? 30 : 90;
   101→  return new Date(now.getTime() - days * 24 * 60 * 60 * 1000);
   102→}

   104→function filterByDate<T extends { created_at: string }>(items: T[], range: DateRange): T[] {
   105→  const start = getStartDate(range);
   106→  if (!start) return items;
   107→  return items.filter((item) => new Date(item.created_at) >= start);
   108→}
