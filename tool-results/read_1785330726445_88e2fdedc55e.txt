    82→// Per-game error boundary so one crashing game does not kill the whole page
    83→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean}> {
    84→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false}; }
    85→  static getDerivedStateFromError() { return {hasError: true}; }
    86→  componentDidCatch(err: Error) { console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err); }
    87→  render() {
    88→    if (this.state.hasError) return (
    89→      <div className="flex flex-col items-center justify-center py-16 px-4 rounded-2xl border border-dashed border-destructive/40 bg-destructive/5">
    90→        <div className="text-4xl mb-3">🎮</div>
    91→        <p className="text-sm font-bold mb-1">Erro ao carregar {this.props.gameName}</p>
    92→        <p className="text-xs text-muted-foreground mb-3">Tente selecionar outro jogo</p>
    93→        <button onClick={() => this.setState({hasError: false})} className="text-xs px-3 py-1.5 rounded-full bg-primary text-primary-foreground hover:bg-primary/90">Tentar novamente</button>
    94→      </div>
    95→    );
    96→    return this.props.children;
    97→  }
