   230→const computeQuality = (
   231→  players: number,
   232→  plays: number,
   233→  games: number,
   234→  heat: number,

   236→  if (plays === 0) return 0;

   238→  const volume = Math.min(25, plays * 1.5);

   240→  const momentum = (heat / 100) * 15;

   242→  return Math.round(breadth + volume + diversity + momentum + retention);

   343→ * ★ REVOLUTION: Hype Meter — detects surges in plays over the last 20s vs the
   344→ * previous 20s window. Returns 0-100 representing "is the chat exploding right now".
   345→ */
   346→const computeHype = (entries: LeaderEntry[], now: number): number => {
   347→  if (!entries.length) return 0;
   348→  const recent = entries.filter((e) => e.at >= now - 20_000).length;
   349→  const prior = entries.filter((e) => e.at >= now - 40_000 && e.at < now - 20_000).length;
   350→  if (prior === 0) return recent > 0 ? Math.min(100, recent * 25) : 0;
   351→  const ratio = recent / Math.max(1, prior);
   352→  return Math.max(0, Math.min(100, Math.round((ratio - 1) * 80 + 20)));

   386→/**
   387→ * ★ REVOLUTION: Trajectory forecast.
   388→ * Predicts projected total plays at the current pace, plus ETA in seconds to reach the goal.
   389→ */
   390→const computeTrajectory = (
   391→  plays: number,
   392→  elapsedSec: number,
   393→  goal: number,
   394→): { projected: number; etaSec: number | null; pacePerMin: number } => {
   395→  if (elapsedSec < 5 || plays === 0) return { projected: plays, etaSec: null, pacePerMin: 0 };
   396→  const pacePerMin = (plays / elapsedSec) * 60;
   397→  const projected = Math.round(plays + (pacePerMin / 60) * Math.max(0, (1800 - elapsedSec))); // project to 30min mark
   398→  if (plays >= goal || pacePerMin <= 0) return { projected, etaSec: null, pacePerMin };
   399→  const remaining = goal - plays;
   400→  const etaSec = Math.round((remaining / pacePerMin) * 60);
   401→  return { projected, etaSec, pacePerMin };
   402→};

   888→const CommandPalette = ({
   889→  open,
   890→  onClose,
   891→  items,
   892→}: {
   893→  open: boolean;
   894→  onClose: () => void;
   895→  items: CommandItem[];
   896→}) => {
   897→  const [query, setQuery] = useState("");
   898→  const [active, setActive] = useState(0);
   899→  const inputRef = useRef<HTMLInputElement>(null);
   900→
   901→  useEffect(() => {
   902→    if (open) {
   903→      setQuery("");
   904→      setActive(0);
   905→      setTimeout(() => inputRef.current?.focus(), 50);
   906→    }
   907→  }, [open]);
   908→
   909→  const filtered = useMemo(() => {
   910→    if (!query.trim()) return items;
   911→    const q = query.toLowerCase();
   912→    return items.filter(
   913→      (it) =>
   914→        it.label.toLowerCase().includes(q) ||
   915→        it.group.toLowerCase().includes(q) ||
   916→        it.hint?.toLowerCase().includes(q),
   917→    );
   918→  }, [items, query]);
   919→
   920→  useEffect(() => {
   921→    setActive(0);
   922→  }, [query]);
   923→
   924→  const groups = useMemo(() => {
   925→    const map = new Map<string, CommandItem[]>();
   926→    filtered.forEach((it) => {
   927→      if (!map.has(it.group)) map.set(it.group, []);
   928→      map.get(it.group)!.push(it);
   929→    });
   930→    return [...map.entries()];
   931→  }, [filtered]);
   932→
   933→  const onKeyDown = (e: React.KeyboardEvent) => {
   934→    if (e.key === "ArrowDown") {
   935→      e.preventDefault();
   936→      setActive((a) => Math.min(a + 1, filtered.length - 1));
   937→    } else if (e.key === "ArrowUp") {
   938→      e.preventDefault();
   939→      setActive((a) => Math.max(a - 1, 0));
   940→    } else if (e.key === "Enter") {
   941→      e.preventDefault();
   942→      const it = filtered[active];
   943→      if (it && !it.disabled) {
   944→        it.run();
   945→        onClose();
   946→      }
   947→    } else if (e.key === "Escape") {
   948→      e.preventDefault();
   949→      onClose();
   950→    }
   951→  };
   952→
   953→  return (
   954→    <AnimatePresence>
   955→      {open && (
   956→        <motion.div
   957→          initial={{ opacity: 0 }}
   958→          animate={{ opacity: 1 }}
   959→          exit={{ opacity: 0 }}
   960→          transition={{ duration: 0.15 }}
   961→          className="fixed inset-0 z-[100] flex items-start justify-center pt-[15vh] px-4 bg-background/70 backdrop-blur-sm"
   962→          onClick={onClose}
   963→        >
   964→          <motion.div
   965→            initial={{ opacity: 0, y: -16, scale: 0.96 }}
   966→            animate={{ opacity: 1, y: 0, scale: 1 }}
   967→            exit={{ opacity: 0, y: -16, scale: 0.96 }}
   968→            transition={{ type: "spring", stiffness: 400, damping: 28 }}
   969→            onClick={(e) => e.stopPropagation()}
   970→            className="w-full max-w-xl rounded-2xl border border-border bg-card shadow-elegant overflow-hidden"
   971→          >
   972→            <div className="flex items-center gap-2 px-3 py-2.5 border-b border-border/60">
   973→              <Search className="h-4 w-4 text-muted-foreground shrink-0" />
   974→              <input
   975→                ref={inputRef}
   976→                value={query}
   977→                onChange={(e) => setQuery(e.target.value)}
   978→                onKeyDown={onKeyDown}
   979→                placeholder="Pesquisar ação... (ex: coroar, exportar, copiar)"
   980→                className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   981→              />
   982→              <kbd className="text-[9px] font-mono px-1.5 py-0.5 rounded border border-border bg-muted/50 text-muted-foreground">
   983→                ESC
   984→              </kbd>
