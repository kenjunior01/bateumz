    54→function pickPattern(total: number, count: number): Set<number> {
    55→  const set = new Set<number>();
    56→  while (set.size < count) set.add(Math.floor(Math.random() * total));
    57→  return set;
    58→}
    59→
    60→function calcRoundScore(taps: Set<number>, pattern: Set<number>): number {
    61→  let correct = 0;
    62→  let wrong = 0;
    63→  taps.forEach((c) => (pattern.has(c) ? correct++ : wrong++));
    64→  return Math.max(0, correct * 10 - wrong * 5);
    65→}

   150→function Cell({
   151→  index,
   152→  inPattern,
   153→  patternStaggerIndex,
   154→  isTapped,
   155→  isWrongTap,
   156→  isCorrectTap,
   157→  isMissed,
   158→  showPattern,
   159→  interactive,
   160→  tapColor,
   161→  phase,
   162→  onTap,
   163→}: {
   164→  index: number;
   165→  inPattern: boolean;
   166→  patternStaggerIndex: number;
   167→  isTapped: boolean;
   168→  isWrongTap: boolean;
   169→  isCorrectTap: boolean;
   170→  isMissed: boolean;
   171→  showPattern: boolean;
   172→  interactive: boolean;
   173→  tapColor: "cyan" | "pink" | null;
   174→  phase: Phase;
   175→  onTap: (i: number) => void;
   176→}) {
   177→  /* bg class */
   178→  let bg = "bg-zinc-800/70 border-zinc-700/40";
   179→  let glow = "";
   180→  let shouldShake = false;
   181→  let shouldPulse = false;
   182→  let animDelay = 0;
   183→
   184→  if (phase === "memorize") {
   185→    if (inPattern) {
   186→      bg = "bg-violet-500 border-violet-400/60";
   187→      glow = "shadow-[0_0_22px_4px_rgba(139,92,246,0.55)]";
   188→      shouldPulse = true;
   189→      animDelay = patternStaggerIndex * 0.07;
   190→    } else {
   191→      bg = "bg-zinc-800/50 border-zinc-700/30";
   192→    }
   193→  } else if (phase === "recall_p1" || phase === "recall_p2") {
   194→    if (isTapped && tapColor === "cyan") {
   195→      bg = "bg-cyan-500 border-cyan-400/60";
   196→      glow = "shadow-[0_0_16px_3px_rgba(6,182,212,0.45)]";
   197→    } else if (isTapped && tapColor === "pink") {
   198→      bg = "bg-pink-500 border-pink-400/60";
   199→      glow = "shadow-[0_0_16px_3px_rgba(236,72,153,0.45)]";
   200→    } else {
   201→      bg = "bg-zinc-800/70 border-zinc-700/40";
   202→    }
   203→  } else if (phase === "feedback_p1" || phase === "feedback_p2") {
   204→    if (isCorrectTap) {
   205→      bg = "bg-emerald-500 border-emerald-400/60";
   206→      glow = "shadow-[0_0_16px_3px_rgba(16,185,129,0.5)]";
   207→    } else if (isWrongTap) {
   208→      bg = "bg-red-500 border-red-400/60";
   209→      shouldShake = true;
   210→    } else if (isMissed) {
   211→      bg = "bg-violet-700/35 border-violet-600/30";
   212→    } else {
   213→      bg = "bg-zinc-800/40 border-zinc-700/20";
   214→    }
   215→  } else if (phase === "round_result" || phase === "game_over") {
   216→    if (showPattern && inPattern) {
   217→      bg = "bg-violet-600/40 border-violet-500/30";
   218→    } else {
   219→      bg = "bg-zinc-800/30 border-zinc-700/20";
   220→    }
   221→  } else {
   222→    bg = "bg-zinc-800/70 border-zinc-700/40";
   223→  }
   224→
   225→  return (
   226→    <motion.button
   227→      type="button"
   228→      onClick={() => interactive && onTap(index)}
   229→      className={cn(
   230→        "w-14 h-14 sm:w-18 sm:h-18 rounded-xl border-2 transition-[background-color,border-color,box-shadow] duration-150",
   231→        bg,
   232→        glow,
   233→        interactive && "cursor-pointer hover:brightness-125",
   234→        !interactive && "cursor-default",
   235→        shouldPulse && phase === "memorize" && "animate-pulse"
   236→      )}
   237→      animate={{
   238→        ...(shouldShake ? shakeKeyframes : { x: 0 }),
   239→        scale: phase === "memorize" && inPattern
   240→          ? [0.6, 1.08, 1]
   241→          : shouldShake
   242→            ? [1, 0.95, 1.02, 0.98, 1]
   243→            : 1,
   244→      }}
   245→      transition={{
   246→        x: shouldShake ? { duration: 0.5, ease: "easeOut" } : { duration: 0.1 },
   247→        scale: { duration: 0.35, ease: "easeOut" },
   248→        delay: animDelay,
   249→      }}
   250→      whileTap={interactive ? { scale: 0.82 } : undefined}
   251→      whileHover={interactive ? { scale: 1.06 } : undefined}
   252→    />
   253→  );
   254→}

   259→
   260→export default function PatternMemory({ onScore, liveCode }: Props) {
   261→  /* ── state ── */
   262→  const [phase, setPhase] = useState<Phase>("idle");
   263→  const [round, setRound] = useState(0);
   264→  const [pattern, setPattern] = useState<Set<number>>(new Set());
   265→  const [p1Taps, setP1Taps] = useState<Set<number>>(new Set());
   266→  const [p2Taps, setP2Taps] = useState<Set<number>>(new Set());
   267→  const [p1Total, setP1Total] = useState(0);
   268→  const [p2Total, setP2Total] = useState(0);
   269→  const [p1RoundPts, setP1RoundPts] = useState(0);
   270→  const [p2RoundPts, setP2RoundPts] = useState(0);
   271→  const [memProgress, setMemProgress] = useState(100);
   272→
   273→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   274→  const onScoreFired = useRef(false);
   275→  const scoresRef = useRef({ p1: 0, p2: 0 });
   276→
   277→  const cfg = ROUNDS[round] ?? ROUNDS[0];
   278→  const totalCells = cfg.gridSize * cfg.gridSize;
   279→  const patternArr = Array.from(pattern).sort((a, b) => a - b);
   280→
   281→  const recalling = phase === "recall_p1" || phase === "recall_p2";
   282→  const isP1 = phase === "recall_p1";
   283→
   284→  /* ── timer cleanup ── */
   285→  const clearTimer = useCallback(() => {
   286→    if (timerRef.current) {
   287→      clearInterval(timerRef.current);
   288→      timerRef.current = null;
   289→    }
   290→  }, []);
   291→
   292→  useEffect(() => clearTimer, [clearTimer]);
   293→
   294→  /* ── memorize timer ── */
   295→  useEffect(() => {
   296→    if (phase !== "memorize") return;
   297→    clearTimer();
   298→    const start = Date.now();
   299→    const durationMs = cfg.memorizeTime * 1000;
   300→    timerRef.current = setInterval(() => {
   301→      const elapsed = Date.now() - start;
   302→      const pct = Math.max(0, 100 - (elapsed / durationMs) * 100);
   303→      setMemProgress(pct);
   304→      if (elapsed >= durationMs) {
   305→        clearTimer();
   306→        setPhase("recall_p1");
   307→      }
   308→    }, 40);
   309→    return clearTimer;
   310→  }, [phase, cfg.memorizeTime, clearTimer]);
   311→
   312→  /* ── feedback auto-advance timers ── */
   313→  useEffect(() => {
   314→    if (phase === "feedback_p1") {
   315→      const id = setTimeout(() => setPhase("recall_p2"), 1800);
   316→      return () => clearTimeout(id);
   317→    }
   318→    if (phase === "feedback_p2") {
   319→      const id = setTimeout(() => setPhase("round_result"), 1800);
   320→      return () => clearTimeout(id);
   321→    }
   322→  }, [phase]);
   323→
   324→  /* ── start round ── */
   325→  const startRound = useCallback(() => {
   326→    const pat = pickPattern(totalCells, cfg.patternSize);
   327→    setPattern(pat);
   328→    setP1Taps(new Set());
   329→    setP2Taps(new Set());
   330→    setP1RoundPts(0);
   331→    setP2RoundPts(0);
   332→    setMemProgress(100);
   333→    setPhase("memorize");
   334→  }, [totalCells, cfg.patternSize]);
   335→
   336→  /* ── begin full game ── */
   337→  const handleStart = useCallback(() => {
   338→    scoresRef.current = { p1: 0, p2: 0 };
   339→    onScoreFired.current = false;
   340→    setRound(0);
   341→    setP1Total(0);
   342→    setP2Total(0);
   343→    setP1Taps(new Set());
   344→    setP2Taps(new Set());
   345→    setP1RoundPts(0);
   346→    setP2RoundPts(0);
   347→    setPattern(new Set());
   348→    setPhase("idle");
   349→    // startRound runs after setState flushes, so we schedule it
   350→    setTimeout(() => startRound(), 0);
   351→  }, [startRound]);
   352→
   353→  /* ── tap cell ── */
   354→  const handleTap = useCallback(
   355→    (idx: number) => {
   356→      if (phase === "recall_p1") {
   357→        setP1Taps((prev) => {
   358→          const next = new Set(prev);
   359→          next.has(idx) ? next.delete(idx) : next.add(idx);
   360→          return next;
   361→        });
   362→      } else if (phase === "recall_p2") {
   363→        setP2Taps((prev) => {
   364→          const next = new Set(prev);
   365→          next.has(idx) ? next.delete(idx) : next.add(idx);
   366→          return next;
   367→        });
   368→      }
   369→    },
   370→    [phase]
   371→  );
   372→
   373→  /* ── confirm recall ── */
   374→  const handleConfirm = useCallback(() => {
   375→    if (phase === "recall_p1") {
   376→      const pts = calcRoundScore(p1Taps, pattern);
   377→      setP1RoundPts(pts);
   378→      scoresRef.current.p1 += pts;
   379→      setP1Total(scoresRef.current.p1);
   380→      setPhase("feedback_p1");
   381→    } else if (phase === "recall_p2") {
   382→      const pts = calcRoundScore(p2Taps, pattern);
   383→      setP2RoundPts(pts);
   384→      scoresRef.current.p2 += pts;
   385→      setP2Total(scoresRef.current.p2);
   386→      setPhase("feedback_p2");
   387→    }
   388→  }, [phase, p1Taps, p2Taps, pattern]);
   389→
   390→  /* ── next round / game over ── */
   391→  const handleNextRound = useCallback(() => {
   392→    const nextRound = round + 1;
   393→    if (nextRound >= TOTAL_ROUNDS) {
   394→      setPhase("game_over");
   395→      if (onScore && !onScoreFired.current) {
   396→        onScoreFired.current = true;
   397→        const { p1, p2 } = scoresRef.current;
   398→        if (p1 > p2) {
   399→          onScore("Jogador 1", p1);
   400→        } else if (p2 > p1) {
   401→          onScore("Jogador 2", p2);
   402→        } else {
   403→          onScore("Jogador 1", p1);
   404→          onScore("Jogador 2", p2);
   405→        }
   406→      }
   407→    } else {
   408→      setRound(nextRound);
   409→      setPhase("idle");
   410→      setTimeout(() => startRound(), 0);
   411→    }
   412→  }, [round, onScore, startRound]);
   413→
   414→  /* ── restart ── */
   415→  const handleRestart = useCallback(() => {
   416→    clearTimer();
   417→    handleStart();
   418→  }, [clearTimer, handleStart]);
   419→
   420→  /* ── derived sets for feedback ── */
   421→  const fbTaps =
   422→    phase === "feedback_p1" ? p1Taps : phase === "feedback_p2" ? p2Taps : null;
   423→  const correctSet = new Set<number>();
   424→  const wrongSet = new Set<number>();
   425→  const missedSet = new Set<number>();
   426→  if (fbTaps) {
   427→    fbTaps.forEach((c) => (pattern.has(c) ? correctSet.add(c) : wrongSet.add(c)));
   428→    pattern.forEach((c) => {
   429→      if (!fbTaps.has(c)) missedSet.add(c);
   430→    });
   431→  }
   432→
   433→  const currentTaps =
   434→    phase === "recall_p1" ? p1Taps : phase === "recall_p2" ? p2Taps : new Set<number>();
   435→  const activePlayer: 1 | 2 = phase === "recall_p1" ? 1 : 2;
   436→
   437→  /* ================================================================ */
   438→  /*  RENDER                                                           */
   439→  /* ================================================================ */
   440→
   441→  return (
   442→    <div className="min-h-[540px] bg-gradient-to-b from-zinc-900 to-zinc-950 text-white flex flex-col items-center justify-center p-4 select-none overflow-hidden rounded-2xl border border-zinc-800/60 shadow-2xl">
   443→      <AnimatePresence>
   444→        {phase !== "idle" && (
   445→          <motion.div
   446→            initial={{ opacity: 0, y: -12 }}
   447→            animate={{ opacity: 1, y: 0 }}
   448→            exit={{ opacity: 0 }}
   449→            className="w-full flex justify-center mb-4"
   450→          >
   451→            <ScoreBar
   452→              p1={p1Total}
   453→              p2={p2Total}
   454→              round={round}
   455→              highlight={phase === "recall_p1" ? 1 : phase === "recall_p2" ? 2 : undefined}
   456→            />
   457→          </motion.div>
   458→        )}
   459→      </AnimatePresence>
   460→
   461→      <AnimatePresence mode="wait">
   462→        {phase === "idle" && (
   463→          <motion.div
   464→            key="idle"
   465→            {...fadeUp}
   466→            transition={{ duration: 0.35 }}
   467→            className="flex flex-col items-center gap-5 text-center max-w-sm"
   468→          >
   469→            <motion.div
   470→              initial={{ scale: 0, rotate: -15 }}
   471→              animate={{ scale: 1, rotate: 0 }}
   472→              transition={{ delay: 0.1, type: "spring", stiffness: 200, damping: 15 }}
   473→              className="w-20 h-20 rounded-2xl bg-gradient-to-br from-cyan-500/20 to-pink-500/20 border border-zinc-700/60 flex items-center justify-center"
   474→            >
   475→              <Grid3X3 className="w-10 h-10 text-violet-400" />
   476→            </motion.div>
   477→
   478→            <h1 className="text-3xl sm:text-4xl font-extrabold tracking-tight">
   479→              <span className="bg-gradient-to-r from-cyan-400 to-cyan-300 bg-clip-text text-transparent">
   480→                Memória
   481→              </span>{" "}
   482→              <span className="bg-gradient-to-r from-pink-400 to-pink-300 bg-clip-text text-transparent">
   483→                de Padrão
   484→              </span>
   485→            </h1>
   486→
   487→            <p className="text-zinc-400 text-sm leading-relaxed">
   488→              Um padrão acende brevemente na grade.
   489→              <br />
   490→              Cada jogador recria de memória.
   491→              <br />
   492→              <span className="text-emerald-400/80">Acerto = +10 pts</span>{" "}
   493→              <span className="text-zinc-600">•</span>{" "}
   494→              <span className="text-red-400/80">Erro = −5 pts</span>
   495→            </p>
   496→
   497→            <div className="flex items-center gap-3 text-xs">
   498→              <Badge className="bg-cyan-500/20 text-cyan-300 border border-cyan-500/30 px-3 py-0.5">
   499→                Jogador 1
   500→              </Badge>
   501→              <span className="text-zinc-500 font-extrabold text-sm">VS</span>
   502→              <Badge className="bg-pink-500/20 text-pink-300 border border-pink-500/30 px-3 py-0.5">
   503→                Jogador 2
   504→              </Badge>
   505→            </div>
   506→
   507→            <div className="flex flex-col gap-0.5 text-[11px] text-zinc-500 mt-1">
   508→              <span>
   509→                {TOTAL_ROUNDS} rodadas • Grade 3×3 → 5×5
   510→              </span>
   511→              <span>
   512→                Padrão cresce de {ROUNDS[0].patternSize} a {ROUNDS[ROUNDS.length - 1].patternSize}{" "}
   513→                células
   514→              </span>
   515→              <span>
   516→                Memorização: {ROUNDS[0].memorizeTime}s → {ROUNDS[ROUNDS.length - 1].memorizeTime}s
   517→              </span>
   518→            </div>
   519→
   520→            <Button
   521→              size="lg"
   522→              onClick={handleStart}
   523→              className="mt-2 bg-gradient-to-r from-cyan-600 to-pink-600 hover:from-cyan-500 hover:to-pink-500 text-white font-bold px-8 shadow-lg shadow-cyan-500/20 transition-all"
   524→            >
   525→              <Grid3X3 className="w-5 h-5 mr-2" />
   526→              Iniciar Jogo
   527→            </Button>
   528→          </motion.div>
   529→        )}
   530→
   531→        {phase === "memorize" && (
   532→          <motion.div
   533→            key={`mem-${round}`}
   534→            {...popIn}
   535→            transition={{ type: "spring", stiffness: 400, damping: 25 }}
   536→            className="flex flex-col items-center gap-4"
   537→          >
   538→            <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-center">
   539→              <div className="flex items-center justify-center gap-2 mb-1">
   540→                <Eye className="w-5 h-5 text-violet-400" />
   541→                <p className="text-violet-300 font-bold text-lg">
   542→                  Memorize o padrão! ({cfg.memorizeTime.toFixed(1)}s)
   543→                </p>
   544→              </div>
   545→              <p className="text-zinc-500 text-xs">
   546→                {cfg.gridSize}×{cfg.gridSize} • {cfg.patternSize} células
   547→              </p>
   548→            </motion.div>
