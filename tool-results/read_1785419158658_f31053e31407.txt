   289→      mouseX.set(e.clientX - rect.left);
   290→      mouseY.set(e.clientY - rect.top);
   291→    };
   292→    el.addEventListener("mousemove", handler);
   293→    return () => el.removeEventListener("mousemove", handler);
   294→  }, [containerRef, mouseX, mouseY]);
   295→
   296→  return (
   297→    <motion.div
   298→      className="pointer-events-none absolute w-[400px] h-[400px] rounded-full opacity-0 hover:opacity-100 transition-opacity duration-500"
   299→      style={{
   300→        left: smoothX, top: smoothY,

   725→                    <Radio className="h-12 w-12 text-white/[0.06]" />

   772→                      </div>
