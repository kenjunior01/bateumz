   177→/* ─── Animated Counter ──────────────────────────────── */
   178→const CountUp = ({ target, duration = 1.5 }: { target: number; duration?: number }) => {
   179→  const [val, setVal] = useState(0);
   180→  const ref = useRef<HTMLSpanElement>(null);
   181→  const hasAnimated = useRef(false);
   182→
   183→  useEffect(() => {
   184→    if (hasAnimated.current || !ref.current) return;
   185→    const obs = new IntersectionObserver(
   186→      ([entry]) => {
   187→        if (entry.isIntersecting && !hasAnimated.current) {
   188→          hasAnimated.current = true;
   189→          const start = performance.now();
   190→          const ease = (t: number) => 1 - Math.pow(1 - t, 3);
   191→          const tick = (now: number) => {
   192→            const t = Math.min((now - start) / (duration * 1000), 1);
   193→            setVal(Math.round(ease(t) * target));
   194→            if (t < 1) requestAnimationFrame(tick);
   195→          };
   196→          requestAnimationFrame(tick);
   197→        }
   198→      },
   199→      { threshold: 0.3 }
   200→    );
   201→    obs.observe(ref.current);
   202→    return () => obs.disconnect();
   203→  }, [target, duration]);
   204→
   205→  return <span ref={ref}>{val}</span>;
   206→};

   209→const RingStat = ({ icon: Icon, label, value, color, delay, suffix }: { icon: any; label: string; value: number; color: string; delay: number; suffix?: string }) => {

   223→  }, []);

   276→/* ─── Mouse Glow (follows cursor) ───────────────────── */
   277→const MouseGlow = ({ color, containerRef }: { color: string; containerRef: React.RefObject<HTMLDivElement | null> }) => {
   278→  const mouseX = useMotionValue(0);
   279→  const mouseY = useMotionValue(0);
   280→  const smoothX = useSpring(mouseX, { stiffness: 50, damping: 30 });
   281→  const smoothY = useSpring(mouseY, { stiffness: 50, damping: 30 });
   282→
   283→  useEffect(() => {
   284→    const el = containerRef.current;
   285→    if (!el) return;
   286→    const handler = (e: MouseEvent) => {
   287→      const rect = el.getBoundingClientRect();
   288→      mouseX.set(e.clientX - rect.left);
   289→      mouseY.set(e.clientY - rect.top);
   290→    };
   291→    el.addEventListener("mousemove", handler);
   292→    return () => el.removeEventListener("mousemove", handler);
   293→  }, [containerRef, mouseX, mouseY]);
   294→
   295→  return (
   296→    <motion.div
   297→      className="pointer-events-none absolute w-[400px] h-[400px] rounded-full opacity-0 hover:opacity-100 transition-opacity duration-500"
   298→      style={{
   299→        left: smoothX, top: smoothY,
   300→        x: "-50%", y: "-50%",
   301→        background: `radial-gradient(circle, ${color}08, transparent 60%)`,
   302→        filter: "blur(2px)",

   771→                      </div>

   781→                      <div className="p-1.5 rounded-lg bg-white/[0.04]"><Globe className="h-4 w-4 text-white/30" /></div>

   787→                      {company.created_at && <div className="flex items-center gap-2 text-white/30"><Calendar className="h-3.5 w-3.5" />{new Date(company.created_at).toLocaleDateString("pt-PT", { month: "long", year: "numeric" })}</div>}
