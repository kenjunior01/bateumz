   158→  const [raffles, setRaffles] = useState<Raffle[]>([]);

   189→          .from("prestacao_products")

   191→          .eq("business_user_id", id)
   192→          .eq("status", "active")

   213→        supabase.from("millionaire_games").select("id, title, description, cover_image_url, is_active, created_at").eq("business_user_id", bizId).order("created_at", { ascending: false }).limit(20),

   270→              .from("contest_submissions")
   271→              .select("id, participant_name, photo_url, votes_count, views_count")
   272→              .eq("contest_id", c.id)
   273→              .eq("status", "approved")
   274→              .order(orderCol, { ascending: false })
   275→              .limit(3);

   277→              contest_id: c.id,
   278→              contest_title: c.title,
   279→              evaluation_type: c.evaluation_type,

   282→                submission_id: s.id,
   283→                name: s.participant_name,
   284→                photo_url: s.photo_url,
   285→                score: c.evaluation_type === "views" ? s.views_count : s.votes_count,
   286→              })),
   287→            });
   288→          }),
   289→        );
   290→        setRankings(results.filter((r) => r.top.length > 0));
   291→      }
   292→    };

   297→  const allGames = useMemo(() => [

   307→    const activeGames = [...spinGames, ...millionaireGames].filter((g) => g.is_active).length;
   308→    return {
   309→      activeRaffles,
   310→      totalRaffles: raffles.length,
   311→      contests: contests.length,
   312→      products: products.length,
   313→      totalSold,
   314→      totalGames,
   315→      activeGames,
   316→    };
   317→  }, [raffles, contests, products, spinGames, millionaireGames, challengeRoulettes]);
   318→

   320→    if (!guestName) {
   321→      setPendingGame({ type, id: gameId });

   325→        spin: `/games/spin-wheel/${gameId}`,
   326→        millionaire: `/games/millionaire/${gameId}`,

   337→      const routes: Record<string, string> = {
   338→        spin: `/games/spin-wheel/${pendingGame.id}`,
   339→        millionaire: `/games/millionaire/${pendingGame.id}`,
   340→      };
   341→      navigate(routes[pendingGame.type] || `/jogos`);
   342→      setPendingGame(null);
   343→    }
   344→  }, [pendingGame, navigate]);
   345→

   347→    const url = window.location.href;
   348→    const title = business?.company_name || business?.display_name || "Empresa";

   350→      if (navigator.share) {
   351→        await navigator.share({ title: `${title} • Bateu`, url });

   353→        await navigator.clipboard.writeText(url);

   357→      // user cancelled
   358→    }

   360→

   362→    return (
   363→      <div className="min-h-screen bg-background">
   364→        <Navbar />
   365→        <div className="container mx-auto px-4 py-6 sm:py-10 max-w-6xl">
   366→          <div className="flex items-center gap-4 mb-6">
   367→            <Skeleton className="h-20 w-20 sm:h-24 sm:w-24 rounded-2xl" />
   368→            <div className="flex-1 space-y-2">
   369→              <Skeleton className="h-6 w-2/3 max-w-xs" />

   409→      <Navbar />

   411→      <div className="relative">
   412→        <div className="relative h-32 sm:h-48 overflow-hidden">
   413→          <div className="absolute inset-0 bg-gradient-to-br from-primary via-primary/80 to-accent" />

   417→              backgroundImage:
   418→                "radial-gradient(circle at 20% 80%, rgba(255,255,255,0.3) 0, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.2) 0, transparent 50%)",

  1274→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */
  1275→
  1276→interface StatusOption { value: string; label: string }
  1277→interface SortOption { value: string; label: string }
  1278→
  1279→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1280→  items,
  1281→  statuses,
  1282→  sorts,
  1283→  renderItem,
  1284→  getStatus,
  1285→  getSortValue,
  1286→  formatScore,
  1287→  emptyMessage,
  1288→  extraAction,
  1289→  pageSize = 9,
  1290→}: {
  1291→  items: T[];
  1292→  statuses: StatusOption[];
  1293→  sorts: SortOption[];
  1294→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1295→  getStatus: (item: T) => string;
  1296→  getSortValue: (item: T, sort: string) => number;
  1297→  formatScore?: (item: T, sort: string) => string | undefined;
  1298→  emptyMessage: string;
  1299→  extraAction?: React.ReactNode;
  1300→  pageSize?: number;
  1301→}) {
  1302→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1303→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1304→  const [page, setPage] = useState(1);
  1305→  const [transitioning, setTransitioning] = useState(false);
  1306→
  1307→  // Reset page + show brief skeleton when filter/sort changes
  1308→  useEffect(() => {
  1309→    setPage(1);
  1310→    setTransitioning(true);
  1311→    const t = setTimeout(() => setTransitioning(false), 280);
  1312→    return () => clearTimeout(t);
  1313→  }, [status, sort]);
  1314→
  1315→  const filtered = useMemo(() => {
  1316→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1317→    if (sort === "recent") {
  1318→      list = [...list].sort((a, b) => {
  1319→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1320→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1321→        return tb - ta;
  1322→      });
  1323→    } else {
  1324→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1325→    }
  1326→    return list;
  1327→  }, [items, status, sort, getStatus, getSortValue]);
  1328→
  1329→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  1330→  const visible = filtered.slice(0, page * pageSize);
  1331→  const canLoadMore = page < totalPages;
  1332→
  1333→  const counts = useMemo(() => {
  1334→    const map: Record<string, number> = { all: items.length };
  1335→    for (const i of items) {
  1336→      const s = getStatus(i);
  1337→      map[s] = (map[s] ?? 0) + 1;
  1338→    }
  1339→    return map;
  1340→  }, [items, getStatus]);
  1341→
  1342→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1343→
  1344→  if (items.length === 0) {
  1345→    return <EmptyState message={emptyMessage} />;
  1346→  }
  1347→
  1348→  return (
  1349→    <div>
  1350→      <div className="mb-4 flex flex-col gap-3">
  1351→        <div className="flex flex-wrap gap-1.5">
  1352→          {statuses.map((s) => {
  1353→            const active = status === s.value;
  1354→            const count = counts[s.value] ?? 0;
  1355→            return (
  1356→              <button
  1357→                key={s.value}
  1358→                onClick={() => setStatus(s.value)}
  1359→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1360→                  active
  1361→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1362→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1363→                }`}
  1364→              >
  1365→                {s.label}
  1366→                <span className="ml-1 opacity-70">({count})</span>
  1367→              </button>
  1368→            );
  1369→          })}
  1370→        </div>
  1371→
  1372→        <div className="flex items-center justify-between gap-2">
  1373→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1374→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1375→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1376→          </p>
  1377→          <div className="flex items-center gap-2">
  1378→            {extraAction}
  1379→            <Select value={sort} onValueChange={setSort}>
  1380→              <SelectTrigger className="h-8 text-xs w-[140px] sm:w-[170px]">
  1381→                <SelectValue />
  1382→              </SelectTrigger>
  1383→              <SelectContent>
  1384→                {sorts.map((s) => (
  1385→                  <SelectItem key={s.value} value={s.value} className="text-xs">
  1386→                    {s.label}
  1387→                  </SelectItem>
  1388→                ))}
  1389→              </SelectContent>
  1390→            </Select>
  1391→          </div>
  1392→        </div>
  1393→      </div>
  1394→
  1395→      {transitioning ? (
  1396→        <SkeletonCardGrid count={Math.min(pageSize, 6)} />
  1397→      ) : filtered.length === 0 ? (
  1398→        <EmptyState message="Nenhum resultado para os filtros selecionados." />
  1399→      ) : (
  1400→        <>
  1401→          <Grid>
  1402→            {visible.map((item, idx) =>
  1403→              renderItem(item, {
  1404→                rank: idx + 1,
  1405→                total: filtered.length,
  1406→                criterion: sortLabel,
  1407→                scoreLabel: formatScore?.(item, sort),
  1408→              }),
  1409→            )}
  1410→          </Grid>
  1411→
  1412→          {canLoadMore && (
  1413→            <div className="mt-6 flex justify-center">
  1414→              <Button variant="outline" size="sm" onClick={() => setPage((p) => p + 1)} className="gap-2">
  1415→                Carregar mais
  1416→                <span className="text-xs text-muted-foreground">
  1417→                  ({visible.length}/{filtered.length})
  1418→                </span>
  1419→              </Button>
  1420→            </div>
  1421→          )}
  1422→          {!canLoadMore && filtered.length > pageSize && (
  1423→            <p className="mt-6 text-center text-[11px] text-muted-foreground">
  1424→              Mostrando todos os {filtered.length} itens
  1425→            </p>
  1426→          )}
  1427→        </>
  1428→      )}
  1429→    </div>
  1430→  );
  1431→}
