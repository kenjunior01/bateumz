    34→    34→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    35→    35→  const [display, setDisplay] = useState(0);
    36→    36→  useEffect(() => {
    37→    37→    if (target === 0) { setDisplay(0); return; }
    38→    38→    const startTime = performance.now();
    39→    39→    const step = (now: number) => {
    40→    40→      const elapsed = (now - startTime) / 1000;
    41→    41→      const progress = Math.min(elapsed / duration, 1);
    42→    42→      const eased = 1 - Math.pow(1 - progress, 4);
    43→    43→      setDisplay(Math.round(eased * target));
    44→    44→      if (progress < 1) requestAnimationFrame(step);
    45→    45→    };
    46→    46→    requestAnimationFrame(step);
    47→    47→  }, [target, duration]);
    48→    48→  return <>{display.toLocaleString("pt-PT")}</>;
    49→    49→}
