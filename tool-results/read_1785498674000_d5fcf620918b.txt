  2967→  if (!vars) return template;
  2968→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);

  2980→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  2981→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  2982→    return resolveLangFromCountry(getStoredCountry());
  2983→  });
  2984→
  2985→  useEffect(() => {
  2986→    localStorage.setItem("bateu_lang", lang);

  2989→

  2994→      localStorage.setItem("bateu_lang_explicit", "1");

  2999→    // Prioritize the currently SELECTED LANGUAGE first, then English as fallback, then PT
