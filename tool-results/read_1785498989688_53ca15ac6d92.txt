  3173→function format(template: string, vars?: Record<string, string>) {
  3174→  if (!vars) return template;
  3175→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
  3176→}

  3178→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
  3179→
  3180→export function LanguageProvider({ children }: { children: ReactNode }) {
  3181→  // English is the universal default. Only honor an explicit non-English pick
  3182→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
  3183→  // PT defaults from previous releases are reset to English automatically.
  3184→  const [lang, setLangState] = useState<Lang>(() => {
  3185→    if (typeof window === "undefined") return "en";
  3186→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
  3187→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  3188→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  3189→    return resolveLangFromCountry(getStoredCountry());
  3190→  });
  3191→
  3192→  useEffect(() => {
  3193→    localStorage.setItem("bateu_lang", lang);
  3194→    document.documentElement.lang = lang;
  3195→  }, [lang]);
  3196→
  3197→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
  3198→    if (!SUPPORTED.includes(l)) return;
  3199→    setLangState(l);
  3200→    if (typeof window !== "undefined" && options?.explicit !== false) {
  3201→      localStorage.setItem("bateu_lang_explicit", "1");
  3202→    }
  3203→  };
  3204→
  3205→  const t = (key: string, vars?: Record<string, string>) => {
  3206→    // Prioritize the currently SELECTED LANGUAGE first, then English as fallback, then PT
  3207→    const en = translations.en!;
  3208→    const dict = translations[lang] ?? en;
  3209→    const value = dict[key] ?? en[key] ?? translations.pt?.[key] ?? key;
  3210→    return format(value, vars);
  3211→  };
  3212→
  3213→  return (
  3214→    <LanguageContext.Provider value={{ lang, setLang, t }}>
  3215→      {children}
  3216→    </LanguageContext.Provider>
  3217→  );
  3218→}

  3227→        format(translations.en?.[k] ?? k, vars),
  3228→    };
  3229→  }
  3230→  return ctx;
  3231→}
