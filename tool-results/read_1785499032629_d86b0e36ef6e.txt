  3234→interface LanguageContextValue {
  3235→  lang: Lang;
  3236→  setLang: (l: Lang, options?: { explicit?: boolean }) => void;
  3237→  t: (key: string, vars?: Record<string, string>) => string;
  3238→}
  3239→
  3240→const LanguageContext = createContext<LanguageContextValue | null>(null);
  3241→
  3242→function format(template: string, vars?: Record<string, string>) {
  3243→  if (!vars) return template;
  3244→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
  3245→}
  3246→
  3247→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
  3248→
  3249→export function LanguageProvider({ children }: { children: ReactNode }) {
  3250→  // English is the universal default. Only honor an explicit non-English pick
  3251→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
  3252→  // PT defaults from previous releases are reset to English automatically.
  3253→  const [lang, setLangState] = useState<Lang>(() => {
  3254→    if (typeof window === "undefined") return "en";
  3255→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
  3256→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  3257→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  3258→    return resolveLangFromCountry(getStoredCountry());
  3259→  });
  3260→
  3261→  useEffect(() => {
  3262→    localStorage.setItem("bateu_lang", lang);
  3263→    document.documentElement.lang = lang;
  3264→  }, [lang]);
  3265→
  3266→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
  3267→    if (!SUPPORTED.includes(l)) return;
  3268→    setLangState(l);
  3269→    if (typeof window !== "undefined" && options?.explicit !== false) {
  3270→      localStorage.setItem("bateu_lang_explicit", "1");
  3271→    }
  3272→  };
  3273→
  3274→  const t = (key: string, vars?: Record<string, string>) => {
  3275→    // Prioritize the currently SELECTED LANGUAGE first, then English as fallback, then PT
  3276→    const en = translations.en!;
  3277→    const dict = translations[lang] ?? en;
  3278→    const value = dict[key] ?? en[key] ?? translations.pt?.[key] ?? key;
  3279→    return format(value, vars);
  3280→  };
  3281→
  3282→  return (
  3283→    <LanguageContext.Provider value={{ lang, setLang, t }}>
  3284→      {children}
  3285→    </LanguageContext.Provider>
  3286→  );
  3287→}
  3288→
  3289→export function useLanguage() {
  3290→  const ctx = useContext(LanguageContext);
  3291→  if (!ctx) {
  3292→    return {
  3293→      lang: "en" as Lang,
  3294→      setLang: () => {},
  3295→      t: (k: string, vars?: Record<string, string>) =>
  3296→        format(translations.en?.[k] ?? k, vars),
  3297→    };
  3298→  }
