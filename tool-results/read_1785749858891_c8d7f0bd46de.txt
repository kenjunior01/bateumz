    39→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    40→  const [display, setDisplay] = useState(0);
    41→  useEffect(() => {
    42→    if (target === 0) { setDisplay(0); return; }
    43→    const startTime = performance.now();
    44→    const step = (now: number) => {
    45→      const elapsed = (now - startTime) / 1000;
    46→      const progress = Math.min(elapsed / duration, 1);
    47→      const eased = 1 - Math.pow(1 - progress, 4);
    48→      setDisplay(Math.round(eased * target));
    49→      if (progress < 1) requestAnimationFrame(step);
    50→    };
    51→    requestAnimationFrame(step);
    52→  }, [target, duration]);
    53→  return <>{display.toLocaleString("pt-PT")}</>;
    54→}

    56→function DirectoryParticles() {
    57→  const canvasRef = useRef<HTMLCanvasElement>(null);
    58→  useEffect(() => {
    59→    const canvas = canvasRef.current;
    60→    if (!canvas) return;
    61→    const ctx = canvas.getContext("2d");
    62→    if (!ctx) return;
    63→    let animId = 0;
    64→    let w = 0;
    65→    let h = 0;
    66→    const dots: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
    67→    const resize = () => {
    68→      w = canvas.width = canvas.offsetWidth * 2;
    69→      h = canvas.height = canvas.offsetHeight * 2;
    70→    };
    71→    resize();
    72→    window.addEventListener("resize", resize);
    73→    for (let i = 0; i < 30; i++) {
    74→      dots.push({
    75→        x: Math.random() * w,
    76→        y: Math.random() * h,
    77→        vx: (Math.random() - 0.5) * 0.4,
    78→        vy: (Math.random() - 0.5) * 0.4,
    79→        r: Math.random() * 2 + 1,
    80→        o: Math.random() * 0.3 + 0.1,
    81→      });
    82→    }
    83→    const draw = () => {
    84→      ctx.clearRect(0, 0, w, h);
    85→      for (const d of dots) {
    86→        d.x += d.vx;
    87→        d.y += d.vy;
    88→        if (d.x < 0 || d.x > w) d.vx *= -1;
    89→        if (d.y < 0 || d.y > h) d.vy *= -1;
    90→        ctx.beginPath();
    91→        ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
    92→        ctx.fillStyle = "rgba(255,255,255," + d.o + ")";
    93→        ctx.fill();
    94→      }
    95→      for (let i = 0; i < dots.length; i++) {
    96→        for (let j = i + 1; j < dots.length; j++) {
    97→          const dx = dots[i].x - dots[j].x;
    98→          const dy = dots[i].y - dots[j].y;
    99→          const dist = Math.sqrt(dx * dx + dy * dy);
   100→          if (dist < 180) {
   101→            ctx.beginPath();
   102→            ctx.moveTo(dots[i].x, dots[i].y);
   103→            ctx.lineTo(dots[j].x, dots[j].y);
   104→            ctx.strokeStyle = "rgba(255,255,255," + (0.04 * (1 - dist / 180)) + ")";
   105→            ctx.lineWidth = 1;
   106→            ctx.stroke();
   107→          }
   108→        }
   109→      }
   110→      animId = requestAnimationFrame(draw);
   111→    };
   112→    draw();
   113→    return () => {
   114→      cancelAnimationFrame(animId);
   115→      window.removeEventListener("resize", resize);
   116→    };
   117→  }, []);
   118→  return (
   119→    <canvas
   120→      ref={canvasRef}
   121→      className="absolute inset-0 w-full h-full pointer-events-none"
   122→      style={{ opacity: 0.6 }}
   123→    />
   124→  );
   125→}

   162→function TiltCard({ children, className = "", ...props }: { children: React.ReactNode; className?: string } & Record<string, unknown>) {
   163→  const ref = useRef<HTMLDivElement>(null);
   164→  const x = useMotionValue(0);
   165→  const y = useMotionValue(0);
   166→  const rotateX = useSpring(useTransform(y, [-0.5, 0.5], [6, -6]), SPRING);
   167→  const rotateY = useSpring(useTransform(x, [-0.5, 0.5], [-6, 6]), SPRING);
   168→  const handleMouse = useCallback((e: React.MouseEvent) => {
   169→    if (!ref.current) return;
   170→    const r = ref.current.getBoundingClientRect();
   171→    x.set((e.clientX - r.left) / r.width - 0.5);
   172→    y.set((e.clientY - r.top) / r.height - 0.5);
   173→  }, [x, y]);
   174→  const handleLeave = useCallback(() => {
   175→    x.set(0);
   176→    y.set(0);
   177→  }, [x, y]);
   178→  return (
   179→    <motion.div
   180→      ref={ref}
   181→      onMouseMove={handleMouse}
   182→      onMouseLeave={handleLeave}
   183→      style={{ rotateX, rotateY, transformStyle: "preserve-3d", perspective: 800 }}
   184→      className={className}
   185→      {...props}
   186→    >
   187→      {children}
   188→    </motion.div>
   189→  );
   190→}

   192→function ActivityRing({ value, max, color, size = 40 }: { value: number; max: number; color: string; size?: number }) {
   193→  const [pct, setPct] = useState(0);
   194→  const safeMax = Math.max(max, 1);
   195→  useEffect(() => {
   196→    const timer = setTimeout(() => setPct(Math.min(value / safeMax, 1)), 400);
   197→    return () => clearTimeout(timer);
   198→  }, [value, safeMax]);
   199→  const r = (size - 6) / 2;
   200→  const circ = 2 * Math.PI * r;
   201→  const offset = circ * (1 - pct);
   202→  return (
   203→    <svg width={size} height={size} className="dirv2-activity-ring">
   204→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth={3} />
   205→      <circle
   206→        cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={3}
   207→        strokeDasharray={circ} strokeDashoffset={offset}
   208→        strokeLinecap="round"
   209→        style={{ transition: "stroke-dashoffset 1.2s cubic-bezier(0.22,1,0.36,1)" }}
   210→      />
   211→      <text x={size / 2} y={size / 2} textAnchor="middle" dominantBaseline="central"
   212→        fill={color} fontSize={size < 36 ? 9 : 11} fontWeight={800} fontFamily="inherit">
   213→        {value}
   214→      </text>
   215→    </svg>
   216→  );
   217→}
