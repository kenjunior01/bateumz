    69→    69→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    70→    70→  const [display, setDisplay] = useState(0);
    71→    71→  useEffect(() => {
    72→    72→    if (target === 0) { setDisplay(0); return; }
    73→    73→    const startTime = performance.now();
    74→    74→    const step = (now: number) => {
    75→    75→      const elapsed = (now - startTime) / 1000;
    76→    76→      const progress = Math.min(elapsed / duration, 1);
    77→    77→      const eased = 1 - Math.pow(1 - progress, 4);
    78→    78→      setDisplay(Math.round(eased * target));
    79→    79→      if (progress < 1) requestAnimationFrame(step);
    80→    80→    };
    81→    81→    requestAnimationFrame(step);
    82→    82→  }, [target, duration]);
    83→    83→  return <>{display.toLocaleString("en-US")}</>;
    84→    84→}

    86→    86→function DirectoryParticles() {
    87→    87→  const canvasRef = useRef<HTMLCanvasElement>(null);
    88→    88→  useEffect(() => {
    89→    89→    const canvas = canvasRef.current;
    90→    90→    if (!canvas) return;
    91→    91→    const ctx = canvas.getContext("2d");
    92→    92→    if (!ctx) return;
    93→    93→    let animId = 0;
    94→    94→    let w = 0;
    95→    95→    let h = 0;
    96→    96→    const dots: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
    97→    97→    const resize = () => {
    98→    98→      w = canvas.width = canvas.offsetWidth * 2;
    99→    99→      h = canvas.height = canvas.offsetHeight * 2;
   100→   100→    };
   101→   101→    resize();
   102→   102→    window.addEventListener("resize", resize);
   103→   103→    for (let i = 0; i < 30; i++) {
   104→   104→      dots.push({
   105→   105→        x: Math.random() * w,
   106→   106→        y: Math.random() * h,
   107→   107→        vx: (Math.random() - 0.5) * 0.4,
   108→   108→        vy: (Math.random() - 0.5) * 0.4,
   109→   109→        r: Math.random() * 2 + 1,
   110→   110→        o: Math.random() * 0.3 + 0.1,
   111→   111→      });
   112→   112→    }
   113→   113→    const draw = () => {
   114→   114→      ctx.clearRect(0, 0, w, h);
   115→   115→      for (const d of dots) {
   116→   116→        d.x += d.vx;
   117→   117→        d.y += d.vy;
   118→   118→        if (d.x < 0 || d.x > w) d.vx *= -1;
   119→   119→        if (d.y < 0 || d.y > h) d.vy *= -1;
   120→   120→        ctx.beginPath();
   121→   121→        ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
   122→   122→        ctx.fillStyle = "rgba(255,255,255," + d.o + ")";
   123→   123→        ctx.fill();
   124→   124→      }
   125→   125→      for (let i = 0; i < dots.length; i++) {
   126→   126→        for (let j = i + 1; j < dots.length; j++) {
   127→   127→          const dx = dots[i].x - dots[j].x;
   128→   128→          const dy = dots[i].y - dots[j].y;
   129→   129→          const dist = Math.sqrt(dx * dx + dy * dy);
   130→   130→          if (dist < 180) {
   131→   131→            ctx.beginPath();
   132→   132→            ctx.moveTo(dots[i].x, dots[i].y);
   133→   133→            ctx.lineTo(dots[j].x, dots[j].y);
   134→   134→            ctx.strokeStyle = "rgba(255,255,255," + (0.04 * (1 - dist / 180)) + ")";
   135→   135→            ctx.lineWidth = 1;
   136→   136→            ctx.stroke();
   137→   137→          }
   138→   138→        }
   139→   139→      }
   140→   140→      animId = requestAnimationFrame(draw);
   141→   141→    };
   142→   142→    draw();
   143→   143→    return () => {
   144→   144→      cancelAnimationFrame(animId);
   145→   145→      window.removeEventListener("resize", resize);
   146→   146→    };
   147→   147→  }, []);
   148→   148→  return (
   149→   149→    <canvas
   150→   150→      ref={canvasRef}
   151→   151→      className="absolute inset-0 w-full h-full pointer-events-none"
   152→   152→      style={{ opacity: 0.6 }}
   153→   153→    />
   154→   154→  );
   155→   155→}

   192→   192→function TiltCard({ children, className = "", ...props }: { children: React.ReactNode; className?: string } & Record<string, unknown>) {
   193→   193→  const ref = useRef<HTMLDivElement>(null);
   194→   194→  const x = useMotionValue(0);
   195→   195→  const y = useMotionValue(0);
   196→   196→  const rotateX = useSpring(useTransform(y, [-0.5, 0.5], [6, -6]), SPRING);
   197→   197→  const rotateY = useSpring(useTransform(x, [-0.5, 0.5], [-6, 6]), SPRING);
   198→   198→  const handleMouse = useCallback((e: React.MouseEvent) => {
   199→   199→    if (!ref.current) return;
   200→   200→    const r = ref.current.getBoundingClientRect();
   201→   201→    x.set((e.clientX - r.left) / r.width - 0.5);
   202→   202→    y.set((e.clientY - r.top) / r.height - 0.5);
   203→   203→  }, [x, y]);
   204→   204→  const handleLeave = useCallback(() => {
   205→   205→    x.set(0);
   206→   206→    y.set(0);
   207→   207→  }, [x, y]);
   208→   208→  return (
   209→   209→    <motion.div
   210→   210→      ref={ref}
   211→   211→      onMouseMove={handleMouse}
   212→   212→      onMouseLeave={handleLeave}
   213→   213→      style={{ rotateX, rotateY, transformStyle: "preserve-3d", perspective: 800 }}
   214→   214→      className={className}
   215→   215→      {...props}
   216→   216→    >
   217→   217→      {children}
   218→   218→    </motion.div>
   219→   219→  );
   220→   220→}

   222→   222→function ActivityRing({ value, max, color, size = 40 }: { value: number; max: number; color: string; size?: number }) {
   223→   223→  const [pct, setPct] = useState(0);
   224→   224→  const safeMax = Math.max(max, 1);

   226→   226→    const timer = setTimeout(() => setPct(Math.min(value / safeMax, 1)), 400);
   227→   227→    return () => clearTimeout(timer);
   228→   228→  }, [value, safeMax]);
   229→   229→  const r = (size - 6) / 2;
   230→   230→  const circ = 2 * Math.PI * r;
   231→   231→  const offset = circ * (1 - pct);
   232→   232→  return (
   233→   233→    <svg width={size} height={size} className="dirv2-activity-ring">
   234→   234→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth={3} />

   238→   238→        strokeLinecap="round"
