  3729→  if (!vars) return template;
  3730→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
  3731→}

  3733→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
  3734→
  3735→export function LanguageProvider({ children }: { children: ReactNode }) {
  3736→  // English is the universal default. Only honor an explicit non-English pick
  3737→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
  3738→  // PT defaults from previous releases are reset to English automatically.
  3739→  const [lang, setLangState] = useState<Lang>(() => {
  3740→    if (typeof window === "undefined") return "pt";
  3741→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
  3742→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  3743→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  3744→    return resolveLangFromCountry(getStoredCountry());
  3745→  });
  3746→
  3747→  useEffect(() => {
  3748→    localStorage.setItem("bateu_lang", lang);
  3749→    document.documentElement.lang = lang;
  3750→  }, [lang]);
  3751→
  3752→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
  3753→    if (!SUPPORTED.includes(l)) return;
  3754→    setLangState(l);
  3755→    if (typeof window !== "undefined" && options?.explicit !== false) {
  3756→      localStorage.setItem("bateu_lang_explicit", "1");
  3757→    }
  3758→  };
  3759→
  3760→  const t = (key: string, vars?: Record<string, string>) => {
  3761→    // For PT/PT-BR: selected → pt → en → raw key
  3762→    // For other langs (hi, es, fr): selected → en → raw key (skip PT)
  3763→    const en = translations.en!;
  3764→    const dict = translations[lang] ?? en;
  3765→    const isPT = lang === "pt" || lang === "pt-BR";
  3766→    const pt = translations.pt!;
  3767→    const value = dict[key] ?? (isPT ? pt[key] : undefined) ?? en[key] ?? key;
  3768→    return format(value, vars);
  3769→  };
  3770→
  3771→  return (
  3772→    <LanguageContext.Provider value={{ lang, setLang, t }}>
  3773→      {children}
  3774→    </LanguageContext.Provider>
  3775→  );
  3776→}
  3777→
  3778→export function useLanguage() {
  3779→  const ctx = useContext(LanguageContext);
  3780→  if (!ctx) {
  3781→    return {
  3782→      lang: "pt" as Lang,
  3783→      setLang: () => {},
  3784→      t: (k: string, vars?: Record<string, string>) =>
  3785→        format(translations.pt?.[k] ?? translations.en?.[k] ?? k, vars),
  3786→    };
  3787→  }
  3788→  return ctx;
  3789→}
