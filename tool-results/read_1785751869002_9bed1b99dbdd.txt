    70→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    71→  const [display, setDisplay] = useState(0);
    72→  useEffect(() => {
    73→    if (target === 0) { setDisplay(0); return; }
    74→    const startTime = performance.now();
    75→    const step = (now: number) => {
    76→      const elapsed = (now - startTime) / 1000;
    77→      const progress = Math.min(elapsed / duration, 1);
    78→      const eased = 1 - Math.pow(1 - progress, 4);
    79→      setDisplay(Math.round(eased * target));
    80→      if (progress < 1) requestAnimationFrame(step);
    81→    };
    82→    requestAnimationFrame(step);
    83→  }, [target, duration]);
    84→  return <>{display.toLocaleString("en-US")}</>;
    85→}

    87→function DirectoryParticles() {
    88→  const canvasRef = useRef<HTMLCanvasElement>(null);
    89→  useEffect(() => {
    90→    const canvas = canvasRef.current;
    91→    if (!canvas) return;
    92→    const ctx = canvas.getContext("2d");
    93→    if (!ctx) return;
    94→    let animId = 0;
    95→    let w = 0;
    96→    let h = 0;
    97→    const dots: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
    98→    const resize = () => {
    99→      w = canvas.width = canvas.offsetWidth * 2;
   100→      h = canvas.height = canvas.offsetHeight * 2;
   101→    };
   102→    resize();
   103→    window.addEventListener("resize", resize);
   104→    for (let i = 0; i < 30; i++) {
   105→      dots.push({
   106→        x: Math.random() * w,
   107→        y: Math.random() * h,
   108→        vx: (Math.random() - 0.5) * 0.4,
   109→        vy: (Math.random() - 0.5) * 0.4,
   110→        r: Math.random() * 2 + 1,
   111→        o: Math.random() * 0.3 + 0.1,
   112→      });
   113→    }
   114→    const draw = () => {
   115→      ctx.clearRect(0, 0, w, h);
   116→      for (const d of dots) {
   117→        d.x += d.vx;
   118→        d.y += d.vy;
   119→        if (d.x < 0 || d.x > w) d.vx *= -1;
   120→        if (d.y < 0 || d.y > h) d.vy *= -1;
   121→        ctx.beginPath();
   122→        ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
   123→        ctx.fillStyle = "rgba(255,255,255," + d.o + ")";
   124→        ctx.fill();
   125→      }
   126→      for (let i = 0; i < dots.length; i++) {
   127→        for (let j = i + 1; j < dots.length; j++) {
   128→          const dx = dots[i].x - dots[j].x;
   129→          const dy = dots[i].y - dots[j].y;
   130→          const dist = Math.sqrt(dx * dx + dy * dy);
   131→          if (dist < 180) {
   132→            ctx.beginPath();
   133→            ctx.moveTo(dots[i].x, dots[i].y);
   134→            ctx.lineTo(dots[j].x, dots[j].y);
   135→            ctx.strokeStyle = "rgba(255,255,255," + (0.04 * (1 - dist / 180)) + ")";
   136→            ctx.lineWidth = 1;
   137→            ctx.stroke();
   138→          }
   139→        }
   140→      }
   141→      animId = requestAnimationFrame(draw);
   142→    };
   143→    draw();
   144→    return () => {
   145→      cancelAnimationFrame(animId);
   146→      window.removeEventListener("resize", resize);
   147→    };
   148→  }, []);
   149→  return (
   150→    <canvas
   151→      ref={canvasRef}
   152→      className="absolute inset-0 w-full h-full pointer-events-none"
   153→      style={{ opacity: 0.6 }}
   154→    />
   155→  );
   156→}

   193→function TiltCard({ children, className = "", ...props }: { children: React.ReactNode; className?: string } & Record<string, unknown>) {
   194→  const ref = useRef<HTMLDivElement>(null);
   195→  const x = useMotionValue(0);
   196→  const y = useMotionValue(0);
   197→  const rotateX = useSpring(useTransform(y, [-0.5, 0.5], [6, -6]), SPRING);
   198→  const rotateY = useSpring(useTransform(x, [-0.5, 0.5], [-6, 6]), SPRING);
   199→  const handleMouse = useCallback((e: React.MouseEvent) => {
   200→    if (!ref.current) return;
   201→    const r = ref.current.getBoundingClientRect();
   202→    x.set((e.clientX - r.left) / r.width - 0.5);
   203→    y.set((e.clientY - r.top) / r.height - 0.5);
   204→  }, [x, y]);
   205→  const handleLeave = useCallback(() => {
   206→    x.set(0);
   207→    y.set(0);
   208→  }, [x, y]);
   209→  return (
   210→    <motion.div
   211→      ref={ref}
   212→      onMouseMove={handleMouse}
   213→      onMouseLeave={handleLeave}
   214→      style={{ rotateX, rotateY, transformStyle: "preserve-3d", perspective: 800 }}
   215→      className={className}
   216→      {...props}
   217→    >
   218→      {children}
   219→    </motion.div>
   220→  );
   221→}

   223→function ActivityRing({ value, max, color, size = 40 }: { value: number; max: number; color: string; size?: number }) {
   224→  const [pct, setPct] = useState(0);
   225→  const safeMax = Math.max(max, 1);
   226→  useEffect(() => {
   227→    const timer = setTimeout(() => setPct(Math.min(value / safeMax, 1)), 400);
   228→    return () => clearTimeout(timer);
   229→  }, [value, safeMax]);
   230→  const r = (size - 6) / 2;
   231→  const circ = 2 * Math.PI * r;
   232→  const offset = circ * (1 - pct);
   233→  return (
   234→    <svg width={size} height={size} className="dirv2-activity-ring">
   235→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth={3} />
   236→      <circle
   237→        cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={3}
   238→        strokeDasharray={circ} strokeDashoffset={offset}
   239→        strokeLinecap="round"
   240→        style={{ transition: "stroke-dashoffset 1.2s cubic-bezier(0.22,1,0.36,1)" }}
   241→      />
   242→      <text x={size / 2} y={size / 2} textAnchor="middle" dominantBaseline="central"
   243→        fill={color} fontSize={size < 36 ? 9 : 11} fontWeight={800} fontFamily="inherit">
   244→        {value}
   245→      </text>
   246→    </svg>
   247→  );
   248→}
