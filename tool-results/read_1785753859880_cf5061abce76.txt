    84→    84→function useCountUp(end: number, duration: number = 2000, enabled: boolean = true) {
    85→    85→  const [count, setCount] = useState(0);
    86→    86→  const rafRef = useRef<number | null>(null);
    87→    87→  const startTimeRef = useRef<number | null>(null);
    88→    88→
    89→    89→  const animate = useCallback(
    90→    90→    (timestamp: number) => {
    91→    91→      if (!startTimeRef.current) startTimeRef.current = timestamp;
    92→    92→      const elapsed = timestamp - startTimeRef.current;
    93→    93→      const progress = Math.min(elapsed / duration, 1);
    94→    94→      const eased = 1 - Math.pow(1 - progress, 3);
    95→    95→      setCount(Math.floor(eased * end));
    96→    96→      if (progress < 1) {
    97→    97→        rafRef.current = requestAnimationFrame(animate);
    98→    98→      }
    99→    99→    },
   100→   100→    [end, duration]
   101→   101→  );
   102→   102→
   103→   103→  useEffect(() => {
   104→   104→    if (!enabled) {
   105→   105→      setCount(end);
   106→   106→      return;
   107→   107→    }
   108→   108→    startTimeRef.current = null;
   109→   109→    rafRef.current = requestAnimationFrame(animate);
   110→   110→    return () => {
   111→   111→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   112→   112→    };
   113→   113→  }, [animate, enabled]);
   114→   114→
   115→   115→  return count;
   116→   116→}
