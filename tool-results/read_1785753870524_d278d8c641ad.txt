    84→function useCountUp(end: number, duration: number = 2000, enabled: boolean = true) {
    85→  const [count, setCount] = useState(0);
    86→  const rafRef = useRef<number | null>(null);
    87→  const startTimeRef = useRef<number | null>(null);
    88→
    89→  const animate = useCallback(
    90→    (timestamp: number) => {
    91→      if (!startTimeRef.current) startTimeRef.current = timestamp;
    92→      const elapsed = timestamp - startTimeRef.current;
    93→      const progress = Math.min(elapsed / duration, 1);
    94→      const eased = 1 - Math.pow(1 - progress, 3);
    95→      setCount(Math.floor(eased * end));
    96→      if (progress < 1) {
    97→        rafRef.current = requestAnimationFrame(animate);
    98→      }
    99→    },
   100→    [end, duration]
   101→  );
   102→
   103→  useEffect(() => {
   104→    if (!enabled) {
   105→      setCount(end);
   106→      return;
   107→    }
   108→    startTimeRef.current = null;
   109→    rafRef.current = requestAnimationFrame(animate);
   110→    return () => {
   111→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   112→    };
   113→  }, [animate, enabled]);
   114→
   115→  return count;
   116→}

   119→  const [index, setIndex] = useState(0);
   120→  useEffect(() => {
   121→    const id = setInterval(() => {
   122→      setIndex((prev) => (prev + 1) % words.length);
   123→    }, intervalMs);
   124→    return () => clearInterval(id);
   125→  }, [words.length, intervalMs]);
   126→  return words[index];
   127→}
