    70→    70→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    71→    71→  const [display, setDisplay] = useState(0);
    72→    72→  useEffect(() => {
    73→    73→    if (target === 0) { setDisplay(0); return; }
    74→    74→    const startTime = performance.now();
    75→    75→    const step = (now: number) => {
    76→    76→      const elapsed = (now - startTime) / 1000;
    77→    77→      const progress = Math.min(elapsed / duration, 1);
    78→    78→      const eased = 1 - Math.pow(1 - progress, 4);
    79→    79→      setDisplay(Math.round(eased * target));
    80→    80→      if (progress < 1) requestAnimationFrame(step);
    81→    81→    };
    82→    82→    requestAnimationFrame(step);
    83→    83→  }, [target, duration]);
    84→    84→  return <>{display.toLocaleString("en-US")}</>;
    85→    85→}

    87→    87→function DirectoryParticles() {
    88→    88→  const canvasRef = useRef<HTMLCanvasElement>(null);
    89→    89→  useEffect(() => {
    90→    90→    const canvas = canvasRef.current;
    91→    91→    if (!canvas) return;
    92→    92→    const ctx = canvas.getContext("2d");
    93→    93→    if (!ctx) return;
    94→    94→    let animId = 0;
    95→    95→    let w = 0;
    96→    96→    let h = 0;
    97→    97→    const dots: { x: number; y: number; vx: number; vy: number; r: number; o: number }[] = [];
    98→    98→    const resize = () => {
    99→    99→      w = canvas.width = canvas.offsetWidth * 2;
   100→   100→      h = canvas.height = canvas.offsetHeight * 2;
   101→   101→    };
   102→   102→    resize();
   103→   103→    window.addEventListener("resize", resize);
   104→   104→    for (let i = 0; i < 30; i++) {
   105→   105→      dots.push({
   106→   106→        x: Math.random() * w,
   107→   107→        y: Math.random() * h,
   108→   108→        vx: (Math.random() - 0.5) * 0.4,
   109→   109→        vy: (Math.random() - 0.5) * 0.4,
   110→   110→        r: Math.random() * 2 + 1,
   111→   111→        o: Math.random() * 0.3 + 0.1,
   112→   112→      });
   113→   113→    }
   114→   114→    const draw = () => {
   115→   115→      ctx.clearRect(0, 0, w, h);
   116→   116→      for (const d of dots) {
   117→   117→        d.x += d.vx;
   118→   118→        d.y += d.vy;
   119→   119→        if (d.x < 0 || d.x > w) d.vx *= -1;
   120→   120→        if (d.y < 0 || d.y > h) d.vy *= -1;
   121→   121→        ctx.beginPath();
   122→   122→        ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
   123→   123→        ctx.fillStyle = "rgba(255,255,255," + d.o + ")";
   124→   124→        ctx.fill();
   125→   125→      }
   126→   126→      for (let i = 0; i < dots.length; i++) {
   127→   127→        for (let j = i + 1; j < dots.length; j++) {
   128→   128→          const dx = dots[i].x - dots[j].x;
   129→   129→          const dy = dots[i].y - dots[j].y;
   130→   130→          const dist = Math.sqrt(dx * dx + dy * dy);
   131→   131→          if (dist < 180) {
   132→   132→            ctx.beginPath();
   133→   133→            ctx.moveTo(dots[i].x, dots[i].y);
   134→   134→            ctx.lineTo(dots[j].x, dots[j].y);
   135→   135→            ctx.strokeStyle = "rgba(255,255,255," + (0.04 * (1 - dist / 180)) + ")";
   136→   136→            ctx.lineWidth = 1;
   137→   137→            ctx.stroke();
   138→   138→          }
   139→   139→        }
   140→   140→      }
   141→   141→      animId = requestAnimationFrame(draw);
   142→   142→    };
   143→   143→    draw();
   144→   144→    return () => {
   145→   145→      cancelAnimationFrame(animId);
   146→   146→      window.removeEventListener("resize", resize);
   147→   147→    };
   148→   148→  }, []);
   149→   149→  return (
   150→   150→    <canvas
   151→   151→      ref={canvasRef}
   152→   152→      className="absolute inset-0 w-full h-full pointer-events-none"
   153→   153→      style={{ opacity: 0.6 }}
   154→   154→    />
   155→   155→  );
   156→   156→}

   193→   193→function TiltCard({ children, className = "", ...props }: { children: React.ReactNode; className?: string } & Record<string, unknown>) {
   194→   194→  const ref = useRef<HTMLDivElement>(null);
   195→   195→  const x = useMotionValue(0);
   196→   196→  const y = useMotionValue(0);
   197→   197→  const rotateX = useSpring(useTransform(y, [-0.5, 0.5], [6, -6]), SPRING);
   198→   198→  const rotateY = useSpring(useTransform(x, [-0.5, 0.5], [-6, 6]), SPRING);
   199→   199→  const handleMouse = useCallback((e: React.MouseEvent) => {
   200→   200→    if (!ref.current) return;
   201→   201→    const r = ref.current.getBoundingClientRect();
   202→   202→    x.set((e.clientX - r.left) / r.width - 0.5);
   203→   203→    y.set((e.clientY - r.top) / r.height - 0.5);
   204→   204→  }, [x, y]);
   205→   205→  const handleLeave = useCallback(() => {
   206→   206→    x.set(0);
   207→   207→    y.set(0);
   208→   208→  }, [x, y]);
   209→   209→  return (
   210→   210→    <motion.div
   211→   211→      ref={ref}
   212→   212→      onMouseMove={handleMouse}
   213→   213→      onMouseLeave={handleLeave}
   214→   214→      style={{ rotateX, rotateY, transformStyle: "preserve-3d", perspective: 800 }}
   215→   215→      className={className}
   216→   216→      {...props}
   217→   217→    >
   218→   218→      {children}
   219→   219→    </motion.div>
   220→   220→  );
   221→   221→}

   223→   223→function ActivityRing({ value, max, color, size = 40 }: { value: number; max: number; color: string; size?: number }) {
   224→   224→  const [pct, setPct] = useState(0);
   225→   225→  const safeMax = Math.max(max, 1);
   226→   226→  useEffect(() => {
   227→   227→    const timer = setTimeout(() => setPct(Math.min(value / safeMax, 1)), 400);
   228→   228→    return () => clearTimeout(timer);
   229→   229→  }, [value, safeMax]);
   230→   230→  const r = (size - 6) / 2;
   231→   231→  const circ = 2 * Math.PI * r;
   232→   232→  const offset = circ * (1 - pct);
   233→   233→  return (
   234→   234→    <svg width={size} height={size} className="dirv2-activity-ring">
   235→   235→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth={3} />

   238→   238→        strokeDasharray={circ} strokeDashoffset={offset}
   239→   239→        strokeLinecap="round"
   240→   240→        style={{ transition: "stroke-dashoffset 1.2s cubic-bezier(0.22,1,0.36,1)" }}
