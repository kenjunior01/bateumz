    85→// Per-game error boundary so one crashing game does not kill the whole page
    86→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean}> {
    87→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false}; }
    88→  static getDerivedStateFromError() { return {hasError: true}; }
    89→  componentDidCatch(err: Error) { console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err); }
    90→  render() {
    91→    if (this.state.hasError) return (
    92→      <div className="flex flex-col items-center justify-center py-16 px-4 rounded-2xl border border-dashed border-destructive/40 bg-destructive/5">
    93→        <div className="text-4xl mb-3">🎮</div>
    94→        <p className="text-sm font-bold mb-1">Erro ao carregar {this.props.gameName}</p>
    95→        <p className="text-xs text-muted-foreground mb-3">Tente selecionar outro jogo</p>
    96→        <button onClick={() => this.setState({hasError: false})} className="text-xs px-3 py-1.5 rounded-full bg-primary text-primary-foreground hover:bg-primary/90">Tentar novamente</button>
    97→      </div>
    98→    );
    99→    return this.props.children;
   100→  }
