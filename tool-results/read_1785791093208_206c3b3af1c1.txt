    66→const getBranding = async (code: string): Promise<OverlayBranding> => {
    67→  try {
    68→    // Try scheduled_lives first (for planned/scheduled events)
    69→    const { data: session } = await sb
    70→      .from('scheduled_lives')
    71→      .select('business_user_id')
    72→      .eq('live_code', code)
    73→      .single();
    74→    let userId = session?.business_user_id;
    75→    
    76→    // Fallback: try live_sessions table (for ad-hoc LiveHub sessions)
    77→    if (!userId) {
    78→      const { data: liveSession } = await sb
    79→        .from('live_sessions')
    80→        .select('business_user_id')
    81→        .eq('live_code', code)
    82→        .single();
    83→      userId = (liveSession as any)?.business_user_id;
    84→    }
    85→    
    86→    if (userId) {
    87→      const { data: brand } = await sb
    88→        .from('company_branding')
    89→        .select('*')
    90→        .eq('user_id', userId)
    91→        .single();
    92→      if (brand) {
    93→        return {
    94→          companyName: brand.company_name || undefined,
    95→          companySlogan: brand.company_slogan || undefined,
    96→          companyLogoUrl: brand.company_logo_url || undefined,
    97→          primaryColor: brand.primary_color || DEFAULT_BRANDING.primaryColor,
    98→          secondaryColor: brand.secondary_color || DEFAULT_BRANDING.secondaryColor,
    99→          accentColor: brand.accent_color || DEFAULT_BRANDING.accentColor,
   100→          backgroundColor: 'transparent',
   101→          textColor: brand.text_color || DEFAULT_BRANDING.textColor,
   102→          backgroundImageUrl: brand.background_image_url || undefined,
   103→          overlayStyle: (brand.overlay_style as OverlayStyle) || 'neon',
   104→        };
   105→      }
   106→    }
   107→  } catch { /* noop */ }
   108→  return DEFAULT_BRANDING;

   111→const AnimatedNumber = ({ value, color, size = 'text-2xl' }: { value: number; color?: string; size?: string }) => {
   112→  const prevRef = useRef(value);
   113→  const [displayVal, setDisplayVal] = useState(value);
   114→  const [animating, setAnimating] = useState(false);
   115→
   116→  useEffect(() => {
   117→    if (value !== prevRef.current) {
   118→      const diff = value - prevRef.current;
   119→      prevRef.current = value;
   120→      setAnimating(true);
   121→      let cur = displayVal;
   122→      const steps = 14;
   123→      let step = 0;
   124→      const inc = diff / steps;
   125→      const iv = setInterval(() => {
   126→        step++;
   127→        cur += inc;
   128→        setDisplayVal(Math.round(cur));
   129→        if (step >= steps) {
   130→          clearInterval(iv);
   131→          setDisplayVal(value);
   132→          setTimeout(() => setAnimating(false), 200);
   133→        }
   134→      }, 25);
   135→      return () => clearInterval(iv);
   136→    }
   137→  }, [value]);
   138→
   139→  return (
   140→    <motion.span
   141→      className={`${size} font-black tabular-nums leading-none inline-block`}
   142→      style={{ color }}
   143→      animate={animating ? { scale: [1, 1.18, 1], y: [0, -4, 0] } : { scale: 1, y: 0 }}
   144→      transition={{ type: 'spring', stiffness: 400, damping: 12 }}
   145→    >
   146→      {displayVal.toLocaleString()}
   147→    </motion.span>
   148→  );
