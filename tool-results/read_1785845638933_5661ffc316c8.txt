    69→  const { user, role } = useAuth();
    70→  const { region } = useRegionalTheme();
    71→  const [loading, setLoading] = useState(true);
    72→  const [games, setGames] = useState<Game[]>([]);
    73→  const [selectedGame, setSelectedGame] = useState<Game | null>(null);
    74→  const [segments, setSegments] = useState<Segment[]>([]);
    75→  const [saving, setSaving] = useState(false);
    76→
    77→  // Game form states
    78→  const [gameName, setGameName] = useState("");
    79→  const [gameDescription, setGameDescription] = useState("");
    80→  const [segmentCount, setSegmentCount] = useState(8);
    81→  const [rotationDuration, setRotationDuration] = useState(5);
    82→  const [wheelBackgroundColor, setWheelBackgroundColor] = useState("#334155");
    83→  const [wheelBorderColor, setWheelBorderColor] = useState("#fbbf24");
    84→  const [spinCost, setSpinCost] = useState(0);
    85→  const [soundEnabled, setSoundEnabledGame] = useState(true);
    86→  const [particleEffects, setParticleEffects] = useState(true);
    87→  const [backgroundImageUrl, setBackgroundImageUrl] = useState("");
    88→  const [backgroundColor, setBackgroundColor] = useState("#0f172a");
    89→  const [companyLogoUrl, setCompanyLogoUrl] = useState("");
    90→  const [companySlogan, setCompanySlogan] = useState("");
    91→  const [defaultEffect, setDefaultEffect] = useState("confetti");
    92→  const [rtpMode, setRtpMode] = useState<"normal" | "rain" | "hardcore">("normal");
    93→
    94→  // Segment form
    95→  const [segmentLabel, setSegmentLabel] = useState("");
    96→  const [segmentDescription, setSegmentDescription] = useState("");
    97→  const [segmentColor, setSegmentColor] = useState("#ef4444");
    98→  const [segmentTextColor, setSegmentTextColor] = useState("#ffffff");
    99→  const [rewardType, setRewardType] = useState("points");
   100→  const [rewardValue, setRewardValue] = useState("");
   101→  const [rewardImageUrl, setRewardImageUrl] = useState("");
   102→  const [weight, setWeight] = useState(1);
   103→  const [segmentEffectType, setSegmentEffectType] = useState("confetti");
   104→
   105→  useEffect(() => {
   106→    if (!user) return;
   107→    if (role !== "admin" && role !== "superadmin" && role !== "business") return;
   108→    loadGames();
   109→  }, [user, role]);
   110→
   111→  const loadGames = async () => {
   112→    setLoading(true);
   113→    try {
   114→      let query = supabase.from("spin_wheel_games").select("*").order("created_at", { ascending: false });
   115→      if (role !== "superadmin") {
   116→        query = query.eq("created_by", user!.id);
   117→      }
   118→      const { data, error } = await query;
   119→
   120→      if (error) throw error;
   121→      setGames(data || []);
   122→    } catch (error) {
   123→      console.error("Error loading games:", error);
   124→      toast.error("Erro ao carregar jogos");
   125→    } finally {
   126→      setLoading(false);
   127→    }
   128→  };
   129→
   130→  const loadSegments = async (gameId: string) => {
   131→    try {
   132→      const { data, error } = await supabase
   133→        .from("spin_wheel_segments")
   134→        .select("*")
   135→        .eq("wheel_id", gameId)
   136→        .order("segment_number");
   137→
   138→      if (error) throw error;
   139→      setSegments(data || []);
   140→    } catch (error) {
   141→      console.error("Error loading segments:", error);
   142→      toast.error("Erro ao carregar segmentos");
   143→    }
   144→  };
   145→
   146→  const selectGameForEdit = (game: Game) => {
   147→    setSelectedGame(game);
   148→    setGameName(game.name);
   149→    setGameDescription(game.description || "");
   150→    setSegmentCount(game.segment_count);
   151→    setRotationDuration(game.rotation_duration);
   152→    setWheelBackgroundColor(game.wheel_background_color || "#334155");
   153→    setWheelBorderColor(game.wheel_border_color || "#fbbf24");
   154→    setSpinCost(game.spin_cost || 0);
   155→    setSoundEnabledGame(game.sound_enabled ?? true);
   156→    setParticleEffects(game.particle_effects ?? true);
   157→    setBackgroundImageUrl(game.background_image_url || "");
   158→    setBackgroundColor(game.background_color || "#0f172a");
   159→    setCompanyLogoUrl(game.company_logo_url || "");
   160→    setCompanySlogan(game.company_slogan || "");
   161→    setDefaultEffect(game.default_effect || "confetti");
   162→    setRtpMode((game as any).rtp_mode || "normal");
   163→    loadSegments(game.id);
   164→  };
   165→

   167→    if (!gameName.trim()) {
   168→      toast.error("Nome do jogo é obrigatório");
   169→      return;
   170→    }
   171→
   172→    setSaving(true);
   173→    try {
   174→      const regionId = region?.id || (await supabase.from("regions").select("id").limit(1).single()).data?.id;
   175→
   176→      if (!regionId) {
   177→        toast.error("Erro: Nenhuma região encontrada.");
   178→        setSaving(false);
   179→        return;
   180→      }
   181→
   182→      const { data, error } = await supabase
   183→        .from("spin_wheel_games")
   184→        .insert({
   185→          name: gameName,
   186→          description: gameDescription,
   187→          segment_count: segmentCount,
   188→          rotation_duration: rotationDuration,
   189→          wheel_background_color: wheelBackgroundColor,
   190→          wheel_border_color: wheelBorderColor,
   191→          spin_cost: spinCost,
   192→          sound_enabled: soundEnabled,
   193→          particle_effects: particleEffects,
   194→          background_image_url: backgroundImageUrl,
   195→          background_color: backgroundColor,
   196→          company_logo_url: companyLogoUrl,
   197→          company_slogan: companySlogan,
   198→          default_effect: defaultEffect,
   199→          rtp_mode: rtpMode,
   200→          region_id: regionId,
   201→          created_by: user?.id
   202→        })
   203→        .select()
   204→        .single();
   205→
   206→      if (error) throw error;
   207→
   208→      setGames([data, ...games]);
   209→      setSelectedGame(data);
   210→      resetGameForm();
   211→      toast.success("Jogo criado com sucesso!");
   212→    } catch (error) {
   213→      console.error("Error creating game:", error);
   214→      toast.error("Erro ao criar jogo");
   215→    } finally {
   216→      setSaving(false);
   217→    }
   218→  };
   219→
   220→  const updateGame = async () => {
   221→    if (!selectedGame || !gameName.trim()) {
   222→      toast.error("Nome do jogo é obrigatório");
   223→      return;
   224→    }
   225→
   226→    setSaving(true);
   227→    try {
   228→      const { error } = await supabase
   229→        .from("spin_wheel_games")
   230→        .update({
   231→          name: gameName,
   232→          description: gameDescription,
   233→          segment_count: segmentCount,
   234→          rotation_duration: rotationDuration,
   235→          wheel_background_color: wheelBackgroundColor,
   236→          wheel_border_color: wheelBorderColor,
   237→          spin_cost: spinCost,
   238→          sound_enabled: soundEnabled,
   239→          particle_effects: particleEffects,
   240→          background_image_url: backgroundImageUrl,
   241→          background_color: backgroundColor,
   242→          company_logo_url: companyLogoUrl,
   243→          company_slogan: companySlogan,
   244→          default_effect: defaultEffect,
   245→          rtp_mode: rtpMode
   246→        })
   247→        .eq("id", selectedGame.id);
   248→
   249→      if (error) throw error;
   250→
   251→      setGames(games.map(g => g.id === selectedGame.id ? { ...g, name: gameName } : g));
   252→      toast.success("Jogo atualizado com sucesso!");
   253→    } catch (error) {
   254→      console.error("Error updating game:", error);
   255→      toast.error("Erro ao atualizar jogo");
   256→    } finally {
   257→      setSaving(false);
   258→    }
   259→  };
   260→
   261→  const addSegment = async () => {
   262→    if (!selectedGame || !segmentLabel.trim() || !rewardValue.trim()) {
   263→      toast.error("Preenche todos os campos obrigatórios");
   264→      return;
   265→    }
   266→
   267→    setSaving(true);
   268→    try {
   269→      const segmentNumber = segments.length + 1;
   270→
   271→      const { data, error } = await supabase
   272→        .from("spin_wheel_segments")
   273→        .insert({
   274→          wheel_id: selectedGame.id,
   275→          segment_number: segmentNumber,
   276→          label: segmentLabel,
   277→          description: segmentDescription,
   278→          background_color: segmentColor,
   279→          text_color: segmentTextColor,
   280→          reward_type: rewardType,
   281→          reward_value: rewardValue,
   282→          reward_image_url: rewardImageUrl,
   283→          weight: weight,
   284→          effect_type: segmentEffectType
   285→        })
   286→        .select()
   287→        .single();
   288→
   289→      if (error) throw error;
   290→
   291→      setSegments([...segments, data]);
   292→      resetSegmentForm();
   293→      toast.success("Segmento adicionado!");
   294→    } catch (error) {
   295→      console.error("Error adding segment:", error);
   296→      toast.error("Erro ao adicionar segmento");
   297→    } finally {
   298→      setSaving(false);
   299→    }
   300→  };
   301→
   302→  const deleteSegment = async (segmentId: string) => {
   303→    try {
   304→      const { error } = await supabase
   305→        .from("spin_wheel_segments")
   306→        .delete()
   307→        .eq("id", segmentId);
   308→
   309→      if (error) throw error;
   310→
   311→      setSegments(segments.filter((s) => s.id !== segmentId));
   312→      toast.success("Segmento removido");
   313→    } catch (error) {
   314→      console.error("Error deleting segment:", error);
   315→      toast.error("Erro ao remover segmento");
   316→    }
   317→  };
   318→
   319→  const publishGame = async (gameId: string) => {
   320→    try {
   321→      const { error } = await supabase
   322→        .from("spin_wheel_games")
   323→        .update({ is_published: true })
   324→        .eq("id", gameId);
   325→
   326→      if (error) throw error;
   327→
   328→      setGames(
   329→        games.map((g) =>
   330→          g.id === gameId ? { ...g, is_published: true } : g
   331→        )
   332→      );
   333→      toast.success("Jogo publicado!");
   334→    } catch (error) {
   335→      console.error("Error publishing game:", error);
   336→      toast.error("Erro ao publicar jogo");
   337→    }
   338→  };
   339→
   340→  const resetGameForm = () => {
   341→    setGameName("");
   342→    setGameDescription("");
   343→    setSegmentCount(8);
   344→    setRotationDuration(5);
   345→    setWheelBackgroundColor("#334155");
   346→    setWheelBorderColor("#fbbf24");
   347→    setSpinCost(0);
   348→    setSoundEnabledGame(true);
   349→    setParticleEffects(true);
   350→    setBackgroundImageUrl("");
   351→    setBackgroundColor("#0f172a");
   352→    setCompanyLogoUrl("");
   353→    setCompanySlogan("");
   354→    setDefaultEffect("confetti");
   355→    setRtpMode("normal");
   356→  };
   357→
   358→  const resetSegmentForm = () => {
   359→    setSegmentLabel("");
   360→    setSegmentDescription("");
   361→    setSegmentColor("#ef4444");
   362→    setSegmentTextColor("#ffffff");
   363→    setRewardType("points");
   364→    setRewardValue("");
   365→    setRewardImageUrl("");
   366→    setWeight(1);
   367→    setSegmentEffectType("confetti");
   368→  };
   369→
   370→  if (role !== "admin" && role !== "superadmin" && role !== "business") {
   371→    return <Navigate to="/profile" replace />;
   372→  }
   373→
   374→  if (loading) {
   375→    return (
   376→      <div className="flex h-64 items-center justify-center">
   377→        <Loader2 className="h-6 w-6 animate-spin" />
   378→      </div>
   379→    );
   380→  }
   381→
   382→  return (
   383→    <div className="space-y-6">
   384→      <h1 className="font-display text-2xl font-bold">Gerenciar Roda da Sorte</h1>
   385→
   386→      <Tabs defaultValue="games" className="w-full">
   387→        <TabsList className="grid w-full grid-cols-2">
   388→          <TabsTrigger value="games">Meus Jogos</TabsTrigger>
   389→          <TabsTrigger value="create">{selectedGame ? "Editar Jogo" : "Criar Novo Jogo"}</TabsTrigger>
   390→        </TabsList>
   391→
   392→        <TabsContent value="games" className="space-y-4">
   393→          {games.length === 0 ? (
   394→            <Card>
   395→              <CardContent className="py-12 text-center text-muted-foreground">
   396→                Nenhum jogo criado ainda
   397→              </CardContent>
   398→            </Card>
   399→          ) : (
   400→            <div className="grid gap-4">
   401→              {games.map((game) => (
   402→                <Card
   403→                  key={game.id}
   404→                  className="cursor-pointer hover:shadow-lg transition-shadow"
   405→                  onClick={() => selectGameForEdit(game)}
   406→                >
   407→                  <CardHeader>
   408→                    <div className="flex items-start justify-between">
   409→                      <div>
   410→                        <CardTitle>{game.name}</CardTitle>
   411→                        <p className="text-sm text-muted-foreground mt-2">
   412→                          {game.description}
   413→                        </p>
   414→                      </div>
   415→                      <Badge variant={game.is_published ? "default" : "outline"}>
   416→                        {game.is_published ? "Publicado" : "Rascunho"}
   417→                      </Badge>
   418→                    </div>
   419→                  </CardHeader>
   420→                  <CardContent>
   421→                    <div className="flex items-center justify-between">
   422→                      <p className="text-sm text-muted-foreground">
   423→                        {game.segment_count} segmentos
   424→                      </p>
   425→                      <div className="flex gap-2">
   426→                        <Button size="sm" variant="outline" onClick={(e) => {
   427→                          e.stopPropagation();
   428→                          selectGameForEdit(game);
   429→                        }}>
   430→                          <Edit2 className="h-4 w-4 mr-2" />
   431→                          Editar
   432→                        </Button>
   433→                        {!game.is_published && (
   434→                          <Button
   435→                            size="sm"
   436→                            onClick={(e) => {
   437→                              e.stopPropagation();
   438→                              publishGame(game.id);
   439→                            }}
   440→                          >
   441→                            <Eye className="h-4 w-4 mr-2" />
   442→                            Publicar
   443→                          </Button>
   444→                        )}
   445→                        <Button size="sm" variant="secondary" asChild onClick={(e) => e.stopPropagation()}>
   446→                          <a href={`/games/spin-wheel/${game.id}`} target="_blank" rel="noreferrer">
   447→                            <Eye className="h-4 w-4 mr-2" />
   448→                            Pré-visualizar
   449→                          </a>
   450→                        </Button>
   451→                      </div>
   452→                    </div>
   453→                  </CardContent>
   454→                </Card>
   455→              ))}
   456→            </div>
   457→          )}
   458→        </TabsContent>
   459→
   460→        <TabsContent value="create" className="space-y-4">
   461→          <Card>
   462→            <CardHeader><CardTitle>{selectedGame ? "Editar Jogo" : "Criar Nova Roda"}</CardTitle></CardHeader>
   463→            <CardContent className="space-y-4">
   464→              <div>
   465→                <Label>Nome do Jogo</Label>
   466→                <Input
   467→                  placeholder="Nome da roda da sorte"
   468→                  value={gameName}
   469→                  onChange={(e) => setGameName(e.target.value)}
   470→                />
   471→              </div>
   472→
   473→              <div>
   474→                <Label>Descrição</Label>
   475→                <Textarea

   477→                  value={gameDescription}
   478→                  onChange={(e) => setGameDescription(e.target.value)}
   479→                  rows={3}
   480→                />
   481→              </div>
   482→
   483→              <div>
   484→                <Label>Tema</Label>
   485→                <div className="grid grid-cols-5 gap-2 mt-2">
   486→                  {Object.entries(PRESET_THEMES).map(([key, theme]) => (
   487→                    <button
   488→                      key={key}
   489→                      onClick={() => {
   490→                        setBackgroundColor(theme.backgroundColor);
   491→                        setWheelBackgroundColor(theme.wheelBackgroundColor);
   492→                        setWheelBorderColor(theme.primaryColor);
   493→                        setCompanyLogoUrl(theme.primaryColor);
   494→                      }}
   495→                      className="w-full aspect-square rounded-lg border-2 border-border hover:border-primary transition-all p-2 flex flex-col items-center justify-center gap-1"
   496→                      style={{ backgroundColor: theme.backgroundColor }}
   497→                      title={theme.name}
   498→                    >
   499→                      <div
   500→                        className="w-8 h-8 rounded-full"
   501→                        style={{ backgroundColor: theme.primaryColor }}
   502→                      />
   503→                      <span className="text-xs text-white font-medium">{theme.name.substring(0, 3)}</span>
   504→                    </button>
   505→                  ))}
   506→                </div>
   507→              </div>
   508→
   509→              <div className="grid gap-4 md:grid-cols-2">
   510→                <div>
   511→                  <Label>Cor de Fundo (Hex)</Label>
   512→                  <div className="flex gap-2">
   513→                    <input
   514→                      type="color"
   515→                      value={backgroundColor}
   516→                      onChange={(e) => setBackgroundColor(e.target.value)}
   517→                      className="w-12 h-10 rounded-md border border-border cursor-pointer"
   518→                    />
   519→                    <Input
   520→                      value={backgroundColor}
   521→                      onChange={(e) => setBackgroundColor(e.target.value)}
   522→                      className="flex-1"
   523→                    />
   524→                  </div>
   525→                </div>
   526→                <div>
   527→                  <Label>Cor da Roda (Hex)</Label>
   528→                  <div className="flex gap-2">
   529→                    <input
   530→                      type="color"
   531→                      value={wheelBackgroundColor}
   532→                      onChange={(e) => setWheelBackgroundColor(e.target.value)}
   533→                      className="w-12 h-10 rounded-md border border-border cursor-pointer"
   534→                    />
   535→                    <Input
   536→                      value={wheelBackgroundColor}
   537→                      onChange={(e) => setWheelBackgroundColor(e.target.value)}
   538→                      className="flex-1"
   539→                    />
   540→                  </div>
   541→                </div>
   542→              </div>
   543→
   544→              <div>

   547→                  value={backgroundImageUrl}

   550→                  bucketName="game-images"
   551→                />

   553→
   554→              <div className="grid gap-4 md:grid-cols-2">

   557→                    label="Logo da Empresa"

   562→                  />
   563→                </div>
   564→                <div>
   565→                  <Label>Slogan da Empresa</Label>
   566→                  <Input
   567→                    placeholder="Slogan..."
   568→                    value={companySlogan}

   570→                  />
   571→                </div>
