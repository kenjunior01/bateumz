   380→   380→   380→  return text
   381→   381→   381→    .toLowerCase()
   382→   382→   382→    .normalize("NFD")
   383→   383→   383→    .replace(/[\u0300-\u036f]/g, "")

   385→   385→   385→    .replace(/(^-|-$)/g, "");
   386→   386→   386→}

   826→   826→   826→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,

   839→   839→   839→function generateElimination(
   840→   840→   840→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   841→   841→   841→  format: TournamentFormat,
   842→   842→   842→  totalRounds: number
   843→   843→   843→): BracketTree {
   844→   844→   844→  const rounds: BracketRound[] = [];
   845→   845→   845→  let matchCounter = 1;
   846→   846→   846→  const matchIdMap = new Map<string, string>();
   847→   847→   847→
   848→   848→   848→  // Build rounds from first to last
   849→   849→   849→  for (let r = 1; r <= totalRounds; r++) {
   850→   850→   850→    const matchesInRound = Math.ceil(teams.length / Math.pow(2, r));
   851→   851→   851→ const roundMatches: BracketMatch[] = [];
   852→   852→   852→    for (let m = 0; m < matchesInRound; m++) {
   853→   853→   853→      const matchId = `bracket-${r}-${m + 1}`;
   854→   854→   854→      const bracketMatch: BracketMatch = {
   855→   855→   855→        id: matchId,
   856→   856→   856→        match_number: matchCounter++,
   857→   857→   857→        p1: null,
   858→   858→   858→        p2: null,
   859→   859→   859→        p1_name: null,
   860→   860→   860→        p2_name: null,
   861→   861→   861→        p1_logo: null,
   862→   862→   862→        p2_logo: null,
   863→   863→   863→        winner: null,
   864→   864→   864→        score1: null,
   865→   865→   865→        score2: null,
   866→   866→   866→        next_match: null,
   867→   867→   867→      };
   868→   868→   868→
   869→   869→   869→      // First round: seed teams
   870→   870→   870→      if (r === 1) {
   871→   871→   871→        const idx1 = m * 2;
   872→   872→   872→        const idx2 = m * 2 + 1;
   873→   873→   873→        if (idx1 < teams.length) {
   874→   874→   874→          bracketMatch.p1 = teams[idx1].id;
   875→   875→   875→          bracketMatch.p1_name = teams[idx1].name;
   876→   876→   876→          bracketMatch.p1_logo = teams[idx1].logo_url ?? null;
   877→   877→   877→        }
   878→   878→   878→        if (idx2 < teams.length) {
   879→   879→   879→          bracketMatch.p2 = teams[idx2].id;
   880→   880→   880→          bracketMatch.p2_name = teams[idx2].name;
   881→   881→   881→          bracketMatch.p2_logo = teams[idx2].logo_url ?? null;
   882→   882→   882→        }
   883→   883→   883→      }
   884→   884→   884→
   885→   885→   885→      roundMatches.push(bracketMatch);
   886→   886→   886→      matchIdMap.set(`${r}-${m}`, matchId);
   887→   887→   887→    }
   888→   888→   888→    rounds.push({ round: r, label: roundLabel(totalRounds, r), matches: roundMatches });
   889→   889→   889→  }
   890→   890→   890→
   891→   891→   891→  // Wire next_match pointers
   892→   892→   892→  for (let r = 1; r < totalRounds; r++) {
   893→   893→   893→    const currentMatches = rounds[r - 1].matches;
   894→   894→   894→    const nextMatches = rounds[r].matches;
   895→   895→   895→    for (let i = 0; i < currentMatches.length; i++) {
   896→   896→   896→      const nextIdx = Math.floor(i / 2);
   897→   897→   897→      if (nextIdx < nextMatches.length) {
   898→   898→   898→        currentMatches[i].next_match = nextMatches[nextIdx].id;
   899→   899→   899→      }
   900→   900→   900→    }
   901→   901→   901→  }
   902→   902→   902→
   903→   903→   903→  return { format, rounds };
   904→   904→   904→}

   906→   906→   906→function generateRoundRobin(
   907→   907→   907→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   908→   908→   908→  _totalRounds: number
   909→   909→   909→): BracketTree {
   910→   910→   910→  const rounds: BracketRound[] = [];
   911→   911→   911→  const n = teams.length;
   912→   912→   912→  if (n < 2) return { format: "round_robin", rounds: [] };
   913→   913→   913→
   914→   914→   914→  // Standard round-robin algorithm
   915→   915→   915→  const list = [...teams];
   916→   916→   916→  if (n % 2 !== 0) list.push({ id: "bye", name: "Bye", logo_url: null });
   917→   917→   917→  const fixed = list[0];
   918→   918→   918→  const rotating = list.slice(1);
   919→   919→   919→  const totalRounds = rotating.length;
   920→   920→   920→  let matchCounter = 1;
   921→   921→   921→
   922→   922→   922→  for (let r = 0; r < totalRounds; r++) {
   923→   923→   923→    const roundMatches: BracketMatch[] = [];
   924→   924→   924→    const current = [fixed, ...rotating];
   925→   925→   925→    const half = current.length / 2;
   926→   926→   926→    for (let i = 0; i < half; i++) {
   927→   927→   927→      const t1 = current[i];
   928→   928→   928→      const t2 = current[current.length - 1 - i];
   929→   929→   929→      roundMatches.push({
   930→   930→   930→        id: `rr-r${r + 1}-m${i + 1}`,
   931→   931→   931→        match_number: matchCounter++,
   932→   932→   932→        p1: t1.id === "bye" ? null : t1.id,
   933→   933→   933→        p2: t2.id === "bye" ? null : t2.id,
   934→   934→   934→        p1_name: t1.id === "bye" ? null : t1.name,
   935→   935→   935→        p2_name: t2.id === "bye" ? null : t2.name,
   936→   936→   936→        p1_logo: t1.logo_url ?? null,
   937→   937→   937→        p2_logo: t2.logo_url ?? null,
   938→   938→   938→        winner: null,
   939→   939→   939→        score1: null,
   940→   940→   940→        score2: null,
   941→   941→   941→        next_match: null,
   942→   942→   942→      });
   943→   943→   943→    }
   944→   944→   944→    rounds.push({ round: r + 1, label: `Rodada ${r + 1}`, matches: roundMatches });
   945→   945→   945→    // Rotate
   946→   946→   946→    rotating.push(rotating.shift()!);
   947→   947→   947→  }
   948→   948→   948→
   949→   949→   949→  return { format: "round_robin", rounds };
   950→   950→   950→}
