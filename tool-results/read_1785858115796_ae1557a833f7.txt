    41→function ShimmerText({ children, className }: { children: string; className?: string }) {
    42→  return (
    43→    <span className={className}>
    44→      {children.split('').map((char, i) => (
    45→        <motion.span
    46→          key={i}
    47→          initial={{ opacity: 0.3 }}
    48→          animate={{ opacity: [0.3, 1, 0.3] }}
    49→          transition={{ duration: 2.5, repeat: Infinity, delay: i * 0.06 }}
    50→          style={{ display: 'inline-block' }}
    51→        >
    52→          {char === ' ' ? '\u00A0' : char}
    53→        </motion.span>
    54→      ))}
    55→    </span>
    56→  );
    57→}

    84→function getStatusColor(status: ChampStatus): string {
    85→  switch (status) {
    86→    case 'live': return 'bg-red-500/20 text-red-400 border-red-500/40';
    87→    case 'registration_open': return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/40';
    88→    case 'check_in': return 'bg-amber-500/20 text-amber-400 border-amber-500/40';

    90→    case 'completed': return 'bg-zinc-500/20 text-zinc-400 border-zinc-500/40';
    91→    default: return 'bg-zinc-700/30 text-zinc-400 border-zinc-600/30';
    92→  }

   444→function ChampionshipCard({ champ, gameEmoji, gameLabel }: { champ: Championship; gameEmoji: string; gameLabel: string }) {
   445→  const progressPct = champ.max_teams > 0 ? Math.min((champ.registered_teams / champ.max_teams) * 100, 100) : 0;
   446→  const isLive = champ.status === 'live';
   447→
   448→  return (
   449→    <motion.div
