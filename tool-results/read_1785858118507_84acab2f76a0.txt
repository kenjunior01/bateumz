    66→function formatCurrency(value: number, currency: string): string {
    67→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: currency || 'BRL' }).format(value);
    68→}

    70→function formatDate(dateStr: string): string {
    71→  return new Date(dateStr).toLocaleDateString('pt-BR', { day: '2-digit', month: 'short', year: 'numeric' });
    72→}
    73→
    74→function formatDateTime(dateStr: string): string {
    75→  return new Date(dateStr).toLocaleDateString('pt-BR', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' });
    76→}
    77→
    78→function timeAgo(dateStr: string): string {
    79→  const diff = Date.now() - new Date(dateStr).getTime();
    80→  const mins = Math.floor(diff / 60000);
    81→  if (mins < 1) return 'agora';
    82→  if (mins < 60) return `${mins}min atras`;
    83→  const hrs = Math.floor(mins / 60);
    84→  if (hrs < 24) return `${hrs}h atras`;
    85→  const days = Math.floor(hrs / 24);
    86→  return `${days}d atras`;
    87→}

    89→function getStatusColor(status: ChampStatus): string {
    90→  switch (status) {
    91→    case 'live': return 'bg-red-500/20 text-red-400 border-red-500/40';
    92→    case 'registration_open': return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/40';
    93→    case 'check_in': return 'bg-amber-500/20 text-amber-400 border-amber-500/40';
    94→    case 'registration_closed': return 'bg-orange-500/20 text-orange-400 border-orange-500/40';
    95→    case 'completed': return 'bg-zinc-500/20 text-zinc-400 border-zinc-500/40';
    96→    default: return 'bg-zinc-700/30 text-zinc-400 border-zinc-600/30';
    97→  }
    98→}

   100→function getTeamStatusLabel(status: string): { label: string; color: string } {
   101→  switch (status) {
   102→    case 'checked_in': return { label: 'Check-in Feito', color: 'bg-emerald-500/20 text-emerald-400' };
   103→    case 'registered': return { label: 'Registado', color: 'bg-blue-500/20 text-blue-400' };
   104→    case 'eliminated': return { label: 'Eliminado', color: 'bg-red-500/20 text-red-400' };

   106→    default: return { label: status, color: 'bg-zinc-700/30 text-zinc-400' };
   107→  }

  1045→function PredictionCard({ match, user, userId }: { match: EsportMatch; user: any; userId: string }) {

  1047→  const [resolved, setResolved] = useState<boolean | null>(null);
  1048→
  1049→  const handlePredict = async (winnerId: string) => {
  1050→    if (!userId || prediction) return;
  1051→    try {
  1052→      await makePrediction(match.id, userId, winnerId, 10);
  1053→      setPrediction(winnerId);
  1054→    } catch (err: any) {
  1055→      alert(err.message);
  1056→    }
  1057→  };
  1058→
  1059→  return (
  1060→    <Card className="border-zinc-800 bg-zinc-900/60">

  1075→              prediction === match.team1_id ? 'border-emerald-500/50 bg-emerald-500/10'
