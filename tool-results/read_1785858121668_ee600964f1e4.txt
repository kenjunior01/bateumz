   112→  const [loading, setLoading] = useState(true);
   113→  const [membersLoading, setMembersLoading] = useState(false);

   126→      setLoading(true);
   127→      const myTeams = await getTeams({ limit: 50 });

   137→          if (!memberTeamIds.includes(m.team_id)) memberTeamIds.push(m.team_id);
   138→        }

   140→      const memberTeams = myTeams.filter((t) => memberTeamIds.includes(t.id) && t.owner_id !== user.id);
   141→      setTeams([...owned, ...memberTeams]);
   142→    } catch (err: any) {
   143→      toast({ title: "Erro", description: err.message });
   144→    } finally {
   145→      setLoading(false);
   146→    }
   147→  }, [user, toast]);

   150→    loadTeams();

   166→        else profiles.push({ id: m.user_id, display_name: "Desconhecido", avatar_url: null });
   167→      }

   169→        ...m,
   170→        ...profiles[i],
   171→      }));
   172→      setMembers(merged);

   174→        .from("esport_champ_teams")

   177→        .order("registered_at", { ascending: false })
   178→        .limit(20);

   181→      toast({ title: "Erro", description: err.message });

   195→      toast({ title: "Tag invalida", description: "A tag deve ter entre 3 e 5 caracteres." });

   202→        tag: form.tag,

   204→        country: form.country || undefined,
   205→        discord_url: form.discord_url || undefined,
   206→        logo_url: form.logo_url || undefined,

   215→      toast({ title: "Erro ao criar time", description: err.message });

   217→      setFormLoading(false);
   218→    }
   219→  };

   231→        logo_url: form.logo_url || null,
   232→        banner_url: form.banner_url || null,

   248→      await removeTeamMember(selectedTeam.id, userId);

   287→      toast({ title: "Erro ao adicionar", description: err.message });

   289→      setAddMemberLoading(false);

   299→      country: selectedTeam.country || "",
   300→      discord_url: selectedTeam.discord_url || "",
   301→      logo_url: selectedTeam.logo_url || "",
   302→      banner_url: selectedTeam.banner_url || "",

   310→    if (rating >= 1200) return "text-yellow-400";
   311→    if (rating >= 1000) return "text-emerald-400";
   312→    return "text-zinc-400";
   313→  };
   314→

   317→    if (total === 0) return "0%";
   318→    return Math.round((w / total) * 100) + "%";
   319→  };
   320→
   321→  if (loading) {
   322→    return (
   323→      <div className="space-y-4 p-4">
   324→        <Skeleton className="h-8 w-48" />
   325→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   326→          {[1, 2, 3].map((i) => (
   327→            <Skeleton key={i} className="h-40 rounded-xl" />
   328→          ))}
   329→        </div>
   330→      </div>
   331→    );
   332→  }
   333→
   334→  return (
   335→    <div className="min-h-screen bg-background p-4 md:p-6">
   336→      <AnimatePresence mode="wait">

   338→          <motion.div
   339→            key="list"
   340→            initial={{ opacity: 0, y: 12 }}
   341→            animate={{ opacity: 1, y: 0 }}
   342→            exit={{ opacity: 0, y: -12 }}
   343→          >
   344→            <div className="flex items-center justify-between mb-6">
   345→              <div>
   346→                <h1 className="text-2xl font-bold flex items-center gap-2">
   347→                  <Shield className="w-6 h-6 text-primary" />
   348→                  Meus Times
   349→                </h1>
   350→                <p className="text-sm text-muted-foreground mt-1">
   351→                  Gerencie suas equipas de esports
   352→                </p>

   354→              <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   355→                <DialogTrigger asChild>
   356→                  <Button className="gap-2">
   357→                    <Plus className="w-4 h-4" />
   358→                    Criar Time
   359→                  </Button>
   360→                </DialogTrigger>
   361→                <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   362→                  <DialogHeader>
   363→                    <DialogTitle>Criar Novo Time</DialogTitle>

   365→                  <div className="space-y-4 pt-2">

   367→                      <Label>Nome do Time *</Label>
   368→                      <Input

   371→                        onChange={(e) => setForm({ ...form, name: e.target.value })}
   372→                      />
   373→                    </div>
   374→                    <div className="space-y-2">
   375→                      <Label>Tag (3-5 caracteres) *</Label>
   376→                      <Input
   377→                        placeholder="Ex: WLF"
   378→                        maxLength={5}
   379→                        className="uppercase"
   380→                        value={form.tag}
   381→                        onChange={(e) => setForm({ ...form, tag: e.target.value.toUpperCase() })}
   382→                      />
   383→                    </div>
   384→                    <div className="space-y-2">
   385→                      <Label>Descricao</Label>
   386→                      <Textarea
   387→                        placeholder="Descreva o seu time..."
   388→                        value={form.description}
   389→                        onChange={(e) => setForm({ ...form, description: e.target.value })}
   390→                      />
   391→                    </div>
   392→                    <div className="space-y-2">
   393→                      <Label>Pais</Label>
   394→                      <Input
   395→                        placeholder="Ex: Brasil"
   396→                        value={form.country}
   397→                        onChange={(e) => setForm({ ...form, country: e.target.value })}

   399→                    </div>
   400→                    <div className="space-y-2">
   401→                      <Label>Logo URL</Label>
   402→                      <Input
   403→                        placeholder="https://..."
   404→                        value={form.logo_url}
   405→                        onChange={(e) => setForm({ ...form, logo_url: e.target.value })}

   407→                    </div>
   408→                    <div className="space-y-2">
   409→                      <Label>Banner URL</Label>

   411→                        placeholder="https://..."
   412→                        value={form.banner_url}

   435→
   436→            {teams.length === 0 ? (

   440→                <p className="text-sm text-muted-foreground mb-4">

   445→                  Criar Time
   446→                </Button>

   472→                            </CardTitle>
   473→                            {team.tag && (

   478→                          </div>
   479→                        </div>
   480→                      </CardHeader>
   481→                      <CardContent className="space-y-3">

   483→                          <span className="text-muted-foreground flex items-center gap-1">

   485→                            Membros

   487→                          <span className="font-medium">--</span>

   510→                        {team.owner_id === user?.id && (

   515→                      </CardContent>

   517→                  </motion.div>

   519→              </div>

   521→          </motion.div>
