   856→export async function placeBet(data: {
   857→  user_id: string;
   858→  match_id?: string;
   859→  season_match_id?: string;
   860→  championship_id?: string;
   861→  bet_type: PredictionMarketType;
   862→  predicted_outcome: Record<string, any>;
   863→  predicted_label: string;
   864→  odds: number;
   865→  amount_wagered: number;
   866→}): Promise<Bet> {
   867→  const wallet = await getOrCreateWallet(data.user_id);
   868→  if (wallet.balance < data.amount_wagered) {
   869→    throw new Error('Saldo insuficiente');
   870→  }
   871→
   872→  const newBalance = wallet.balance - data.amount_wagered;
   873→  const potentialPayout = Math.floor(data.amount_wagered * data.odds);
   874→
   875→  const { data: bet, error } = await sb
   876→    .from('esport_bets')
   877→    .insert({
   878→      ...data,
   879→      potential_payout: potentialPayout,
   880→    })
   881→    .select()
   882→    .single();
   883→  if (error) throw new Error('Erro ao apostar: ' + error.message);
   884→
   885→  await sb
   886→    .from('user_wallets')
   887→    .update({
   888→      balance: newBalance,
   889→      total_wagered: wallet.total_wagered + data.amount_wagered,
   890→    })
   891→    .eq('user_id', data.user_id);
   892→
   893→  await addWalletTransaction({
   894→    user_id: data.user_id,
   895→    type: 'bet_placed',
   896→    amount: -data.amount_wagered,
   897→    balance_after: newBalance,
   898→    description: `Aposta: ${data.predicted_label} @ ${data.odds}x`,
   899→    reference_id: bet.id,
   900→  });
   901→
   902→  return bet;
   903→}

   905→export async function settleBet(betId: string, isCorrect: boolean): Promise<Bet> {
   906→  const { data: bet } = await sb.from('esport_bets').select('*').eq('id', betId).single();
   907→  if (!bet) throw new Error('Aposta nao encontrada');
   908→  if (bet.status !== 'open') throw new Error('Aposta ja resolvida');
   909→
   910→  const wallet = await getOrCreateWallet(bet.user_id);
   911→  let newBalance = wallet.balance;
   912→  let amountWon = 0;
   913→
   914→  if (isCorrect) {
   915→    amountWon = bet.potential_payout;
   916→    newBalance += amountWon;
   917→    await sb.from('user_wallets').update({
   918→      balance: newBalance,
   919→      total_earned: wallet.total_earned + amountWon,
   920→      total_won: wallet.total_won + amountWon,
   921→    }).eq('user_id', bet.user_id);
   922→
   923→    await addWalletTransaction({
   924→      user_id: bet.user_id,
   925→      type: 'bet_won',
   926→      amount: amountWon,
   927→      balance_after: newBalance,
   928→      description: `Ganhaste ${amountWon} moedas!`,
   929→      reference_id: betId,
   930→    });
   931→  } else {
   932→    await sb.from('user_wallets').update({
   933→      total_lost: wallet.total_lost + bet.amount_wagered,
   934→    }).eq('user_id', bet.user_id);
   935→  }
   936→
   937→  const { data: settled, error } = await sb
   938→    .from('esport_bets')
   939→    .update({
   940→      status: 'settled',
   941→      is_correct: isCorrect,
   942→      amount_won: amountWon,
   943→      settled_at: new Date().toISOString(),
   944→    })
   945→    .eq('id', betId)
   946→    .select()
   947→    .single();
   948→  if (error) throw new Error('Erro ao resolver aposta: ' + error.message);
   949→  return settled;
   950→}

  1434→export async function checkAndUnlockAchievement(userId: string, type: string, currentValue: number): Promise<UserAchievement | null> {
  1435→  const { data: achievements } = await sb
  1436→    .from('esport_achievements')
  1437→    .select('*')
  1438→    .eq('is_active', true);
  1439→  if (!achievements) return null;
  1440→
  1441→  for (const ach of achievements) {
  1442→    const req = ach.requirement as Record<string, any>;
  1443→    if (req.type !== type) continue;
  1444→    if (currentValue < req.count) continue;
  1445→
  1446→    const { data: existing } = await sb
  1447→      .from('user_achievements')
  1448→      .select('*')
  1449→      .eq('user_id', userId)
  1450→      .eq('achievement_id', ach.id)
  1451→      .single();
  1452→
  1453→    if (existing) {
  1454→      if (existing.is_completed) continue;
  1455→      await sb.from('user_achievements')
  1456→        .update({
  1457→          progress: Math.min(currentValue, req.count),
  1458→          is_completed: true,
  1459→          completed_at: new Date().toISOString(),
  1460→        })
  1461→        .eq('id', existing.id);
  1462→
  1463→      if (ach.reward_coins > 0) {
  1464→        await sb.raw(
  1465→          `UPDATE user_wallets SET balance = balance + ?, total_earned = total_earned + ? WHERE user_id = ?`,
  1466→          [ach.reward_coins, ach.reward_coins, userId]
  1467→        );
  1468→        await addWalletTransaction({
  1469→          user_id: userId,
  1470→          type: 'achievement_reward',
  1471→          amount: ach.reward_coins,
  1472→          balance_after: 0,
  1473→          description: `Conquista desbloqueada: ${ach.name}! +${ach.reward_coins} moedas`,
  1474→          reference_id: ach.id,
  1475→        });
  1476→      }
  1477→
  1478→      if (ach.reward_badge) {
  1479→        await sb.raw(
  1480→          `UPDATE user_reputation SET earned_badges = array_append(earned_badges, ?) WHERE user_id = ? AND NOT ? = ANY(earned_badges)`,
  1481→          [ach.reward_badge, userId, ach.reward_badge]
  1482→        );
  1483→      }
  1484→
  1485→      return { ...existing, is_completed: true, completed_at: new Date().toISOString() };
  1486→    } else {
  1487→      const isCompleted = currentValue >= req.count;
  1488→      const { data: ua, error } = await sb
  1489→        .from('user_achievements')
  1490→        .insert({
  1491→          user_id: userId,
  1492→          achievement_id: ach.id,
  1493→          progress: Math.min(currentValue, req.count),
  1494→          target: req.count,
  1495→          is_completed: isCompleted,
  1496→          completed_at: isCompleted ? new Date().toISOString() : undefined,
  1497→        })
  1498→        .select()
  1499→        .single();
  1500→      if (error) continue;
  1501→
  1502→      if (isCompleted && ach.reward_coins > 0) {
  1503→        await sb.raw(
  1504→          `UPDATE user_wallets SET balance = balance + ?, total_earned = total_earned + ? WHERE user_id = ?`,
  1505→          [ach.reward_coins, ach.reward_coins, userId]
  1506→        );
  1507→      }
  1508→
  1509→      return ua;
  1510→    }
  1511→  }
  1512→
  1513→  return null;
  1514→}
