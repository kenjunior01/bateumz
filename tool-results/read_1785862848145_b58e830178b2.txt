    42→function ShimmerText({ children, className }: { children: string; className?: string }) {
    43→  return (
    44→    <span className={className}>
    45→      {children.split('').map((char, i) => (
    46→        <motion.span
    47→          key={i}
    48→          initial={{ opacity: 0.3 }}
    49→          animate={{ opacity: [0.3, 1, 0.3] }}
    50→          transition={{ duration: 2.5, repeat: Infinity, delay: i * 0.06 }}
    51→          style={{ display: 'inline-block' }}
    52→        >
    53→          {char === ' ' ? '\u00A0' : char}
    54→        </motion.span>
    55→      ))}
    56→    </span>
    57→  );
    58→}

    60→function FloatingIcon({ emoji, delay, x, y }: { emoji: string; delay: number; x: number; y: number }) {
    61→  return (
    62→    <motion.div
    63→      className="absolute text-3xl md:text-5xl opacity-10 pointer-events-none select-none"

    71→    >

    73→    </motion.div>
    74→  );

    77→function formatCurrency(value: number, currency: string): string {

    81→function formatDate(dateStr: string): string {

    85→function getStatusColor(status: ChampStatus): string {
    86→  switch (status) {
    87→    case 'live': return 'bg-red-500/20 text-red-400 border-red-500/40';
    88→    case 'registration_open': return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/40';
    89→    case 'check_in': return 'bg-amber-500/20 text-amber-400 border-amber-500/40';
    90→    case 'registration_closed': return 'bg-orange-500/20 text-orange-400 border-orange-500/40';
    91→    case 'completed': return 'bg-zinc-500/20 text-zinc-400 border-zinc-500/40';
    92→    default: return 'bg-zinc-700/30 text-zinc-400 border-zinc-600/30';
    93→  }

   434→                  ) : tabFilter(tab).length === 0 ? (
   435→                    <motion.div
   436→                      initial={{ opacity: 0 }}
   437→                      animate={{ opacity: 1 }}
   438→                      className="text-center py-16"

   441→                      <p className="text-zinc-500 text-lg font-medium">Nenhum campeonato encontrado</p>

   445→                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   446→                      <AnimatePresence mode="popLayout">
   447→                        {tabFilter(tab).map((champ) => (
   448→                          <ChampionshipCard
   449→                            key={champ.id}
   450→                            champ={champ}
   451→                            gameEmoji={gameEmoji(champ.game_id)}
   452→                            gameLabel={gameName(champ.game_id)}
   453→                          />
   454→                        ))}
   455→                      </AnimatePresence>
   456→                    </div>
   457→                  )}
   458→                </TabsContent>
   459→              ))}
   460→            </Tabs>
   461→          </div>
   462→
   463→          {/* Sidebar */}
   464→          <aside className="w-full lg:w-80 flex-shrink-0 space-y-6">
   465→            <TopTeamsSection teams={topTeams} loading={loading} />
   466→            <NextMatchesSection matches={recentMatches} loading={loading} />
   467→            <ActivityFeedSection activities={activityFeed} loading={loading} />
   468→          </aside>
   469→        </div>
   470→      </section>
   471→    </div>
   472→  );
   473→}
