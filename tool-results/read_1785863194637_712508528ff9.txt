    49→// Helper to get a mock user id (in production, this comes from auth)
    50→function useUserId() {
    51→  const [userId, setUserId] = useState<string | null>(null);
    52→
    53→  useEffect(() => {
    54→    async function fetchUser() {
    55→      try {
    56→        const { data: { user } } = await supabase.auth.getUser();
    57→        if (user) {
    58→          setUserId(user.id);
    59→        }
    60→      } catch {
    61→        // silent fail
    62→      }
    63→    }
    64→    fetchUser();
    65→  }, []);
    66→
    67→  return userId;
    68→}

    70→// Format time ago in Portuguese
    71→function timeAgo(dateStr: string): string {
    72→  const now = new Date();
    73→  const date = new Date(dateStr);
    74→  const diffMs = now.getTime() - date.getTime();
    75→  const diffSec = Math.floor(diffMs / 1000);
    76→  const diffMin = Math.floor(diffSec / 60);
    77→  const diffHour = Math.floor(diffMin / 60);
    78→  const diffDay = Math.floor(diffHour / 24);
    79→
    80→  if (diffSec < 60) return 'agora mesmo';
    81→  if (diffMin < 60) return `ha ${diffMin} min`;
    82→  if (diffHour < 24) return `ha ${diffHour}h`;
    83→  if (diffDay < 7) return `ha ${diffDay}d`;
    84→  return `ha ${Math.floor(diffDay / 7)} semanas`;
    85→}

    87→// Get status badge color
    88→function getStatusColor(status: BetStatus): string {
    89→  switch (status) {
    90→    case 'open':
    91→      return 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30';
    92→    case 'settled':
    93→      return 'bg-green-500/20 text-green-400 border-green-500/30';
    94→    case 'refunded':
    95→      return 'bg-blue-500/20 text-blue-400 border-blue-500/30';
    96→    case 'cancelled':
    97→      return 'bg-red-500/20 text-red-400 border-red-500/30';
    98→    default:
    99→      return 'bg-white/10 text-white border-white/20';
   100→  }
   101→}
