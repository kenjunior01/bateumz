    57→    57→        if (user) {

    70→    70→// Format time ago in Portuguese
    71→    71→function timeAgo(dateStr: string): string {
    72→    72→  const now = new Date();
    73→    73→  const date = new Date(dateStr);
    74→    74→  const diffMs = now.getTime() - date.getTime();
    75→    75→  const diffSec = Math.floor(diffMs / 1000);
    76→    76→  const diffMin = Math.floor(diffSec / 60);
    77→    77→  const diffHour = Math.floor(diffMin / 60);
    78→    78→  const diffDay = Math.floor(diffHour / 24);
    79→    79→
    80→    80→  if (diffSec < 60) return 'agora mesmo';
    81→    81→  if (diffMin < 60) return `ha ${diffMin} min`;
    82→    82→  if (diffHour < 24) return `ha ${diffHour}h`;
    83→    83→  if (diffDay < 7) return `ha ${diffDay}d`;
    84→    84→  return `ha ${Math.floor(diffDay / 7)} semanas`;
    85→    85→}
