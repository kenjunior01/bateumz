   103→function getWinRate(wins: number, total: number): number {
   104→  if (total === 0) return 0;
   105→  return Math.round((wins / total) * 100);
   106→}
   107→
   108→function getPositionColor(pos: number): string {
   109→  if (pos === 1) return 'bg-amber-500/20 border-amber-500/30';
   110→  if (pos === 2) return 'bg-zinc-400/15 border-zinc-400/25';
   111→  if (pos === 3) return 'bg-orange-700/15 border-orange-700/25';
   112→  if (pos <= 10) return 'bg-emerald-500/10 border-emerald-500/20';
   113→  return 'bg-white/5 border-white/10';
   114→}
   115→
   116→function getPositionTextColor(pos: number): string {
   117→  if (pos === 1) return 'text-amber-400';
   118→  if (pos === 2) return 'text-zinc-300';
   119→  if (pos === 3) return 'text-orange-400';
   120→  if (pos <= 10) return 'text-emerald-400';

   131→function getRatingColor(rating: number): string {

   147→function getInitials(name: string): string {
   148→  return name
   149→    .split(' ')
   150→    .map((w) => w[0])

   156→function getTrendDirection(index: number): 'up' | 'down' | 'stable' {
   157→  const patterns = [0, 1, 2, 4, 6, 9, 11, 14, 17, 19];
   158→  if (patterns.includes(index % 20)) return 'up';
   159→  if (patterns.includes((index + 5) % 20)) return 'down';
   160→  return 'stable';
   161→}

   213→export default function LeaderboardPage() {

   218→

   220→  const [teams, setTeams] = useState<EsportTeam[]>([]);
   221→  const [reputations, setReputations] = useState<UserReputation[]>([]);
   222→  const [loading, setLoading] = useState(true);
   223→
   224→  // Fetch data on mount
   225→  useEffect(() => {
   226→    async function fetchData() {
   227→      setLoading(true);
   228→      try {
   229→        const [playerData, teamData, repData] = await Promise.all([
   230→          getPlayerLeaderboard(50).catch(() => []),
   231→          getTeams({ limit: 50, sort_by: 'rating', sort_dir: 'desc' }).catch(() => []),
   232→          getTopReputation(50).catch(() => []),
   233→        ]);
   234→
   235→        setPlayers(playerData.length > 0 ? playerData : MOCK_PLAYERS);
   236→        setTeams(teamData.length > 0 ? teamData : MOCK_TEAMS);
   237→        setReputations(repData.length > 0 ? repData : MOCK_REPUTATIONS);
   238→      } catch {
   239→        setPlayers(MOCK_PLAYERS);
   240→        setTeams(MOCK_TEAMS);
   241→        setReputations(MOCK_REPUTATIONS);
   242→      } finally {
   243→        setLoading(false);
   244→      }
   245→    }
   246→    fetchData();
   247→  }, []);
   248→
   249→  // Compute stats
   250→  const stats = useMemo(() => {
   251→    const totalJogadores = players.length;
   252→    const totalEquipas = teams.length;
   253→    const maiorKarma = players.length > 0 ? Math.max(...players.map((p) => p.karma_points)) : 0;
   254→    const mediaMatches =
   255→      players.length > 0
   256→        ? Math.round(players.reduce((acc, p) => acc + p.total_matches, 0) / players.length)
   257→        : 0;
   258→    return { totalJogadores, totalEquipas, maiorKarma, mediaMatches };
   259→  }, [players, teams]);
   260→
   261→  // Sort toggle
   262→  const handleSort = useCallback(
   263→    (field: string) => {
   264→      if (sortField === field) {
   265→        setSortDir((d) => (d === 'desc' ? 'asc' : 'desc'));
   266→      } else {
   267→        setSortField(field);
   268→        setSortDir('desc');
   269→      }
   270→    },
   271→    [sortField]
   272→  );
   273→
   274→  // Filtered + sorted players
   275→  const filteredPlayers = useMemo(() => {
   276→    let result = [...players];
   277→    if (searchQuery.trim()) {
   278→      const q = searchQuery.toLowerCase();
   279→      result = result.filter((p) => (p.username || '').toLowerCase().includes(q));
   280→    }
   281→    if (sortField === 'karma') {
   282→      result.sort((a, b) =>
   283→        sortDir === 'desc' ? b.karma_points - a.karma_points : a.karma_points - b.karma_points
   284→      );
   285→    } else if (sortField === 'matches') {
   286→      result.sort((a, b) =>
   287→        sortDir === 'desc' ? b.total_matches - a.total_matches : a.total_matches - b.total_matches
   288→      );
   289→    } else if (sortField === 'winrate') {
   290→      result.sort((a, b) => {
   291→        const wrA = getWinRate(a.wins, a.total_matches);
   292→        const wrB = getWinRate(b.wins, b.total_matches);
   293→        return sortDir === 'desc' ? wrB - wrA : wrA - wrB;
   294→      });
   295→    }
   296→    return result;
   297→  }, [players, searchQuery, sortField, sortDir]);
   298→
   299→  // Filtered + sorted teams
   300→  const filteredTeams = useMemo(() => {
   301→    let result = [...teams];
   302→    if (searchQuery.trim()) {
   303→      const q = searchQuery.toLowerCase();
   304→      result = result.filter(
   305→        (t) =>
   306→          t.name.toLowerCase().includes(q) ||
   307→          (t.tag || '').toLowerCase().includes(q)
   308→      );
   309→    }
   310→    if (sortField === 'rating' || sortField === 'karma') {
   311→      result.sort((a, b) =>
   312→        sortDir === 'desc' ? b.rating - a.rating : a.rating - b.rating
   313→      );
   314→    } else if (sortField === 'wins') {
   315→      result.sort((a, b) =>
   316→        sortDir === 'desc' ? b.total_wins - a.total_wins : a.total_wins - b.total_wins
   317→      );
   318→    } else if (sortField === 'tournaments') {
   319→      result.sort((a, b) =>
   320→        sortDir === 'desc'
   321→          ? b.total_tournaments - a.total_tournaments
   322→          : a.total_tournaments - b.total_tournaments
   323→      );
   324→    }
   325→    return result;
   326→  }, [teams, searchQuery, sortField, sortDir]);
   327→
   328→  // Top 3 players for podium
   329→  const top3Players = filteredPlayers.slice(0, 3);
   330→  const restPlayers = filteredPlayers.slice(3);
   331→
   332→  // Top 3 teams for podium
   333→  const top3Teams = filteredTeams.slice(0, 3);
   334→  const restTeams = filteredTeams.slice(3);
   335→
   336→  // Build reputation lookup
   337→  const repMap = useMemo(() => {
   338→    const map: Record<string, UserReputation> = {};
   339→    reputations.forEach((r) => {
   340→      map[r.user_id] = r;
   341→    });
   342→    return map;
   343→  }, [reputations]);

   346→    <div className="min-h-screen bg-[#0a0a0f] text-white">
   347→      {/* HERO SECTION */}
   348→      <section className="relative overflow-hidden py-12 md:py-20 px-4">
   349→        <div className="absolute inset-0 bg-gradient-to-b from-purple-500/5 via-transparent to-transparent" />
   350→        <div className="absolute top-20 left-1/4 w-72 h-72 bg-purple-500/5 rounded-full blur-3xl" />
   351→        <div className="absolute top-32 right-1/4 w-56 h-56 bg-amber-500/5 rounded-full blur-3xl" />
   352→
   353→        <div className="relative z-10 max-w-5xl mx-auto text-center">
   354→          <motion.div
   355→            initial={{ opacity: 0, y: -20 }}
   356→            animate={{ opacity: 1, y: 0 }}
   357→            transition={{ duration: 0.6 }}
   358→          >
   359→            <div className="inline-flex items-center gap-2 px-4 py-1.5 rounded-full bg-white/5 border border-white/10 mb-6">
   360→              <Trophy className="w-4 h-4 text-amber-400" />
   361→              <span className="text-xs font-medium text-zinc-300 tracking-wider uppercase">
   362→                Classificacao Global
   363→              </span>
   364→            </div>
   365→
   366→            <h1 className="text-4xl md:text-6xl lg:text-7xl font-black tracking-tight mb-4">
   367→              <span
   368→                className="bg-clip-text text-transparent"
   369→                style={{
   370→                  backgroundImage: 'linear-gradient(135deg, #f59e0b, #ef4444, #ec4899, #8b5cf6, #3b82f6, #f59e0b)',
   371→                  backgroundSize: '300% 300%',
   372→                  animation: 'shimmer 4s ease-in-out infinite',
