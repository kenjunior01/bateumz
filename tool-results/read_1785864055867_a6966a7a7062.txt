   436→function calcularTaxaVitorias(vitorias: number, total: number): number {
   437→  if (total === 0) return 0;
   438→  return Math.round((vitorias / total) * 100);
   439→}

   441→function obterIniciais(nome: string): string {
   442→  return nome
   443→    .split(' ')
   444→    .map((palavra) => palavra[0])
   445→    .join('')
   446→    .toUpperCase()
   447→    .slice(0, 2);
   448→}
   449→
   450→function obterDireccaoTendencia(indice: number): 'up' | 'down' | 'stable' {
   451→  const padroes = [0, 1, 2, 4, 6, 9, 11, 14];
   452→  if (padroes.includes(indice % 20)) return 'up';
   453→  if (padroes.includes((indice + 5) % 20)) return 'down';
   454→  return 'stable';
   455→}

   457→function obterCorReputacao(titulo: string): string {
   458→  if (titulo.includes('Lenda')) return 'bg-red-500/20 text-red-400 border border-red-500/30';
   459→  if (titulo.includes('Mestre')) return 'bg-purple-500/20 text-purple-400 border border-purple-500/30';
   460→  if (titulo.includes('Elite')) return 'bg-amber-500/20 text-amber-400 border border-amber-500/30';
   461→  if (titulo.includes('Veterano')) return 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30';
   462→  if (titulo.includes('Experiente')) return 'bg-sky-500/20 text-sky-400 border border-sky-500/30';
   463→  return 'bg-zinc-500/20 text-zinc-400 border border-zinc-500/30';
   464→}
   465→
   466→function obterCorGlowReputacao(titulo: string): string {
   467→  if (titulo.includes('Lenda')) return '0 0 8px rgba(239, 68, 68, 0.4), 0 0 16px rgba(239, 68, 68, 0.2)';
   468→  if (titulo.includes('Mestre')) return '0 0 8px rgba(168, 85, 247, 0.4), 0 0 16px rgba(168, 85, 247, 0.2)';
   469→  if (titulo.includes('Elite')) return '0 0 8px rgba(245, 158, 11, 0.4), 0 0 16px rgba(245, 158, 11, 0.2)';
   470→  return 'none';
   471→}

   473→function obterCorBarraProgresso(taxa: number): string {
   474→  if (taxa >= 85) return 'bg-gradient-to-r from-amber-400 to-yellow-300';
   475→  if (taxa >= 75) return 'bg-gradient-to-r from-emerald-400 to-green-300';
   476→  if (taxa >= 65) return 'bg-gradient-to-r from-cyan-400 to-sky-300';
   477→  if (taxa >= 50) return 'bg-gradient-to-r from-violet-400 to-purple-300';
   478→  return 'bg-gradient-to-r from-zinc-400 to-zinc-300';
   479→}
