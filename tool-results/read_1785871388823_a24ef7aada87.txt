    79→  const [loading, setLoading] = useState(true);
    80→  const [games, setGames] = useState<Game[]>([]);

    84→
    85→  // Form states
    86→  const [gameName, setGameName] = useState("");
    87→  const [gameDescription, setGameDescription] = useState("");
    88→  const [totalQuestions, setTotalQuestions] = useState(15);
    89→  const [timePerQuestion, setTimePerQuestion] = useState(30);

   103→  const [optionC, setOptionC] = useState("");
   104→  const [optionD, setOptionD] = useState("");
   105→  const [correctAnswer, setCorrectAnswer] = useState("A");
   106→  const [explanation, setExplanation] = useState("");

   110→    if (role !== "admin" && role !== "superadmin" && role !== "business") return;

   123→      if (error) throw error;

   135→      const { data, error } = await supabase
   136→        .from("millionaire_questions")
   137→        .select("*")
   138→        .eq("game_id", gameId)
   139→        .order("question_number");

   141→      if (error) throw error;
   142→      setQuestions(data || []);
   143→    } catch (error) {
   144→      console.error("Error loading questions:", error);
   145→      toast.error("Erro ao carregar perguntas");
   146→    }

   150→    setSelectedGame(game);
   151→    setGameName(game.name);
   152→    setGameDescription(game.description || "");

   158→    setCompanyLogoUrl(game.company_logo_url || "");

   160→    setPrizeStructure(game.prize_structure || DEFAULT_PRIZE_STRUCTURE);

   185→        .from("millionaire_games")

   199→            'ask_audience': lifelines.askAudience,

   217→      console.error("Error creating game:", error);
   218→      toast.error("Erro ao criar jogo");
   219→    } finally {
   220→      setSaving(false);
   221→    }
   222→  };

   225→    if (!selectedGame || !gameName.trim()) {
   226→      toast.error("Nome do jogo é obrigatório");
   227→      return;
   228→    }
   229→
   230→    setSaving(true);

   232→      const adjustedPrizeStructure = prizeStructure.slice(0, totalQuestions);
   233→      
   234→      const { error } = await supabase
   235→        .from("millionaire_games")

   238→          description: gameDescription,
   239→          total_questions: totalQuestions,
   240→          time_per_question: timePerQuestion,
   241→          primary_color: primaryColor,
   242→          background_color: backgroundColor,
   243→          background_image_url: backgroundImage,
   244→          company_logo_url: companyLogoUrl,
   245→          company_slogan: companySlogan,
   246→          prize_structure: adjustedPrizeStructure,
   247→          lifelines: {
   248→            '50_50': lifelines.fiftyFifty,
   249→            'ask_audience': lifelines.askAudience,

   256→
   257→      setGames(games.map(g => g.id === selectedGame.id ? { ...g, name: gameName } : g));

   259→    } catch (error) {
   260→      console.error("Error updating game:", error);
   261→      toast.error("Erro ao atualizar jogo");
   262→    } finally {
   263→      setSaving(false);
   264→    }
   265→  };

   268→    if (!selectedGame || !questionText.trim()) {
   269→      toast.error("Pergunta e opções são obrigatórias");

   271→    }
   272→
   273→    setSaving(true);
   274→    try {
   275→      const questionNumber = questions.length + 1;
   276→

   278→        .from("millionaire_questions")
   279→        .insert({
   280→          game_id: selectedGame.id,
   281→          question_number: questionNumber,
   282→          question_text: questionText,
   283→          option_a: optionA,
   284→          option_b: optionB,
   285→          option_c: optionC,
   286→          option_d: optionD,
   287→          correct_answer: correctAnswer,
   288→          explanation: explanation,
   289→        })
   290→        .select()

   293→      if (error) throw error;
   294→
   295→      setQuestions([...questions, data]);
   296→      resetQuestionForm();
   297→      toast.success("Pergunta adicionada!");

   302→      setSaving(false);
   303→    }
   304→  };

   308→      const { error } = await supabase
   309→        .from("millionaire_questions")

   317→    } catch (error) {
   318→      console.error("Error deleting question:", error);
   319→      toast.error("Erro ao remover pergunta");
   320→    }
   321→  };
   322→

   324→    try {
   325→      const { error } = await supabase
   326→        .from("millionaire_games")
   327→        .update({ is_published: true })
   328→        .eq("id", gameId);
   329→
   330→      if (error) throw error;
   331→
   332→      setGames(
   333→        games.map((g) =>
   334→          g.id === gameId ? { ...g, is_published: true } : g
   335→        )
   336→      );
   337→      toast.success("Jogo publicado!");

   339→      console.error("Error publishing game:", error);

   345→    setGameName("");
   346→    setGameDescription("");
   347→    setTotalQuestions(15);
   348→    setTimePerQuestion(30);
   349→    setPrimaryColor("#fbbf24");
   350→    setBackgroundColor("#0a0e17");
   351→    setBackgroundImage("");
   352→    setCompanyLogoUrl("");
   353→    setCompanySlogan("");

   355→    setLifelines({ fiftyFifty: true, askAudience: false, phoneFriend: false });
   356→  };
   357→
   358→  const resetQuestionForm = () => {
   359→    setQuestionText("");
   360→    setOptionA("");
   361→    setOptionB("");
   362→    setOptionC("");
   363→    setOptionD("");
   364→    setCorrectAnswer("A");
   365→    setExplanation("");
   366→  };
   367→
   368→  const updatePrizeLevel = (index: number, updates: Partial<PrizeLevel>) => {
   369→    setPrizeStructure(prev => prev.map((level, i) => 
   370→      i === index ? { ...level, ...updates } : level

   372→  };
   373→
   374→  if (role !== "admin" && role !== "superadmin" && role !== "business") {
   375→    return <Navigate to="/profile" replace />;

   377→

   380→      <div className="flex h-64 items-center justify-center">

   402→            </Card>

   404→            <div className="grid gap-4">
   405→              {games.map((game) => (
   406→                <Card
   407→                  key={game.id}
   408→                  className="cursor-pointer hover:shadow-lg transition-shadow"
   409→                  onClick={() => selectGameForEdit(game)}
   410→                >
   411→                  <CardHeader>
   412→                    <div className="flex items-start justify-between">
   413→                      <div>
   414→                        <CardTitle>{game.name}</CardTitle>
   415→                        <p className="text-sm text-muted-foreground mt-2">
   416→                          {game.description}
   417→                        </p>
   418→                      </div>
   419→                      <div className="flex gap-2">
   420→                        <Badge variant={game.is_published ? "default" : "outline"}>

   422→                        </Badge>
   423→                      </div>
   424→                    </div>

   426→                  <CardContent>
   427→                    <div className="flex items-center justify-between">
   428→                      <p className="text-sm text-muted-foreground">
   429→                        {game.total_questions} perguntas

   435→                        }}>

   439→                        {!game.is_published && (
   440→                          <Button
   441→                            size="sm"

   444→                              publishGame(game.id);

   448→                            Publicar

   454→                            Pré-visualizar
   455→                          </a>

   541→                    />
   542→                    <Input
