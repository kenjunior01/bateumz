   154→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
   155→  const [count, setCount] = useState(0);
   156→  const ref = useRef<HTMLSpanElement>(null);
   157→  const inView = useInView(ref, { once: true });
   158→  const hasAnimated = useRef(false);
   159→
   160→  useEffect(() => {
   161→    if (!inView || hasAnimated.current) return;
   162→    hasAnimated.current = true;
   163→    const start = performance.now();
   164→    const step = (now: number) => {
   165→      const elapsed = (now - start) / 1000;
   166→      const progress = Math.min(elapsed / duration, 1);
   167→      const eased = 1 - Math.pow(1 - progress, 3);
   168→      setCount(Math.round(eased * target));
   169→      if (progress < 1) requestAnimationFrame(step);
   170→    };
   171→    requestAnimationFrame(step);
   172→  }, [inView, target, duration]);
   173→
   174→  return <span ref={ref}>{count}{suffix}</span>;
   175→}
