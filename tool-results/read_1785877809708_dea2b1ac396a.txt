   165→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
   166→  const [count, setCount] = useState(0);
   167→  const ref = useRef<HTMLSpanElement>(null);
   168→  const inView = useInView(ref, { once: true });
   169→  const hasAnimated = useRef(false);
   170→
   171→  useEffect(() => {
   172→    if (!inView || hasAnimated.current) return;
   173→    hasAnimated.current = true;
   174→    const start = performance.now();
   175→    const step = (now: number) => {
   176→      const elapsed = (now - start) / 1000;
   177→      const progress = Math.min(elapsed / duration, 1);
   178→      const eased = 1 - Math.pow(1 - progress, 3);
   179→      setCount(Math.round(eased * target));
   180→      if (progress < 1) requestAnimationFrame(step);
   181→    };
   182→    requestAnimationFrame(step);
   183→  }, [inView, target, duration]);
   184→
   185→  return <span ref={ref}>{count}{suffix}</span>;
   186→}
