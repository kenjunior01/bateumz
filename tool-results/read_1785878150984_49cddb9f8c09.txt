   118→   118→/** BFS from start to target, returning the first step direction (or null). */
   119→   119→function bfsDirection(
   120→   120→  start: Point,
   121→   121→  target: Point,
   122→   122→  obstacles: Set<string>
   123→   123→): Direction | null {
   124→   124→  const queue: { pos: Point; firstDir: Direction | null }[] = [
   125→   125→    { pos: start, firstDir: null },
   126→   126→  ];
   127→   127→  const visited = new Set<string>();
   128→   128→  visited.add(key(start));
   129→   129→
   130→   130→  while (queue.length > 0) {
   131→   131→    const { pos, firstDir } = queue.shift()!;
   132→   132→
   133→   133→    for (const dir of ALL_DIRECTIONS) {
   134→   134→      const delta = DIRECTION_DELTA[dir];
   135→   135→      const next: Point = { x: pos.x + delta.x, y: pos.y + delta.y };
   136→   136→
   137→   137→      if (next.x < 0 || next.x >= GRID_SIZE || next.y < 0 || next.y >= GRID_SIZE) continue;
   138→   138→      if (obstacles.has(key(next))) continue;
   139→   139→      if (visited.has(key(next))) continue;
   140→   140→
   141→   141→      const newFirstDir = firstDir ?? dir;
   142→   142→
   143→   143→      if (pointsEqual(next, target)) {
   144→   144→        return newFirstDir;
   145→   145→      }
   146→   146→
   147→   147→      visited.add(key(next));
   148→   148→      queue.push({ pos: next, firstDir: newFirstDir });
   149→   149→    }
   150→   150→  }
   151→   151→  return null;
   152→   152→}

   154→   154→/** BFS that avoids obstacles but also avoids cells adjacent to the player snake's head (cut-off avoidance). */
   155→   155→function bfsSmartDirection(
   156→   156→  aiSnake: Snake,
   157→   157→  target: Point,
   158→   158→  obstacles: Set<string>,
   159→   159→  playerSnake: Snake
   160→   160→): Direction | null {
   161→   161→  const playerHead = playerSnake.body[0];
   162→   162→  const dangerZone = new Set<string>();
   163→   163→  // Mark cells adjacent to player head as dangerous (avoid getting cut off)
   164→   164→  for (const dir of ALL_DIRECTIONS) {
   165→   165→    const delta = DIRECTION_DELTA[dir];
   166→   166→    dangerZone.add(`${playerHead.x + delta.x},${playerHead.y + delta.y}`);
   167→   167→  }
   168→   168→
   169→   169→  const queue: { pos: Point; firstDir: Direction | null }[] = [
   170→   170→    { pos: aiSnake.body[0], firstDir: null },
   171→   171→  ];
   172→   172→  const visited = new Set<string>();
   173→   173→  visited.add(key(aiSnake.body[0]));
   174→   174→
   175→   175→  while (queue.length > 0) {
   176→   176→    const { pos, firstDir } = queue.shift()!;
   177→   177→
   178→   178→    for (const dir of ALL_DIRECTIONS) {
   179→   179→      const delta = DIRECTION_DELTA[dir];
   180→   180→      const next: Point = { x: pos.x + delta.x, y: pos.y + delta.y };
   181→   181→
   182→   182→      if (next.x < 0 || next.x >= GRID_SIZE || next.y < 0 || next.y >= GRID_SIZE) continue;
   183→   183→      if (obstacles.has(key(next))) continue;
   184→   184→      if (visited.has(key(next))) continue;
   185→   185→
   186→   186→      // For smart difficulty, deprioritize but don't completely block danger zones
   187→   187→      // We handle this by preferring non-dangerous paths through ordering
   188→   188→
   189→   189→      const newFirstDir = firstDir ?? dir;
   190→   190→
   191→   191→      if (pointsEqual(next, target)) {
   192→   192→        return newFirstDir;
   193→   193→      }
   194→   194→
   195→   195→      visited.add(key(next));
   196→   196→
   197→   197→      // Prioritize safe cells (push dangerous ones to back of queue)
   198→   198→      const entry = { pos: next, firstDir: newFirstDir };
   199→   199→      if (dangerZone.has(key(next))) {
   200→   200→        queue.push(entry); // lower priority
   201→   201→      } else {
   202→   202→        queue.unshift(entry); // higher priority
   203→   203→      }
   204→   204→    }
   205→   205→  }
   206→   206→
   207→   207→  // Fallback: try to find any safe direction (survival mode)
   208→   208→  return bfsSurvivalDirection(aiSnake, obstacles);
   209→   209→}

   211→   211→/** Fallback: find any direction that doesn't immediately kill the snake. */
   212→   212→function bfsSurvivalDirection(
   213→   213→  snake: Snake,
   214→   214→  obstacles: Set<string>
   215→   215→): Direction | null {
   216→   216→  const head = snake.body[0];
   217→   217→  for (const dir of ALL_DIRECTIONS) {
   218→   218→    if (dir === OPPOSITE[snake.direction]) continue;
   219→   219→    const delta = DIRECTION_DELTA[dir];
   220→   220→    const next: Point = { x: head.x + delta.x, y: head.y + delta.y };
   221→   221→    if (
   222→   222→      next.x >= 0 && next.x < GRID_SIZE &&
   223→   223→      next.y >= 0 && next.y < GRID_SIZE &&
   224→   224→      !obstacles.has(key(next))
   225→   225→    ) {
   226→   226→      return dir;
   227→   227→    }
   228→   228→  }
   229→   229→  return null;
   230→   230→}

   232→   232→/** Compute the best AI direction based on difficulty. */
   233→   233→function getAIDirection(
   234→   234→  aiSnake: Snake,
   235→   235→  playerSnake: Snake,
   236→   236→  food: Point,
   237→   237→  difficulty: Difficulty,
   238→   238→  tickCount: number
   239→   239→): Direction {
   240→   240→  const config = AI_CONFIG[difficulty];
   241→   241→
   242→   242→  // Reaction delay: AI doesn't react for the first N ticks
   243→   243→  if (tickCount > 0 && tickCount <= config.reactionDelay) {
   244→   244→    return aiSnake.direction;
   245→   245→  }
   246→   246→
   247→   247→  // Build obstacle set (all snake bodies)
   248→   248→  const obstacles = new Set<string>();
   249→   249→  for (const p of aiSnake.body) obstacles.add(key(p));
   250→   250→  for (const p of playerSnake.body) obstacles.add(key(p));
   251→   251→  // Remove tail tip since it will move (unless growing)
   252→   252→  // Simple: keep full body as obstacle for safety
   253→   253→
   254→   254→  // Random chance to make a random move
   255→   255→  if (Math.random() < config.randomChance) {
   256→   256→    const safeDirs = ALL_DIRECTIONS.filter((dir) => {
   257→   257→      if (dir === OPPOSITE[aiSnake.direction]) return false;
   258→   258→      const delta = DIRECTION_DELTA[dir];
   259→   259→      const next: Point = { x: aiSnake.body[0].x + delta.x, y: aiSnake.body[0].y + delta.y };
   260→   260→      return (
   261→   261→        next.x >= 0 && next.x < GRID_SIZE &&
   262→   262→        next.y >= 0 && next.y < GRID_SIZE &&
   263→   263→        !obstacles.has(key(next))
   264→   264→      );
   265→   265→    });
   266→   266→    if (safeDirs.length > 0) {
   267→   267→      return safeDirs[Math.floor(Math.random() * safeDirs.length)];
   268→   268→    }
   269→   269→  }
   270→   270→
   271→   271→  let chosenDir: Direction | null;
   272→   272→
   273→   273→  if (difficulty === "facil") {
   274→   274→    // Easy: simple greedy — just move towards food if possible
   275→   275→    chosenDir = bfsDirection(aiSnake.body[0], food, obstacles);
   276→   276→  } else if (difficulty === "medio") {
   277→   277→    // Medium: BFS to food
   278→   278→    chosenDir = bfsDirection(aiSnake.body[0], food, obstacles);
   279→   279→  } else {
   280→   280→    // Hard: smart BFS with avoidance and cut-off potential
   281→   281→    chosenDir = bfsSmartDirection(aiSnake, food, obstacles, playerSnake);
   282→   282→  }
   283→   283→
   284→   284→  if (chosenDir && chosenDir !== OPPOSITE[aiSnake.direction]) {
   285→   285→    return chosenDir;
   286→   286→  }
   287→   287→
   288→   288→  // Fallback survival
   289→   289→  return bfsSurvivalDirection(aiSnake, obstacles) ?? aiSnake.direction;
   290→   290→}
