   129→   129→function trackPageVisit(path: string) {
   130→   130→  if (path === "/") return;
   131→   131→  try {
   132→   132→    const raw = sessionStorage.getItem(RECENT_PAGES_KEY);
   133→   133→    const pages: string[] = raw ? JSON.parse(raw) : [];
   134→   134→    const filtered = pages.filter((p) => p !== path);
   135→   135→    filtered.unshift(path);
   136→   136→    sessionStorage.setItem(RECENT_PAGES_KEY, JSON.stringify(filtered.slice(0, MAX_RECENT)));
   137→   137→  } catch { /* noop */ }
   138→   138→}
   139→   139→
   140→   140→export function useTrackRecentPages() {
   141→   141→  const location = useLocation();
   142→   142→  useEffect(() => { trackPageVisit(location.pathname); }, [location.pathname]);
