     1→import { useState, useEffect } from 'react';

     4→import {

     6→  Search, Filter, Zap, Crown, X, ChevronRight, Loader2,
     7→} from 'lucide-react';
     8→import { useAuth } from '@/contexts/AuthContext';
     9→import {
    10→  BATTLE_GAMES, createBattle, getOpenBattles, getMyBattles, acceptBattle,

    13→import { getBalance, formatMZN } from '@/lib/wallet';

    23→
    24→function timeAgo(dateStr: string): string {
    25→  const diff = Date.now() - new Date(dateStr).getTime();
    26→  const mins = Math.floor(diff / 60000);
    27→  if (mins < 1) return 'agora';
    28→  if (mins < 60) return `${mins}min`;
    29→  const hrs = Math.floor(mins / 60);
    30→  if (hrs < 24) return `${hrs}h`;
    31→  return `${Math.floor(hrs / 24)}d`;
    32→}

    36→  completed: 'Concluida', cancelled: 'Cancelada', disputed: 'Disputada',
    37→};
    38→

   266→                            {game?.emoji ?? '🎮'}

   308→                                    </p>
   309→                                  </div>
   310→                                </div>
   311→                                <div className="flex items-center justify-between rounded-lg bg-white/5 px-4 py-3">
   312→                                  <span className="text-sm text-white/60">Aposta</span>
