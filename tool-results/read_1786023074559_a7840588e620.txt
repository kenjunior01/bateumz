    95→// Per-game error boundary so one crashing game does not kill the whole page
    96→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean; resetKey: number; errorMsg: string}> {
    97→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false, resetKey: 0, errorMsg: ""}; }
    98→  static getDerivedStateFromError() { return {hasError: true}; }
    99→  componentDidCatch(err: Error) {
   100→    console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err);
   101→    const msg = err.message || "Erro desconhecido";
   102→    this.setState({ errorMsg: msg.length > 120 ? msg.slice(0, 120) + "..." : msg });
   103→  }
   104→  handleRetry = () => { this.setState((s) => ({ hasError: false, resetKey: (s.resetKey || 0) + 1, errorMsg: "" })); };
   105→  handleSelectOther = () => {
   106→    window.dispatchEvent(new CustomEvent("game-error-select-other"));
   107→  };
   108→  render() {
   109→    if (this.state.hasError) return (
   110→      <div className="flex flex-col items-center justify-center py-20 px-6 rounded-2xl border border-dashed border-destructive/30 bg-gradient-to-b from-destructive/5 to-transparent">
   111→        <div className="text-5xl mb-4">⚠️</div>
   112→        <p className="text-sm font-bold mb-1">Erro ao carregar {this.props.gameName}</p>
   113→        <p className="text-[11px] text-muted-foreground mb-5 text-center max-w-xs">Ocorreu um problema inesperado. Tenta novamente ou escolhe outro jogo.</p>
   114→        <div className="flex items-center gap-2">
   115→          <button onClick={this.handleRetry} className="text-xs px-4 py-2 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 font-semibold transition-colors">Tentar novamente</button>
   116→          <button onClick={this.handleSelectOther} className="text-xs px-4 py-2 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/80 font-semibold transition-colors">Outro jogo</button>
   117→        </div>
   118→      </div>
   119→    );
   120→    return <div key={this.state.resetKey}>{this.props.children}</div>;
   121→  }
   122→}
