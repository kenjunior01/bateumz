   248→   248→              p2.invincible = 60;
   249→   249→              setLives({ p1: p1.lives, p2: p2.lives });

   267→   267→          const dy = b.y - e.y;

   269→   269→          if (dist < e.size + 4) {
   270→   270→            e.hp--;
   271→   271→            bulletsToRemove.add(bi);
   272→   272→            if (e.hp <= 0) {
   273→   273→              enemiesToRemove.add(ei);

   276→   276→              else p2.score += pts;

   278→   278→              scoreChanged = true;
   279→   279→            }
   280→   280→            break;
   281→   281→          }
   282→   282→        }
