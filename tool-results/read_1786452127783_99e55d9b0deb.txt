   150→def make_table(headers, rows, col_widths=None):
   151→    cw = col_widths or [CONTENT_W / len(headers)] * len(headers)
   152→    header_row = [P(h, 'TableHeader') for h in headers]
   153→    data = [header_row]
   154→    for row in rows:
   155→        data.append([P(str(c), 'TableCell') for c in row])
   156→    t = Table(data, colWidths=cw, repeatRows=1)
   157→    style_cmds = [
   158→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   159→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   160→        ('FONTNAME', (0, 0), (-1, 0), 'DejaVuSans-Bold'),
   161→        ('FONTSIZE', (0, 0), (-1, 0), 8.5),
   162→        ('BOTTOMPADDING', (0, 0), (-1, 0), 6),
   163→        ('TOPPADDING', (0, 0), (-1, 0), 6),
   164→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   165→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   166→        ('LEFTPADDING', (0, 0), (-1, -1), 5),
   167→        ('RIGHTPADDING', (0, 0), (-1, -1), 5),
   168→        ('TOPPADDING', (0, 1), (-1, -1), 4),
   169→        ('BOTTOMPADDING', (0, 1), (-1, -1), 4),
   170→    ]
   171→    for i in range(1, len(data)):
   172→        if i % 2 == 0:
   173→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   174→    t.setStyle(TableStyle(style_cmds))
   175→    return t

   180→def score_bar(label, score, max_score=10):
   181→    pct = score / max_score
   182→    bar_w = 60 * mm
   183→    filled = bar_w * pct
   184→    empty = bar_w - filled
   185→    tbl = Table(
   186→        [[P(label, 'Body'), P(f'<font color="{SEM_SUCCESS.hexval() if pct >= 0.7 else SEM_WARNING.hexval() if pct >= 0.5 else SEM_ERROR.hexval()}">{score}/{max_score}</font>', 'TableCell')]],
   187→        colWidths=[CONTENT_W - 70*mm, 70*mm]
   188→    )
   189→    tbl.setStyle(TableStyle([
   190→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   191→        ('LEFTPADDING', (0,0), (-1,-1), 0),
   192→        ('RIGHTPADDING', (0,0), (-1,-1), 0),
   193→        ('TOPPADDING', (0,0), (-1,-1), 1),
   194→        ('BOTTOMPADDING', (0,0), (-1,-1), 1),
   195→    ]))
   196→    return tbl
