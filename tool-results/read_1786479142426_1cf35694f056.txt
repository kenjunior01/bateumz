    97→    97→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean; resetKey: number; errorMsg: string}> {
    98→    98→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false, resetKey: 0, errorMsg: ""}; }
    99→    99→  static getDerivedStateFromError() { return {hasError: true}; }
   100→   100→  componentDidCatch(err: Error) {
   101→   101→    console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err);
   102→   102→    const msg = err.message || "Erro desconhecido";
   103→   103→    this.setState({ errorMsg: msg.length > 120 ? msg.slice(0, 120) + "..." : msg });
   104→   104→  }
   105→   105→  handleRetry = () => { this.setState((s) => ({ hasError: false, resetKey: (s.resetKey || 0) + 1, errorMsg: "" })); };
   106→   106→  handleSelectOther = () => {
   107→   107→    window.dispatchEvent(new CustomEvent("game-error-select-other"));
   108→   108→  };
   109→   109→  render() {
   110→   110→    if (this.state.hasError) return (
   111→   111→      <div className="flex flex-col items-center justify-center py-20 px-6 rounded-2xl border border-dashed border-destructive/30 bg-gradient-to-b from-destructive/5 to-transparent">
   112→   112→        <div className="text-5xl mb-4">⚠️</div>
   113→   113→        <p className="text-sm font-bold mb-1">Error loading {this.props.gameName}</p>
   114→   114→        <p className="text-[11px] text-muted-foreground mb-5 text-center max-w-xs">An unexpected error occurred. Try again or choose another game.</p>
   115→   115→        <div className="flex items-center gap-2">
   116→   116→          <button onClick={this.handleRetry} className="text-xs px-4 py-2 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 font-semibold transition-colors">Try Again</button>
   117→   117→          <button onClick={this.handleSelectOther} className="text-xs px-4 py-2 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/80 font-semibold transition-colors">Other game</button>
   118→   118→        </div>
   119→   119→      </div>
   120→   120→    );
   121→   121→    return <div key={this.state.resetKey}>{this.props.children}</div>;
   122→   122→  }
   123→   123→}
