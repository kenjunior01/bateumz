   114→function getCategory(id: string) {
   115→  return categories.find((c) => c.id === id) ?? categories[0];
   116→}

   118→// Per-category indicative annual rate that decreases with amount
   119→function annualRateFor(categoryId: string, amount: number) {
   120→  const cat = getCategory(categoryId);
   121→  // anchor amount where the "best" rate is reached
   122→  const anchor =
   123→    cat.id === "imoveis"
   124→      ? 5_000_000
   125→      : cat.id === "viaturas"
   126→        ? 2_000_000
   127→        : cat.id === "equipamentos"
   128→          ? 1_000_000
   129→          : 200_000; // eletrónicos
   130→  const t = Math.min(1, Math.max(0, amount / anchor));
   131→  // linear interpolation base → best rate
   132→  return cat.baseRate + (cat.bestRate - cat.baseRate) * t;
   133→}
   134→
   135→function simulate(categoryId: string, value: number, downPayment: number, months: number) {
   136→  const principal = Math.max(0, value - downPayment);
   137→  const annualRate = annualRateFor(categoryId, value);
   138→  const monthlyRate = annualRate / 12;
   139→  const monthly =
   140→    monthlyRate === 0 || months === 0
   141→      ? principal / Math.max(1, months)
   142→      : (principal * monthlyRate) / (1 - Math.pow(1 + monthlyRate, -months));
   143→  const total = monthly * months + downPayment;
   144→  const interest = total - value;
   145→  return { principal, monthly, total, interest, annualRate };
   146→}
