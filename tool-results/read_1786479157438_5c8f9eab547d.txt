   163→// ==================== TAP BATTLE SPECTATOR ====================
   164→const SpectatorTapBattle = ({ liveCode }: { liveCode?: string }) => {
   165→  const { user } = useAuth();
   166→  const [taps, setTaps] = useState(0);
   167→  const [timeLeft, setTimeLeft] = useState(10);
   168→  const [active, setActive] = useState(false);
   169→  const [bestScore, setBestScore] = useState(0);
   170→  const [tapParticles, setTapParticles] = useState<{id: number; x: number; y: number; color: string}[]>([]);
   171→  const [shakeKey, setShakeKey] = useState(0);
   172→  const [flashClass, setFlashClass] = useState("");
   173→  const [gameEnded, setGameEnded] = useState(false);
   174→  const tapRef = useRef<HTMLDivElement>(null);
   175→
   176→  const spawnTapParticle = (e: React.MouseEvent) => {
   177→    if (!tapRef.current) return;
   178→    const rect = tapRef.current.getBoundingClientRect();
   179→    const x = e.clientX - rect.left;
   180→    const y = e.clientY - rect.top;
   181→    const colors = ["#f97316", "#ef4444", "#fbbf24", "#f59e0b", "#ffffff"];
   182→    const newParticles = Array.from({ length: 6 }, (_, i) => ({
   183→      id: Date.now() + i, x: x + (Math.random() - 0.5) * 60, y: y + (Math.random() - 0.5) * 60,
   184→      color: colors[Math.floor(Math.random() * colors.length)]
   185→    }));
   186→    setTapParticles(prev => [...prev, ...newParticles]);
   187→    setTimeout(() => setTapParticles(prev => prev.filter(p => !newParticles.find(np => np.id === p.id))), 600);
   188→  };
   189→
   190→  const handleTap = (e: React.MouseEvent) => {
   191→    if (!active) { startGame(); return; }
   192→    spawnTapParticle(e);
   193→    setTaps(p => {
   194→      const n = p + 1;
   195→      if (n % 10 === 0) {
   196→        setShakeKey(k => k + 1);
   197→        setFlashClass("game-flash-white");
   198→        setTimeout(() => setFlashClass(""), 300);
   199→      }
   200→      if (n % 25 === 0) {
   201→        setFlashClass("game-flash-gold");
   202→        setTimeout(() => setFlashClass(""), 400);
   203→      }
   204→      return n;
   205→    });
   206→  };
   207→
   208→  const startGame = () => { setTaps(0); setTimeLeft(10); setActive(true); setTapParticles([]); setGameEnded(false); setFlashClass(""); };
   209→
   210→  useEffect(() => {
   211→    if (!active) return;
   212→    const t = setInterval(() => {
   213→      setTimeLeft((p) => {
   214→        if (p <= 1) {
   215→          setActive(false);
   216→          setGameEnded(true);
   217→          const finalTaps = taps;
   218→          setBestScore((b) => {
   219→            const newBest = Math.max(b, finalTaps);
   220→            if (finalTaps >= 50) {
   221→              setFlashClass("game-flash-gold");
   222→              setTimeout(() => setFlashClass(""), 600);
   223→              confetti({ particleCount: 100, spread: 80, origin: { y: 0.6 }, colors: ["#f97316", "#fbbf24", "#ffffff"] });
   224→              setTimeout(() => confetti({ particleCount: 60, spread: 60, origin: { x: 0.3, y: 0.5 }, colors: ["#f97316", "#ef4444"] }), 200);
   225→              setTimeout(() => confetti({ particleCount: 60, spread: 60, origin: { x: 0.7, y: 0.5 }, colors: ["#f97316", "#ef4444"] }), 400);
   226→            }
   227→            return newBest;
   228→          });
   229→          return 0;
   230→        }
   231→        return p - 1;
   232→      });
   233→    }, 1000);
   234→    return () => clearInterval(t);
   235→  }, [active, taps]);
   236→
   237→  return (
   238→    <div className="space-y-4">
   239→      <div className="flex items-center justify-between">
   240→        <div className="flex items-center gap-2">
   241→          <Flame className="h-5 w-5 text-orange-500" />
   242→          <h3 className="font-bold text-base">Batalha de Toques</h3>
   243→        </div>
   244→        {bestScore > 0 && (
   245→          <Badge variant="secondary" className="text-[10px]">Recorde: {bestScore}</Badge>
   246→        )}
   247→      </div>
   248→
   249→      <div key={shakeKey} className={"text-center py-2 relative " + (shakeKey > 0 ? "game-screen-shake" : "")}>
   250→        {flashClass && <div className={"absolute inset-0 z-10 pointer-events-none rounded-2xl " + flashClass} />}
   251→        <motion.p key={taps} initial={{ scale: 1.3 }} animate={{ scale: 1 }} transition={{ type: "spring", stiffness: 300, damping: 15 }} className="text-5xl font-black text-primary">{taps}</motion.p>
   252→        <p className="text-xs text-muted-foreground mt-1">toques</p>
   253→      </div>
   254→
   255→      {active && (
   256→        <div className="relative">
   257→          <div className={`absolute inset-0 rounded-full ${timeLeft <= 3 ? 'energy-bar-glow' : ''}`} />
   258→          <Progress value={(timeLeft / 10) * 100} className={`h-3 ${timeLeft <= 3 ? '[&>div]:!bg-red-500' : timeLeft <= 5 ? '[&>div]:!bg-amber-500' : ''}`} />
   259→          {timeLeft <= 3 && <p className="text-center text-[10px] text-red-400 font-bold mt-1 animate-pulse">ULTIMOS SEGUNDOS!</p>}
   260→        </div>
   261→      )}
   262→
   263→      <button
   264→        onClick={handleTap}
   265→        className={cn(
   266→          "w-full h-32 rounded-2xl text-xl font-black transition-all active:scale-90 relative overflow-hidden",
   267→          active
   268→            ? "bg-gradient-to-r from-orange-500 to-red-500 text-white shadow-lg shadow-orange-500/30 spin-btn-glow"
   269→            : "bg-gradient-to-r from-primary/10 to-accent/10 text-primary border-2 border-dashed border-primary/30"
   270→        )}
   271→      >
   272→        {active && taps >= 50 && <div className="absolute inset-0 celebration-rays" />}
   273→        {active ? "\u{1F4AA} TOQUE! (" + timeLeft + "s)" : "\u{1F3C3} Come\u00E7ar"}
   274→      </button>
   275→      {gameEnded && taps >= 50 && (
   276→        <motion.div
   277→          initial={{ opacity: 0, y: 10 }}
   278→          animate={{ opacity: 1, y: 0 }}
   279→          className="text-center mt-2"
   280→        >
   281→          <p className="text-xs font-bold text-primary animate-pulse">✨ INCREDIBLE! ✨</p>
   282→        </motion.div>
   283→      )}
   284→    </div>
   285→  );

   471→                  "flex items-center gap-1.5 px-4 py-2 text-xs font-bold rounded-xl whitespace-nowrap transition-all",
   472→                  activePanel === tab.id
   473→                    ? "bg-primary text-primary-foreground shadow-md"
   474→                    : "text-muted-foreground hover:bg-muted"
   475→                )}
   476→              >
   477→                <Icon className="h-3.5 w-3.5" /> {tab.label}
   478→              </button>
   479→            );
   480→          })}
   481→        </div>
   482→
   483→        <div className="flex-1 overflow-hidden">
   484→          <AnimatePresence mode="wait">
   485→            {activePanel === "chat" && (
   486→              <motion.div
   487→                key="chat"
   488→                initial={{ opacity: 0, x: -10 }}
   489→                animate={{ opacity: 1, x: 0 }}
   490→                exit={{ opacity: 0, x: 10 }}
   491→                className="h-full flex flex-col"
   492→              >
   493→                <div className="flex-1 overflow-y-auto px-4 py-3 space-y-1.5">
   494→                  {messages.length === 0 && (
   495→                    <div className="flex flex-col items-center justify-center h-full text-muted-foreground">
   496→                      <Zap className="h-8 w-8 mb-2 opacity-20" />
   497→                      <p className="text-xs">Aguardando mensagens...</p>
   498→                    </div>
   499→                  )}
   500→                  <AnimatePresence initial={false}>
   501→                    {messages.map((msg) => (
   502→                      <motion.div
   503→                        key={msg.id}
   504→                        initial={{ opacity: 0, y: 6, scale: 0.98 }}
   505→                        animate={{ opacity: 1, y: 0, scale: 1 }}
   506→                        transition={{ duration: 0.15 }}
   507→                        className={cn(
   508→                          "text-xs rounded-xl px-3 py-2",
   509→                          msg.is_system
   510→                            ? "text-center text-primary/70 bg-primary/5"
   511→                            : msg.is_highlighted
   512→                              ? "bg-gradient-to-r from-amber-500/15 to-orange-500/10 border border-amber-500/20"
   513→                              : "hover:bg-muted/40"
   514→                        )}
   515→                      >
   516→                        {msg.is_system ? (
   517→                          <p>{msg.message}</p>
   518→                        ) : (
   519→                          <>
   520→                            {msg.is_highlighted && msg.tip_amount > 0 && (
   521→                              <Badge className="bg-amber-500 text-white text-[8px] border-0 mb-1 px-1.5 rounded-full">
   522→                                $${msg.tip_amount.toFixed(2)}
   523→                              </Badge>
   524→                            )}
   525→                            <div className="flex items-start gap-2">
   526→                              <Avatar className="h-5 w-5 mt-0.5 shrink-0">
   527→                                <AvatarImage src={msg.avatar_url || undefined} />
   528→                                <AvatarFallback className="text-[7px] bg-primary/10">
   529→                                  {msg.display_name?.charAt(0)?.toUpperCase()}
   530→                                </AvatarFallback>
   531→                              </Avatar>
   532→                              <div className="min-w-0">
   533→                                <span className="font-bold text-primary mr-1.5">{msg.display_name}</span>
   534→                                <span className="break-words">{msg.message}</span>
   535→                              </div>
   536→                              <span className="text-[9px] text-muted-foreground/50 shrink-0 ml-auto">{timeAgo(msg.created_at)}</span>
   537→                            </div>
   538→                          </>
   539→                        )}
   540→                      </motion.div>
   541→                    ))}
   542→                  </AnimatePresence>
   543→                  <div ref={chatEndRef} />
   544→                </div>
   545→
   546→                <div className="shrink-0 flex items-center gap-1 px-4 py-1.5 overflow-x-auto">
   547→                  {EMOJI_QUICK.map((emoji) => (
   548→                    <button
   549→                      key={emoji}
   550→                      onClick={() => sendReaction({ scheduled_live_id: scheduledLiveId, live_code: liveCode || undefined, emoji })}
   551→                      className="h-8 w-8 flex items-center justify-center rounded-full hover:bg-muted active:scale-125 transition-all text-lg shrink-0"
   552→                    >
   553→                      {emoji}
   554→                    </button>
   555→                  ))}
   556→                </div>
   557→
   558→                <div className="shrink-0 flex gap-2 px-4 py-3 border-t border-border/40 bg-background">
   559→                  <Input
   560→                    value={chatInput}
   561→                    onChange={(e) => setChatInput(e.target.value)}
   562→                    onKeyDown={(e) => e.key === "Enter" && handleSendChat()}
   563→                    placeholder={joined ? "Enviar mensagem..." : "Entre na live para chat"}
   564→                    className="flex-1 h-10 rounded-xl text-sm"
   565→                    maxLength={500}
   566→                    disabled={!joined}
   567→                  />
   568→                  <Button
   569→                    onClick={handleSendChat}
   570→                    disabled={!joined || !chatInput.trim()}
   571→                    size="sm"
   572→                    className="h-10 w-10 p-0 rounded-xl bg-primary"
   573→                  >
   574→                    <Send className="h-4 w-4" />
   575→                  </Button>
   576→                </div>
   577→              </motion.div>
   578→            )}
   579→
   580→            {activePanel === "games" && (
   581→              <motion.div
   582→                key="games"
   583→                initial={{ opacity: 0, x: 10 }}
   584→                animate={{ opacity: 1, x: 0 }}
   585→                exit={{ opacity: 0, x: -10 }}
   586→                className="h-full overflow-y-auto p-4 space-y-4"
   587→              >
   588→                <SpectatorTapBattle liveCode={liveCode || undefined} />
   589→
   590→                <div className="border-t border-border/40 pt-4">
   591→                  <p className="text-xs text-muted-foreground mb-3 font-medium">Jogos dispon\u00EDveis na live:</p>
   592→                  <div className="grid grid-cols-3 gap-2">
   593→                    {QUICK_GAMES.map((g) => (
   594→                      <div
   595→                        key={g.id}
   596→                        className="flex flex-col items-center gap-2 p-4 rounded-2xl border border-border/60 hover:border-primary/30 hover:bg-muted/20 transition-all cursor-pointer active:scale-95"
   597→                      >
   598→                        <span className="text-3xl">{g.emoji}</span>
   599→                        <span className="text-[11px] font-medium">{g.label}</span>
   600→                      </div>
   601→                    ))}
   602→                  </div>
   603→                </div>
   604→              </motion.div>
   605→            )}
   606→
   607→            {activePanel === "bingo" && (
   608→              <motion.div
   609→                key="bingo"
   610→                initial={{ opacity: 0, y: 10 }}
   611→                animate={{ opacity: 1, y: 0 }}
   612→                exit={{ opacity: 0, y: -10 }}
   613→                className="h-full overflow-y-auto p-4"
   614→              >
   615→                <SpectatorBingo scheduledLiveId={scheduledLiveId} liveCode={liveCode || undefined} />
   616→              </motion.div>
   617→            )}
   618→          </AnimatePresence>
   619→        </div>
   620→      </div>
   621→
   622→      <BottomTabBar />
   623→    </div>
   624→  );
   625→};
