   125→const RECENT_PAGES_KEY = "bateu_recent_pages";

   127→const MAX_RECENT = 6;
   128→
   129→function trackPageVisit(path: string) {
   130→  if (path === "/") return;
   131→  try {
   132→    const raw = sessionStorage.getItem(RECENT_PAGES_KEY);
   133→    const pages: string[] = raw ? JSON.parse(raw) : [];
   134→    const filtered = pages.filter((p) => p !== path);
   135→    filtered.unshift(path);
   136→    sessionStorage.setItem(RECENT_PAGES_KEY, JSON.stringify(filtered.slice(0, MAX_RECENT)));
   137→  } catch { /* noop */ }
   138→}
   139→
   140→export function useTrackRecentPages() {
   141→  const location = useLocation();
   142→  useEffect(() => { trackPageVisit(location.pathname); }, [location.pathname]);
   143→}
