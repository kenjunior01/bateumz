   782→export interface BracketMatch {
   783→  id: string;
   784→  match_number: number;
   785→  p1: string | null;
   786→  p2: string | null;
   787→  p1_name: string | null;
   788→  p2_name: string | null;
   789→  p1_logo: string | null;
   790→  p2_logo: string | null;
   791→  winner: string | null;
   792→  score1: number | null;
   793→  score2: number | null;
   794→  next_match: string | null;
   795→}
   796→
   797→export interface BracketRound {
   798→  round: number;
   799→  label: string;
   800→  matches: BracketMatch[];
   801→}
   802→
   803→export interface BracketTree {
   804→  format: TournamentFormat;
   805→  rounds: BracketRound[];
   806→}
   807→
   808→const ROUND_LABELS: Record<number, string> = {
   809→  1: "Rodada 1",
   810→  2: "Oitavas de Final",
   811→  3: "Quartas de Final",
   812→  4: "Semifinal",
   813→  5: "Final",
   814→};
   815→
   816→function roundLabel(totalRounds: number, currentRound: number): string {
   817→  const offset = totalRounds - currentRound;
   818→  if (offset === 0) return "Final";
   819→  if (offset === 1) return "Semifinal";
   820→  if (offset === 2) return "Quartas de Final";
   821→  if (offset === 3) return "Oitavas de Final";
   822→  return `Rodada ${currentRound}`;
   823→}
   824→
   825→export function generateBracket(
   826→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   827→  format: TournamentFormat,
   828→  totalRounds: number
   829→): BracketTree {
   830→  if (format === "round_robin") {
   831→    return generateRoundRobin(teams, totalRounds);
   832→  }
   833→  if (format === "battle_royale") {
   834→    return generateBRRounds(teams, totalRounds);
   835→  }
   836→  return generateElimination(teams, format, totalRounds);
   837→}
   838→
   839→function generateElimination(
   840→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   841→  format: TournamentFormat,
   842→  totalRounds: number
   843→): BracketTree {
   844→  const rounds: BracketRound[] = [];
   845→  let matchCounter = 1;
   846→  const matchIdMap = new Map<string, string>();
   847→
   848→  // Build rounds from first to last
   849→  for (let r = 1; r <= totalRounds; r++) {
   850→    const matchesInRound = Math.ceil(teams.length / Math.pow(2, r));
   851→ const roundMatches: BracketMatch[] = [];
   852→    for (let m = 0; m < matchesInRound; m++) {
   853→      const matchId = `bracket-${r}-${m + 1}`;
   854→      const bracketMatch: BracketMatch = {
   855→        id: matchId,
   856→        match_number: matchCounter++,
   857→        p1: null,
   858→        p2: null,
   859→        p1_name: null,
   860→        p2_name: null,
   861→        p1_logo: null,
   862→        p2_logo: null,
   863→        winner: null,
   864→        score1: null,
   865→        score2: null,
   866→        next_match: null,
   867→      };
   868→
   869→      // First round: seed teams
   870→      if (r === 1) {
   871→        const idx1 = m * 2;
   872→        const idx2 = m * 2 + 1;
   873→        if (idx1 < teams.length) {
   874→          bracketMatch.p1 = teams[idx1].id;
   875→          bracketMatch.p1_name = teams[idx1].name;
   876→          bracketMatch.p1_logo = teams[idx1].logo_url ?? null;
   877→        }
   878→        if (idx2 < teams.length) {
   879→          bracketMatch.p2 = teams[idx2].id;
   880→          bracketMatch.p2_name = teams[idx2].name;
   881→          bracketMatch.p2_logo = teams[idx2].logo_url ?? null;
   882→        }
   883→      }
   884→
   885→      roundMatches.push(bracketMatch);
   886→      matchIdMap.set(`${r}-${m}`, matchId);
   887→    }
   888→    rounds.push({ round: r, label: roundLabel(totalRounds, r), matches: roundMatches });
   889→  }
   890→
   891→  // Wire next_match pointers
   892→  for (let r = 1; r < totalRounds; r++) {
   893→    const currentMatches = rounds[r - 1].matches;
   894→    const nextMatches = rounds[r].matches;
   895→    for (let i = 0; i < currentMatches.length; i++) {
   896→      const nextIdx = Math.floor(i / 2);
   897→      if (nextIdx < nextMatches.length) {
   898→        currentMatches[i].next_match = nextMatches[nextIdx].id;
   899→      }
   900→    }
   901→  }
   902→
   903→  return { format, rounds };
   904→}

   906→function generateRoundRobin(
   907→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   908→  _totalRounds: number
   909→): BracketTree {
   910→  const rounds: BracketRound[] = [];
   911→  const n = teams.length;
   912→  if (n < 2) return { format: "round_robin", rounds: [] };
   913→
   914→  // Standard round-robin algorithm
   915→  const list = [...teams];
   916→  if (n % 2 !== 0) list.push({ id: "bye", name: "Bye", logo_url: null });
   917→  const fixed = list[0];
   918→  const rotating = list.slice(1);
   919→  const totalRounds = rotating.length;
   920→  let matchCounter = 1;
   921→
   922→  for (let r = 0; r < totalRounds; r++) {
   923→    const roundMatches: BracketMatch[] = [];
   924→    const current = [fixed, ...rotating];
   925→    const half = current.length / 2;
   926→    for (let i = 0; i < half; i++) {
   927→      const t1 = current[i];
   928→      const t2 = current[current.length - 1 - i];
   929→      roundMatches.push({
   930→        id: `rr-r${r + 1}-m${i + 1}`,
   931→        match_number: matchCounter++,
   932→        p1: t1.id === "bye" ? null : t1.id,
   933→        p2: t2.id === "bye" ? null : t2.id,
   934→        p1_name: t1.id === "bye" ? null : t1.name,
   935→        p2_name: t2.id === "bye" ? null : t2.name,
   936→        p1_logo: t1.logo_url ?? null,
   937→        p2_logo: t2.logo_url ?? null,
   938→        winner: null,
   939→        score1: null,
   940→        score2: null,
   941→        next_match: null,
   942→      });
   943→    }
   944→    rounds.push({ round: r + 1, label: `Rodada ${r + 1}`, matches: roundMatches });
   945→    // Rotate
   946→    rotating.push(rotating.shift()!);
   947→  }
   948→
   949→  return { format: "round_robin", rounds };
   950→}

   952→function generateBRRounds(
   953→  teams: Array<{ id: string; name: string; logo_url?: string | null; seed?: number | null }>,
   954→  totalRounds: number
   955→): BracketTree {
   956→  const rounds: BracketRound[] = [];
   957→  let matchCounter = 1;
   958→
   959→  for (let r = 1; r <= totalRounds; r++) {
   960→    const roundMatches: BracketMatch[] = [];
   961→    // In BR, each round is a single match/lobby with all teams
   962→    roundMatches.push({
   963→      id: `br-r${r}-m1`,
   964→      match_number: matchCounter++,
   965→      p1: null,
   966→      p2: null,
   967→      p1_name: null,
   968→      p2_name: null,
   969→      p1_logo: null,
   970→      p2_logo: null,
   971→      winner: null,
   972→      score1: null,
   973→      score2: null,
   974→      next_match: r < totalRounds ? `br-r${r + 1}-m1` : null,
   975→    });
   976→    rounds.push({ round: r, label: `Partida ${r} de ${totalRounds}`, matches: roundMatches });
   977→  }
   978→
   979→  return { format: "battle_royale", rounds };
   980→}

  1155→    .from("esport_mvp_votes")
  1156→    .select("candidate_id, candidate_name")
  1157→    .eq("match_id", matchId);

  1162→  for (const v of data) {

  1165→    counts[cid].total++;
  1166→  }
  1167→
  1168→  let best = "";
  1169→  let bestCount = 0;
  1170→  for (const [cid, info] of Object.entries(counts)) {
  1171→    if (info.total > bestCount) {
  1172→      best = cid;
  1173→      bestCount = info.total;
  1174→    }
  1175→  }
  1176→
  1177→  if (!best) return null;
  1178→  return { candidate_id: best, candidate_name: counts[best].name, total_votes: bestCount };
  1179→}

  1223→export function calculateElo(currentRating: number, opponentRating: number, result: "win" | "loss" | "draw"): number {
  1224→  const K = 32;
  1225→  const expectedScore = 1 / (1 + Math.pow(10, (opponentRating - currentRating) / 400));
  1226→  const actualScore = result === "win" ? 1 : result === "loss" ? 0 : 0.5;
  1227→  return Math.round(currentRating + K * (actualScore - expectedScore));
  1228→}
