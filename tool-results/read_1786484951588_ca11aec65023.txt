   228→const computeQuality = (
   229→  players: number,
   230→  plays: number,
   231→  games: number,
   232→  heat: number,
   233→): number => {
   234→  if (plays === 0) return 0;

   238→  const momentum = (heat / 100) * 15;

   240→  return Math.round(breadth + volume + diversity + momentum + retention);

   341→ * ★ REVOLUTION: Hype Meter — detects surges in plays over the last 20s vs the
   342→ * previous 20s window. Returns 0-100 representing "is the chat exploding right now".
   343→ */
   344→const computeHype = (entries: LeaderEntry[], now: number): number => {
   345→  if (!entries.length) return 0;
   346→  const recent = entries.filter((e) => e.at >= now - 20_000).length;
   347→  const prior = entries.filter((e) => e.at >= now - 40_000 && e.at < now - 20_000).length;
   348→  if (prior === 0) return recent > 0 ? Math.min(100, recent * 25) : 0;
   349→  const ratio = recent / Math.max(1, prior);
   350→  return Math.max(0, Math.min(100, Math.round((ratio - 1) * 80 + 20)));

   384→/**
   385→ * ★ REVOLUTION: Trajectory forecast.
   386→ * Predicts projected total plays at the current pace, plus ETA in seconds to reach the goal.
   387→ */
   388→const computeTrajectory = (
   389→  plays: number,
   390→  elapsedSec: number,
   391→  goal: number,
   392→): { projected: number; etaSec: number | null; pacePerMin: number } => {
   393→  if (elapsedSec < 5 || plays === 0) return { projected: plays, etaSec: null, pacePerMin: 0 };
   394→  const pacePerMin = (plays / elapsedSec) * 60;
   395→  const projected = Math.round(plays + (pacePerMin / 60) * Math.max(0, (1800 - elapsedSec))); // project to 30min mark
   396→  if (plays >= goal || pacePerMin <= 0) return { projected, etaSec: null, pacePerMin };
   397→  const remaining = goal - plays;
   398→  const etaSec = Math.round((remaining / pacePerMin) * 60);
   399→  return { projected, etaSec, pacePerMin };
   400→};

   886→const CommandPalette = ({
   887→  open,
   888→  onClose,
   889→  items,
   890→}: {
   891→  open: boolean;
   892→  onClose: () => void;
   893→  items: CommandItem[];
   894→}) => {
   895→  const [query, setQuery] = useState("");
   896→  const [active, setActive] = useState(0);
   897→  const inputRef = useRef<HTMLInputElement>(null);
   898→
   899→  useEffect(() => {
   900→    if (open) {
   901→      setQuery("");
   902→      setActive(0);
   903→      setTimeout(() => inputRef.current?.focus(), 50);
   904→    }
   905→  }, [open]);
   906→
   907→  const filtered = useMemo(() => {
   908→    if (!query.trim()) return items;
   909→    const q = query.toLowerCase();
   910→    return items.filter(
   911→      (it) =>
   912→        it.label.toLowerCase().includes(q) ||
   913→        it.group.toLowerCase().includes(q) ||
   914→        it.hint?.toLowerCase().includes(q),
   915→    );
   916→  }, [items, query]);
   917→
   918→  useEffect(() => {
   919→    setActive(0);
   920→  }, [query]);
   921→
   922→  const groups = useMemo(() => {
   923→    const map = new Map<string, CommandItem[]>();
   924→    filtered.forEach((it) => {
   925→      if (!map.has(it.group)) map.set(it.group, []);
   926→      map.get(it.group)!.push(it);
   927→    });
   928→    return [...map.entries()];
   929→  }, [filtered]);
   930→
   931→  const onKeyDown = (e: React.KeyboardEvent) => {
   932→    if (e.key === "ArrowDown") {
   933→      e.preventDefault();
   934→      setActive((a) => Math.min(a + 1, filtered.length - 1));
   935→    } else if (e.key === "ArrowUp") {
   936→      e.preventDefault();
   937→      setActive((a) => Math.max(a - 1, 0));
   938→    } else if (e.key === "Enter") {
   939→      e.preventDefault();
   940→      const it = filtered[active];
   941→      if (it && !it.disabled) {
   942→        it.run();
   943→        onClose();
   944→      }
   945→    } else if (e.key === "Escape") {
   946→      e.preventDefault();
   947→      onClose();
   948→    }
   949→  };
   950→
   951→  return (
   952→    <AnimatePresence>
   953→      {open && (
   954→        <motion.div
   955→          initial={{ opacity: 0 }}
   956→          animate={{ opacity: 1 }}
   957→          exit={{ opacity: 0 }}
   958→          transition={{ duration: 0.15 }}
   959→          className="fixed inset-0 z-[100] flex items-start justify-center pt-[15vh] px-4 bg-background/70 backdrop-blur-sm"
   960→          onClick={onClose}
   961→        >
   962→          <motion.div
   963→            initial={{ opacity: 0, y: -16, scale: 0.96 }}
   964→            animate={{ opacity: 1, y: 0, scale: 1 }}
   965→            exit={{ opacity: 0, y: -16, scale: 0.96 }}
   966→            transition={{ type: "spring", stiffness: 400, damping: 28 }}
   967→            onClick={(e) => e.stopPropagation()}
   968→            className="w-full max-w-xl rounded-2xl border border-border bg-card shadow-elegant overflow-hidden"
   969→          >
   970→            <div className="flex items-center gap-2 px-3 py-2.5 border-b border-border/60">
   971→              <Search className="h-4 w-4 text-muted-foreground shrink-0" />
   972→              <input
   973→                ref={inputRef}
   974→                value={query}
   975→                onChange={(e) => setQuery(e.target.value)}
   976→                onKeyDown={onKeyDown}
   977→                placeholder="Pesquisar ação... (ex: coroar, exportar, copiar)"
   978→                className="flex-1 bg-transparent text-sm outline-none placeholder:text-muted-foreground"
   979→              />
   980→              <kbd className="text-[9px] font-mono px-1.5 py-0.5 rounded border border-border bg-muted/50 text-muted-foreground">
   981→                ESC
   982→              </kbd>
   983→            </div>
