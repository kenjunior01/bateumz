   168→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
   169→  const [count, setCount] = useState(0);
   170→  const ref = useRef<HTMLSpanElement>(null);
   171→  const inView = useInView(ref, { once: true });
   172→  const hasAnimated = useRef(false);
   173→
   174→  useEffect(() => {
   175→    if (!inView || hasAnimated.current) return;
   176→    hasAnimated.current = true;
   177→    const start = performance.now();
   178→    const step = (now: number) => {
   179→      const elapsed = (now - start) / 1000;
   180→      const progress = Math.min(elapsed / duration, 1);
   181→      const eased = 1 - Math.pow(1 - progress, 3);
   182→      setCount(Math.round(eased * target));
   183→      if (progress < 1) requestAnimationFrame(step);
   184→    };
   185→    requestAnimationFrame(step);
   186→  }, [inView, target, duration]);
   187→
   188→  return <span ref={ref}>{count}{suffix}</span>;
   189→}
