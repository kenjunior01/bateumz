    39→function tempoAgo(dateStr: string): string {
    40→  const agora = Date.now();
    41→  const data = new Date(dateStr).getTime();
    42→  const diff = agora - data;
    43→  const minutos = Math.floor(diff / 60000);
    44→  if (minutos < 1) return 'Agora mesmo';
    45→  if (minutos < 60) return `${minutos}min atr\u00e1s`;
    46→  const horas = Math.floor(minutos / 60);
    47→  if (horas < 24) return `${horas}h atr\u00e1s`;
    48→  const dias = Math.floor(horas / 24);
    49→  if (dias < 7) return `${dias}d atr\u00e1s`;
    50→  return new Date(dateStr).toLocaleDateString('pt-BR');
    51→}

    53→function formatarPremio(valor: number): string {
    54→  if (valor >= 1000000) return `${(valor / 1000000).toFixed(1)}M`;
    55→  if (valor >= 1000) return `${(valor / 1000).toFixed(1)}K`;
    56→  return String(valor);
    57→}

    59→function formatarHorario(dateStr: string | null): string {
    60→  if (!dateStr) return '--:--';
    61→  const d = new Date(dateStr);
    62→  return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
    63→}
