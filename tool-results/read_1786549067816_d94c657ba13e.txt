   169→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
   170→  const [count, setCount] = useState(0);
   171→  const ref = useRef<HTMLSpanElement>(null);
   172→  const inView = useInView(ref, { once: true });
   173→  const hasAnimated = useRef(false);
   174→
   175→  useEffect(() => {
   176→    if (!inView || hasAnimated.current) return;
   177→    hasAnimated.current = true;
   178→    const start = performance.now();
   179→    const step = (now: number) => {
   180→      const elapsed = (now - start) / 1000;
   181→      const progress = Math.min(elapsed / duration, 1);
   182→      const eased = 1 - Math.pow(1 - progress, 3);
   183→      setCount(Math.round(eased * target));
   184→      if (progress < 1) requestAnimationFrame(step);
   185→    };
   186→    requestAnimationFrame(step);
   187→  }, [inView, target, duration]);
   188→
   189→  return <span ref={ref}>{count}{suffix}</span>;
   190→}
