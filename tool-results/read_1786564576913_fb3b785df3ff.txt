   171→   171→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
   172→   172→  const [count, setCount] = useState(0);
   173→   173→  const ref = useRef<HTMLSpanElement>(null);
   174→   174→  const inView = useInView(ref, { once: true });
   175→   175→  const hasAnimated = useRef(false);
   176→   176→
   177→   177→  useEffect(() => {
   178→   178→    if (!inView || hasAnimated.current) return;
   179→   179→    hasAnimated.current = true;
   180→   180→    const start = performance.now();
   181→   181→    const step = (now: number) => {
   182→   182→      const elapsed = (now - start) / 1000;
   183→   183→      const progress = Math.min(elapsed / duration, 1);
   184→   184→      const eased = 1 - Math.pow(1 - progress, 3);
   185→   185→      setCount(Math.round(eased * target));
   186→   186→      if (progress < 1) requestAnimationFrame(step);
   187→   187→    };
   188→   188→    requestAnimationFrame(step);
   189→   189→  }, [inView, target, duration]);
   190→   190→
   191→   191→  return <span ref={ref}>{count}{suffix}</span>;
   192→   192→}
