    94→function timeAgo(dateStr: string): string {
    95→  const now = new Date();
    96→  const date = new Date(dateStr);
    97→  const diffMs = now.getTime() - date.getTime();
    98→  const diffSec = Math.floor(diffMs / 1000);
    99→  const diffMin = Math.floor(diffSec / 60);
   100→  const diffHour = Math.floor(diffMin / 60);
   101→  const diffDay = Math.floor(diffHour / 24);
   102→
   103→  if (diffSec < 60) return 'agora mesmo';
   104→  if (diffMin < 60) return `ha ${diffMin} min`;
   105→  if (diffHour < 24) return `ha ${diffHour}h`;
   106→  if (diffDay < 7) return `ha ${diffDay}d`;
   107→  return `ha ${Math.floor(diffDay / 7)} semanas`;
   108→}

   110→function getStatusColor(status: BetStatus): string {
   111→  switch (status) {
   112→    case 'open':
   113→      return 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30';
   114→    case 'settled':
   115→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   116→    case 'refunded':
   117→      return 'bg-sky-500/20 text-sky-400 border-sky-500/30';
   118→    case 'cancelled':
   119→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   120→    default:
   121→      return 'bg-white/10 text-white border-white/20';
   122→  }
   123→}
   124→
   125→function getBetStatusLabel(bet: Bet): string {
   126→  if (bet.status === 'settled' && bet.is_correct) return 'GANHOU';
   127→  if (bet.status === 'settled' && !bet.is_correct) return 'PERDEU';
   128→  return BET_STATUS_LABELS[bet.status];
   129→}
   130→
   131→function getBetStatusBadgeStyle(bet: Bet): string {
   132→  if (bet.status === 'settled' && bet.is_correct) {
   133→    return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   134→  }
   135→  if (bet.status === 'settled' && !bet.is_correct) {
   136→    return 'bg-red-500/20 text-red-400 border-red-500/30';
   137→  }
   138→  return getStatusColor(bet.status);
   139→}
