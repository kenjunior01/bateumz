    56→/* ─── Animated Section Wrapper ─── */
    57→function AnimatedSection({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
    58→  const ref = useRef<HTMLDivElement>(null);
    59→  const inView = useInView(ref, { once: true, margin: "-80px" });
    60→  return (
    61→    <motion.section ref={ref} initial="hidden" animate={inView ? "visible" : "hidden"} variants={sectionReveal} transition={{ delay }} className={className}>
    62→      {children}
    63→    </motion.section>
    64→  );
    65→}

    67→/* ─── Counting Number ─── */
    68→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    69→  const [count, setCount] = useState(0);
    70→  const ref = useRef<HTMLSpanElement>(null);
    71→  const inView = useInView(ref, { once: true });
    72→  const hasAnimated = useRef(false);
    73→  useEffect(() => {
    74→    if (!inView || hasAnimated.current) return;
    75→    hasAnimated.current = true;
    76→    const start = performance.now();
    77→    const step = (now: number) => {
    78→      const elapsed = (now - start) / 1000;
    79→      const progress = Math.min(elapsed / duration, 1);
    80→      const eased = 1 - Math.pow(1 - progress, 3);
    81→      setCount(Math.round(eased * target));
    82→      if (progress < 1) requestAnimationFrame(step);
    83→    };
    84→    requestAnimationFrame(step);
    85→  }, [inView, target, duration]);
    86→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
    87→}
