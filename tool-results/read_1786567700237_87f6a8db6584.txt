    26→const sb: any = supabase;

    59→
    60→const Marketplace = () => {
    61→  const { t } = useLanguage();
    62→  const [raffles, setRaffles] = useState<Raffle[]>([]);
    63→  const [contests, setContests] = useState<Contest[]>([]);
    64→  const [games, setGames] = useState<any[]>([]);
    65→  const [search, setSearch] = useState("");
    66→  const [loading, setLoading] = useState(true);
    67→  const [sortBy, setSortBy] = useState<"newest" | "ending" | "popular">("newest");
    68→  const [typeFilter, setTypeFilter] = useState<"all" | "paid" | "free" | "points">("all");
    69→  const [country, setCountry] = useState("");
    70→  const [region, setRegion] = useState("");
    71→  const [searchParams] = useSearchParams();
    72→  const [contentType, setContentType] = useState<ContentType>((searchParams.get("tab") as ContentType) || "all");
    73→  const [filterSheetOpen, setFilterSheetOpen] = useState(false);
    74→  const isMobile = useIsMobile();
    75→
    76→  useEffect(() => {
    77→    const tab = searchParams.get("tab") as ContentType;
    78→    if (tab && (tab === "all" || tab === "raffles" || tab === "contests" || tab === "games")) {
    79→      setContentType(tab);
    80→    }
    81→    const q = searchParams.get("q");
    82→    if (q) setSearch(q);
    83→    const filter = searchParams.get("filter");
    84→    if (filter) {
    85→      switch (filter) {
    86→        case "trending":
    87→          setSortBy("popular");
    88→          break;
    89→        case "new":
    90→          setSortBy("newest");
    91→          break;
    92→        case "ending":
    93→          setSortBy("ending");
    94→          break;
    95→        case "cheap":
    96→          setTypeFilter("free");
    97→          setContentType("raffles");
    98→          break;
    99→        case "premium":
   100→          setSortBy("popular");
   101→          setContentType("raffles");
   102→          break;
   103→      }
   104→    }
   105→  }, [searchParams]);
   106→
   107→  useEffect(() => {

   109→      const [rafflesRes, contestsRes, spinRes, millRes] = await Promise.all([
   110→        sb.from("raffles").select("*").eq("status", "active").order("created_at", { ascending: false }),
   111→        sb.from("contests").select("*").in("status", ["active", "voting", "completed"]).order("created_at", { ascending: false }),
   112→        sb.from("spin_wheel_games").select("*").eq("is_published", true),
   113→        sb.from("millionaire_games").select("*").eq("is_published", true),
   114→      ]);

   116→      if (contestsRes.data) setContests(contestsRes.data as Contest[]);
   117→      
   118→      const allGames = [
   119→        ...(spinRes.data || []).map(g => ({ ...g, type: 'spin' })),
   120→        ...(millRes.data || []).map(g => ({ ...g, type: 'millionaire' }))
   121→      ];
   122→      setGames(allGames);
   123→      setLoading(false);
   124→    };
   125→    fetchData();
   126→  }, []);
   127→
   128→  const filteredRaffles = raffles
   129→    .filter((r) => {
   130→      if (!r.title.toLowerCase().includes(search.toLowerCase()) && !r.prize_title.toLowerCase().includes(search.toLowerCase())) return false;
   131→      if (typeFilter !== "all" && r.raffle_type !== typeFilter) return false;
   132→      if (region && r.province !== region) return false;
   133→      if (country && !region) {
   134→        const regs = getRegions(country).map((x) => x.value);
   135→        if (r.province && !regs.includes(r.province)) return false;
   136→      }
   137→      return true;
   138→    })
   139→    .sort((a, b) => {
   140→      if (sortBy === "ending") return (a.end_date || "z").localeCompare(b.end_date || "z");
   141→      if (sortBy === "popular") return b.sold_tickets - a.sold_tickets;
   142→      return 0;
   143→    });
   144→
   145→  const filteredContests = contests.filter((c) =>
   146→    c.title.toLowerCase().includes(search.toLowerCase()) ||
   147→    (c.description || "").toLowerCase().includes(search.toLowerCase()) ||
   148→    (c.prize_description || "").toLowerCase().includes(search.toLowerCase())
   149→  );
   150→
   151→  const timeLeft = (date: string | null) => {
   152→    if (!date) return "Sem prazo";
   153→    const diff = new Date(date).getTime() - Date.now();
   154→    if (diff <= 0) return "Encerrado";
   155→    const days = Math.floor(diff / 86400000);
   156→    if (days > 0) return `${days}d restantes`;
   157→    const hours = Math.floor(diff / 3600000);
   158→    return `${hours}h restantes`;
   159→  };
   160→
   161→  const getRaffleUrl = (raffle: Raffle) => `/raffle/${raffle.slug || raffle.id}`;
   162→
   163→  const showRaffles = contentType === "all" || contentType === "raffles";
   164→  const showContests = contentType === "all" || contentType === "contests";
   165→  const showGames = contentType === "all" || contentType === "games";
   166→
   167→  const contestStatusMap: Record<string, { label: string; color: string }> = {
   168→    active: { label: "Aberto", color: "bg-primary text-primary-foreground" },
   169→    voting: { label: "Em Votação", color: "bg-accent text-accent-foreground" },
   170→    completed: { label: "Encerrado", color: "bg-muted text-muted-foreground" },
   171→  };
   172→
   173→  const activeFilterCount =
   174→    (typeFilter !== "all" ? 1 : 0) +
   175→    (contentType !== "all" ? 1 : 0) +
   176→    (country ? 1 : 0) +
   177→    (region ? 1 : 0) +
   178→    (sortBy !== "newest" ? 1 : 0);
   179→
   180→  const sortLabel = sortBy === "newest" ? "Recentes" : sortBy === "ending" ? "A terminar" : "Populares";
   181→  const typeLabel = typeFilter === "all" ? null : typeFilter === "paid" ? "Pagos" : typeFilter === "free" ? "Gratuitos" : "Pontos";
   182→  const contentLabel = contentType === "all" ? null : contentType === "raffles" ? "Sorteios" : "Concursos";
   183→
   184→  const FiltersBody = () => (
   185→    <div className="space-y-5">
   186→      <div>
   187→        <p className="mb-2 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Tipo de conteúdo</p>
   188→        <div className="flex flex-wrap gap-2">
   189→          {([
   190→            { value: "all" as ContentType, label: "🔥 Tudo" },
   191→            { value: "raffles" as ContentType, label: "🎟️ Sorteios" },
   192→            { value: "contests" as ContentType, label: "🏆 Concursos" },
   193→            { value: "games" as ContentType, label: "🎮 Jogos" },
   194→          ]).map((t) => (
   195→            <Button key={t.value} variant={contentType === t.value ? "default" : "outline"} size="sm" onClick={() => setContentType(t.value)}>
   196→              {t.label}
   197→            </Button>
   198→          ))}
   199→        </div>
   200→      </div>
   201→      <div>
   202→        <p className="mb-2 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Ordenar</p>
   203→        <div className="flex flex-wrap gap-2">
   204→          {(["newest", "ending", "popular"]).map((s) => (
   205→            <Button key={s} variant={sortBy === s ? "default" : "outline"} size="sm" onClick={() => setSortBy(s as any)}>
   206→              {s === "newest" ? "Recentes" : s === "ending" ? "A terminar" : "Populares"}
   207→            </Button>
   208→          ))}
   209→        </div>
   210→      </div>
   211→      {showRaffles && (
   212→        <div>
   213→          <p className="mb-2 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Modalidade de sorteio</p>
   214→          <div className="flex flex-wrap gap-2">
   215→            {(["all", "paid", "free", "points"]).map((t) => (
   216→              <Button key={t} variant={typeFilter === t ? "default" : "outline"} size="sm" onClick={() => setTypeFilter(t as any)} className="gap-1">
   217→                {t === "all" ? "Todos" : t === "paid" ? <><Ticket className="h-3 w-3" /> Pagos</> : t === "free" ? <><Gift className="h-3 w-3" /> Gratuitos</> : <><Star className="h-3 w-3" /> Pontos</>}
   218→              </Button>
   219→            ))}
   220→          </div>
   221→        </div>
   222→      )}
   223→      <div>
   224→        <p className="mb-2 text-[11px] font-bold uppercase tracking-wider text-muted-foreground">Região</p>
   225→        <CountryRegionFilter country={country} region={region} onCountry={setCountry} onRegion={setRegion} />
   226→      </div>
   227→    </div>
   228→  );
   229→
   230→  const chipCategories = [
   231→    { id: "all", label: "Tudo", icon: "🔥", count: raffles.length + contests.length + games.length },
   232→    { id: "raffles", label: "Sorteios", icon: "🎟️", count: raffles.length },
   233→    { id: "contests", label: "Concursos", icon: "🏆", count: contests.length },
   234→    { id: "games", label: "Jogos", icon: "🎮", count: games.length },
   235→    { id: "paid", label: "Pagos", icon: "💎" },
   236→    { id: "free", label: "Grátis", icon: "🎁" },
   237→    { id: "points", label: "Pontos", icon: "⭐" },
   238→    { id: "ending", label: "A terminar", icon: "⏰" },
   239→    { id: "popular", label: "Populares", icon: "🚀" },
   240→  ];
   241→
   242→  const handleChipChange = (id: string) => {
   243→    if (id === "all") { setContentType("all"); setTypeFilter("all"); setSortBy("newest"); return; }
   244→    if (id === "raffles" || id === "contests" || id === "games") { setContentType(id); return; }
   245→    if (id === "paid" || id === "free" || id === "points") { setTypeFilter(id); setContentType("raffles"); return; }
   246→    if (id === "ending") { setSortBy("ending"); return; }
   247→    if (id === "popular") { setSortBy("popular"); return; }
   248→  };
   249→  const activeChip =
   250→    sortBy === "ending" ? "ending" :
   251→    sortBy === "popular" ? "popular" :
   252→    typeFilter !== "all" ? typeFilter :
   253→    contentType !== "all" ? contentType :
   254→    "all";
   255→
   256→  const isCatalogEmpty = !loading && raffles.length === 0 && contests.length === 0 && games.length === 0;
   257→  const hasFilteredResults =
   258→    (showRaffles && filteredRaffles.length > 0) ||
   259→    (showContests && filteredContests.length > 0) ||
   260→    (showGames && games.length > 0);
   261→
   262→  return (
   263→    <div className="min-h-screen bg-background">
   264→      <div className="hidden md:block">
   265→        <Navbar />
   266→      </div>
   267→
   268→      <div className="container mx-auto px-4 md:pt-28 pb-20">
   269→        <MobileDiscoveryHeader
   270→          title={t("marketplace.title")}
   271→          searchValue={search}
   272→          onSearchChange={setSearch}
   273→          searchPlaceholder={t("marketplace.searchPlaceholder")}
   274→          categories={chipCategories}
   275→          activeCategory={activeChip}
   276→          onCategoryChange={handleChipChange}
   277→          onOpenFilters={() => setFilterSheetOpen(true)}
   278→        />
   279→
   280→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="mb-10 hidden md:block">
   281→          <h1 className="font-display text-4xl font-bold text-foreground mb-2">{t("marketplace.title")}</h1>
   282→          <p className="text-muted-foreground text-lg">{t("marketplace.subtitle")}</p>
   283→        </motion.div>
   284→
   285→        <div className="hidden md:flex flex-col gap-3 mb-8">
   286→          <div className="flex gap-2">
   287→            <div className="relative flex-1">
   288→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   289→              <Input placeholder="Pesquisar..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-10 glass border-border h-10" />
   290→            </div>
   291→            <Sheet open={filterSheetOpen} onOpenChange={setFilterSheetOpen}>
   292→              <SheetTrigger asChild>
   293→                <Button variant="outline" size="sm" className="h-10 gap-1.5 shrink-0 relative">
   294→                  <SlidersHorizontal className="h-4 w-4" />
   295→                  <span>Filtros</span>
   296→                  {activeFilterCount > 0 && (
   297→                    <span className="absolute -top-1 -right-1 h-4 w-4 rounded-full bg-primary text-[10px] font-bold text-primary-foreground flex items-center justify-center">
   298→                      {activeFilterCount}
   299→                    </span>
   300→                  )}
   301→                </Button>
   302→              </SheetTrigger>
   303→              <SheetContent side="bottom" className="rounded-t-3xl max-h-[85vh] overflow-y-auto">
   304→                <SheetHeader>
   305→                  <div className="mx-auto mb-2 h-1.5 w-12 rounded-full bg-border" />
   306→                  <SheetTitle className="text-left flex items-center gap-2">
   307→                    <SlidersHorizontal className="h-4 w-4 text-primary" /> Filtros
   308→                  </SheetTitle>
   309→                </SheetHeader>
   310→                <div className="py-4">
   311→                  <FiltersBody />
   312→                </div>
   313→                <div className="sticky bottom-0 bg-background pt-3 pb-2 flex gap-2 border-t border-border">
   314→                  <Button variant="outline" className="flex-1" onClick={() => {
   315→                    setTypeFilter("all"); setContentType("all"); setCountry(""); setRegion(""); setSortBy("newest");
   316→                  }}>Limpar</Button>
   317→                  <Button className="flex-1" onClick={() => setFilterSheetOpen(false)}>Aplicar</Button>
   318→                </div>
   319→              </SheetContent>
   320→            </Sheet>
   321→          </div>
   322→
   323→          {activeFilterCount > 0 && (
   324→            <div className="flex flex-wrap gap-1.5">
   325→              {contentLabel && (<Chip onRemove={() => setContentType("all")}>{contentLabel}</Chip>)}
   326→              {typeLabel && (<Chip onRemove={() => setTypeFilter("all")}>{typeLabel}</Chip>)}
   327→              {sortBy !== "newest" && (<Chip onRemove={() => setSortBy("newest")}>{sortLabel}</Chip>)}
   328→              {region && (<Chip onRemove={() => setRegion("")}>{PROVINCES.find(p => p.value === region)?.label || region}</Chip>)}
   329→              {country && !region && (<Chip onRemove={() => setCountry("")}>{country.toUpperCase()}</Chip>)}
   330→            </div>
   331→          )}
   332→        </div>
   333→
   334→        <MobileFilterSheet
   335→          open={filterSheetOpen && isMobile}
   336→          onOpenChange={setFilterSheetOpen}
   337→          title="Filtrar"
   338→          onReset={() => { setTypeFilter("all"); setContentType("all"); setCountry(""); setRegion(""); setSortBy("newest"); }}
   339→          onApply={() => {}}
   340→        >
   341→          <FiltersBody />
   342→        </MobileFilterSheet>
   343→
   344→        <div className="mt-3 md:mt-0" />
   345→
   346→        {loading ? (
   347→          <div className="flex justify-center py-20">
   348→            <div className="h-8 w-8 animate-spin rounded-full border-2 border-primary border-t-transparent" />
   349→          </div>
   350→        ) : isCatalogEmpty ? (
   351→          <MarketplaceEmptyState />
   352→        ) : !hasFilteredResults ? (
   353→          <div className="text-center py-20">
   354→            <Ticket className="h-16 w-16 text-muted-foreground/30 mx-auto mb-4" />
   355→            <p className="text-xl text-muted-foreground">{t("marketplace.noResults")}</p>
   356→          </div>
   357→        ) : (
   358→          <div className="space-y-10">
   359→            {showGames && games.length > 0 && (
   360→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   361→                <div className="flex items-center gap-2 mb-4">
   362→                  <Zap className="h-5 w-5 text-primary" />
   363→                  <h2 className="font-display text-xl font-bold text-foreground">Jogos & Diversão</h2>
   364→                  <Badge variant="secondary">{games.length}</Badge>
   365→                </div>
   366→                <div className="grid grid-cols-2 gap-3 sm:gap-6 lg:grid-cols-4">
   367→                  {games.map((game, i) => (
   368→                    <Link key={game.id} to={`/games/${game.type === 'spin' ? 'spin-wheel' : 'millionaire'}/${game.id}`}>
   369→                      <Card className="glass group hover:border-primary/30 transition-all overflow-hidden h-full">
   370→                        <div className="relative aspect-video bg-secondary overflow-hidden">
   371→                          {game.background_image_url ? (
   372→                            <OptimizedImage
   373→                              src={game.background_image_url}
   374→                              alt={game.name}
   375→                              optimizeWidth={480}
   376→                              className="w-full h-full object-cover group-hover:scale-110 transition-transform duration-500"
   377→                            />
   378→                          ) : (
   379→                            <div className="w-full h-full flex items-center justify-center bg-primary/10">
   380→                              <Gamepad2 className="h-12 w-12 text-primary/40" />
   381→                            </div>
   382→                          )}
   383→                          <div className="absolute top-2 right-2">
   384→                            <Badge className="bg-black/60 backdrop-blur-md border-none">
   385→                              {game.type === 'spin' ? 'Roda' : 'Milionário'}
   386→                            </Badge>
   387→                          </div>
   388→                        </div>
   389→                        <CardContent className="p-3">
   390→                          <h3 className="font-bold text-sm line-clamp-1 group-hover:text-primary transition-colors">{game.name}</h3>
   391→                          <p className="text-[10px] text-muted-foreground line-clamp-2 mt-1">{game.description}</p>
   392→                          <div className="flex items-center justify-between mt-3">
   393→                            <span className="text-[10px] font-bold text-primary">
   394→                              {game.spin_cost > 0 ? `${game.spin_cost} MZN` : 'GRÁTIS'}

   396→                            <Button size="sm" className="h-6 text-[10px] px-2">Jogar</Button>
   397→                          </div>
   398→                        </CardContent>
   399→                      </Card>
   400→                    </Link>
   401→                  ))}
   402→                </div>
   403→              </motion.div>
   404→            )}
   405→
   406→            {showContests && filteredContests.length > 0 && (
   407→              <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   408→                {contentType === "all" && (
   409→                  <div className="flex items-center gap-2 mb-4">
   410→                    <Trophy className="h-5 w-5 text-primary" />
   411→                    <h2 className="font-display text-xl font-bold text-foreground">Concursos</h2>
   412→                    <Badge variant="secondary">{filteredContests.length}</Badge>

   415→                <div className="grid grid-cols-2 gap-3 sm:gap-6 lg:grid-cols-3">

   421→                      transition={{ delay: i * 0.05, type: "spring", stiffness: 200 }}
   422→                      whileHover={{ y: -4 }}

   440→                              {contestStatusMap[contest.status]?.label || contest.status}
