    97→function timeAgo(dateStr: string): string {
    98→  const now = new Date();
    99→  const date = new Date(dateStr);
   100→  const diffMs = now.getTime() - date.getTime();
   101→  const diffSec = Math.floor(diffMs / 1000);
   102→  const diffMin = Math.floor(diffSec / 60);
   103→  const diffHour = Math.floor(diffMin / 60);
   104→  const diffDay = Math.floor(diffHour / 24);
   105→
   106→  if (diffSec < 60) return 'agora mesmo';
   107→  if (diffMin < 60) return `ha ${diffMin} min`;
   108→  if (diffHour < 24) return `ha ${diffHour}h`;
   109→  if (diffDay < 7) return `ha ${diffDay}d`;
   110→  return `ha ${Math.floor(diffDay / 7)} semanas`;
   111→}

   113→function getStatusColor(status: BetStatus): string {
   114→  switch (status) {
   115→    case 'open':
   116→      return 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30';
   117→    case 'settled':
   118→      return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   119→    case 'refunded':
   120→      return 'bg-sky-500/20 text-sky-400 border-sky-500/30';
   121→    case 'cancelled':
   122→      return 'bg-red-500/20 text-red-400 border-red-500/30';
   123→    default:
   124→      return 'bg-white/10 text-white border-white/20';
   125→  }
   126→}
   127→
   128→function getBetStatusLabel(bet: Bet): string {
   129→  if (bet.status === 'settled' && bet.is_correct) return 'GANHOU';
   130→  if (bet.status === 'settled' && !bet.is_correct) return 'PERDEU';
   131→  return BET_STATUS_LABELS[bet.status];
   132→}
   133→
   134→function getBetStatusBadgeStyle(bet: Bet): string {
   135→  if (bet.status === 'settled' && bet.is_correct) {
   136→    return 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
   137→  }
   138→  if (bet.status === 'settled' && !bet.is_correct) {
   139→    return 'bg-red-500/20 text-red-400 border-red-500/30';
   140→  }
   141→  return getStatusColor(bet.status);
   142→}
