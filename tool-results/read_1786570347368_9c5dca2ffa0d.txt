   508→   508→function slugify(text: string): string {
   509→   509→  return text
   510→   510→    .toLowerCase()
   511→   511→    .normalize('NFD')
   512→   512→    .replace(/[\u0300-\u036f]/g, '')
   513→   513→    .replace(/[^a-z0-9]+/g, '-')
   514→   514→    .replace(/(^-|-$)/g, '');
   515→   515→}

   521→   521→export async function createSeason(data: CreateSeasonData): Promise<Season> {
   522→   522→  const slug = slugify(data.name) + '-' + Date.now().toString(36);
   523→   523→  const { data: season, error } = await sb
   524→   524→    .from('esport_seasons')
   525→   525→    .insert({
   526→   526→      ...data,

   724→   724→export async function generateSeasonSchedule(seasonId: string, teamIds: string[], teamNames: string[], teamLogos: string[]): Promise<SeasonMatch[]> {
   725→   725→  const season = await getSeason(seasonId);
   726→   726→  if (!season) throw new Error('Temporada nao encontrada');
   727→   727→
   728→   728→  const totalRounds = season.total_rounds || 10;
   729→   729→  const matches: SeasonMatch[] = [];
   730→   730→  let matchNum = 0;
   731→   731→
   732→   732→  for (let round = 1; round <= totalRounds; round++) {
   733→   733→    const shuffled = [...teamIds]
   734→   734→      .map((id, i) => ({ id, name: teamNames[i], logo: teamLogos[i] }))
   735→   735→      .sort(() => Math.random() - 0.5);
   736→   736→
   737→   737→    for (let i = 0; i < shuffled.length - 1; i += 2) {
   738→   738→      matchNum++;
   739→   739→      const match = await createSeasonMatch({
   740→   740→        season_id: seasonId,
   741→   741→        round_number: round,
   742→   742→        match_number: matchNum,
   743→   743→        team1_id: shuffled[i].id,
   744→   744→        team1_name: shuffled[i].name,
   745→   745→        team1_logo: shuffled[i].logo,
   746→   746→        team2_id: shuffled[i + 1].id,
   747→   747→        team2_name: shuffled[i + 1].name,
   748→   748→        team2_logo: shuffled[i + 1].logo,
   749→   749→      });
   750→   750→      matches.push(match);
   751→   751→    }
   752→   752→  }
   753→   753→  return matches;
   754→   754→}

   756→   756→export async function updateSeasonStandings(seasonId: string, matchId: string): Promise<SeasonTeam[]> {
   757→   757→  const match = await sb.from('esport_season_matches').select('*').eq('id', matchId).single();
   758→   758→  const season = await getSeason(seasonId);
   759→   759→  if (!match || !season) throw new Error('Jogo ou temporada nao encontrados');
   760→   760→
   761→   761→  if (match.team1_id && match.winner_id === match.team1_id) {
   762→   762→    await sb.raw(
   763→   763→      `UPDATE esport_season_teams SET 
   764→   764→        matches_played = matches_played + 1, matches_won = matches_won + 1,
   765→   765→        points = points + ?, total_kills = total_kills + ?, total_deaths = total_deaths + ?,
   766→   766→        win_streak = win_streak + 1, loss_streak = 0,
   767→   767→        best_win_streak = GREATEST(best_win_streak, win_streak + 1)
   768→   768→        WHERE season_id = ? AND team_id = ?`,
   769→   769→      [season.points_win, match.team1_kills || 0, match.team2_kills || 0, seasonId, match.team1_id]
   770→   770→    );

   772→   772→      `UPDATE esport_season_teams SET 
   773→   773→        matches_played = matches_played + 1, matches_lost = matches_lost + 1,

   775→   775→        loss_streak = loss_streak + 1, win_streak = 0
   776→   776→        WHERE season_id = ? AND team_id = ?`,
   777→   777→      [season.points_loss, match.team2_kills || 0, match.team1_kills || 0, seasonId, match.team2_id]
   778→   778→    );
   779→   779→  } else if (match.team2_id && match.winner_id === match.team2_id) {
   780→   780→    await sb.raw(
   781→   781→      `UPDATE esport_season_teams SET 
   782→   782→        matches_played = matches_played + 1, matches_won = matches_won + 1,
   783→   783→        points = points + ?, total_kills = total_kills + ?, total_deaths = total_deaths + ?,
   784→   784→        win_streak = win_streak + 1, loss_streak = 0,
   785→   785→        best_win_streak = GREATEST(best_win_streak, win_streak + 1)
   786→   786→        WHERE season_id = ? AND team_id = ?`,
   787→   787→      [season.points_win, match.team2_kills || 0, match.team1_kills || 0, seasonId, match.team2_id]
   788→   788→    );
   789→   789→    await sb.raw(
   790→   790→      `UPDATE esport_season_teams SET 
   791→   791→        matches_played = matches_played + 1, matches_lost = matches_lost + 1,
   792→   792→        points = points + ?, total_kills = total_kills + ?, total_deaths = total_deaths + ?,
   793→   793→        loss_streak = loss_streak + 1, win_streak = 0
   794→   794→        WHERE season_id = ? AND team_id = ?`,
   795→   795→      [season.points_loss, match.team1_kills || 0, match.team2_kills || 0, seasonId, match.team1_id]
   796→   796→    );
   797→   797→  } else if (match.is_draw) {
   798→   798→    for (const tid of [match.team1_id, match.team2_id]) {
   799→   799→      if (!tid) continue;
   800→   800→      await sb.raw(
   801→   801→        `UPDATE esport_season_teams SET 
   802→   802→          matches_played = matches_played + 1, matches_drawn = matches_drawn + 1,
   803→   803→          points = points + ?, win_streak = 0, loss_streak = 0
   804→   804→          WHERE season_id = ? AND team_id = ?`,
   805→   805→        [season.points_draw, seasonId, tid]
   806→   806→      );
   807→   807→    }
   808→   808→  }
   809→   809→
   810→   810→  return getSeasonStandings(seasonId);
   811→   811→}

   856→   856→export async function placeBet(data: {
   857→   857→  user_id: string;
   858→   858→  match_id?: string;

   860→   860→  championship_id?: string;

   866→   866→}): Promise<Bet> {
   867→   867→  const wallet = await getOrCreateWallet(data.user_id);
   868→   868→  if (wallet.balance < data.amount_wagered) {
   869→   869→    throw new Error('Saldo insuficiente');
   870→   870→  }
   871→   871→
   872→   872→  const newBalance = wallet.balance - data.amount_wagered;
   873→   873→  const potentialPayout = Math.floor(data.amount_wagered * data.odds);
   874→   874→
   875→   875→  const { data: bet, error } = await sb
   876→   876→    .from('esport_bets')
   877→   877→    .insert({

   882→   882→    .single();
   883→   883→  if (error) throw new Error('Erro ao apostar: ' + error.message);
   884→   884→
   885→   885→  await sb
   886→   886→    .from('user_wallets')
   887→   887→    .update({
   888→   888→      balance: newBalance,
   889→   889→      total_wagered: wallet.total_wagered + data.amount_wagered,
   890→   890→    })
   891→   891→    .eq('user_id', data.user_id);
   892→   892→
   893→   893→  await addWalletTransaction({
   894→   894→    user_id: data.user_id,
   895→   895→    type: 'bet_placed',
   896→   896→    amount: -data.amount_wagered,
   897→   897→    balance_after: newBalance,
   898→   898→    description: `Aposta: ${data.predicted_label} @ ${data.odds}x`,
   899→   899→    reference_id: bet.id,
   900→   900→  });
   901→   901→
   902→   902→  return bet;
   903→   903→}

   905→   905→export async function settleBet(betId: string, isCorrect: boolean): Promise<Bet> {
   906→   906→  const { data: bet } = await sb.from('esport_bets').select('*').eq('id', betId).single();
   907→   907→  if (!bet) throw new Error('Aposta nao encontrada');
   908→   908→  if (bet.status !== 'open') throw new Error('Aposta ja resolvida');
   909→   909→
   910→   910→  const wallet = await getOrCreateWallet(bet.user_id);
   911→   911→  let newBalance = wallet.balance;
   912→   912→  let amountWon = 0;
   913→   913→
   914→   914→  if (isCorrect) {
   915→   915→    amountWon = bet.potential_payout;
   916→   916→    newBalance += amountWon;
   917→   917→    await sb.from('user_wallets').update({
   918→   918→      balance: newBalance,
   919→   919→      total_earned: wallet.total_earned + amountWon,
   920→   920→      total_won: wallet.total_won + amountWon,
   921→   921→    }).eq('user_id', bet.user_id);
   922→   922→
   923→   923→    await addWalletTransaction({
   924→   924→      user_id: bet.user_id,
   925→   925→      type: 'bet_won',
   926→   926→      amount: amountWon,
   927→   927→      balance_after: newBalance,
   928→   928→      description: `Ganhaste ${amountWon} moedas!`,
   929→   929→      reference_id: betId,
   930→   930→    });
   931→   931→  } else {
   932→   932→    await sb.from('user_wallets').update({
   933→   933→      total_lost: wallet.total_lost + bet.amount_wagered,
   934→   934→    }).eq('user_id', bet.user_id);
   935→   935→  }
   936→   936→
   937→   937→  const { data: settled, error } = await sb
   938→   938→    .from('esport_bets')
   939→   939→    .update({
   940→   940→      status: 'settled',
   941→   941→      is_correct: isCorrect,
   942→   942→      amount_won: amountWon,
   943→   943→      settled_at: new Date().toISOString(),
   944→   944→    })
   945→   945→    .eq('id', betId)
   946→   946→    .select()
   947→   947→    .single();
   948→   948→  if (error) throw new Error('Erro ao resolver aposta: ' + error.message);
   949→   949→  return settled;
   950→   950→}
