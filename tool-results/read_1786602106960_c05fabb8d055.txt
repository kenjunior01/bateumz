   232→function getInitials(name: string): string {
   233→  return name.split(' ').map(w => w[0]).join('').toUpperCase().slice(0, 2);
   234→}
   235→
   236→function getAvatarColor(name: string): string {
   237→  const colors = ['#00d4ff', '#7b2ff7', '#ff4655', '#fbbf24', '#2ea043', '#ff6b6b', '#58a6ff', '#f97316'];
   238→  let hash = 0;
   239→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   240→  return colors[Math.abs(hash) % colors.length];
   241→}
   242→
   243→function getTimeRemaining(expiresAt: string): string {
   244→  const diff = new Date(expiresAt).getTime() - Date.now();
   245→  if (diff <= 0) return 'Expirado';
   246→  const hours = Math.floor(diff / 3600000);
   247→  const minutes = Math.floor((diff % 3600000) / 60000);
   248→  if (hours >= 24) return `${Math.floor(hours / 24)}d ${hours % 24}h`;
   249→  if (hours > 0) return `${hours}h ${minutes}m`;
   250→  return `${minutes}m`;
   251→}
