    59→function AnimatedSection({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {

    69→/* ─── Counting Number ─── */
    70→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    71→  const [count, setCount] = useState(0);
    72→  const ref = useRef<HTMLSpanElement>(null);
    73→  const inView = useInView(ref, { once: true });
    74→  const hasAnimated = useRef(false);
    75→  useEffect(() => {
    76→    if (!inView || hasAnimated.current) return;
    77→    hasAnimated.current = true;
    78→    const start = performance.now();
    79→    const step = (now: number) => {
    80→      const elapsed = (now - start) / 1000;
    81→      const progress = Math.min(elapsed / duration, 1);
    82→      const eased = 1 - Math.pow(1 - progress, 3);
    83→      setCount(Math.round(eased * target));
    84→      if (progress < 1) requestAnimationFrame(step);
    85→    };
    86→    requestAnimationFrame(step);
    87→  }, [inView, target, duration]);
    88→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
    89→}
