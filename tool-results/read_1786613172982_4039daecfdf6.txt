    53→const fireConfetti = (type: string = "confetti") => {
    54→  const duration = 3 * 1000;
    55→  const animationEnd = Date.now() + duration;
    56→  const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };
    57→
    58→  const randomInRange = (min: number, max: number) => Math.random() * (max - min) + min;
    59→
    60→  const interval = setInterval(() => {
    61→    const timeLeft = animationEnd - Date.now();
    62→    if (timeLeft <= 0) {
    63→      return clearInterval(interval);
    64→    }
    65→
    66→    const particleCount = 50 * (timeLeft / duration);
    67→
    68→    switch (type) {
    69→      case "fireworks":
    70→        confetti({
    71→          ...defaults,
    72→          particleCount,
    73→          spread: 70,
    74→          origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
    75→          colors: ["#ff0000", "#00ff00", "#0000ff", "#ffff00"],
    76→        });
    77→        confetti({
    78→          ...defaults,
    79→          particleCount,
    80→          spread: 70,
    81→          origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
    82→          colors: ["#ff00ff", "#00ffff", "#ff8800", "#88ff00"],
    83→        });
    84→        break;
    85→      case "stars":
    86→        confetti({
    87→          ...defaults,
    88→          particleCount: particleCount * 0.5,
    89→          shapes: ["star"],
    90→          gravity: 0.8,
    91→          scalar: 1.2,
    92→          origin: { y: 0.4 },
    93→          colors: ["#ffd700", "#ffed4e", "#ffffff"],
    94→        });
    95→        break;
    96→      case "poppers":
    97→        for (let i = 0; i < 5; i++) {
    98→          setTimeout(() => {
    99→            confetti({
   100→              ...defaults,
   101→              particleCount: 20,
   102→              origin: { x: 0.2 + i * 0.15, y: 0.6 },
   103→              spread: 60,
   104→              colors: ["#ff4444", "#44ff44", "#4444ff", "#ffff44"],
   105→            });
   106→          }, i * 200);
   107→        }
   108→        break;
   109→      case "zap":
   110→        confetti({
   111→          ...defaults,
   112→          particleCount: 100,
   113→          shapes: ["circle", "square"],
   114→          gravity: 1.2,
   115→          scalar: 0.9,
   116→          origin: { y: 0.2 },
   117→          colors: ["#ff00ff", "#00ffff", "#ff0000", "#00ff00"],
   118→        });
   119→        break;
   120→      case "confetti":
   121→      default:
   122→        confetti({
   123→          ...defaults,
   124→          particleCount,
   125→          origin: { x: randomInRange(0.1, 0.9), y: Math.random() - 0.2 },
   126→        });
   127→        confetti({
   128→          ...defaults,
   129→          particleCount,
   130→          origin: { x: randomInRange(0.2, 0.8), y: Math.random() - 0.1 },
   131→          spread: 180,
   132→        });
   133→        break;
   134→    }
   135→  }, 250);

   153→const pickWeighted = (prizes: WheelPrize[]) => {
   154→  const total = prizes.reduce((s, p) => s + Math.max(0, p.weight), 0) || 1;
   155→  let r = Math.random() * total;
   156→  for (let i = 0; i < prizes.length; i++) {
   157→    r -= Math.max(0, prizes[i].weight);
   158→    if (r <= 0) return i;
   159→  }
   160→  return prizes.length - 1;
   161→};

   521→        )}

   523→        {mode === "vertical" && (
   524→          <div className="w-full max-w-md space-y-4 text-center">
   525→            <div className="w-full h-40 rounded-2xl bg-black/30 border-2 border-white/20 flex items-center justify-center">
   526→              <p className="text-white/50 text-sm">{t("wheel.webcam")}</p>
   527→            </div>
   528→            {(branding?.companyLogoUrl || branding?.companySlogan) && (
   529→              <div className="flex flex-col items-center gap-2">
   530→                {branding?.companyLogoUrl && (
   531→                  <img src={branding.companyLogoUrl} alt="Logo" className="h-12 object-contain" />
   532→                )}
   533→                {branding?.companySlogan && (
   534→                  <p className="text-xs font-black uppercase tracking-[0.2em]" style={{ color: `${currentTheme.primaryColor}80` }}>{branding.companySlogan}</p>
   535→                )}
   536→              </div>
   537→            )}
   538→          </div>
   539→        )}
   540→
   541→        <div className="relative group">
   542→          <div className="absolute -top-8 left-1/2 -translate-x-1/2 z-20 drop-shadow-2xl">
   543→            <div className="w-16 h-16 bg-white rounded-full flex items-center justify-center border-4 border-primary shadow-lg relative" style={{ borderColor: currentTheme.primaryColor }}>
   544→              <div className="absolute -bottom-1 left-1/2 -translate-x-1/2">
   545→                <div className="w-0 h-0 border-l-[16px] border-r-[16px] border-t-[28px] border-l-transparent border-r-transparent" style={{ borderTopColor: currentTheme.primaryColor }} />
   546→              </div>
   547→              <Sparkles className="w-6 h-6" style={{ color: currentTheme.primaryColor }} />
   548→            </div>
   549→          </div>
   550→
   551→          <div className="relative p-6 rounded-full bg-white/5 border-2 border-white/10 backdrop-blur-sm shadow-[0_0_80px_rgba(0,0,0,0.6)]" style={{ borderColor: `${currentTheme.primaryColor}40` }}>
   552→            <canvas
   553→              ref={canvasRef}
   554→              width="600"
   555→              height="600"
   556→              className="max-w-[300px] sm:max-w-[400px] md:max-w-[600px] h-auto rounded-full"
   557→              style={{
   558→                transform: `rotate(${rotation}deg)`,
   559→                transition: spinning ? 'none' : 'transform 0.5s cubic-bezier(0.1, 0, 0.3, 1)'
   560→              }}
   561→            />
   562→            
   563→            <button
   564→                onClick={spin}
   565→                disabled={spinning || loading}
   566→                className={`absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-24 h-24 rounded-full 
   567→                flex flex-col items-center justify-center font-black text-sm transition-all duration-300 z-20
   568→                ${spinning || loading ? 'bg-gray-500 scale-95 opacity-50 cursor-not-allowed' : 'bg-white hover:scale-110 shadow-2xl active:scale-90 cursor-pointer'}`}
   569→                style={{ color: wheelConfig?.wheel_background_color || currentTheme.primaryColor }}
   570→              >
   571→                {loading || spinning ? (
   572→                  <span className="flex items-center gap-2">
   573→                    <Sparkles className="w-6 h-6 animate-spin" />
   574→                    {t("wheel.spinning")}
   575→                  </span>
   576→                ) : (
   577→                  <span className="flex items-center gap-2">
   578→                    <PartyPopper className="w-6 h-6" />
   579→                    {t("wheel.spin")}
   580→                  </span>
   581→                )}
   582→              </button>
   583→          </div>
   584→        </div>
   585→
   586→        <div className={`${mode === "horizontal" ? "w-full md:w-96" : "w-full max-w-md"} space-y-6`}>
   587→          <AnimatePresence>
   588→            {result && (
   589→              <motion.div
   590→                initial={{ opacity: 0, scale: 0.3, y: 100 }}
   591→                animate={{ opacity: 1, scale: 1, y: 0 }}
   592→                exit={{ opacity: 0, scale: 0.3, y: -100 }}
   593→                transition={{ type: "spring", bounce: 0.6, duration: 0.8 }}
   594→                className={`rounded-3xl p-8 text-center space-y-4 shadow-2xl border-2 ${
   595→                  result.rewardType === "none" || isNoWinLabel(result.label)
   596→                    ? "bg-secondary/90 text-muted-foreground border-white/20"
   597→                    : "border-white/20"
   598→                }`}
   599→                style={
   600→                  result.rewardType !== "none" && !isNoWinLabel(result.label)
   601→                    ? {
   602→                        background: `linear-gradient(135deg, ${currentTheme.primaryColor}66, ${currentTheme.primaryColor}22)`,
   603→                        borderColor: `${currentTheme.primaryColor}66`,
   604→                        color: currentTheme.textColor,
   605→                      }
   606→                    : undefined
   607→                }
   608→              >
   609→                {result.rewardType !== "none" && !isNoWinLabel(result.label) ? (
   610→                  <div className="flex flex-col items-center gap-4">
   611→                    <motion.div
   612→                      animate={{
   613→                        rotate: [0, 15, -15, 0],
   614→                        scale: [1, 1.4, 1],
   615→                      }}
   616→                      transition={{ duration: 0.6, repeat: Infinity, repeatDelay: 2 }}
   617→                      className="text-yellow-400"
   618→                    >
   619→                      {getResultIcon(result)}
   620→                    </motion.div>
   621→                    <motion.p
   622→                      initial={{ y: 40, opacity: 0 }}
   623→                      animate={{ y: 0, opacity: 1 }}
   624→                      transition={{ delay: 0.2 }}
   625→                      className="font-black text-3xl"
   626→                    >
   627→                      🎉 {t("wheel.win")}! 🎉
   628→                    </motion.p>
   629→                    <motion.p
   630→                      initial={{ scale: 0 }}
   631→                      animate={{ scale: 1 }}
   632→                      transition={{ delay: 0.4, type: "spring", stiffness: 200 }}
   633→                      className="text-6xl font-black tracking-tighter"
   634→                    >
   635→                      {result.rewardValue || result.label}
   636→                    </motion.p>
   637→                    {result.description && (
   638→                      <p className="opacity-90 text-lg">{result.description}</p>
   639→                    )}
   640→                    {result.rewardImageUrl && (
   641→                      <img 
   642→                        src={result.rewardImageUrl} 
   643→                        alt="Prémio" 
   644→                        className="w-32 h-32 object-cover rounded-2xl shadow-lg"
   645→                      />
   646→                    )}
   647→                  </div>
   648→                ) : (
   649→                  <p className="font-bold text-3xl flex items-center gap-3 justify-center">
   650→                    😅 {t("tryAgain")}
   651→                  </p>
   652→                )}
   653→              </motion.div>
   654→            )}
   655→          </AnimatePresence>
   656→
   657→          {prizes.length < 4 && (
   658→            <div className="flex items-start gap-3 p-3 rounded-xl bg-amber-500/10 border border-amber-500/20 text-amber-200">
   659→              <AlertTriangle className="h-5 w-5 flex-shrink-0 text-amber-400" />
   660→              <p className="text-sm">{t("wheel.minSegmentsWarning")}</p>
   661→            </div>
   662→          )}
   663→
   664→          <div className="rounded-2xl bg-black/20 border border-white/10 backdrop-blur-xl text-white overflow-hidden">
   665→            <div className="bg-white/5 p-4 border-b border-white/10 flex items-center justify-between">
   666→              <span className="text-xs uppercase tracking-wider opacity-70 font-bold">{t("wheel.prizes")}</span>
   667→              {editable && onChange && (
   668→                <Sheet>
   669→                  <SheetTrigger asChild>
   670→                    <button className="inline-flex items-center gap-1 px-3 py-1 rounded-full bg-card border border-border text-[11px] font-medium hover:bg-secondary">
   671→                      <Settings2 className="h-3.5 w-3.5" /> {t("wheel.config")}
   672→                    </button>
   673→                  </SheetTrigger>
   674→                  <SheetContent side="bottom" className="rounded-t-3xl max-h-[85vh] overflow-y-auto">
   675→                    <SheetHeader className="mb-4"><SheetTitle>{t("wheel.configureTitle")}</SheetTitle></SheetHeader>
   676→                    <div className="space-y-4 pb-6">
   677→                      {prizes.map((p) => {
   678→                        const pct = ((Math.max(0, p.weight) / totalWeight) * 100).toFixed(1);
   679→                        return (
   680→                          <div key={p.id} className="rounded-2xl border border-border p-3 space-y-2">
   681→                            <div className="flex items-center gap-2">
   682→                              <input
   683→                                type="color"
   684→                                value={p.color}
   685→                                onChange={(e) => patchPrize(p.id, { color: e.target.value })}
   686→                                className="h-9 w-9 rounded-md border border-border bg-transparent cursor-pointer"
   687→                              />
   688→                              <Input
   689→                                value={p.label}
   690→                                onChange={(e) => patchPrize(p.id, { label: e.target.value })}
   691→                                placeholder={t("wheel.prizeName")}
   692→                                className="flex-1"
   693→                              />
   694→                              <button
   695→                                onClick={() => removePrize(p.id)}
   696→                                className="p-2 rounded-md hover:bg-destructive/10 text-destructive"
   697→                                aria-label="Remove"
   698→                              >
   699→                                <Trash2 className="h-4 w-4" />
   700→                              </button>
   701→                            </div>
   702→                            <div className="grid grid-cols-2 gap-2">
   703→                              <div className="space-y-1">
   704→                                <Label className="text-[11px] text-muted-foreground">{t("wheel.prizeValue")}</Label>
   705→                                <Input
   706→                                  value={p.rewardValue || ""}
   707→                                  onChange={(e) => patchPrize(p.id, { rewardValue: e.target.value })}
   708→                                  placeholder="Ex: 100 MZN"
   709→                                  className="text-xs"
   710→                                />
   711→                              </div>
   712→                              <div className="space-y-1">
   713→                                <Label className="text-[11px] text-muted-foreground">{t("wheel.weight")} ({pct}%)</Label>
   714→                                <Input
   715→                                  type="number"
   716→                                  min={0}
   717→                                  max={1000}
   718→                                  value={p.weight}
   719→                                  onChange={(e) => patchPrize(p.id, { weight: Math.max(0, Number(e.target.value) || 0) })}
   720→                                  className="text-xs"
   721→                                />
   722→                              </div>
   723→                            </div>
   724→                            <div className="space-y-1">
   725→                              <Label className="text-[11px] text-muted-foreground">{t("wheel.effectType")}</Label>
   726→                              <select
   727→                                value={p.effectType || "confetti"}
   728→                                onChange={(e) => patchPrize(p.id, { effectType: e.target.value as any })}
   729→                                className="w-full px-3 py-2 rounded-xl bg-card border border-border text-xs"
   730→                              >
   731→                                <option value="confetti">Confetti</option>
   732→                                <option value="fireworks">Fireworks</option>
   733→                                <option value="stars">Stars</option>
   734→                                <option value="poppers">Poppers</option>
   735→                                <option value="zap">Zap</option>
   736→                              </select>
   737→                            </div>
   738→                          </div>
   739→                        );
   740→                      })}
   741→                      <button
   742→                        onClick={addPrize}
   743→                        className="w-full flex items-center justify-center gap-2 py-3 rounded-2xl border-2 border-dashed border-border text-sm text-muted-foreground hover:bg-secondary"
   744→                      >
   745→                        <Plus className="h-4 w-4" /> {t("wheel.addPrize")}
   746→                      </button>
   747→                      <p className="text-[11px] text-muted-foreground text-center">
   748→                        {t("wheel.weightHint")}
   749→                      </p>
   750→                    </div>
   751→                  </SheetContent>
   752→                </Sheet>
   753→              )}
   754→            </div>
   755→            <div className="p-0 max-h-[300px] overflow-y-auto custom-scrollbar">
   756→              {prizes.map((p, i) => (
   757→                <div key={i} className="flex items-center gap-4 p-4 hover:bg-white/5 transition-colors border-b border-white/5 last:border-0">
   758→                  <div className="w-3 h-3 rounded-full" style={{ backgroundColor: p.color }} />
   759→                  <div className="flex-1">
   760→                    <p className="font-bold text-sm">{p.label}</p>
   761→                    {p.rewardValue && <p className="text-xs opacity-50">{p.rewardValue}</p>}
   762→                  </div>
   763→                  {p.rewardImageUrl && (
   764→                    <img src={p.rewardImageUrl} alt="" className="w-8 h-8 rounded object-cover" />
   765→                  )}
   766→                </div>
   767→              ))}
   768→            </div>
   769→          </div>
   770→        </div>
   771→      </div>
   772→
   773→      <div className="fixed bottom-8 right-8 flex gap-3 z-50">
   774→        <button 
   775→          onClick={() => setSoundEnabled(!soundEnabled)}
   776→          className="p-3 rounded-full bg-white/10 border border-white/20 text-white hover:bg-white/20 transition-all"
   777→        >
   778→          {soundEnabled ? <Volume2 className="h-5 w-5" /> : <VolumeX className="h-5 w-5" />}
   779→        </button>
   780→        <button className="p-3 rounded-full bg-white/10 border border-white/20 text-white hover:bg-white/20 transition-all">
   781→          <Share2 className="h-5 w-5" />
   782→        </button>
   783→      </div>
   784→    </div>
   785→  );
   786→};
