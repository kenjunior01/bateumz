    91→export default function MillionairePage() {
    92→  const { gameId } = useParams();
    93→  const navigate = useNavigate();
    94→  const { user } = useAuth();
    95→  const { region } = useRegionalTheme();
    96→
    97→  const [screen, setScreen] = useState<Screen>("lobby");
    98→  const [phase, setPhase] = useState<Phase>("question-in");
    99→  const [questions, setQuestions] = useState<Question[]>([]);
   100→  const [currentLevel, setCurrentLevel] = useState(1);
   101→  const [selectedAnswer, setSelectedAnswer] = useState<string | null>(null);
   102→  const [answered, setAnswered] = useState(false);
   103→  const [timeLeft, setTimeLeft] = useState(45);
   104→  const [lifelinesUsed, setLifelinesUsed] = useState<Record<string, boolean>>({});
   105→  const [soundEnabled, setSoundEnabled] = useState(true);
   106→  const [disabledOptions, setDisabledOptions] = useState<string[]>([]);
   107→  const [loading, setLoading] = useState(true);
   108→  const [gameName, setGameName] = useState("");
   109→  const [bgColor, setBgColor] = useState("#020817");
   110→  const [primaryColor, setPrimaryColor] = useState("#fbbf24");
   111→  const [leaderboard, setLeaderboard] = useState<LeaderboardEntry[]>([]);
   112→  const [audienceVotes, setAudienceVotes] = useState<number[]>([]);
   113→  const [showAudience, setShowAudience] = useState(false);
   114→  const [phoneFriend, setPhoneFriend] = useState("");
   115→  const [showPhone, setShowPhone] = useState(false);
   116→  const [questionKey, setQuestionKey] = useState(0);
   117→  const timerRef = useRef<ReturnType<typeof setInterval>>();
   118→
   119→  const totalQuestions = 15;
   120→  const currentQuestion = useMemo(() => questions[currentLevel - 1], [questions, currentLevel]);
   121→  const currentPrize = PRIZE_STRUCTURE[currentLevel - 1];
   122→
   123→  useEffect(() => {
   124→    loadGameData();
   125→    loadLeaderboard();
   126→  }, []);
   127→
   128→  const loadGameData = async () => {
   129→    if (gameId) {
   130→      const { data } = await supabase.from("millionaire_games").select("*").eq("id", gameId).single();
   131→      if (data) {
   132→        setGameName(data.name || "Quem Quer Ser Milionario?");
   133→        setBgColor(data.background_color || "#020817");
   134→        setPrimaryColor(data.primary_color || "#fbbf24");
   135→      }
   136→    }
   137→    setLoading(false);
   138→  };
   139→
   140→  const loadLeaderboard = async () => {
   141→    if (!gameId) return;
   142→    const { data } = await (supabase as any)
   143→      .from("millionaire_leaderboard")
   144→      .select("*, profiles!inner(display_name, avatar_url)")
   145→      .eq("game_id", gameId)
   146→      .order("highest_prize", { ascending: false })
   147→      .limit(10);
   148→    if (data) {
   149→      setLeaderboard(data.map((d: any) => ({
   150→        user_id: d.user_id,
   151→        display_name: d.profiles?.display_name || "Anonymous",
   152→        avatar_url: d.profiles?.avatar_url,
   153→        highest_level: d.highest_level,
   154→        highest_prize: d.highest_prize,
   155→        total_plays: d.total_plays,
   156→        total_wins: d.total_wins,
   157→        average_level: d.average_level,
   158→        last_played_at: d.last_played_at,
   159→      })));
   160→    }
   161→  };
   162→
   163→  const startGame = async () => {
   164→    setLoading(true);
   165→    const trivia = await fetchTriviaQuestions(totalQuestions, undefined, "easy", "multiple");
   166→    if (trivia.length > 0) {
   167→      const formatted: Question[] = trivia.map((tq, i) => {
   168→        const s = shuffleAnswers(decodeHTMLEntities(tq.correct_answer), tq.incorrect_answers.map(a => decodeHTMLEntities(a)));
   169→        return { id: `t-${i}`, question_number: i + 1, question_text: decodeHTMLEntities(tq.question), ...s };
   170→      });
   171→      setQuestions(formatted);
   172→    } else {
   173→      setQuestions(defaultQuestions);
   174→      toast.warning("Perguntas padrao carregadas");
   175→    }
   176→    setLoading(false);
   177→    setScreen("playing");
   178→    setCurrentLevel(1);
   179→    setSelectedAnswer(null);
   180→    setAnswered(false);
   181→    setTimeLeft(45);
   182→    setLifelinesUsed({});
   183→    setDisabledOptions([]);
   184→    setAudienceVotes([]);
   185→    setPhoneFriend("");
   186→  };
   187→
   188→  useEffect(() => {
   189→    if (screen !== "playing" || answered) return;
   190→    timerRef.current = setInterval(() => setTimeLeft(p => p - 1), 1000);
   191→    return () => { if (timerRef.current) clearInterval(timerRef.current); };
   192→  }, [screen, answered, currentLevel]);
   193→
   194→  useEffect(() => {
   195→    if (timeLeft === 0 && !answered && screen === "playing") {
   196→      setAnswered(true);
   197→      setPhase("reveal");
   198→      setTimeout(() => {
   199→        setScreen("lost");
   200→        toast.error("Tempo esgotado!");
   201→      }, 2500);
   202→    }
   203→  }, [timeLeft, answered, screen]);
   204→
   205→  const handleAnswer = (choice: string) => {
   206→    if (answered || screen !== "playing") return;
   207→    setSelectedAnswer(choice);
   208→    setAnswered(true);
   209→    setPhase("thinking");
   210→
   211→    setTimeout(() => {
   212→      setPhase("reveal");
   213→      const isCorrect = choice === currentQuestion.correct_answer;
   214→
   215→      if (isCorrect) {
   216→        fireCorrectEffects();
   217→        if (currentLevel === totalQuestions) {
   218→          setTimeout(() => {
   219→            fireWinEffects();
   220→            setScreen("won");
   221→            saveSession("completed", currentPrize.amount);
   222→          }, 2000);
   223→        } else {
   224→          setTimeout(() => {
   225→            setPhase("transition");
   226→            setTimeout(() => {
   227→              setCurrentLevel(p => p + 1);
   228→              setAnswered(false);
   229→              setSelectedAnswer(null);
   230→              setDisabledOptions([]);
   231→              setTimeLeft(45);
   232→              setAudienceVotes([]);
   233→              setPhoneFriend("");
   234→              setQuestionKey(k => k + 1);
   235→              setPhase("question-in");
   236→            }, 800);
   237→          }, 2000);
   238→        }
   239→      } else {
   240→        fireWrongEffects();
   241→        setTimeout(() => {
   242→          setScreen("lost");
   243→          saveSession("abandoned", calculateSafePrize());
   244→        }, 2500);
   245→      }
   246→    }, 3000);
   247→  };
   248→
   249→  const calculateSafePrize = () => {
   250→    let safe = 0;
   251→    for (let i = 0; i < currentLevel - 1; i++) {
   252→      if (PRIZE_STRUCTURE[i]?.is_safe_haven) safe = PRIZE_STRUCTURE[i].amount;
   253→    }
   254→    return safe;
   255→  };
   256→
   257→  const saveSession = async (status: string, prize: number) => {
   258→    if (!user || !region || !gameId) return;
   259→    await supabase.from("millionaire_sessions").insert({
   260→      game_id: gameId,
   261→      user_id: user.id,
   262→      region_id: region.id,
   263→      current_level: currentLevel,
   264→      prize_won: prize,
   265→      status,
   266→    });
   267→    loadLeaderboard();
   268→  };
   269→
   270→  const handle5050 = () => {
   271→    if (lifelinesUsed["50_50"] || answered) return;
   272→    setLifelinesUsed(p => ({ ...p, "50_50": true }));
   273→    const wrong = ["A", "B", "C", "D"].filter(l => l !== currentQuestion.correct_answer);
   274→    const toDisable = wrong.sort(() => 0.5 - Math.random()).slice(0, 2);
   275→    setDisabledOptions(toDisable);
   276→    toast.info("50:50 - Duas respostas removidas!");
   277→  };
   278→
   279→  const handleAudience = () => {
   280→    if (lifelinesUsed.audience || answered) return;
   281→    setLifelinesUsed(p => ({ ...p, audience: true }));
   282→    setShowAudience(true);
   283→    const correct = currentQuestion.correct_answer.charCodeAt(0) - 65;
   284→    const votes = [0, 0, 0, 0].map((_, i) => {
   285→      if (i === correct) return 40 + Math.floor(Math.random() * 40);
   286→      return Math.floor(Math.random() * 20);
   287→    });
   288→    const total = votes.reduce((a, b) => a + b, 0);
   289→    setAudienceVotes(votes.map(v => Math.round((v / total) * 100)));
   290→    toast.info("O Publico votou!");
   291→    setTimeout(() => setShowAudience(false), 5000);
   292→  };
   293→
   294→  const handlePhone = () => {
   295→    if (lifelinesUsed.phone || answered) return;
   296→    setLifelinesUsed(p => ({ ...p, phone: true }));
   297→    setShowPhone(true);
   298→    const friends = ["Ana (Professora)", "Carlos (Engenheiro)", "Maria (Medica)", "Joao (Programador)"];
   299→    const friend = friends[Math.floor(Math.random() * friends.length)];
   300→    const correct = currentQuestion.correct_answer;
   301→    const correctText = currentQuestion[`option_${correct.toLowerCase()}` as keyof Question] as string;
   302→    const confidence = Math.random() > 0.2 ? "bastante seguro" : "nao tenho certeza";
   303→    setPhoneFriend(`${friend} diz: \"Acho que e ${correctText}... estou ${confidence}!\"`);
   304→    toast.info(`Ligacao para ${friend.split(" (")[0]}...`);
   305→    setTimeout(() => setShowPhone(false), 5000);
   306→  };
   307→
   308→  const fireCorrectEffects = () => {
   309→    confetti({ particleCount: 50, spread: 60, origin: { y: 0.7 }, colors: ["#22c55e", "#4ade80", "#ffffff"] });
   310→  };
   311→
   312→  const fireWrongEffects = () => {
   313→    setTimeout(() => confetti({ particleCount: 30, spread: 60, origin: { y: 0.6 }, colors: ["#ef4444", "#991b1b"] }), 500);
   314→  };
   315→
   316→  const fireWinEffects = () => {
   317→    confetti({ particleCount: 300, spread: 120, origin: { y: 0.5 } });
   318→    setTimeout(() => confetti({ particleCount: 200, spread: 150, origin: { x: 0.2, y: 0.5 }, colors: ["#fbbf24", "#f59e0b", "#ffffff", "#fde68a"] }), 200);
   319→    setTimeout(() => confetti({ particleCount: 200, spread: 150, origin: { x: 0.8, y: 0.5 }, colors: ["#fbbf24", "#f59e0b", "#ffffff", "#fde68a"] }), 400);
   320→    setTimeout(() => {
   321→      const end = Date.now() + 5000;
   322→      const iv = setInterval(() => {
   323→        if (Date.now() > end) return clearInterval(iv);
   324→        confetti({ particleCount: 3, angle: 60, spread: 55, origin: { x: 0, y: 0.6 }, colors: ["#fbbf24", "#ffffff"] });
   325→        confetti({ particleCount: 3, angle: 120, spread: 55, origin: { x: 1, y: 0.6 }, colors: ["#fbbf24", "#ffffff"] });
   326→      }, 60);
   327→    }, 800);
   328→  };
   329→
   330→  const formatPrize = (n: number) => n.toLocaleString("pt-MZ");
   331→
   332→  if (loading) {
   333→    return (
   334→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: bgColor }}>
   335→        <div className="text-center">
   336→          <div className="relative inline-block">
   337→            <div className="w-20 h-20 rounded-full border-4 border-amber-400/30 border-t-amber-400 animate-spin" />
   338→            <Crown className="w-8 h-8 text-amber-400 absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2" />
   339→          </div>
   340→          <p className="text-amber-400/60 mt-4 text-sm uppercase tracking-[0.3em]">A carregar...</p>
   341→        </div>
   342→      </div>
   343→    );
   344→  }
   345→
   346→  return (
   347→    <div className="min-h-screen relative overflow-hidden font-sans" style={{ backgroundColor: bgColor }}>
   348→      <div className="absolute inset-0 bg-gradient-to-b from-blue-950/30 via-transparent to-black/40" />
   349→      <div className="absolute top-0 left-1/2 -translate-x-1/2 w-[800px] h-[800px] rounded-full bg-amber-500/5 blur-[120px]" />
   350→      <div className="absolute bottom-0 left-0 w-[600px] h-[600px] rounded-full bg-blue-600/5 blur-[100px]" />
   351→      <div className="absolute bottom-0 right-0 w-[600px] h-[600px] rounded-full bg-purple-600/5 blur-[100px]" />
   352→
   353→      <div className="relative z-10">
   354→        {screen === "lobby" && <LobbyScreen gameName={gameName} primaryColor={primaryColor} leaderboard={leaderboard} onStart={startGame} user={user} onLogin={() => navigate("/login")} formatPrize={formatPrize} />}
   355→        {screen === "playing" && currentQuestion && (
   356→          <GameScreen
   357→            key={questionKey}
   358→            gameName={gameName}
   359→            primaryColor={primaryColor}
   360→            currentLevel={currentLevel}
   361→            totalQuestions={totalQuestions}
   362→            currentQuestion={currentQuestion}
   363→            currentPrize={currentPrize}
   364→            timeLeft={timeLeft}
   365→            maxTime={45}
   366→            selectedAnswer={selectedAnswer}
   367→            answered={answered}
   368→            phase={phase}
   369→            lifelinesUsed={lifelinesUsed}
   370→            disabledOptions={disabledOptions}
   371→            prizeStructure={PRIZE_STRUCTURE}
   372→            soundEnabled={soundEnabled}
   373→            audienceVotes={audienceVotes}
   374→            showAudience={showAudience}
   375→            phoneFriend={phoneFriend}
   376→            showPhone={showPhone}
   377→            onAnswer={handleAnswer}
   378→            on5050={handle5050}
   379→            onAudience={handleAudience}
   380→            onPhone={handlePhone}
   381→            onToggleSound={() => setSoundEnabled(p => !p)}
   382→            onQuit={() => setScreen("lobby")}
   383→            formatPrize={formatPrize}
   384→          />
   385→        )}
   386→        {(screen === "won" || screen === "lost") && (
   387→          <ResultScreen
   388→            screen={screen}
   389→            gameName={gameName}
   390→            primaryColor={primaryColor}
   391→            currentLevel={currentLevel}
   392→            currentPrize={currentPrize}
   393→            safePrize={calculateSafePrize()}
   394→            prizeStructure={PRIZE_STRUCTURE}
   395→            onRestart={() => { setScreen("lobby"); loadLeaderboard(); }}
   396→            onQuit={() => setScreen("lobby")}
   397→            formatPrize={formatPrize}
   398→          />
   399→        )}
   400→      </div>
   401→    </div>
   402→  );
   403→}
