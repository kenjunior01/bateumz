    70→/* ─── Counting Number ─── */
    71→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    72→  const [count, setCount] = useState(0);
    73→  const ref = useRef<HTMLSpanElement>(null);
    74→  const inView = useInView(ref, { once: true });
    75→  const hasAnimated = useRef(false);
    76→  useEffect(() => {
    77→    if (!inView || hasAnimated.current) return;
    78→    hasAnimated.current = true;
    79→    const start = performance.now();
    80→    const step = (now: number) => {
    81→      const elapsed = (now - start) / 1000;
    82→      const progress = Math.min(elapsed / duration, 1);
    83→      const eased = 1 - Math.pow(1 - progress, 3);
    84→      setCount(Math.round(eased * target));
    85→      if (progress < 1) requestAnimationFrame(step);
    86→    };
    87→    requestAnimationFrame(step);
    88→  }, [inView, target, duration]);
    89→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
    90→}
