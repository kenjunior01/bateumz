    75→/* ─── Counting Number ─── */
    76→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    77→  const [count, setCount] = useState(0);
    78→  const ref = useRef<HTMLSpanElement>(null);
    79→  const inView = useInView(ref, { once: true });
    80→  const hasAnimated = useRef(false);
    81→  useEffect(() => {
    82→    if (!inView || hasAnimated.current) return;
    83→    hasAnimated.current = true;
    84→    const start = performance.now();
    85→    const step = (now: number) => {
    86→      const elapsed = (now - start) / 1000;
    87→      const progress = Math.min(elapsed / duration, 1);
    88→      const eased = 1 - Math.pow(1 - progress, 3);
    89→      setCount(Math.round(eased * target));
    90→      if (progress < 1) requestAnimationFrame(step);
    91→    };
    92→    requestAnimationFrame(step);
    93→  }, [inView, target, duration]);
    94→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
    95→}
