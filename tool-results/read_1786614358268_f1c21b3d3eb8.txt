    81→    81→function formatMinutes(min: number): string {
    82→    82→  if (min < 60) return `${min} min`;
    83→    83→  const hours = Math.floor(min / 60);
    84→    84→  const remaining = min % 60;
    85→    85→  return remaining > 0 ? `${hours}h ${remaining}min` : `${hours}h`;
    86→    86→}
    87→    87→
    88→    88→function getLimitIndex<T>(value: T | null, options: { value: T | null }[]): number {
    89→    89→  const idx = options.findIndex(o => o.value === value);
    90→    90→  return idx >= 0 ? idx : options.length - 1;
    91→    91→}
    92→    92→
    93→    93→function getSliderPosition(index: number, total: number): number {
    94→    94→  if (total <= 1) return 0;
    95→    95→  return (index / (total - 1)) * 100;
    96→    96→}
