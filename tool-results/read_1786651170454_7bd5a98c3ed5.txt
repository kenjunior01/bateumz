    44→function tempoAgo(dateStr: string): string {
    45→  const agora = Date.now();
    46→  const data = new Date(dateStr).getTime();
    47→  const diff = agora - data;
    48→  const minutos = Math.floor(diff / 60000);
    49→  if (minutos < 1) return 'Agora mesmo';
    50→  if (minutos < 60) return `${minutos}min atr\u00e1s`;
    51→  const horas = Math.floor(minutos / 60);
    52→  if (horas < 24) return `${horas}h atr\u00e1s`;
    53→  const dias = Math.floor(horas / 24);
    54→  if (dias < 7) return `${dias}d atr\u00e1s`;
    55→  return new Date(dateStr).toLocaleDateString('pt-BR');
    56→}

    58→function formatarPremio(valor: number): string {
    59→  if (valor >= 1000000) return `${(valor / 1000000).toFixed(1)}M`;
    60→  if (valor >= 1000) return `${(valor / 1000).toFixed(1)}K`;
    61→  return String(valor);
    62→}
    63→
    64→function formatarHorario(dateStr: string | null): string {
    65→  if (!dateStr) return '--:--';
    66→  const d = new Date(dateStr);
    67→  return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
    68→}
