    91→    91→function useCountUp(end: number, duration: number = 2000, enabled: boolean = true) {
    92→    92→  const [count, setCount] = useState(0);
    93→    93→  const rafRef = useRef<number | null>(null);
    94→    94→  const startTimeRef = useRef<number | null>(null);
    95→    95→
    96→    96→  const animate = useCallback(
    97→    97→    (timestamp: number) => {
    98→    98→      if (!startTimeRef.current) startTimeRef.current = timestamp;
    99→    99→      const elapsed = timestamp - startTimeRef.current;
   100→   100→      const progress = Math.min(elapsed / duration, 1);
   101→   101→      const eased = 1 - Math.pow(1 - progress, 3);
   102→   102→      setCount(Math.floor(eased * end));
   103→   103→      if (progress < 1) {
   104→   104→        rafRef.current = requestAnimationFrame(animate);
   105→   105→      }
   106→   106→    },
   107→   107→    [end, duration]
   108→   108→  );
   109→   109→
   110→   110→  useEffect(() => {
   111→   111→    if (!enabled) {
   112→   112→      setCount(end);
   113→   113→      return;
   114→   114→    }
   115→   115→    startTimeRef.current = null;
   116→   116→    rafRef.current = requestAnimationFrame(animate);
   117→   117→    return () => {
   118→   118→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   119→   119→    };
   120→   120→  }, [animate, enabled]);
   121→   121→
   122→   122→  return count;
   123→   123→}
