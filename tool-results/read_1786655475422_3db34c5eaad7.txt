   228→function getInitials(name: string): string {
   229→  return name.split(' ').map(w => w[0]).join('').toUpperCase().slice(0, 2);
   230→}
   231→
   232→function getAvatarColor(name: string): string {
   233→  const colors = ['#00d4ff', '#7b2ff7', '#ff4655', '#fbbf24', '#2ea043', '#ff6b6b', '#58a6ff', '#f97316'];
   234→  let hash = 0;
   235→  for (let i = 0; i < name.length; i++) hash = name.charCodeAt(i) + ((hash << 5) - hash);
   236→  return colors[Math.abs(hash) % colors.length];
   237→}
   238→
   239→function getTimeRemaining(expiresAt: string): string {
   240→  const diff = new Date(expiresAt).getTime() - Date.now();
   241→  if (diff <= 0) return 'Expirado';
   242→  const hours = Math.floor(diff / 3600000);
   243→  const minutes = Math.floor((diff % 3600000) / 60000);
   244→  if (hours >= 24) return `${Math.floor(hours / 24)}d ${hours % 24}h`;
   245→  if (hours > 0) return `${hours}h ${minutes}m`;
   246→  return `${minutes}m`;
   247→}
