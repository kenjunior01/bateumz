    15→// ===================== TYPES =====================
    16→type GameScreen =
    17→  | 'menu' | 'classSelect' | 'campaignMap' | 'combat'
    18→  | 'inventory' | 'shop' | 'pvp' | 'victory'
    19→  | 'defeat' | 'bossIntro' | 'levelUp';
    20→
    21→type Element = 'fire' | 'lightning' | 'poison' | 'shadow' | 'holy';
    22→type AttackPattern = 'melee' | 'ranged' | 'charge' | 'summon' | 'aoe';
    23→type Rarity = 'common' | 'rare' | 'epic' | 'legendary';
    24→type EquipSlot = 'weapon' | 'armor' | 'accessory';
    25→
    26→interface CharClass {
    27→  id: string; name: string; emoji: string; color: string;
    28→  element: Element; baseHP: number; baseATK: number;
    29→  baseDEF: number; baseSPD: number;
    30→  abilityName: string; abilityDesc: string;
    31→  abilityCooldown: number;
    32→  passiveDesc: string;
    33→}
    34→
    35→interface Equipment {
    36→  id: string; name: string; type: EquipSlot; rarity: Rarity;
    37→  atk: number; def: number; hp: number; spd: number;
    38→  element?: Element; special?: string; sellPrice: number;
    39→}
    40→
    41→interface EnemyDef {
    42→  name: string; emoji: string; hp: number; atk: number;
    43→  def: number; spd: number; color: string;
    44→  attackPattern: AttackPattern; size: number;
    45→  isBoss?: boolean; abilities?: string[];
    46→  xpReward: number; goldReward: number;
    47→}
    48→
    49→interface LevelDef {
    50→  enemy: EnemyDef;
    51→  reward: Equipment | null;
    52→  starThresholds: number[];
    53→}
    54→
    55→interface WorldDef {
    56→  id: string; name: string; emoji: string;
    57→  bgColor1: string; bgColor2: string; accentColor: string;
    58→  unlockRequirement: string; description: string;
    59→  levels: LevelDef[];
    60→}
    61→
    62→interface ShopItem extends Equipment { buyPrice: number; }
    63→
    64→interface PlayerData {
    65→  classId: string; level: number; xp: number; xpToNext: number;
    66→  gold: number; kills: number; pvpWins: number;
    67→  equipment: (Equipment | null)[];
    68→  inventory: Equipment[];
    69→  completedLevels: Record<string, number>;
    70→}
    71→
    72→interface Particle {
    73→  x: number; y: number; vx: number; vy: number;
    74→  life: number; maxLife: number; color: string; size: number;
    75→  type: 'circle' | 'spark' | 'ring' | 'text' | 'star';
    76→  text?: string; gravity?: number; shrink?: boolean;
    77→}
    78→
    79→interface Projectile {
    80→  x: number; y: number; vx: number; vy: number;
    81→  damage: number; isPlayer: boolean; color: string;
    82→  trail: { x: number; y: number }[]; size: number;
    83→}
    84→
    85→interface Summon {
    86→  x: number; y: number; hp: number; atk: number;
    87→  cooldown: number; vx: number; vy: number;
    88→}
    89→
    90→interface CombatState {
    91→  px: number; py: number; php: number; pmaxhp: number;
    92→  patk: number; pdef: number; pspd: number;
    93→  pvx: number; pvy: number; pAtkCD: number; pAbilCD: number;
    94→  pHurt: number; pAtkAnim: number; pFaceR: boolean; pBob: number;
    95→  pCritNext: boolean; pFuryNext: boolean;
    96→  ex: number; ey: number; ehp: number; emaxhp: number;
    97→  eatk: number; edef: number; espd: number;
    98→  eHurt: number; eAtkAnim: number; eAtkCD: number;
    99→  eFaceR: boolean; eCharging: boolean; eChargeTimer: number;
   100→  eChargeVX: number; eChargeVY: number; ePause: number;
   101→  eSummonCD: number; eAoeCD: number;
   102→  projectiles: Projectile[]; particles: Particle[]; summons: Summon[];
   103→  ambient: { x: number; y: number; vy: number; size: number; alpha: number }[];
   104→  keys: Set<string>; frame: number; time: number;
   105→  shake: number; shakeI: number;
   106→  worldIdx: number; levelIdx: number;
   107→  won: boolean; lost: boolean;
   108→}
   109→
   110→interface BattleResult {
   111→  stars: number; time: number; score: number;
   112→  reward: Equipment | null; xp: number; gold: number;
   113→  leveledUp: boolean; newLevel: number;
   114→}

   362→function drawStar(ctx: CanvasRenderingContext2D, cx: number, cy: number, spikes: number, outerR: number, innerR: number, color: string) {
   363→  ctx.fillStyle = color;
   364→  let rot = (Math.PI / 2) * 3;
   365→  const step = Math.PI / spikes;
   366→  ctx.beginPath();
   367→  ctx.moveTo(cx, cy - outerR);
   368→  for (let i = 0; i < spikes; i++) {
   369→    ctx.lineTo(cx + Math.cos(rot) * outerR, cy + Math.sin(rot) * outerR);
   370→    rot += step;
   371→    ctx.lineTo(cx + Math.cos(rot) * innerR, cy + Math.sin(rot) * innerR);
   372→    rot += step;
   373→  }
   374→  ctx.closePath();
   375→  ctx.fill();
   376→}

   378→function drawHealthBar(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, current: number, max: number, color1: string, color2: string) {
   379→  ctx.fillStyle = 'rgba(0,0,0,0.6)';
   380→  ctx.fillRect(x - 1, y - 1, w + 2, h + 2);
   381→  const pct = Math.max(0, current / max);
   382→  const grad = ctx.createLinearGradient(x, y, x, y + h);
   383→  grad.addColorStop(0, color1);
   384→  grad.addColorStop(1, color2);
   385→  ctx.fillStyle = grad;
   386→  ctx.fillRect(x, y, w * pct, h);
   387→  ctx.strokeStyle = 'rgba(255,255,255,0.3)';
   388→  ctx.lineWidth = 0.5;
   389→  ctx.strokeRect(x - 1, y - 1, w + 2, h + 2);
   390→}

   402→function calcStats(pd: PlayerData) {
   403→  const cls = CLASSES.find(c => c.id === pd.classId);
   404→  if (!cls) return { hp: 100, atk: 10, def: 5, spd: 3 };
   405→  let hp = cls.baseHP + (pd.level - 1) * 15;
   406→  let atk = cls.baseATK + (pd.level - 1) * 3;
   407→  let def = cls.baseDEF + (pd.level - 1) * 2;
   408→  let spd = cls.baseSPD + (pd.level - 1) * 0.5;
   409→  for (const item of pd.equipment) {
   410→    if (item) { hp += item.hp; atk += item.atk; def += item.def; spd += item.spd; }
   411→  }
   412→  return { hp, atk, def, spd };
   413→}
