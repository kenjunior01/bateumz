    70→function AnimatedSection({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
    71→  const ref = useRef<HTMLDivElement>(null);

    75→      {children}
    76→    </motion.section>
    77→  );
    78→}

    80→/* ─── Counting Number ─── */
    81→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    82→  const [count, setCount] = useState(0);
    83→  const ref = useRef<HTMLSpanElement>(null);
    84→  const inView = useInView(ref, { once: true });
    85→  const hasAnimated = useRef(false);
    86→  useEffect(() => {
    87→    if (!inView || hasAnimated.current) return;
    88→    hasAnimated.current = true;
    89→    const start = performance.now();
    90→    const step = (now: number) => {
    91→      const elapsed = (now - start) / 1000;
    92→      const progress = Math.min(elapsed / duration, 1);
    93→      const eased = 1 - Math.pow(1 - progress, 3);
    94→      setCount(Math.round(eased * target));
    95→      if (progress < 1) requestAnimationFrame(step);
    96→    };
    97→    requestAnimationFrame(step);
    98→  }, [inView, target, duration]);
    99→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
   100→}
