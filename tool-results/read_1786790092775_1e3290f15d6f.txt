    47→function tempoAgo(dateStr: string): string {
    48→  const agora = Date.now();
    49→  const data = new Date(dateStr).getTime();
    50→  const diff = agora - data;
    51→  const minutos = Math.floor(diff / 60000);
    52→  if (minutos < 1) return 'Agora mesmo';
    53→  if (minutos < 60) return `${minutos}min atr\u00e1s`;
    54→  const horas = Math.floor(minutos / 60);
    55→  if (horas < 24) return `${horas}h atr\u00e1s`;
    56→  const dias = Math.floor(horas / 24);
    57→  if (dias < 7) return `${dias}d atr\u00e1s`;
    58→  return new Date(dateStr).toLocaleDateString('pt-BR');
    59→}
    60→
    61→function formatarPremio(valor: number): string {
    62→  if (valor >= 1000000) return `${(valor / 1000000).toFixed(1)}M`;
    63→  if (valor >= 1000) return `${(valor / 1000).toFixed(1)}K`;
    64→  return String(valor);
    65→}
    66→
    67→function formatarHorario(dateStr: string | null): string {
    68→  if (!dateStr) return '--:--';
    69→  const d = new Date(dateStr);
    70→  return d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
    71→}
