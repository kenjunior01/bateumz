   100→// Per-game error boundary so one crashing game does not kill the whole page
   101→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean; resetKey: number; errorMsg: string}> {
   102→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false, resetKey: 0, errorMsg: ""}; }
   103→  static getDerivedStateFromError() { return {hasError: true}; }
   104→  componentDidCatch(err: Error) {
   105→    console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err);
   106→    const msg = err.message || "Erro desconhecido";
   107→    this.setState({ errorMsg: msg.length > 120 ? msg.slice(0, 120) + "..." : msg });
   108→  }
   109→  handleRetry = () => { this.setState((s) => ({ hasError: false, resetKey: (s.resetKey || 0) + 1, errorMsg: "" })); };
   110→  handleSelectOther = () => {
   111→    window.dispatchEvent(new CustomEvent("game-error-select-other"));
   112→  };
   113→  render() {
   114→    if (this.state.hasError) return (
   115→      <div className="flex flex-col items-center justify-center py-20 px-6 rounded-2xl border border-dashed border-destructive/30 bg-gradient-to-b from-destructive/5 to-transparent">
   116→        <div className="text-5xl mb-4">⚠️</div>
   117→        <p className="text-sm font-bold mb-1">Error loading {this.props.gameName}</p>
   118→        <p className="text-[11px] text-muted-foreground mb-2 text-center max-w-xs">{this.state.errorMsg}</p>
   119→        <div className="flex items-center gap-2">
   120→          <button onClick={this.handleRetry} className="text-xs px-4 py-2 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 font-semibold transition-colors">Try Again</button>
   121→          <button onClick={this.handleSelectOther} className="text-xs px-4 py-2 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/80 font-semibold transition-colors">Other game</button>
   122→        </div>
   123→      </div>
   124→    );
   125→    return <div key={this.state.resetKey}>{this.props.children}</div>;
   126→  }
   127→}
