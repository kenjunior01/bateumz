   101→// Per-game error boundary so one crashing game does not kill the whole page
   102→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean; resetKey: number; errorMsg: string}> {
   103→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false, resetKey: 0, errorMsg: ""}; }
   104→  static getDerivedStateFromError() { return {hasError: true}; }
   105→  componentDidCatch(err: Error) {
   106→    console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err);
   107→    const msg = err.message || "Erro desconhecido";
   108→    this.setState({ errorMsg: msg.length > 120 ? msg.slice(0, 120) + "..." : msg });
   109→  }
   110→  handleRetry = () => { this.setState((s) => ({ hasError: false, resetKey: (s.resetKey || 0) + 1, errorMsg: "" })); };
   111→  handleSelectOther = () => {
   112→    window.dispatchEvent(new CustomEvent("game-error-select-other"));
   113→  };
   114→  render() {
   115→    if (this.state.hasError) return (
   116→      <div className="flex flex-col items-center justify-center py-20 px-6 rounded-2xl border border-dashed border-destructive/30 bg-gradient-to-b from-destructive/5 to-transparent">
   117→        <div className="text-5xl mb-4">⚠️</div>
   118→        <p className="text-sm font-bold mb-1">Error loading {this.props.gameName}</p>
   119→        <p className="text-[11px] text-muted-foreground mb-2 text-center max-w-xs">{this.state.errorMsg}</p>
   120→        <div className="flex items-center gap-2">
   121→          <button onClick={this.handleRetry} className="text-xs px-4 py-2 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 font-semibold transition-colors">Try Again</button>
   122→          <button onClick={this.handleSelectOther} className="text-xs px-4 py-2 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/80 font-semibold transition-colors">Other game</button>
   123→        </div>
   124→      </div>
   125→    );
   126→    return <div key={this.state.resetKey}>{this.props.children}</div>;
   127→  }
   128→}
