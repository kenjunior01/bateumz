    71→function AnimatedSection({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {

    81→/* ─── Counting Number ─── */
    82→function CountingNumber({ target, suffix = "", duration = 2 }: { target: number; suffix?: string; duration?: number }) {
    83→  const [count, setCount] = useState(0);
    84→  const ref = useRef<HTMLSpanElement>(null);
    85→  const inView = useInView(ref, { once: true });
    86→  const hasAnimated = useRef(false);
    87→  useEffect(() => {
    88→    if (!inView || hasAnimated.current) return;
    89→    hasAnimated.current = true;
    90→    const start = performance.now();
    91→    const step = (now: number) => {
    92→      const elapsed = (now - start) / 1000;
    93→      const progress = Math.min(elapsed / duration, 1);
    94→      const eased = 1 - Math.pow(1 - progress, 3);
    95→      setCount(Math.round(eased * target));
    96→      if (progress < 1) requestAnimationFrame(step);
    97→    };
    98→    requestAnimationFrame(step);
    99→  }, [inView, target, duration]);
   100→  return <span ref={ref}>{count.toLocaleString("pt-BR")}{suffix}</span>;
   101→}
