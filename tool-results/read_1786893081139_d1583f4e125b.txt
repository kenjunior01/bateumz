   102→   102→// Per-game error boundary so one crashing game does not kill the whole page
   103→   103→class GameErrorBoundary extends Component<{children: ReactNode; gameName: string}, {hasError: boolean; resetKey: number; errorMsg: string}> {
   104→   104→  constructor(props: {children: ReactNode; gameName: string}) { super(props); this.state = {hasError: false, resetKey: 0, errorMsg: ""}; }
   105→   105→  static getDerivedStateFromError() { return {hasError: true}; }
   106→   106→  componentDidCatch(err: Error) {
   107→   107→    console.error(`[GameErrorBoundary] ${this.props.gameName}:`, err);
   108→   108→    const msg = err.message || "Erro desconhecido";
   109→   109→    this.setState({ errorMsg: msg.length > 120 ? msg.slice(0, 120) + "..." : msg });
   110→   110→  }
   111→   111→  handleRetry = () => { this.setState((s) => ({ hasError: false, resetKey: (s.resetKey || 0) + 1, errorMsg: "" })); };
   112→   112→  handleSelectOther = () => {
   113→   113→    window.dispatchEvent(new CustomEvent("game-error-select-other"));
   114→   114→  };
   115→   115→  render() {
   116→   116→    if (this.state.hasError) return (
   117→   117→      <div className="flex flex-col items-center justify-center py-20 px-6 rounded-2xl border border-dashed border-destructive/30 bg-gradient-to-b from-destructive/5 to-transparent">
   118→   118→        <div className="text-5xl mb-4">⚠️</div>
   119→   119→        <p className="text-sm font-bold mb-1">Error loading {this.props.gameName}</p>
   120→   120→        <p className="text-[11px] text-muted-foreground mb-2 text-center max-w-xs">{this.state.errorMsg}</p>
   121→   121→        <div className="flex items-center gap-2">
   122→   122→          <button onClick={this.handleRetry} className="text-xs px-4 py-2 rounded-full bg-primary text-primary-foreground hover:bg-primary/90 font-semibold transition-colors">Try Again</button>
   123→   123→          <button onClick={this.handleSelectOther} className="text-xs px-4 py-2 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/80 font-semibold transition-colors">Other game</button>
   124→   124→        </div>
   125→   125→      </div>
   126→   126→    );
   127→   127→    return <div key={this.state.resetKey}>{this.props.children}</div>;
   128→   128→  }
   129→   129→}
