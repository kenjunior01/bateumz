   113→/** Predict where the ball will be when it reaches targetX */
   114→function predictBallY(ball: Ball, targetX: number): number {
   115→  if (
   116→    (targetX > ball.x && ball.vx <= 0) ||
   117→    (targetX < ball.x && ball.vx >= 0)
   118→  ) {
   119→    return ball.y;
   120→  }
   121→
   122→  let x = ball.x;
   123→  let y = ball.y;
   124→  let vy = ball.vy;
   125→  const step = Math.abs(ball.vx) || 1;
   126→  const dir = ball.vx > 0 ? 1 : -1;
   127→
   128→  for (let i = 0; i < 2000; i++) {
   129→    x += dir * step;
   130→    y += vy;
   131→
   132→    if (y <= BALL_R) {
   133→      y = BALL_R;
   134→      vy = Math.abs(vy);
   135→    }
   136→    if (y >= CANVAS_H - BALL_R) {
   137→      y = CANVAS_H - BALL_R;
   138→      vy = -Math.abs(vy);
   139→    }
   140→
   141→    if ((dir > 0 && x >= targetX) || (dir < 0 && x <= targetX)) {
   142→      return y;
   143→    }
   144→  }
   145→
   146→  return ball.y;
   147→}

   200→    const b = ballRef.current;
   201→    b.x = CANVAS_W / 2;
   202→    b.y = CANVAS_H / 2;
   203→    b.speed = BALL_SPEED_INIT;

   205→    b.vx = Math.cos(angle) * b.speed * direction;
   206→    b.vy = Math.sin(angle) * b.speed;

   209→  const botAI = useCallback(() => {
   210→    const p = p2Ref.current;
   211→    const b = ballRef.current;
   212→    const cfg = BOT_CONFIGS[difficultyRef.current];
   213→    const now = performance.now();
   214→
   215→    // Only recalculate target at reaction interval
   216→    if (now - botLastUpdateRef.current >= cfg.reactionMs) {
   217→      botLastUpdateRef.current = now;
   218→
   219→      // Decide whether to "miss" this update cycle
   220→      const shouldMiss = Math.random() < cfg.missChance;
   221→      if (shouldMiss) {
   222→        botMovingRef.current = false;
   223→        return;
   224→      }
   225→      botMovingRef.current = true;
   226→
   227→      // Calculate target y position
   228→      let targetY: number;
   229→      if (cfg.predictAhead) {
   230→        targetY = predictBallY(b, p.x);
   231→      } else {
   232→        // Only track when ball is heading toward the bot
   233→        if (b.vx > 0) {
   234→          targetY = b.y;
   235→        } else {
   236→          // Ball going away, drift toward center slowly
   237→          targetY = CANVAS_H / 2;
   238→        }
   239→      }
   240→
   241→      // Add noise
   242→      const noise = (Math.random() - 0.5) * 2 * cfg.noisePx;
   243→      botTargetRef.current = Math.max(
   244→        PADDLE_H / 2,
   245→        Math.min(CANVAS_H - PADDLE_H / 2, targetY + noise)
   246→      );
   247→    }
   248→
   249→    // Move paddle toward target
   250→    if (!botMovingRef.current) return;
   251→
   252→    const center = p.y + PADDLE_H / 2;
   253→    const diff = botTargetRef.current - center;
   254→
   255→    if (Math.abs(diff) > cfg.deadzonePx) {
   256→      const moveAmount = Math.min(
   257→        Math.abs(diff),
   258→        cfg.speed * PADDLE_SPEED
   259→      );
   260→      p.y += diff > 0 ? moveAmount : -moveAmount;
   261→    }
   262→
   263→    p.y = Math.max(0, Math.min(CANVAS_H - PADDLE_H, p.y));
   264→  }, []);

   367→      b.y = CANVAS_H - BALL_R;
   368→      b.vy = -Math.abs(b.vy);
   369→    }
   370→
   371→    // Paddle 1 collision (left)
   372→    if (
   373→      b.vx < 0 &&
   374→      b.x - BALL_R <= p1.x + PADDLE_W &&
   375→      b.x - BALL_R >= p1.x &&

   381→      const angle = relY * (Math.PI / 3);
   382→      b.speed = Math.min(b.speed + BALL_SPEED_INC, 10);

   386→
   387→    // Paddle 2 collision (right)

   389→      b.vx > 0 &&
   390→      b.x + BALL_R >= p2.x &&
   391→      b.x + BALL_R <= p2.x + PADDLE_W &&
   392→      b.y >= p2.y &&
   393→      b.y <= p2.y + PADDLE_H

   397→      const angle = relY * (Math.PI / 3);
   398→      b.speed = Math.min(b.speed + BALL_SPEED_INC, 10);
   399→      b.vx = -Math.cos(angle) * b.speed;
   400→      b.vy = Math.sin(angle) * b.speed;

   413→          particleCount: 100,
   414→          spread: 70,

   417→      } else {

   508→    botTargetRef.current = CANVAS_H / 2;
   509→    botLastUpdateRef.current = 0;
   510→    botMovingRef.current = true;
   511→    setPhase("countdown");

   514→
   515→  useEffect(() => {
   516→    if (phase !== "countdown") return;

   530→          animate={{ opacity: 1, y: 0 }}

   532→          className="w-full flex flex-col gap-4"

   542→              Escolha o Modo de Jogo

   546→              animate={{ opacity: 1 }}
   547→              transition={{ delay: 0.15 }}
   548→              className="text-xs text-muted-foreground mt-1"
   549→            >
   550→              Selecione como deseja jogar
   551→            </motion.p>
   552→          </div>
   553→
   554→          <div className="grid grid-cols-2 gap-3 w-full">
   555→            <motion.button

   559→              className={`relative flex flex-col items-center gap-2 p-4 rounded-2xl border-2 transition-all duration-200 cursor-pointer ${
   560→                mode === "bot"
   561→                  ? "border-primary bg-primary/5 shadow-lg shadow-primary/20"
   562→                  : "border-border bg-card hover:border-primary/30"

   564→            >
