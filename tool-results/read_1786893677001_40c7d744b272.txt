   113→   113→/** Predict where the ball will be when it reaches targetX */
   114→   114→function predictBallY(ball: Ball, targetX: number): number {
   115→   115→  if (
   116→   116→    (targetX > ball.x && ball.vx <= 0) ||
   117→   117→    (targetX < ball.x && ball.vx >= 0)
   118→   118→  ) {
   119→   119→    return ball.y;
   120→   120→  }
   121→   121→
   122→   122→  let x = ball.x;
   123→   123→  let y = ball.y;
   124→   124→  let vy = ball.vy;
   125→   125→  const step = Math.abs(ball.vx) || 1;
   126→   126→  const dir = ball.vx > 0 ? 1 : -1;
   127→   127→
   128→   128→  for (let i = 0; i < 2000; i++) {
   129→   129→    x += dir * step;
   130→   130→    y += vy;
   131→   131→
   132→   132→    if (y <= BALL_R) {
   133→   133→      y = BALL_R;
   134→   134→      vy = Math.abs(vy);
   135→   135→    }
   136→   136→    if (y >= CANVAS_H - BALL_R) {
   137→   137→      y = CANVAS_H - BALL_R;
   138→   138→      vy = -Math.abs(vy);
   139→   139→    }
   140→   140→
   141→   141→    if ((dir > 0 && x >= targetX) || (dir < 0 && x <= targetX)) {
   142→   142→      return y;
   143→   143→    }
   144→   144→  }
   145→   145→
   146→   146→  return ball.y;
   147→   147→}
