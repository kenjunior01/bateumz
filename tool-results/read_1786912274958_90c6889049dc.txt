  1380→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */

  1382→interface StatusOption { value: string; label: string }
  1383→interface SortOption { value: string; label: string }
  1384→
  1385→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1386→  items,
  1387→  statuses,
  1388→  sorts,
  1389→  renderItem,
  1390→  getStatus,
  1391→  getSortValue,
  1392→  formatScore,
  1393→  emptyMessage,
  1394→  extraAction,
  1395→  pageSize = 9,
  1396→}: {
  1397→  items: T[];
  1398→  statuses: StatusOption[];
  1399→  sorts: SortOption[];
  1400→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1401→  getStatus: (item: T) => string;
  1402→  getSortValue: (item: T, sort: string) => number;
  1403→  formatScore?: (item: T, sort: string) => string | undefined;
  1404→  emptyMessage: string;
  1405→  extraAction?: React.ReactNode;
  1406→  pageSize?: number;
  1407→}) {
  1408→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1409→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1410→  const [page, setPage] = useState(1);
  1411→  const [transitioning, setTransitioning] = useState(false);
  1412→
  1413→  // Reset page + show brief skeleton when filter/sort changes
  1414→  useEffect(() => {
  1415→    setPage(1);
  1416→    setTransitioning(true);
  1417→    const t = setTimeout(() => setTransitioning(false), 280);
  1418→    return () => clearTimeout(t);
  1419→  }, [status, sort]);
  1420→
  1421→  const filtered = useMemo(() => {
  1422→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1423→    if (sort === "recent") {
  1424→      list = [...list].sort((a, b) => {
  1425→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1426→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1427→        return tb - ta;
  1428→      });
  1429→    } else {
  1430→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1431→    }
  1432→    return list;
  1433→  }, [items, status, sort, getStatus, getSortValue]);
  1434→
  1435→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  1436→  const visible = filtered.slice(0, page * pageSize);
  1437→  const canLoadMore = page < totalPages;
  1438→
  1439→  const counts = useMemo(() => {
  1440→    const map: Record<string, number> = { all: items.length };
  1441→    for (const i of items) {
  1442→      const s = getStatus(i);
  1443→      map[s] = (map[s] ?? 0) + 1;
  1444→    }
  1445→    return map;
  1446→  }, [items, getStatus]);
  1447→
  1448→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1449→
  1450→  if (items.length === 0) {
  1451→    return <EmptyState message={emptyMessage} />;
  1452→  }
  1453→
  1454→  return (
  1455→    <div>
  1456→      <div className="mb-4 flex flex-col gap-3">
  1457→        <div className="flex flex-wrap gap-1.5">
  1458→          {statuses.map((s) => {
  1459→            const active = status === s.value;
  1460→            const count = counts[s.value] ?? 0;
  1461→            return (
  1462→              <button
  1463→                key={s.value}
  1464→                onClick={() => setStatus(s.value)}
  1465→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1466→                  active
  1467→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1468→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1469→                }`}
  1470→              >
  1471→                {s.label}
  1472→                <span className="ml-1 opacity-70">({count})</span>
  1473→              </button>
  1474→            );
  1475→          })}
  1476→        </div>
  1477→
  1478→        <div className="flex items-center justify-between gap-2">
  1479→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1480→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1481→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1482→          </p>
  1483→          <div className="flex items-center gap-2">
  1484→            {extraAction}
  1485→            <Select value={sort} onValueChange={setSort}>
  1486→              <SelectTrigger className="h-8 text-xs w-[140px] sm:w-[170px]">
  1487→                <SelectValue />
  1488→              </SelectTrigger>
  1489→              <SelectContent>
  1490→                {sorts.map((s) => (
  1491→                  <SelectItem key={s.value} value={s.value} className="text-xs">
  1492→                    {s.label}
  1493→                  </SelectItem>
  1494→                ))}
  1495→              </SelectContent>
  1496→            </Select>
  1497→          </div>
  1498→        </div>
