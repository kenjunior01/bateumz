   108→const FloatingParticles = ({ color, count = 30 }: { color: string; count?: number }) => {
   109→  const particles = useMemo(
   110→    () =>
   111→      Array.from({ length: count }, (_, i) => ({
   112→        id: i,
   113→        x: Math.random() * 100,
   114→        y: Math.random() * 100,
   115→        size: 2 + Math.random() * 4,
   116→        duration: 6 + Math.random() * 8,
   117→        delay: Math.random() * 4,
   118→        opacity: 0.15 + Math.random() * 0.35,
   119→      })),
   120→    [count],
   121→  );
   122→  return (
   123→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
   124→      {particles.map((p) => (
   125→        <motion.div
   126→          key={p.id}
   127→          className="absolute rounded-full"
   128→          style={{
   129→            left: `${p.x}%`,
   130→            width: p.size,
   131→            height: p.size,
   132→            backgroundColor: color,
   133→          }}
   134→          animate={{
   135→            y: [0, -30, 0, 20, 0],
   136→            x: [0, 10, -10, 5, 0],
   137→            opacity: [p.opacity, p.opacity * 1.5, p.opacity * 0.5, p.opacity],
   138→          }}
   139→          transition={{
   140→            duration: p.duration,
   141→            repeat: Infinity,
   142→            delay: p.delay,
   143→            ease: "easeInOut",
   144→          }}
   145→        />
   146→      ))}
   147→    </div>
   148→  );
