   644→              <motion.div initial={{ y: 15, opacity: 0 }} whileInView={{ y: 0, opacity: 1 }} viewport={{ once: true }} className="rounded-2xl border border-border/40 bg-card/40 p-6">
   645→                <div className="flex items-center gap-2 mb-4">

   648→                </div>

   650→                  <div className="flex items-center gap-2"><span className="text-muted-foreground text-xs">Nicho:</span><span className="font-bold text-xs px-2 py-0.5 rounded-full" style={{ backgroundColor: `${primary}15`, color: primary }}>{niche.label}</span></div>

   665→                </div>

   669→        </Tabs>
