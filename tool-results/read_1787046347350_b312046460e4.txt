    97→function formatDate(iso: string) {
    98→  return new Date(iso).toLocaleDateString('pt-BR', {
    99→    day: '2-digit',
   100→    month: 'short',
   101→    year: 'numeric',
   102→  });
   103→}
   104→
   105→function formatCount(n: number): string {
   106→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   107→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
   108→  return String(n);
   109→}

   274→export default function Blog() {
   275→  const { region } = useRegionalTheme();
   276→
   277→  /* ---------- state ---------- */
   278→  const [posts, setPosts] = useState<BlogPost[]>([]);
   279→  const [categories, setCategories] = useState<BlogCategory[]>([]);
   280→  const [trendingPosts, setTrendingPosts] = useState<BlogPost[]>([]);
   281→  const [featuredPost, setFeaturedPost] = useState<BlogPost | null>(null);
   282→
   283→  const [loading, setLoading] = useState(true);
   284→  const [loadingMore, setLoadingMore] = useState(false);
   285→  const [error, setError] = useState<string | null>(null);
   286→
   287→  const [searchTerm, setSearchTerm] = useState('');
   288→  const [activeCategory, setActiveCategory] = useState<string | null>(null);
   289→
   290→  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
   291→  const [hasMore, setHasMore] = useState(false);
   292→
   293→  /* ---------- data fetching ---------- */
   294→  const fetchData = useCallback(async () => {
   295→    setLoading(true);
   296→    setError(null);
   297→    try {
   298→      // Categorias
   299→      const { data: catData } = await (supabase as any)
   300→        .from('blog_categories')
   301→        .select('*')
   302→        .order('name');
   303→      setCategories((catData as BlogCategory[]) || []);
   304→
   305→      // Posts principais
   306→      let postQuery = (supabase as any)
   307→        .from('blog_posts')
   308→        .select(
   309→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   310→        )
   311→        .eq('published', true);
   312→
   313→      if (region) {
   314→        postQuery = postQuery.or(
   315→          `region_id.eq.${region.id},region_id.is.null`
   316→        );
   317→      }
   318→
   319→      const { data: postData, error: postErr } = await postQuery.order(
   320→        'published_at',
   321→        { ascending: false }
   322→      );
   323→      if (postErr) throw postErr;
   324→      setPosts((postData as BlogPost[]) || []);
   325→
   326→      // Posts em alta
   327→      let trendQuery = (supabase as any)
   328→        .from('blog_posts')
   329→        .select(
   330→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   331→        )
   332→        .eq('published', true)
   333→        .eq('is_trending', true);
   334→
   335→      if (region) {
   336→        trendQuery = trendQuery.or(
   337→          `region_id.eq.${region.id},region_id.is.null`
   338→        );
   339→      }
   340→
   341→      const { data: trendData } = await trendQuery
   342→        .order('trending_score', { ascending: false })
   343→        .limit(5);
   344→      setTrendingPosts((trendData as BlogPost[]) || []);
   345→
   346→      // Post destaque
   347→      let featQuery = (supabase as any)
   348→        .from('blog_posts')
   349→        .select(
   350→          `id, title, slug, summary, image_url, published_at, seo_keywords, view_count, like_count, share_count, reading_time_min, is_featured, is_trending, trending_score, category:blog_categories(name, slug, color, icon)`
   351→        )
   352→        .eq('published', true)
   353→        .eq('is_featured', true);
   354→
   355→      if (region) {
   356→        featQuery = featQuery.or(
   357→          `region_id.eq.${region.id},region_id.is.null`
   358→        );
   359→      }
   360→
   361→      const { data: featData } = await featQuery
   362→        .order('published_at', { ascending: false })
   363→        .limit(1)
   364→        .single();
   365→      setFeaturedPost((featData as BlogPost) || null);
   366→    } catch (err: any) {
   367→      console.error('Erro ao carregar blog:', err);
   368→      setError('Não foi possível carregar os posts. Tente novamente mais tarde.');
   369→    } finally {
   370→      setLoading(false);
   371→    }
   372→  }, [region]);
   373→
   374→  useEffect(() => {
   375→    fetchData();
   376→  }, [fetchData]);
   377→
   378→  /* ---------- filtering ---------- */
   379→  const filteredPosts = useMemo(() => {
   380→    let result = posts;
   381→
   382→    // Remover o post destaque da listagem principal para evitar duplicidade
   383→    if (featuredPost) {
   384→      result = result.filter((p) => p.id !== featuredPost.id);
   385→    }
   386→
   387→    // Filtro por categoria
   388→    if (activeCategory) {
   389→      result = result.filter((p) => p.category?.slug === activeCategory);
   390→    }
   391→
   392→    // Filtro por busca
   393→    if (searchTerm.trim()) {
   394→      const term = searchTerm.toLowerCase();
   395→      result = result.filter(
   396→        (p) =>
   397→          p.title.toLowerCase().includes(term) ||
   398→          p.summary.toLowerCase().includes(term) ||
   399→          p.seo_keywords?.some((k: string) => k.toLowerCase().includes(term))
   400→      );
   401→    }
   402→
   403→    return result;
   404→  }, [posts, activeCategory, searchTerm, featuredPost]);
   405→
   406→  const visiblePosts = useMemo(
   407→    () => filteredPosts.slice(0, visibleCount),
   408→    [filteredPosts, visibleCount]
   409→  );
   410→
   411→  useEffect(() => {
   412→    setHasMore(visibleCount < filteredPosts.length);
   413→  }, [visibleCount, filteredPosts.length]);
   414→
   415→  // Resetar contagem ao mudar filtros
   416→  useEffect(() => {
   417→    setVisibleCount(PAGE_SIZE);
   418→  }, [activeCategory, searchTerm]);
   419→
   420→  const handleLoadMore = () => {
   421→    setLoadingMore(true);
   422→    setTimeout(() => {
   423→      setVisibleCount((prev) => prev + PAGE_SIZE);
   424→      setLoadingMore(false);
   425→    }, 400);
   426→  };

   430→    <>
   431→      <Helmet>
   432→        <title>Blog – Bateu Online | Notícias, Dicas e Sorteios</title>
   433→        <meta
   434→          name="description"
   435→          content="Fique por dentro das últimas novidades sobre sorteios online, dicas de participação, resultados e muito mais no blog do Bateu Online."
   436→        />
   437→        <link rel="canonical" href={CANONICAL} />
   438→        <meta property="og:title" content="Blog – Bateu Online | Notícias e Sorteios" />
   439→        <meta

   456→
   457→      <script

   473→      />
   474→
   475→      <div className="min-h-screen bg-background">
   476→        <section className="relative overflow-hidden">

   482→              backgroundSize: '400% 400%',
