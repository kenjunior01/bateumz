    26→function timeAgo(dateStr: string): string {
    27→  const diff = Date.now() - new Date(dateStr).getTime();
    28→  const mins = Math.floor(diff / 60000);
    29→  if (mins < 1) return 'agora';
    30→  if (mins < 60) return `${mins}min`;
    31→  const hrs = Math.floor(mins / 60);
    32→  if (hrs < 24) return `${hrs}h`;
    33→  return `${Math.floor(hrs / 24)}d`;
    34→}

    98→// Animated counter hook
    99→function useAnimatedCounter(target: number, duration: number = 2000, startOnMount: boolean = true) {
   100→  const [count, setCount] = useState(0);
   101→
   102→  useEffect(() => {
   103→    if (!startOnMount) return;
   104→    let startTime: number;
   105→    let raf: number;
   106→
   107→    const animate = (timestamp: number) => {
   108→      if (!startTime) startTime = timestamp;
   109→      const progress = Math.min((timestamp - startTime) / duration, 1);
   110→      const eased = 1 - Math.pow(1 - progress, 3);
   111→      setCount(Math.floor(eased * target));
   112→      if (progress < 1) {
   113→        raf = requestAnimationFrame(animate);
   114→      }
   115→    };
   116→
   117→    raf = requestAnimationFrame(animate);
   118→    return () => cancelAnimationFrame(raf);
   119→  }, [target, duration, startOnMount]);
   120→
   121→  return count;
   122→}
