    56→function formatDate(iso?: string) {
    57→  if (!iso) return "--";
    58→  return new Date(iso).toLocaleDateString("pt-BR", {
    59→    day: "2-digit",
    60→    month: "short",
    61→    year: "numeric",
    62→    hour: "2-digit",
    63→    minute: "2-digit",
    64→  });
    65→}
    66→
    67→function timeAgo(iso: string) {
    68→  const diff = Date.now() - new Date(iso).getTime();
    69→  const mins = Math.floor(diff / 60000);
    70→  if (mins < 1) return "agora";
    71→  if (mins < 60) return `${mins}m atras`;
    72→  const hrs = Math.floor(mins / 60);
    73→  if (hrs < 24) return `${hrs}h atras`;
    74→  const days = Math.floor(hrs / 24);
    75→  return `${days}d atras`;
    76→}
