   113→function getCategory(id: string) {
   114→  return categories.find((c) => c.id === id) ?? categories[0];
   115→}
   116→
   117→// Per-category indicative annual rate that decreases with amount
   118→function annualRateFor(categoryId: string, amount: number) {
   119→  const cat = getCategory(categoryId);
   120→  // anchor amount where the "best" rate is reached
   121→  const anchor =
   122→    cat.id === "imoveis"
   123→      ? 5_000_000
   124→      : cat.id === "viaturas"
   125→        ? 2_000_000
   126→        : cat.id === "equipamentos"
   127→          ? 1_000_000
   128→          : 200_000; // eletrónicos
   129→  const t = Math.min(1, Math.max(0, amount / anchor));
   130→  // linear interpolation base → best rate
   131→  return cat.baseRate + (cat.bestRate - cat.baseRate) * t;
   132→}
   133→
   134→function simulate(categoryId: string, value: number, downPayment: number, months: number) {
   135→  const principal = Math.max(0, value - downPayment);
   136→  const annualRate = annualRateFor(categoryId, value);
   137→  const monthlyRate = annualRate / 12;
   138→  const monthly =
   139→    monthlyRate === 0 || months === 0
   140→      ? principal / Math.max(1, months)
   141→      : (principal * monthlyRate) / (1 - Math.pow(1 + monthlyRate, -months));
   142→  const total = monthly * months + downPayment;
   143→  const interest = total - value;
   144→  return { principal, monthly, total, interest, annualRate };
   145→}
