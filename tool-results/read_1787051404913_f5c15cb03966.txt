   146→const FloatingParticles = ({ color, count = 30 }: { color: string; count?: number }) => {
   147→  const particles = useMemo(
   148→    () =>
   149→      Array.from({ length: count }, (_, i) => ({
   150→        id: i,
   151→        x: Math.random() * 100,
   152→        y: Math.random() * 100,
   153→        size: 2 + Math.random() * 4,
   154→        duration: 6 + Math.random() * 8,
   155→        delay: Math.random() * 4,
   156→        opacity: 0.15 + Math.random() * 0.35,
   157→      })),
   158→    [count],
   159→  );
   160→  return (
   161→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
   162→      {particles.map((p) => (
   163→        <motion.div
   164→          key={p.id}
   165→          className="absolute rounded-full"
   166→          style={{
   167→            left: `${p.x}%`,
   168→            width: p.size,
   169→            height: p.size,
   170→            backgroundColor: color,
   171→          }}
   172→          animate={{
   173→            y: [0, -30, 0, 20, 0],
   174→            x: [0, 10, -10, 5, 0],
   175→            opacity: [p.opacity, p.opacity * 1.5, p.opacity * 0.5, p.opacity],
   176→          }}
   177→          transition={{
   178→            duration: p.duration,
   179→            repeat: Infinity,
   180→            delay: p.delay,
   181→            ease: "easeInOut",
   182→          }}
   183→        />
   184→      ))}
   185→    </div>
   186→  );
