    70→function CountingStat({ target, duration = 2 }: { target: number; duration?: number }) {
    71→  const [display, setDisplay] = useState(0);
    72→  useEffect(() => {
    73→    if (target === 0) { setDisplay(0); return; }
    74→    const startTime = performance.now();
    75→    const step = (now: number) => {
    76→      const elapsed = (now - startTime) / 1000;
    77→      const progress = Math.min(elapsed / duration, 1);
    78→      const eased = 1 - Math.pow(1 - progress, 4);
    79→      setDisplay(Math.round(eased * target));
    80→      if (progress < 1) requestAnimationFrame(step);
    81→    };
    82→    requestAnimationFrame(step);
    83→  }, [target, duration]);
    84→  return <>{display.toLocaleString("en-US")}</>;
    85→}

    87→function DirectoryParticles() {
    88→  const canvasRef = useRef<HTMLCanvasElement>(null);
    89→  useEffect(() => {
    90→    const canvas = canvasRef.current;
    91→    if (!canvas) return;
    92→    const ctx = canvas.getContext("2d");
    93→    if (!ctx) return;
    94→    let animId = 0;
    95→    let w = 0;
    96→    let h = 0;
    97→    const dots: { x: number; y: number; vx: number; vy: number; r: number; o: number; od: number }[] = [];
    98→    const resize = () => {
    99→      w = canvas.width = canvas.offsetWidth * 2;
   100→      h = canvas.height = canvas.offsetHeight * 2;
   101→    };
   102→    resize();
   103→    window.addEventListener("resize", resize);
   104→    for (let i = 0; i < 18; i++) {
   105→      dots.push({
   106→        x: Math.random() * w,
   107→        y: Math.random() * h,
   108→        vx: (Math.random() - 0.5) * 0.2,
   109→        vy: (Math.random() - 0.5) * 0.2,
   110→        r: Math.random() * 2.5 + 1,
   111→        o: Math.random() * 0.2 + 0.05,
   112→        od: (Math.random() - 0.5) * 0.002,
   113→      });
   114→    }
   115→    let lastTime = 0;
   116→    const draw = (timestamp: number) => {
   117→      animId = requestAnimationFrame(draw);
   118→      if (timestamp - lastTime < 33) return;
   119→      lastTime = timestamp;
   120→      ctx.clearRect(0, 0, w, h);
   121→      for (const d of dots) {
   122→        d.x += d.vx;
   123→        d.y += d.vy;
   124→        d.o += d.od;
   125→        if (d.o > 0.3) { d.o = 0.3; d.od *= -1; }
   126→        if (d.o < 0.03) { d.o = 0.03; d.od *= -1; }
   127→        if (d.x < 0 || d.x > w) d.vx *= -1;
   128→        if (d.y < 0 || d.y > h) d.vy *= -1;
   129→        const grad = ctx.createRadialGradient(d.x, d.y, 0, d.x, d.y, d.r * 3);
   130→        grad.addColorStop(0, `rgba(255,255,255,${d.o})`);
   131→        grad.addColorStop(1, "transparent");
   132→        ctx.beginPath();
   133→        ctx.arc(d.x, d.y, d.r * 3, 0, Math.PI * 2);
   134→        ctx.fillStyle = grad;
   135→        ctx.fill();
   136→      }
   137→      for (let i = 0; i < dots.length; i++) {
   138→        for (let j = i + 1; j < dots.length; j++) {
   139→          const dx = dots[i].x - dots[j].x;
   140→          const dy = dots[i].y - dots[j].y;
   141→          const dist = Math.sqrt(dx * dx + dy * dy);
   142→          if (dist < 220) {
   143→            ctx.beginPath();
   144→            ctx.moveTo(dots[i].x, dots[i].y);
   145→            ctx.lineTo(dots[j].x, dots[j].y);
   146→            ctx.strokeStyle = `rgba(255,255,255,${0.025 * (1 - dist / 220)})`;
   147→            ctx.lineWidth = 0.8;
   148→            ctx.stroke();
   149→          }
   150→        }
   151→      }
   152→    };
   153→    animId = requestAnimationFrame(draw);
   154→    return () => {
   155→      cancelAnimationFrame(animId);
   156→      window.removeEventListener("resize", resize);
   157→    };
   158→  }, []);
   159→  return (
   160→    <canvas
   161→      ref={canvasRef}
   162→      className="absolute inset-0 w-full h-full pointer-events-none"
   163→      style={{ opacity: 0.5 }}
   164→    />
   165→  );
   166→}

   203→function TiltCard({ children, className = "", ...props }: { children: React.ReactNode; className?: string } & Record<string, unknown>) {
   204→  const ref = useRef<HTMLDivElement>(null);
   205→  const x = useMotionValue(0);
   206→  const y = useMotionValue(0);
   207→  const rotateX = useSpring(useTransform(y, [-0.5, 0.5], [6, -6]), SPRING);
   208→  const rotateY = useSpring(useTransform(x, [-0.5, 0.5], [-6, 6]), SPRING);
   209→  const handleMouse = useCallback((e: React.MouseEvent) => {
   210→    if (!ref.current) return;
   211→    const r = ref.current.getBoundingClientRect();
   212→    x.set((e.clientX - r.left) / r.width - 0.5);
   213→    y.set((e.clientY - r.top) / r.height - 0.5);
   214→  }, [x, y]);
   215→  const handleLeave = useCallback(() => {
   216→    x.set(0);
   217→    y.set(0);
   218→  }, [x, y]);
   219→  return (
   220→    <motion.div
   221→      ref={ref}
   222→      onMouseMove={handleMouse}
   223→      onMouseLeave={handleLeave}
   224→      style={{ rotateX, rotateY, transformStyle: "preserve-3d", perspective: 800 }}
   225→      className={className}
   226→      {...props}
   227→    >
   228→      {children}
   229→    </motion.div>
   230→  );
   231→}

   233→function ActivityRing({ value, max, color, size = 40 }: { value: number; max: number; color: string; size?: number }) {
   234→  const [pct, setPct] = useState(0);
   235→  const safeMax = Math.max(max, 1);
   236→  useEffect(() => {
   237→    const timer = setTimeout(() => setPct(Math.min(value / safeMax, 1)), 400);
   238→    return () => clearTimeout(timer);
   239→  }, [value, safeMax]);
   240→  const r = (size - 6) / 2;
   241→  const circ = 2 * Math.PI * r;
   242→  const offset = circ * (1 - pct);
   243→  return (
   244→    <svg width={size} height={size} className="dirv2-activity-ring">
   245→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth={3} />
   246→      <circle
   247→        cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={3}
   248→        strokeDasharray={circ} strokeDashoffset={offset}
   249→        strokeLinecap="round"
   250→        style={{ transition: "stroke-dashoffset 1.2s cubic-bezier(0.22,1,0.36,1)" }}
   251→      />
   252→      <text x={size / 2} y={size / 2} textAnchor="middle" dominantBaseline="central"
   253→        fill={color} fontSize={size < 36 ? 9 : 11} fontWeight={800} fontFamily="inherit">
   254→        {value}
   255→      </text>
   256→    </svg>
   257→  );
   258→}
