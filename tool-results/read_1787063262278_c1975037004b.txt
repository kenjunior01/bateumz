   146→/* ─── Floating Particles Component ──────────────────── */
   147→const FloatingParticles = ({ color, count = 30 }: { color: string; count?: number }) => {
   148→  const particles = useMemo(
   149→    () =>
   150→      Array.from({ length: count }, (_, i) => ({
   151→        id: i,
   152→        x: Math.random() * 100,
   153→        y: Math.random() * 100,
   154→        size: 2 + Math.random() * 4,
   155→        duration: 6 + Math.random() * 8,
   156→        delay: Math.random() * 4,
   157→        opacity: 0.15 + Math.random() * 0.35,
   158→      })),
   159→    [count],
   160→  );
   161→  return (
   162→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
   163→      {particles.map((p) => (
   164→        <motion.div
   165→          key={p.id}
   166→          className="absolute rounded-full"
   167→          style={{
   168→            left: `${p.x}%`,
   169→            width: p.size,
   170→            height: p.size,
   171→            backgroundColor: color,
   172→          }}
   173→          animate={{
   174→            y: [0, -30, 0, 20, 0],
   175→            x: [0, 10, -10, 5, 0],
   176→            opacity: [p.opacity, p.opacity * 1.5, p.opacity * 0.5, p.opacity],
   177→          }}
   178→          transition={{
   179→            duration: p.duration,
   180→            repeat: Infinity,
   181→            delay: p.delay,
   182→            ease: "easeInOut",
   183→          }}
   184→        />
   185→      ))}
   186→    </div>
   187→  );
