   163→   163→function calcStats(p: PlayerState): Stats {
   164→   164→  const c = CLASSES[p.classId];
   165→   165→  const lvl = p.level;
   166→   166→  let hp = c.baseHp + lvl * 12;
   167→   167→  let maxHp = hp;
   168→   168→  let atk = c.baseAtk + lvl * 2;
   169→   169→  let def = c.baseDef + lvl * 1;
   170→   170→  let spd = c.baseSpd + lvl * 1;
   171→   171→  let mp = c.baseMp + lvl * 5;
   172→   172→  let maxMp = mp;
   173→   173→  for (const eq of p.equipment) {
   174→   174→    if (!eq) continue;
   175→   175→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   176→   176→    atk += eq.atk || 0;
   177→   177→    def += eq.def || 0;
   178→   178→    spd += eq.spd || 0;
   179→   179→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   180→   180→  }
   181→   181→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   182→   182→}
