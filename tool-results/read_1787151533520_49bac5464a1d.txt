   185→  const isUuid = (v: string) =>

   219→          .select("*")

   226→          .eq("business_user_id", bizId)
   227→          .eq("status", "active")
   228→          .order("created_at", { ascending: false }),

   232→      const contestsData = (contestsRes.data as Contest[]) || [];

   235→      setRaffles(rafflesData);
   236→      setContests(contestsData);

   249→        sb.from("challenge_roulettes").select("id, title, is_published, created_at").eq("business_user_id", bizId).order("created_at", { ascending: false }).limit(20),

   295→              raffle_title: r?.title || "Sorteio",

   298→              display_name: profMap.get(w.user_id) || null,
   299→              slug: r?.slug || null,

   301→          });

   303→        }

   306→      // Rankings: top 3 submissions per active/voting contest

   314→              .from("contest_submissions")

   316→              .eq("contest_id", c.id)
   317→              .eq("status", "approved")

   321→              contest_id: c.id,
   322→              contest_title: c.title,
   323→              evaluation_type: c.evaluation_type,
   324→              status: c.status,

   327→                name: s.participant_name,
   328→                photo_url: s.photo_url,
   329→                score: c.evaluation_type === "views" ? s.views_count : s.votes_count,
   330→              })),

   488→
   489→      <div className="relative">

   496→                "radial-gradient(circle at 20% 80%, rgba(255,255,255,0.3) 0, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.2) 0, transparent 50%)",

  1388→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */
  1389→
  1390→interface StatusOption { value: string; label: string }
  1391→interface SortOption { value: string; label: string }
  1392→
  1393→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1394→  items,
  1395→  statuses,
  1396→  sorts,
  1397→  renderItem,
  1398→  getStatus,
  1399→  getSortValue,
  1400→  formatScore,
  1401→  emptyMessage,
  1402→  extraAction,
  1403→  pageSize = 9,
  1404→}: {
  1405→  items: T[];
  1406→  statuses: StatusOption[];
  1407→  sorts: SortOption[];
  1408→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1409→  getStatus: (item: T) => string;
  1410→  getSortValue: (item: T, sort: string) => number;
  1411→  formatScore?: (item: T, sort: string) => string | undefined;
  1412→  emptyMessage: string;
  1413→  extraAction?: React.ReactNode;
  1414→  pageSize?: number;
  1415→}) {
  1416→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1417→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1418→  const [page, setPage] = useState(1);
  1419→  const [transitioning, setTransitioning] = useState(false);
  1420→
  1421→  // Reset page + show brief skeleton when filter/sort changes
  1422→  useEffect(() => {
  1423→    setPage(1);
  1424→    setTransitioning(true);
  1425→    const t = setTimeout(() => setTransitioning(false), 280);
  1426→    return () => clearTimeout(t);
  1427→  }, [status, sort]);
  1428→
  1429→  const filtered = useMemo(() => {
  1430→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1431→    if (sort === "recent") {
  1432→      list = [...list].sort((a, b) => {
  1433→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1434→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1435→        return tb - ta;
  1436→      });
  1437→    } else {
  1438→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1439→    }
  1440→    return list;
  1441→  }, [items, status, sort, getStatus, getSortValue]);
  1442→
  1443→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  1444→  const visible = filtered.slice(0, page * pageSize);
  1445→  const canLoadMore = page < totalPages;
  1446→
  1447→  const counts = useMemo(() => {
  1448→    const map: Record<string, number> = { all: items.length };
  1449→    for (const i of items) {
  1450→      const s = getStatus(i);
  1451→      map[s] = (map[s] ?? 0) + 1;
  1452→    }
  1453→    return map;
  1454→  }, [items, getStatus]);
  1455→
  1456→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1457→
  1458→  if (items.length === 0) {
  1459→    return <EmptyState message={emptyMessage} />;
  1460→  }
  1461→
  1462→  return (
  1463→    <div>
  1464→      <div className="mb-4 flex flex-col gap-3">
  1465→        <div className="flex flex-wrap gap-1.5">
  1466→          {statuses.map((s) => {
  1467→            const active = status === s.value;
  1468→            const count = counts[s.value] ?? 0;
  1469→            return (
  1470→              <button
  1471→                key={s.value}
  1472→                onClick={() => setStatus(s.value)}
  1473→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1474→                  active
  1475→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1476→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1477→                }`}
  1478→              >
  1479→                {s.label}
  1480→                <span className="ml-1 opacity-70">({count})</span>
  1481→              </button>
  1482→            );
  1483→          })}
  1484→        </div>
  1485→
  1486→        <div className="flex items-center justify-between gap-2">
  1487→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1488→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1489→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1490→          </p>
  1491→          <div className="flex items-center gap-2">
  1492→            {extraAction}
  1493→            <Select value={sort} onValueChange={setSort}>
  1494→              <SelectTrigger className="h-8 text-xs w-[140px] sm:w-[170px]">
  1495→                <SelectValue />
