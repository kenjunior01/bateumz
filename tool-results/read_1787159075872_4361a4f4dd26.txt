  1389→/* -------------------- FilteredList: status filter + sort + pagination -------------------- */
  1390→
  1391→interface StatusOption { value: string; label: string }
  1392→interface SortOption { value: string; label: string }
  1393→
  1394→function FilteredList<T extends { id: string; created_at?: string | null }>({
  1395→  items,
  1396→  statuses,
  1397→  sorts,
  1398→  renderItem,
  1399→  getStatus,
  1400→  getSortValue,
  1401→  formatScore,
  1402→  emptyMessage,
  1403→  extraAction,
  1404→  pageSize = 9,
  1405→}: {
  1406→  items: T[];
  1407→  statuses: StatusOption[];
  1408→  sorts: SortOption[];
  1409→  renderItem: (item: T, meta: { rank: number; total: number; criterion: string; scoreLabel?: string }) => React.ReactNode;
  1410→  getStatus: (item: T) => string;
  1411→  getSortValue: (item: T, sort: string) => number;
  1412→  formatScore?: (item: T, sort: string) => string | undefined;
  1413→  emptyMessage: string;
  1414→  extraAction?: React.ReactNode;
  1415→  pageSize?: number;
  1416→}) {
  1417→  const [status, setStatus] = useState<string>(statuses[0]?.value ?? "all");
  1418→  const [sort, setSort] = useState<string>(sorts[0]?.value ?? "recent");
  1419→  const [page, setPage] = useState(1);
  1420→  const [transitioning, setTransitioning] = useState(false);
  1421→
  1422→  // Reset page + show brief skeleton when filter/sort changes
  1423→  useEffect(() => {
  1424→    setPage(1);
  1425→    setTransitioning(true);
  1426→    const t = setTimeout(() => setTransitioning(false), 280);
  1427→    return () => clearTimeout(t);
  1428→  }, [status, sort]);
  1429→
  1430→  const filtered = useMemo(() => {
  1431→    let list = status === "all" ? items : items.filter((i) => getStatus(i) === status);
  1432→    if (sort === "recent") {
  1433→      list = [...list].sort((a, b) => {
  1434→        const ta = a.created_at ? new Date(a.created_at).getTime() : 0;
  1435→        const tb = b.created_at ? new Date(b.created_at).getTime() : 0;
  1436→        return tb - ta;
  1437→      });
  1438→    } else {
  1439→      list = [...list].sort((a, b) => getSortValue(b, sort) - getSortValue(a, sort));
  1440→    }
  1441→    return list;
  1442→  }, [items, status, sort, getStatus, getSortValue]);
  1443→
  1444→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
  1445→  const visible = filtered.slice(0, page * pageSize);
  1446→  const canLoadMore = page < totalPages;
  1447→
  1448→  const counts = useMemo(() => {
  1449→    const map: Record<string, number> = { all: items.length };
  1450→    for (const i of items) {
  1451→      const s = getStatus(i);
  1452→      map[s] = (map[s] ?? 0) + 1;
  1453→    }
  1454→    return map;
  1455→  }, [items, getStatus]);
  1456→
  1457→  const sortLabel = sorts.find((s) => s.value === sort)?.label || "Ordem";
  1458→
  1459→  if (items.length === 0) {
  1460→    return <EmptyState message={emptyMessage} />;
  1461→  }
  1462→
  1463→  return (
  1464→    <div>
  1465→      <div className="mb-4 flex flex-col gap-3">
  1466→        <div className="flex flex-wrap gap-1.5">
  1467→          {statuses.map((s) => {
  1468→            const active = status === s.value;
  1469→            const count = counts[s.value] ?? 0;
  1470→            return (
  1471→              <button
  1472→                key={s.value}
  1473→                onClick={() => setStatus(s.value)}
  1474→                className={`text-[11px] sm:text-xs px-2.5 py-1.5 rounded-full border transition-all ${
  1475→                  active
  1476→                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
  1477→                    : "bg-card border-border/60 text-muted-foreground hover:text-foreground hover:border-primary/40"
  1478→                }`}
  1479→              >
  1480→                {s.label}
  1481→                <span className="ml-1 opacity-70">({count})</span>
  1482→              </button>
  1483→            );
  1484→          })}
  1485→        </div>
  1486→
  1487→        <div className="flex items-center justify-between gap-2">
  1488→          <p className="text-[11px] sm:text-xs text-muted-foreground">
  1489→            {filtered.length} {filtered.length === 1 ? "item" : "itens"}
  1490→            <span className="hidden sm:inline"> · ordenado por <span className="text-foreground font-medium">{sortLabel.toLowerCase()}</span></span>
  1491→          </p>
  1492→          <div className="flex items-center gap-2">
  1493→            {extraAction}
