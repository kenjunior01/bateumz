   111→
   112→function emptyForm(category: PrestacaoCategory = "outros"): FormState {
   113→  const def = CATEGORY_DEFAULTS[category];
   114→  return {
   115→    title: "",
   116→    category,
   117→    description: "",
   118→    total_price: "",
   119→    min_down_payment: "",
   120→    max_months: String(def.maxMonths),
   121→    annual_rate: String(def.annualRate),
   122→    images: "",
   123→    province: "",
   124→    city: "",
   125→    brand: "",
   126→    model: "",
   127→    year: "",
   128→    whatsapp: "",
   129→    stock: "1",
   130→    status: "draft",
   131→    featured: false,
   132→  };
   133→}
   134→
   135→export default function DashboardPrestacoes() {
   136→  const { user } = useAuth();
   137→  const { formatMoney } = useCurrency();
   138→  const [loading, setLoading] = useState(true);
   139→  const [products, setProducts] = useState<Product[]>([]);
   140→  const [leads, setLeads] = useState<Lead[]>([]);
   141→  const [form, setForm] = useState<FormState | null>(null);
   142→  const [saving, setSaving] = useState(false);
   143→
   144→  async function load() {
   145→    if (!user) return;
   146→    setLoading(true);
   147→    const { data: prods } = await supabase
   148→      .from("prestacao_products")
   149→      .select("*")
   150→      .eq("business_user_id", user.id)
   151→      .order("created_at", { ascending: false });
   152→
   153→    const productIds = (prods ?? []).map((p: any) => p.id);
   154→    const orFilter = productIds.length > 0
   155→      ? `business_user_id.eq.${user.id},product_id.in.(${productIds.join(",")})`
   156→      : `business_user_id.eq.${user.id}`;
   157→
   158→    const { data: ls } = await supabase
   159→      .from("prestacao_product_leads")
   160→      .select("*")
   161→      .or(orFilter)
   162→      .order("created_at", { ascending: false })
   163→      .limit(500);
   164→
   165→    setProducts(
   166→      (prods ?? []).map((p: any) => ({
   167→        ...p,
   168→        images: Array.isArray(p.images) ? p.images : [],
   169→      })),
   170→    );
   171→    setLeads((ls ?? []) as Lead[]);
   172→    setLoading(false);
   173→  }
   174→
   175→  useEffect(() => {
   176→    load();
   177→    // eslint-disable-next-line react-hooks/exhaustive-deps
   178→  }, [user?.id]);
   179→
   180→  const stats = useMemo(() => {
   181→    const active = products.filter((p) => p.status === "active").length;
   182→    const draft = products.filter((p) => p.status === "draft").length;
   183→    const views = products.reduce((s, p) => s + (p.views_count ?? 0), 0);
   184→    return { total: products.length, active, draft, views, leads: leads.length };
   185→  }, [products, leads]);
   186→
   187→  function openCreate() {
   188→    setForm(emptyForm());
   189→  }
   190→  function openEdit(p: Product) {
   191→    setForm({
   192→      id: p.id,
   193→      title: p.title,
   194→      category: (p.category as PrestacaoCategory) ?? "outros",
   195→      description: p.description ?? "",
   196→      total_price: String(p.total_price),
   197→      min_down_payment: String(p.min_down_payment),
   198→      max_months: String(p.max_months),
   199→      annual_rate: String(p.annual_rate),
   200→      images: (p.images ?? []).join("\n"),
   201→      province: p.province ?? "",
   202→      city: p.city ?? "",
   203→      brand: p.brand ?? "",
   204→      model: p.model ?? "",
   205→      year: p.year ? String(p.year) : "",
   206→      whatsapp: p.whatsapp,
   207→      stock: String(p.stock),
   208→      status: (p.status as PrestacaoStatus) ?? "draft",
   209→      featured: p.featured,
   210→    });
   211→  }
   212→
   213→  async function save() {
   214→    if (!form || !user) return;
   215→    const images = form.images
   216→      .split(/\n|,/)
   217→      .map((s) => s.trim())
   218→      .filter(Boolean);
   219→    const payload = {
   220→      title: form.title,
   221→      category: form.category,
   222→      description: form.description || undefined,
   223→      total_price: Number(form.total_price),
   224→      min_down_payment: Number(form.min_down_payment),
   225→      max_months: Number(form.max_months),
   226→      annual_rate: Number(form.annual_rate),
   227→      images,
   228→      province: form.province,
   229→      city: form.city,
   230→      brand: form.brand,
   231→      model: form.model,
   232→      year: form.year ? Number(form.year) : null,
   233→      whatsapp: form.whatsapp,
   234→      stock: Number(form.stock),
   235→      status: form.status,
   236→      featured: form.featured,
   237→    };
   238→
   239→    const parsed = productSchema.safeParse(payload);
   240→    if (!parsed.success) {
   241→      const first = parsed.error.issues[0];
   242→      toast.error(first?.message ?? "Dados inválidos");
   243→      return;
   244→    }
   245→
   246→    setSaving(true);
   247→    const dbRow = {
   248→      ...parsed.data,
   249→      description: parsed.data.description || null,
   250→      province: parsed.data.province || null,
   251→      city: parsed.data.city || null,
   252→      brand: parsed.data.brand || null,
   253→      model: parsed.data.model || null,
   254→      business_user_id: user.id,
   255→    };
   256→
   257→    const { error } = form.id
   258→      ? await supabase
   259→          .from("prestacao_products")
   260→          .update(dbRow as any)
   261→          .eq("id", form.id)
   262→      : await supabase.from("prestacao_products").insert(dbRow as any);
   263→
   264→    setSaving(false);
   265→    if (error) {
   266→      toast.error(error.message);
   267→      return;
   268→    }
   269→    toast.success(form.id ? "Produto atualizado" : "Produto criado");
   270→    setForm(null);
   271→    load();
   272→  }
   273→
   274→  async function remove(id: string) {
   275→    if (!confirm("Eliminar este produto?")) return;
   276→    const { error } = await supabase.from("prestacao_products").delete().eq("id", id);
   277→    if (error) {
   278→      toast.error(error.message);
   279→      return;
   280→    }
   281→    toast.success("Produto eliminado");
   282→    load();
   283→  }
   284→
   285→  async function toggleStatus(p: Product) {
   286→    const next = p.status === "active" ? "draft" : "active";
   287→    const { error } = await supabase
   288→      .from("prestacao_products")
   289→      .update({ status: next })
   290→      .eq("id", p.id);
   291→    if (error) {
   292→      toast.error(error.message);
   293→      return;
   294→    }
   295→    toast.success(`Produto ${next === "active" ? "publicado" : "ocultado"}`);
   296→    load();
   297→  }
   298→
   299→  return (
   300→    <div className="space-y-6">
   301→      <div className="flex items-center justify-between flex-wrap gap-3">
   302→        <div>
   303→          <h1 className="text-2xl font-bold">Prestações</h1>
   304→          <p className="text-sm text-muted-foreground">
   305→            Publique e gira os seus produtos a prestações no catálogo Bateu.
   306→          </p>
   307→        </div>
   308→        <Button onClick={openCreate}>
   309→          <Plus className="h-4 w-4 mr-2" />
   310→          Novo produto
   311→        </Button>
   312→      </div>
   313→
   314→      <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   315→        {[
   316→          { label: "Total", value: stats.total },
   317→          { label: "Ativos", value: stats.active },
   318→          { label: "Rascunhos", value: stats.draft },
   319→          { label: "Visualizações", value: stats.views },
   320→          { label: "Leads", value: stats.leads },
   321→        ].map((s) => (
   322→          <Card key={s.label}>
   323→            <CardContent className="p-4">
   324→              <p className="text-xs text-muted-foreground">{s.label}</p>
   325→              <p className="text-2xl font-bold">{s.value}</p>
   326→            </CardContent>
   327→          </Card>
   328→        ))}
   329→      </div>
   330→
   331→      <Tabs defaultValue="products" className="w-full">
   332→        <TabsList className="grid w-full grid-cols-2 max-w-sm">
   333→          <TabsTrigger value="products" className="flex items-center gap-1.5">
   334→            <Package className="h-3.5 w-3.5" /> Produtos
   335→          </TabsTrigger>
   336→          <TabsTrigger value="leads" className="flex items-center gap-1.5">
   337→            <Inbox className="h-3.5 w-3.5" /> Leads
   338→            {leads.length > 0 && (
   339→              <Badge variant="secondary" className="ml-1 h-5 px-1.5 text-[10px]">
   340→                {leads.length}
   341→              </Badge>
   342→            )}
   343→          </TabsTrigger>
   344→        </TabsList>
   345→
   346→        <TabsContent value="products" className="mt-4">
   347→          <Card>
   348→            <CardContent className="p-3 sm:p-4">
   349→              <h2 className="font-semibold mb-3 text-sm sm:text-base">Os meus produtos</h2>
   350→              {loading ? (
   351→                <div className="py-10 text-center">
   352→                  <Loader2 className="h-5 w-5 animate-spin mx-auto text-primary" />
   353→                </div>
   354→              ) : products.length === 0 ? (
   355→                <div className="py-10 text-center text-sm text-muted-foreground">
   356→                  Ainda não tem produtos. Clique em "Novo produto" para começar.
   357→                </div>
   358→              ) : (
   359→                <div className="space-y-2">
   360→                  {products.map((p) => (
   361→                    <motion.div
   362→                      key={p.id}
   363→                      initial={{ opacity: 0, y: 6 }}
   364→                      animate={{ opacity: 1, y: 0 }}
   365→                      className="flex items-center gap-2 sm:gap-3 p-2 sm:p-3 rounded-lg border bg-card"
   366→                    >
   367→                      <div className="w-12 h-12 sm:w-14 sm:h-14 rounded-md bg-muted overflow-hidden flex items-center justify-center shrink-0">
   368→                        {p.images[0] ? (
   369→                          <img src={p.images[0]} alt="" className="w-full h-full object-cover" />
   370→                        ) : (
   371→                          <Package className="h-5 w-5 text-muted-foreground" />
   372→                        )}
   373→                      </div>
   374→                      <div className="flex-1 min-w-0">
   375→                        <div className="flex items-center gap-1.5 flex-wrap">
   376→                          <p className="font-medium truncate text-sm sm:text-base">{p.title}</p>
   377→                          <Badge
   378→                            variant={p.status === "active" ? "default" : "secondary"}
   379→                            className="capitalize text-[10px] px-1.5 py-0"
   380→                          >
   381→                            {p.status}
   382→                          </Badge>
   383→                          {p.featured && <Badge variant="outline" className="text-[10px] px-1.5 py-0">Destaque</Badge>}
   384→                        </div>
   385→                        <div className="text-[11px] sm:text-xs text-muted-foreground flex gap-2 sm:gap-3 flex-wrap mt-1">
   386→                          <span>{formatMoney(Number(p.total_price))}</span>
   387→                          <span>Stock: {p.stock}</span>
   388→                          <span className="inline-flex items-center gap-1">
   389→                            <Eye className="h-3 w-3" />
   390→                            {p.views_count ?? 0}
   391→                          </span>
   392→                        </div>
   393→                      </div>
   394→                      <div className="flex items-center gap-0.5 sm:gap-1 shrink-0">
   395→                        <Button
   396→                          size="sm"
   397→                          variant="outline"
   398→                          className="h-8 px-2 text-xs"
   399→                          onClick={() => toggleStatus(p)}
   400→                        >
   401→                          {p.status === "active" ? "Ocultar" : "Publicar"}
   402→                        </Button>
   403→                        <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => openEdit(p)}>
   404→                          <Pencil className="h-3.5 w-3.5" />
   405→                        </Button>
   406→                        <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => remove(p.id)}>
   407→                          <Trash2 className="h-3.5 w-3.5 text-destructive" />
   408→                        </Button>
   409→                      </div>
   410→                    </motion.div>
   411→                  ))}
   412→                </div>
   413→              )}
   414→            </CardContent>
   415→          </Card>
   416→        </TabsContent>
   417→
   418→        <TabsContent value="leads" className="mt-4">
   419→          <LeadsPanel leads={leads} products={products} />
   420→        </TabsContent>
   421→      </Tabs>
   422→
   423→      <Dialog open={!!form} onOpenChange={(o) => !o && setForm(null)}>
   424→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   425→          <DialogHeader>
   426→            <DialogTitle>
   427→              {form?.id ? "Editar produto" : "Novo produto a prestações"}
   428→            </DialogTitle>
   429→          </DialogHeader>
   430→          {form && (
   431→            <div className="space-y-4">
   432→              <div className="grid md:grid-cols-2 gap-3">
   433→                <div className="md:col-span-2">
   434→                  <Label>Título *</Label>
   435→                  <Input
   436→                    value={form.title}
   437→                    onChange={(e) => setForm({ ...form, title: e.target.value })}
   438→                  />
   439→                </div>
   440→                <div>
   441→                  <Label>Categoria *</Label>
   442→                  <Select
   443→                    value={form.category}
   444→                    onValueChange={(v) => {
   445→                      const cat = v as PrestacaoCategory;
   446→                      const def = CATEGORY_DEFAULTS[cat];
   447→                      setForm({
   448→                        ...form,
   449→                        category: cat,
   450→                        annual_rate: form.annual_rate || String(def.annualRate),
   451→                        max_months: form.max_months || String(def.maxMonths),
   452→                      });
   453→                    }}
   454→                  >
   455→                    <SelectTrigger>
   456→                      <SelectValue />
   457→                    </SelectTrigger>
   458→                    <SelectContent>
   459→                      {PRESTACAO_CATEGORIES.map((c) => (
   460→                        <SelectItem key={c.id} value={c.id}>
   461→                          {c.label}
   462→                        </SelectItem>
   463→                      ))}
   464→                    </SelectContent>
   465→                  </Select>
   466→                </div>
   467→                <div>
   468→                  <Label>Estado *</Label>
   469→                  <Select
   470→                    value={form.status}
   471→                    onValueChange={(v) =>
   472→                      setForm({ ...form, status: v as PrestacaoStatus })
   473→                    }
   474→                  >
   475→                    <SelectTrigger>
   476→                      <SelectValue />
   477→                    </SelectTrigger>
   478→                    <SelectContent>
   479→                      {PRESTACAO_STATUSES.map((s) => (
   480→                        <SelectItem key={s.value} value={s.value}>
   481→                          {s.label}
   482→                        </SelectItem>
   483→                      ))}
   484→                    </SelectContent>
   485→                  </Select>
   486→                </div>
   487→                <div className="md:col-span-2">
   488→                  <Label>Descrição</Label>
   489→                  <Textarea
   490→                    rows={3}
   491→                    value={form.description}
   492→                    onChange={(e) =>
   493→                      setForm({ ...form, description: e.target.value })
   494→                    }
   495→                  />
   496→                </div>
   497→                <div>
   498→                  <Label>Preço total (MZN) *</Label>
   499→                  <Input
   500→                    type="number"
   501→                    inputMode="numeric"
   502→                    value={form.total_price}
   503→                    onChange={(e) =>
   504→                      setForm({ ...form, total_price: e.target.value })
   505→                    }
   506→                  />
   507→                </div>
   508→                <div>
   509→                  <Label>Entrada mínima (MZN) *</Label>
   510→                  <Input
   511→                    type="number"
   512→                    inputMode="numeric"
   513→                    value={form.min_down_payment}
   514→                    onChange={(e) =>
   515→                      setForm({ ...form, min_down_payment: e.target.value })
   516→                    }
   517→                  />
   518→                </div>
   519→                <div>
   520→                  <Label>Prazo máximo (meses) *</Label>
   521→                  <Input
   522→                    type="number"
   523→                    inputMode="numeric"
   524→                    value={form.max_months}
   525→                    onChange={(e) =>
   526→                      setForm({ ...form, max_months: e.target.value })
   527→                    }
   528→                  />
   529→                </div>
   530→                <div>
   531→                  <Label>Taxa anual (ex.: 0.18) *</Label>
   532→                  <Input
   533→                    type="number"
   534→                    step="0.01"
   535→                    value={form.annual_rate}
   536→                    onChange={(e) =>
   537→                      setForm({ ...form, annual_rate: e.target.value })
   538→                    }
   539→                  />
   540→                </div>
   541→                <div>
   542→                  <Label>Stock *</Label>
   543→                  <Input
   544→                    type="number"
   545→                    inputMode="numeric"
   546→                    value={form.stock}
   547→                    onChange={(e) => setForm({ ...form, stock: e.target.value })}
   548→                  />
   549→                </div>
   550→                <div>
   551→                  <Label>WhatsApp *</Label>
   552→                  <Input
   553→                    placeholder="+258..."
   554→                    value={form.whatsapp}
   555→                    onChange={(e) => setForm({ ...form, whatsapp: e.target.value })}
   556→                  />
   557→                </div>
   558→                <div>
   559→                  <Label>Província</Label>
   560→                  <Select
   561→                    value={form.province || "_none"}
   562→                    onValueChange={(v) =>
   563→                      setForm({ ...form, province: v === "_none" ? "" : v })
   564→                    }
   565→                  >
   566→                    <SelectTrigger>
   567→                      <SelectValue placeholder="Selecionar" />
   568→                    </SelectTrigger>
   569→                    <SelectContent>
   570→                      <SelectItem value="_none">—</SelectItem>
   571→                      {PROVINCES.map((p) => (
   572→                        <SelectItem key={p.value} value={p.label}>
   573→                          {p.label}
   574→                        </SelectItem>
   575→                      ))}
   576→                    </SelectContent>
   577→                  </Select>
   578→                </div>
   579→                <div>
   580→                  <Label>Cidade</Label>
   581→                  <Input
   582→                    value={form.city}
   583→                    onChange={(e) => setForm({ ...form, city: e.target.value })}
   584→                  />
   585→                </div>
   586→                <div>
   587→                  <Label>Marca</Label>
   588→                  <Input
   589→                    value={form.brand}
   590→                    onChange={(e) => setForm({ ...form, brand: e.target.value })}
   591→                  />
   592→                </div>
   593→                <div>
   594→                  <Label>Modelo</Label>
   595→                  <Input
   596→                    value={form.model}
   597→                    onChange={(e) => setForm({ ...form, model: e.target.value })}
   598→                  />
   599→                </div>
   600→                <div>
   601→                  <Label>Ano</Label>
   602→                  <Input
   603→                    type="number"
   604→                    inputMode="numeric"
   605→                    value={form.year}
   606→                    onChange={(e) => setForm({ ...form, year: e.target.value })}
   607→                  />
   608→                </div>
   609→                <div className="md:col-span-2">
   610→                  <Label>Imagens (uma URL por linha, máx. 8)</Label>
   611→                  <Textarea
   612→                    rows={3}
   613→                    placeholder="https://...\nhttps://..."
   614→                    value={form.images}
   615→                    onChange={(e) => setForm({ ...form, images: e.target.value })}
   616→                  />
   617→                </div>
   618→                <div className="flex items-center gap-2">
   619→                  <Switch
   620→                    checked={form.featured}

   624→                </div>
   625→              </div>
   626→            </div>
   627→          )}
   628→          <DialogFooter>
   629→            <Button variant="ghost" onClick={() => setForm(null)} disabled={saving}>
   630→              <X className="h-4 w-4 mr-1" /> Cancelar
   631→            </Button>
   632→            <Button onClick={save} disabled={saving}>
   633→              {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   634→              Guardar
   635→            </Button>
   636→          </DialogFooter>
   637→        </DialogContent>
   638→      </Dialog>
   639→    </div>
   640→  );

   643→// ─── Leads panel with filters ───────────────────────────────────────────
   644→type LeadsPanelProps = {
   645→  leads: Lead[];
   646→  products: Product[];
   647→};
   648→
   649→function LeadsPanel({ leads, products }: LeadsPanelProps) {
   650→  const [category, setCategory] = useState<string>("all");
   651→  const [status, setStatus] = useState<string>("all");
   652→  const [period, setPeriod] = useState<string>("all");
   653→  const [query, setQuery] = useState("");
   654→
   655→  const filtered = useMemo(() => {
   656→    const now = Date.now();
   657→    const cutoff: Record<string, number> = {
   658→      "7": 7 * 24 * 3600 * 1000,
   659→      "30": 30 * 24 * 3600 * 1000,
   660→      "90": 90 * 24 * 3600 * 1000,
   661→    };
   662→    return leads.filter((l) => {
   663→      const product = products.find((p) => p.id === l.product_id);
   664→      const cat = l.category ?? product?.category ?? null;
   665→      if (category !== "all" && cat !== category) return false;
   666→      if (status !== "all" && (l.status ?? "new") !== status) return false;
   667→      if (period !== "all") {
   668→        const ms = cutoff[period];
   669→        if (ms && now - new Date(l.created_at).getTime() > ms) return false;
   670→      }
   671→      if (query.trim()) {
   672→        const q = query.toLowerCase();
   673→        const haystack = [
   674→          l.visitor_name,
   675→          l.visitor_whatsapp,
   676→          product?.title,
   677→          l.notes,
   678→        ]
   679→          .filter(Boolean)
   680→          .join(" ")
   681→          .toLowerCase();
   682→        if (!haystack.includes(q)) return false;
   683→      }
   684→      return true;
   685→    });
   686→  }, [leads, products, category, status, period, query]);
   687→
   688→  const generalCount = leads.filter((l) => !l.product_id).length;
   689→  const productCount = leads.filter((l) => l.product_id).length;
   690→
   691→  return (
   692→    <Card>
   693→      <CardContent className="p-3 sm:p-4 space-y-4">
   694→        <div className="flex items-center gap-2 flex-wrap">
   695→          <h2 className="font-semibold text-sm sm:text-base flex items-center gap-1.5">
   696→            <MessageSquare className="h-4 w-4" /> Leads recebidos
   697→          </h2>
   698→          <Badge variant="outline" className="text-[10px]">
   699→            {productCount} por produto
   700→          </Badge>
   701→          <Badge variant="outline" className="text-[10px]">
   702→            {generalCount} gerais
   703→          </Badge>
   704→        </div>
   705→
   706→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   707→          <Select value={category} onValueChange={setCategory}>
   708→            <SelectTrigger className="h-9 text-xs">
   709→              <Filter className="h-3 w-3 mr-1" />
   710→              <SelectValue placeholder="Categoria" />
   711→            </SelectTrigger>
   712→            <SelectContent>
   713→              <SelectItem value="all">Todas categorias</SelectItem>
   714→              {PRESTACAO_CATEGORIES.map((c) => (
   715→                <SelectItem key={c.id} value={c.id}>
   716→                  {c.label}
   717→                </SelectItem>
   718→              ))}
   719→            </SelectContent>
   720→          </Select>
   721→
   722→          <Select value={status} onValueChange={setStatus}>
   723→            <SelectTrigger className="h-9 text-xs">
   724→              <SelectValue placeholder="Estado" />
   725→            </SelectTrigger>
   726→            <SelectContent>
   727→              <SelectItem value="all">Todos estados</SelectItem>
   728→              <SelectItem value="new">Novo</SelectItem>
   729→              <SelectItem value="contacted">Contactado</SelectItem>
   730→              <SelectItem value="closed">Fechado</SelectItem>
   731→              <SelectItem value="lost">Perdido</SelectItem>
   732→            </SelectContent>
   733→          </Select>
   734→
   735→          <Select value={period} onValueChange={setPeriod}>
   736→            <SelectTrigger className="h-9 text-xs">
   737→              <SelectValue placeholder="Período" />
   738→            </SelectTrigger>
   739→            <SelectContent>
   740→              <SelectItem value="all">Sempre</SelectItem>
   741→              <SelectItem value="7">Últimos 7 dias</SelectItem>
   742→              <SelectItem value="30">Últimos 30 dias</SelectItem>
   743→              <SelectItem value="90">Últimos 90 dias</SelectItem>
   744→            </SelectContent>
   745→          </Select>
   746→
   747→          <Input
   748→            placeholder="Buscar nome, WhatsApp..."
   749→            value={query}
   750→            onChange={(e) => setQuery(e.target.value)}
   751→            className="h-9 text-xs"
   752→          />
   753→        </div>
   754→
   755→        {filtered.length === 0 ? (
   756→          <p className="text-xs sm:text-sm text-muted-foreground py-6 text-center">
   757→            Nenhum lead corresponde aos filtros.
   758→          </p>
   759→        ) : (
   760→          <div className="space-y-2">
   761→            {filtered.map((l) => {
   762→              const product = products.find((p) => p.id === l.product_id);
   763→              const isGeneral = !l.product_id;
   764→              return (
   765→                <div
   766→                  key={l.id}
   767→                  className="text-xs sm:text-sm flex items-start justify-between gap-2 sm:gap-3 p-2 sm:p-3 rounded-lg border bg-card"
   768→                >
   769→                  <div className="min-w-0 flex-1">
   770→                    <div className="flex items-center gap-1.5 flex-wrap">
   771→                      <p className="font-medium truncate">
   772→                        {product?.title ?? l.visitor_name ?? "Interesse geral"}
   773→                      </p>
   774→                      <Badge
   775→                        variant={isGeneral ? "secondary" : "default"}
   776→                        className="text-[10px] px-1.5 py-0"
   777→                      >
   778→                        {isGeneral ? "Geral" : "Produto"}
   779→                      </Badge>
   780→                      {(l.category ?? product?.category) && (
   781→                        <Badge variant="outline" className="text-[10px] px-1.5 py-0 capitalize">
   782→                          {l.category ?? product?.category}
   783→                        </Badge>
   784→                      )}
   785→                    </div>
   786→                    <p className="text-[11px] sm:text-xs text-muted-foreground mt-1 truncate">
   787→                      {l.visitor_name && <span>{l.visitor_name} · </span>}
   788→                      {l.visitor_whatsapp && <span>{l.visitor_whatsapp} · </span>}
   789→                      Entrada {formatMoney(Number(l.down_payment))} · {l.months}m · ~
   790→                      {formatMoney(Number(l.monthly_estimate))}/mês
   791→                    </p>
   792→                    {l.notes && (
   793→                      <p className="text-[11px] text-muted-foreground mt-1 italic line-clamp-2">
   794→                        "{l.notes}"
   795→                      </p>
   796→                    )}
   797→                  </div>
   798→                  <div className="text-[10px] sm:text-xs text-muted-foreground shrink-0 text-right">
   799→                    {new Date(l.created_at).toLocaleDateString("pt-PT")}
   800→                  </div>
   801→                </div>
   802→              );
   803→            })}
   804→          </div>
   805→        )}
   806→      </CardContent>
   807→    </Card>
   808→  );
   809→}
