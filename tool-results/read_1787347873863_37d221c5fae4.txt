   180→   180→function calcStats(char: CharacterData): CharStats {
   181→   181→  const c = CLASSES[char.classId];
   182→   182→  const lvl = char.level;
   183→   183→  let hp = c.baseHp + lvl * 12, maxHp = hp;
   184→   184→  let atk = c.baseAtk + lvl * 2;
   185→   185→  let def = c.baseDef + lvl * 1;
   186→   186→  let spd = c.baseSpd + lvl * 1;
   187→   187→  let mp = c.baseMp + lvl * 5, maxMp = mp;
   188→   188→  for (const eq of char.equipment) {
   189→   189→    if (!eq) continue;
   190→   190→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   191→   191→    atk += eq.atk || 0; def += eq.def || 0; spd += eq.spd || 0;
   192→   192→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   193→   193→  }
   194→   194→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   195→   195→}

   203→   203→function timeAgo(dateStr: string): string {
   204→   204→  const s = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000);
   205→   205→  if (s < 60) return "agora";
   206→   206→  if (s < 3600) return `${Math.floor(s / 60)}m atras`;
   207→   207→  if (s < 86400) return `${Math.floor(s / 3600)}h atras`;
   208→   208→  return `${Math.floor(s / 86400)}d atras`;
   209→   209→}

   215→   215→async function dbUpsertChar(char: CharacterData, guestId?: string) {
   216→   216→  try {
   217→   217→    const stats = calcStats(char);
   218→   218→    const row = {
   219→   219→      guest_id: guestId || null,
   220→   220→      name: char.name, class_id: char.classId, level: char.level, xp: char.xp, gold: char.gold,
   221→   221→      hp: stats.maxHp, max_hp: stats.maxHp, mp: stats.maxMp, max_mp: stats.maxMp,
   222→   222→      atk: stats.atk, def: stats.def, spd: stats.spd,
   223→   223→      equipment: JSON.stringify(char.equipment), inventory: JSON.stringify(char.inventory),
   224→   224→      total_stars: 0, total_kills: char.totalKills, total_deaths: char.totalDeaths,
   225→   225→      total_duels_won: char.duelsWon, total_duels_lost: char.duelsLost, total_earned: char.totalEarned,
   226→   226→      current_zone: char.currentZone, is_online: true, last_online: new Date().toISOString(),
   227→   227→    };
   228→   228→    const { error } = await (supabase as any).from("rpg_characters").upsert(row, { onConflict: "guest_id" });
   229→   229→    if (error) throw error;
   230→   230→  } catch { saveLocalChar(char); }
   231→   231→}
