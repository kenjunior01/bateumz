   154→   154→const pickWeighted = (prizes: WheelPrize[]) => {
   155→   155→  const total = prizes.reduce((s, p) => s + Math.max(0, p.weight), 0) || 1;
   156→   156→  let r = Math.random() * total;
   157→   157→  for (let i = 0; i < prizes.length; i++) {
   158→   158→    r -= Math.max(0, prizes[i].weight);
   159→   159→    if (r <= 0) return i;
   160→   160→  }
   161→   161→  return prizes.length - 1;
