     5→import { useAuth } from "@/contexts/AuthContext";

     7→import { Card, CardContent } from "@/components/ui/card";
     8→import { Button } from "@/components/ui/button";
     9→import { Badge } from "@/components/ui/badge";
    10→import { Loader2, Volume2, VolumeX, Lightbulb, Users, Phone, CheckCircle2, XCircle, Trophy, Timer, RefreshCw } from "lucide-react";
    11→import { toast } from "sonner";
    12→import confetti from "canvas-confetti";
    13→import { fetchTriviaQuestions, decodeHTMLEntities, shuffleAnswers, TriviaQuestion } from "@/lib/openTriviaDB";
    14→

    16→  level: number;
    17→  amount: number;
    18→  currency: string;

    21→

    24→  name: string;

    27→  primary_color?: string;
    28→  company_logo_url?: string;
    29→  company_slogan?: string;
    30→  total_questions: number;
    31→  time_per_question: number;
    32→  prize_structure?: PrizeLevel[];
    33→  lifelines?: Record<string, boolean>;
    34→  use_trivia_db?: boolean;
    35→}
    36→
    37→interface Question {
    38→  id: string;
    39→  question_number: number;
    40→  question_text: string;
    41→  option_a: string;
    42→  option_b: string;
    43→  option_c: string;
    44→  option_d: string;
    45→  correct_answer: string;
    46→  explanation?: string;
    47→}
    48→
    49→interface Props {

    53→

    72→
    73→export default function EnhancedMillionaireGame({ gameId: propGameId, onComplete }: Props) {
    74→  const { gameId: urlGameId } = useParams();
    75→  const gameId = propGameId || urlGameId;
    76→  const { user } = useAuth();
    77→  const { region } = useRegionalTheme();
    78→  const [loading, setLoading] = useState(true);
    79→  const [game, setGame] = useState<Game | null>(null);
    80→  const [questions, setQuestions] = useState<Question[]>([]);
    81→  const [currentLevel, setCurrentLevel] = useState(1);
    82→  const [selectedAnswer, setSelectedAnswer] = useState<string | null>(null);
    83→  const [answered, setAnswered] = useState(false);
    84→  const [timeLeft, setTimeLeft] = useState(30);
    85→  const [lifelinesUsed, setLifelinesUsed] = useState<Record<string, boolean>>({});
    86→  const [status, setStatus] = useState<'playing' | 'won' | 'lost'>('playing');
    87→  const [soundEnabled, setSoundEnabled] = useState(true);
    88→  const [disabledOptions, setDisabledOptions] = useState<string[]>([]);
    89→  const [triviaLoading, setTriviaLoading] = useState(false);
    90→  const [flashClass, setFlashClass] = useState("");
    91→  const [shakeKey, setShakeKey] = useState(0);
    92→  const [questionKey, setQuestionKey] = useState(0);
    93→  const [showCelebrationRays, setShowCelebrationRays] = useState(false);
    94→
    95→  // Default questions for fallback
    96→  const defaultQuestions: Question[] = [
    97→    { id: "1", question_number: 1, question_text: "Qual é a capital de Moçambique?", option_a: "Beira", option_b: "Maputo", option_c: "Nampula", option_d: "Tete", correct_answer: "B" },
    98→    { id: "2", question_number: 2, question_text: "Quantas províncias tem Moçambique?", option_a: "9", option_b: "10", option_c: "11", option_d: "12", correct_answer: "C" },
    99→    { id: "3", question_number: 3, question_text: "Qual é a moeda oficial?", option_a: "Kwanza", option_b: "Real", option_c: "Metical", option_d: "Escudo", correct_answer: "C" },
   100→  ];
   101→
   102→  const loadGame = useCallback(async () => {
   103→    if (!gameId) {
   104→      setLoading(false);
   105→      return;
   106→    }
   107→
   108→    setLoading(true);
   109→    try {
   110→      const { data: gameData, error: gameError } = await supabase
   111→        .from("millionaire_games")
   112→        .select("*")
   113→        .eq("id", gameId)
   114→        .single();
   115→
   116→      if (gameError) {
   117→        console.warn("Falling back to default game config:", gameError);
   118→        // Create a default game object
   119→        setGame({
   120→          id: gameId,
   121→          name: "Quem Quer Ser Milionário?",
   122→          total_questions: 15,
   123→          time_per_question: 30,
   124→          background_color: "#0a0e17",
   125→          primary_color: "#fbbf24",
   126→          use_trivia_db: true,
   127→        });
   128→      } else {
   129→        setGame(gameData as unknown as Game);
   130→        setTimeLeft(gameData.time_per_question || 30);
   131→      }
   132→
   133→
   134→      // Check if we should use Open Trivia DB
   135→      if (game?.use_trivia_db || true) { // Default to true for demo purposes
   136→        await loadTriviaQuestions(game?.total_questions || 15);
   137→      } else {
   138→        const { data: qData, error: qError } = await supabase
   139→          .from("millionaire_questions")
   140→          .select("*")
   141→          .eq("game_id", gameId)
   142→          .order("question_number", { ascending: true });
   143→
   144→        if (qError || !qData || qData.length === 0) {
   145→          console.warn("Falling back to default questions:", qError);
   146→          setQuestions(defaultQuestions);
   147→          if (!game) {
   148→            setGame({
   149→              id: gameId,
   150→              name: "Quem Quer Ser Milionário?",
   151→              total_questions: 3,
   152→              time_per_question: 30,
   153→              background_color: "#0a0e17",
   154→              primary_color: "#fbbf24",
   155→            });
   156→          }
   157→        } else {
   158→          setQuestions(qData);
   159→        }
   160→      }
   161→    } catch (err) {
   162→      console.error("Error loading millionaire game:", err);
   163→      // Fallback to defaults
   164→      setGame({
   165→        id: gameId,
   166→        name: "Quem Quer Ser Milionário?",
   167→        total_questions: 3,
   168→        time_per_question: 30,
   169→        background_color: "#0a0e17",
   170→        primary_color: "#fbbf24",
   171→      });
   172→      setQuestions(defaultQuestions);
   173→    } finally {
   174→      setLoading(false);
   175→    }
   176→  }, [gameId, game?.use_trivia_db, game?.total_questions]);
   177→
   178→  const loadTriviaQuestions = async (count: number) => {
   179→    setTriviaLoading(true);
   180→    try {
   181→      const triviaQuestions = await fetchTriviaQuestions(count, undefined, 'easy', 'multiple');
   182→      
   183→      if (triviaQuestions.length === 0) {
   184→        setQuestions(defaultQuestions);
   185→        toast.warning('Não foi possível carregar perguntas da base de dados, usando padrão');
   186→      } else {
   187→        const formattedQuestions: Question[] = triviaQuestions.map((tq, index) => {
   188→          const shuffled = shuffleAnswers(
   189→            decodeHTMLEntities(tq.correct_answer), 
   190→            tq.incorrect_answers.map(a => decodeHTMLEntities(a))
   191→          );
   192→          
   193→          return {
   194→            id: `trivia-${index}`,
   195→            question_number: index + 1,
   196→            question_text: decodeHTMLEntities(tq.question),
   197→            ...shuffled,
   198→          };
   199→        });
   200→        setQuestions(formattedQuestions);
   201→        toast.success(`Carregadas ${formattedQuestions.length} perguntas da Open Trivia DB!`);
   202→      }
   203→    } catch (error) {
   204→      console.error('Failed to load trivia:', error);
   205→      setQuestions(defaultQuestions);
   206→    } finally {
   207→      setTriviaLoading(false);
   208→    }
   209→  };
   210→
   211→  useEffect(() => { loadGame(); }, [loadGame]);
   212→
   213→  // Timer logic
   214→  useEffect(() => {
   215→    if (status !== 'playing' || answered || timeLeft <= 0) return;
   216→    const timer = setInterval(() => setTimeLeft(prev => prev - 1), 1000);
   217→    return () => clearInterval(timer);
   218→  }, [status, answered, timeLeft]);
   219→
   220→  // Auto-fail on timeout
   221→  useEffect(() => {
   222→    if (timeLeft === 0 && !answered && status === 'playing') {
   223→      setStatus('lost');
   224→      toast.error("Tempo esgotado!");
   225→    }
   226→  }, [timeLeft, answered, status]);
   227→
   228→  const currentQuestion = useMemo(() => questions[currentLevel - 1], [questions, currentLevel]);
   229→  
   230→  const prizeStructure = useMemo(() => {
   231→    if (game?.prize_structure && game.prize_structure.length > 0) {
   232→      return game.prize_structure;
   233→    }
   234→    // Use default prize structure adjusted to match total questions
   235→    return DEFAULT_PRIZE_STRUCTURE.slice(0, game?.total_questions || 15);
   236→  }, [game]);
   237→
   238→  const currentPrize = useMemo(() => prizeStructure[currentLevel - 1], [prizeStructure, currentLevel]);
   239→
   240→  const handleAnswer = async (choice: string) => {
   241→    if (answered || status !== 'playing') return;
   242→    setSelectedAnswer(choice);
   243→    setAnswered(true);
   244→
   245→    const isCorrect = choice === currentQuestion.correct_answer;
   246→
   247→    setFlashClass("");
   248→    setTimeout(() => {
   249→      setFlashClass(isCorrect ? "game-flash-green" : "game-flash-red");
   250→      setShakeKey(k => k + 1);
   251→      setTimeout(() => setFlashClass(""), 600);
   252→    }, 100);
   253→
   254→    setTimeout(async () => {
   255→      if (isCorrect) {
   256→        if (currentLevel === (game?.total_questions || prizeStructure.length)) {
   257→          setStatus('won');
   258→          setShakeKey(k => k + 1);
   259→          setShowCelebrationRays(true);
   260→          setFlashClass("game-flash-gold");
   261→          setTimeout(() => setFlashClass(""), 1200);
   262→          confetti({ particleCount: 200, spread: 100, origin: { y: 0.55 } });
   263→          setTimeout(() => confetti({ particleCount: 150, spread: 140, origin: { x: 0.2, y: 0.5 }, colors: ["#fbbf24", "#f59e0b", "#ffffff", "#fde68a"] }), 150);
   264→          setTimeout(() => confetti({ particleCount: 150, spread: 140, origin: { x: 0.8, y: 0.5 }, colors: ["#fbbf24", "#f59e0b", "#ffffff", "#fde68a"] }), 300);
   265→          setTimeout(() => confetti({ particleCount: 100, spread: 80, origin: { y: 0.3 }, colors: ["#fbbf24", "#ffffff"], shapes: ["star"] }), 500);
   266→          setTimeout(() => {
   267→            const end = Date.now() + 4000;
   268→            const iv = setInterval(() => {
   269→              if (Date.now() > end) return clearInterval(iv);
   270→              confetti({ particleCount: 3, angle: 60, spread: 55, origin: { x: 0, y: 0.6 }, colors: ["#fbbf24", "#ffffff"] });
   271→              confetti({ particleCount: 3, angle: 120, spread: 55, origin: { x: 1, y: 0.6 }, colors: ["#fbbf24", "#ffffff"] });
   272→            }, 60);
   273→          }, 800);
   274→          saveSession('completed', currentPrize?.amount || 0);
   275→          onComplete?.(currentPrize?.amount || 0, currentLevel, 'won');
   276→        } else {
   277→          toast.success("Resposta Correta!");
   278→          confetti({ particleCount: 40, spread: 50, origin: { y: 0.7 }, colors: ["#22c55e", "#4ade80", "#ffffff"] });
   279→          setTimeout(() => confetti({ particleCount: 20, spread: 40, origin: { x: 0.5, y: 0.6 }, colors: ["#22c55e", "#ffffff"] }), 150);
   280→          // Advance after delay
   281→          setTimeout(() => {
   282→            setCurrentLevel(prev => prev + 1);
   283→            setAnswered(false);
   284→            setSelectedAnswer(null);
   285→            setDisabledOptions([]);
   286→            setTimeLeft(game?.time_per_question || 30);
   287→            setQuestionKey(k => k + 1);
   288→          }, 1500);
   289→        }
   290→      } else {
   291→        setStatus('lost');
   292→        setShakeKey(k => k + 1);
   293→        setTimeout(() => {
   294→          confetti({ particleCount: 30, spread: 60, origin: { y: 0.6 }, colors: ["#ef4444", "#991b1b"] });
   295→        }, 500);
   296→        const safePrize = calculateSafePrize();
   297→        saveSession('abandoned', safePrize);
   298→        onComplete?.(safePrize, currentLevel, 'lost');
   299→      }
   300→    }, 2000); // Dramatic pause
   301→  };
   302→
   303→  const calculateSafePrize = () => {
   304→    const prizes = prizeStructure;
   305→    let safeAmount = 0;
   306→    // Níveis de segurança padrão: 5 e 10
   307→    const defaultSafeLevels = [5, 10];
   308→    for (let i = 0; i < currentLevel - 1; i++) {
   309→      if (prizes[i]?.is_safe_haven || defaultSafeLevels.includes(prizes[i]?.level)) {
   310→        safeAmount = prizes[i]?.amount || 0;
   311→      }
   312→    }
   313→    return safeAmount;
   314→  };
   315→
   316→  const handleLifeline = (type: string) => {
   317→    if (lifelinesUsed[type] || answered) return;
   318→    setLifelinesUsed(prev => ({ ...prev, [type]: true }));
   319→    
   320→    if (type === '50_50') {
   321→      const wrongAnswers = ['A', 'B', 'C', 'D'].filter(l => l !== currentQuestion.correct_answer);
   322→      const toDisable = wrongAnswers.sort(() => 0.5 - Math.random()).slice(0, 2);
   323→      setDisabledOptions(toDisable);
   324→      toast.info("50/50 Ativado: Duas respostas erradas removidas.");
   325→    }
   326→  };
   327→
   328→  const saveSession = async (finalStatus: string, prize: number) => {
   329→    if (!user || !region || !gameId) return;
   330→    try {
   331→      await supabase.from("millionaire_sessions").insert({
   332→        game_id: gameId,
   333→        user_id: user.id,
   334→        region_id: region.id,
   335→        current_level: currentLevel,
   336→        prize_won: prize,
   337→        status: finalStatus
   338→      });
   339→    } catch (err) { console.error(err); }
   340→  };
   341→
   342→  const restartGame = () => {
   343→    setCurrentLevel(1);
   344→    setStatus('playing');
   345→    setAnswered(false);
   346→    setSelectedAnswer(null);
   347→    setDisabledOptions([]);
   348→    setTimeLeft(game?.time_per_question || 30);
   349→    setLifelinesUsed({});
   350→    setShowCelebrationRays(false);
   351→  };
   352→
   353→  if (loading || triviaLoading) return (
   354→    <div className="h-screen flex flex-col items-center justify-center bg-[#0a0e17] gap-4 relative overflow-hidden">
   355→      <div className="game-particle game-particle-1" style={{ top: '20%', left: '30%' }} />
   356→      <div className="game-particle game-particle-3" style={{ bottom: '30%', right: '20%' }} />
   357→      <div className="game-particle game-particle-5" style={{ top: '60%', left: '10%' }} />
   358→      <div className="relative">
   359→        <div className="absolute inset-0 rounded-full bg-primary/20 blur-xl animate-pulse" />
   360→        <Loader2 className="h-12 w-12 animate-spin text-primary relative" />
   361→      </div>
   362→      <motion.p
   363→        initial={{ opacity: 0 }}
   364→        animate={{ opacity: 1 }}
   365→        transition={{ delay: 0.3 }}
   366→        className="text-white text-lg"
   367→      >Carregando perguntas...</motion.p>
   368→    </div>
   369→  );
   370→  
   371→  if (!game || !currentQuestion) return <Navigate to="/" replace />;
   372→
   373→  return (
   374→    <div key={shakeKey} className={`min-h-screen relative flex flex-col bg-[#0a0e17] text-white overflow-hidden ${shakeKey > 0 && status !== 'playing' ? 'game-screen-shake' : ''}`}
   375→      style={{ 
   376→        backgroundColor: game.background_color || '#0a0e17',
   377→        backgroundImage: game.background_image_url ? `url(${game.background_image_url})` : 'none',
   378→        backgroundSize: 'cover',
   379→        backgroundPosition: 'center'
   380→      }}
   381→    >
   382→      <div className="absolute inset-0 bg-gradient-to-b from-[#0a0e17]/80 via-[#0a0e17]/60 to-[#0a0e17]/95"></div>
   383→      {flashClass && <div className={`fixed inset-0 z-[100] pointer-events-none ${flashClass}`} />}
   384→      {showCelebrationRays && <div className="celebration-rays" />}
   385→      <div className="game-particle game-particle-1" style={{ top: '15%', left: '10%' }} />
   386→      <div className="game-particle game-particle-2" style={{ top: '25%', right: '15%' }} />
   387→      <div className="game-particle game-particle-3" style={{ bottom: '20%', left: '20%' }} />
   388→      <div className="game-particle game-particle-4" style={{ top: '50%', right: '8%' }} />
   389→      <div className="game-particle game-particle-5" style={{ bottom: '10%', right: '25%' }} />
   390→      <div className="game-particle game-particle-6" style={{ top: '70%', left: '5%' }} />
   391→
   392→      <div className="relative z-10 p-6 flex justify-between items-center border-b border-white/10 backdrop-blur-md">
   393→        <div className="flex items-center gap-4">
   394→          <div className="w-12 h-12 rounded-full bg-primary flex items-center justify-center shadow-[0_0_20px_rgba(var(--primary),0.5)]">
   395→            <Trophy className="text-black w-6 h-6" />
   396→          </div>
   397→          <div>
   398→            <h1 className="text-xl font-black uppercase tracking-widest">{game.name}</h1>
   399→            <p className="text-xs text-white/50 uppercase">Nível {currentLevel} • {currentPrize?.amount} {currentPrize?.currency}</p>
   400→          </div>
   401→        </div>
   402→
   403→        <div className="flex items-center gap-6">
   404→          <Button 
   405→            variant="ghost" 
   406→            size="icon" 
   407→            onClick={() => loadTriviaQuestions(game?.total_questions || 15)} 
   408→            title="Recarregar perguntas"
   409→          >
   410→            <RefreshCw className="w-5 h-5" />
   411→          </Button>
   412→          <div className={`flex items-center gap-2 px-4 py-2 rounded-full border transition-all duration-500 ${
   413→            timeLeft <= 5 ? 'bg-red-500/20 border-red-500/50 timer-urgent' :
   414→            timeLeft <= 10 ? 'bg-amber-500/20 border-amber-500/40' :
   415→            'bg-black/40 border-white/10'
   416→          }`}>
   417→            <Timer className={`w-5 h-5 transition-colors duration-500 ${
   418→              timeLeft <= 5 ? 'text-red-400' :
   419→              timeLeft <= 10 ? 'text-amber-400' :
   420→              'text-primary'
   421→            }`} />
   422→            <span className={`text-xl font-mono font-bold transition-colors duration-500 ${
   423→              timeLeft <= 5 ? 'text-red-400' :
   424→              timeLeft <= 10 ? 'text-amber-400' :
   425→              'text-white'
   426→            }`}>{timeLeft}s</span>
   427→          </div>
   428→          <Button variant="ghost" size="icon" onClick={() => setSoundEnabled(!soundEnabled)}>
   429→            {soundEnabled ? <Volume2 className="w-5 h-5" /> : <VolumeX className="w-5 h-5" />}
   430→          </Button>
   431→        </div>
   432→      </div>
   433→
   434→      <div className="absolute top-[72px] left-0 right-0 h-1 bg-white/5 z-20">
   435→        <div
   436→          className="h-full transition-colors duration-1000"
   437→          style={{
   438→            width: status === 'playing' && !answered ? `${(timeLeft / (game?.time_per_question || 30)) * 100}%` : '0%',
   439→            background: timeLeft <= 5 ? 'linear-gradient(90deg, #ef4444, #f87171)' : timeLeft <= 10 ? 'linear-gradient(90deg, #f59e0b, #fbbf24)' : 'linear-gradient(90deg, hsl(var(--primary)), #fbbf24)',
   440→            transition: 'width 1s linear, background 1s ease',
   441→            boxShadow: timeLeft <= 5 ? '0 0 15px rgba(239, 68, 68, 0.5)' : timeLeft <= 10 ? '0 0 10px rgba(251, 191, 36, 0.3)' : '0 0 5px rgba(251, 191, 36, 0.2)'
   442→          }}
   443→        />
   444→      </div>
   445→
   446→      <div className="relative z-10 flex-1 container mx-auto px-4 py-8 grid lg:grid-cols-[1fr_320px] gap-8">
   447→        
   448→        <div className="flex flex-col justify-center space-y-12">
   449→          
   450→          <div className="relative space-y-6">
   451→            {(game.company_logo_url || game.company_slogan) && (
   452→              <motion.div 
   453→                initial={{ opacity: 0, y: -20 }}
   454→                animate={{ opacity: 1, y: 0 }}
   455→                className="flex flex-col items-center"
   456→              >
   457→                {game.company_logo_url && (
   458→                  <img src={game.company_logo_url} alt="Logo" className="h-16 object-contain mb-2 drop-shadow-[0_0_10px_rgba(255,255,255,0.3)]" />
   459→                )}
   460→                {game.company_slogan && (
   461→                  <p className="text-xs font-black uppercase tracking-[0.3em] text-primary/80">{game.company_slogan}</p>
   462→                )}
   463→              </motion.div>
   464→            )}
   465→            
   466→            <div className="relative">
   467→              <div className="absolute -left-4 top-1/2 -translate-y-1/2 w-8 h-[2px] bg-primary"></div>
   468→              <div className="absolute -right-4 top-1/2 -translate-y-1/2 w-8 h-[2px] bg-primary"></div>
   469→              <motion.div 
   470→                initial={{ opacity: 0, scale: 0.85, y: 20 }}
   471→                animate={{ opacity: 1, scale: 1, y: 0 }}
