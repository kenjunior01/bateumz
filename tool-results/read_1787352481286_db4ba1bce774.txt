   226→function rollLoot(): LootDrop | null {
   227→  const roll = Math.random();
   228→  let cumulative = 0;
   229→  let rarity = "common";
   230→  for (const [key, val] of Object.entries(LOOT_RARITY)) {
   231→    cumulative += val.chance;
   232→    if (roll < cumulative) { rarity = key; break; }
   233→  }
   234→  // Only drop 35% of the time
   235→  if (Math.random() > 0.35) return null;
   236→  const item = { ...LOOT_ITEMS[Math.floor(Math.random() * LOOT_ITEMS.length)] };
   237→  // Boost stats based on rarity
   238→  const mult = rarity === "legendary" ? 3 : rarity === "epic" ? 2.2 : rarity === "rare" ? 1.6 : rarity === "uncommon" ? 1.3 : 1;
   239→  if (item.atk) item.atk = Math.round(item.atk * mult);
   240→  if (item.def) item.def = Math.round(item.def * mult);
   241→  if (item.hp) item.hp = Math.round(item.hp * mult);
   242→  if (item.mp) item.mp = Math.round(item.mp * mult);
   243→  if (item.spd) item.spd = Math.round(item.spd * mult);
   244→  item.price = Math.round(item.price * mult);
   245→  item.id += "_" + rarity;
   246→  return { item, rarity };
   247→}

   266→function calcStats(char: CharacterData): CharStats {
   267→  const c = CLASSES[char.classId];
   268→  const lvl = char.level;
   269→  let hp = c.baseHp + lvl * 12, maxHp = hp;
   270→  let atk = c.baseAtk + lvl * 2;
   271→  let def = c.baseDef + lvl * 1;
   272→  let spd = c.baseSpd + lvl * 1;
   273→  let mp = c.baseMp + lvl * 5, maxMp = mp;
   274→  for (const eq of char.equipment) {
   275→    if (!eq) continue;
   276→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   277→    atk += eq.atk || 0; def += eq.def || 0; spd += eq.spd || 0;
   278→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   279→  }
   280→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   281→}

   287→}

   289→function timeAgo(dateStr: string): string {
   290→  const s = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000);
   291→  if (s < 60) return "agora";
   292→  if (s < 3600) return `${Math.floor(s / 60)}m atras`;
   293→  if (s < 86400) return `${Math.floor(s / 3600)}h atras`;
   294→  return `${Math.floor(s / 86400)}d atras`;
   295→}

   301→async function dbUpsertChar(char: CharacterData, guestId?: string) {
   302→  try {
   303→    const stats = calcStats(char);
   304→    const row = {
   305→      guest_id: guestId || null,
   306→      name: char.name, class_id: char.classId, level: char.level, xp: char.xp, gold: char.gold,
   307→      hp: stats.maxHp, max_hp: stats.maxHp, mp: stats.maxMp, max_mp: stats.maxMp,
   308→      atk: stats.atk, def: stats.def, spd: stats.spd,
   309→      equipment: JSON.stringify(char.equipment), inventory: JSON.stringify(char.inventory),
   310→      total_stars: 0, total_kills: char.totalKills, total_deaths: char.totalDeaths,
   311→      total_duels_won: char.duelsWon, total_duels_lost: char.duelsLost, total_earned: char.totalEarned,
   312→      current_zone: char.currentZone, is_online: true, last_online: new Date().toISOString(),
   313→    };
   314→    const { error } = await (supabase as any).from("rpg_characters").upsert(row, { onConflict: "guest_id" });
   315→    if (error) throw error;
   316→  } catch { saveLocalChar(char); }
   317→}
