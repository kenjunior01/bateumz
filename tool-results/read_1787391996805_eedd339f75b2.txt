   153→const SAVE_KEY = "bateu_rpg_save";

   155→function loadSave(): PlayerState | null {
   156→  try { const r = localStorage.getItem(SAVE_KEY); return r ? JSON.parse(r) : null; } catch { return null; }
   157→}

   159→function persistSave(s: PlayerState) {
   160→  try { localStorage.setItem(SAVE_KEY, JSON.stringify(s)); } catch {}

   163→function calcStats(p: PlayerState): Stats {
   164→  const c = CLASSES[p.classId];
   165→  const lvl = p.level;
   166→  let hp = c.baseHp + lvl * 12;
   167→  let maxHp = hp;
   168→  let atk = c.baseAtk + lvl * 2;
   169→  let def = c.baseDef + lvl * 1;
   170→  let spd = c.baseSpd + lvl * 1;
   171→  let mp = c.baseMp + lvl * 5;
   172→  let maxMp = mp;
   173→  for (const eq of p.equipment) {
   174→    if (!eq) continue;
   175→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   176→    atk += eq.atk || 0;
   177→    def += eq.def || 0;
   178→    spd += eq.spd || 0;
   179→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   180→  }
   181→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   182→}

   289→        ctx.fillStyle = "#1e1b4b";
   290→        ctx.fillRect(px - 50, py + 40, 100, 6);
   291→        ctx.fillStyle = "#6366f1";

   297→          const hit = animatingHit === i + 1;
   298→          const ex = cx + 120 + (i > 0 ? i * 60 : 0) + (hit ? (Math.random()-0.5)*16 : 0);
   299→          const ey = cy - 20 + (i > 0 ? i * -30 : 0);

   307→          // Enemy HP bar
   308→          const ehp = e.hp / e.maxHp;
   309→          ctx.fillStyle = "#1f2937";
   310→          ctx.fillRect(ex - 40, ey + 8, 80, 6);
   311→          ctx.fillStyle = e.color || "#ef4444";
   312→          ctx.fillRect(ex - 40, ey + 8, 80 * ehp, 6);

   314→          ctx.font = "12px sans-serif";

   316→        });

   319→      // Draw particles
   320→      particlesRef.current = particlesRef.current.filter(p => {
   321→        p.x += p.vx;
   322→        p.y += p.vy;
   323→        p.vy += 0.15;
   324→        p.life -= 0.02;
   325→        if (p.life <= 0) return false;
   326→        ctx.save();
   327→        ctx.globalAlpha = p.life;
