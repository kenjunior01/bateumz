   226→   226→function rollLoot(): LootDrop | null {
   227→   227→  const roll = Math.random();
   228→   228→  let cumulative = 0;
   229→   229→  let rarity = "common";
   230→   230→  for (const [key, val] of Object.entries(LOOT_RARITY)) {
   231→   231→    cumulative += val.chance;
   232→   232→    if (roll < cumulative) { rarity = key; break; }
   233→   233→  }
   234→   234→  // Only drop 35% of the time
   235→   235→  if (Math.random() > 0.35) return null;
   236→   236→  const item = { ...LOOT_ITEMS[Math.floor(Math.random() * LOOT_ITEMS.length)] };
   237→   237→  // Boost stats based on rarity
   238→   238→  const mult = rarity === "legendary" ? 3 : rarity === "epic" ? 2.2 : rarity === "rare" ? 1.6 : rarity === "uncommon" ? 1.3 : 1;
   239→   239→  if (item.atk) item.atk = Math.round(item.atk * mult);
   240→   240→  if (item.def) item.def = Math.round(item.def * mult);
   241→   241→  if (item.hp) item.hp = Math.round(item.hp * mult);
   242→   242→  if (item.mp) item.mp = Math.round(item.mp * mult);
   243→   243→  if (item.spd) item.spd = Math.round(item.spd * mult);
   244→   244→  item.price = Math.round(item.price * mult);
   245→   245→  item.id += "_" + rarity;
   246→   246→  return { item, rarity };
   247→   247→}

   266→   266→function calcStats(char: CharacterData): CharStats {
   267→   267→  const c = CLASSES[char.classId];
   268→   268→  const lvl = char.level;
   269→   269→  let hp = c.baseHp + lvl * 12, maxHp = hp;
   270→   270→  let atk = c.baseAtk + lvl * 2;
   271→   271→  let def = c.baseDef + lvl * 1;
   272→   272→  let spd = c.baseSpd + lvl * 1;
   273→   273→  let mp = c.baseMp + lvl * 5, maxMp = mp;
   274→   274→  for (const eq of char.equipment) {
   275→   275→    if (!eq) continue;
   276→   276→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   277→   277→    atk += eq.atk || 0; def += eq.def || 0; spd += eq.spd || 0;
   278→   278→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   279→   279→  }
   280→   280→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   281→   281→}
