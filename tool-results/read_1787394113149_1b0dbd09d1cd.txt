     8→interface Props {

    20→
    21→interface Bullet {
    22→  x: number;
    23→  y: number;
    24→  speed: number;

    27→
    28→interface Enemy {
    29→  x: number;
    30→  y: number;
    31→  speed: number;
    32→  size: number;
    33→  type: "triangle" | "diamond";
    34→  color: string;
    35→  hp: number;
    36→  bonus: boolean;
    37→  rotation: number;
    38→  rotSpeed: number;

    42→
    43→interface Particle {
    44→  x: number;
    45→  y: number;
    46→  vx: number;
    47→  vy: number;
    48→  life: number;
    49→  maxLife: number;
    50→  color: string;
    51→  size: number;

    53→
    54→interface PlayerState {
    55→  x: number;
    56→  y: number;
    57→  score: number;
    58→  lives: number;

    65→
    66→const CANVAS_W = 600;
    67→const CANVAS_H = 500;
    68→const SHIP_SIZE = 18;
    69→const BULLET_SPEED = 7;
    70→const BULLET_COOLDOWN = 12;
    71→const PLAYER_SPEED = 5;
    72→const STAR_COUNT = 120;
    73→const INITIAL_ENEMY_INTERVAL = 60;
    74→const MIN_ENEMY_INTERVAL = 15;
    75→const INITIAL_ENEMY_SPEED = 1.5;
    76→const MAX_ENEMY_SPEED = 4.5;
    77→const BONUS_CHANCE = 0.08;
    78→const WAVE_DURATION = 900;
    79→const POINTS_NORMAL = 10;
    80→const POINTS_BONUS = 30;
    81→
    82→const createStars = (): Star[] =>
    83→  Array.from({ length: STAR_COUNT }, () => ({
    84→    x: Math.random() * CANVAS_W,
    85→    y: Math.random() * CANVAS_H,
    86→    speed: 0.2 + Math.random() * 1.2,
    87→    size: 0.5 + Math.random() * 1.8,
    88→    brightness: 0.3 + Math.random() * 0.7,
    89→  }));
    90→
    91→const createExplosion = (x: number, y: number, color: string): Particle[] =>
    92→  Array.from({ length: 14 }, () => {
    93→    const angle = Math.random() * Math.PI * 2;
    94→    const speed = 1 + Math.random() * 3.5;
    95→    return {
    96→      x,
    97→      y,
    98→      vx: Math.cos(angle) * speed,
    99→      vy: Math.sin(angle) * speed,
   100→      life: 20 + Math.random() * 20,
   101→      maxLife: 40,
   102→      color,
   103→      size: 1.5 + Math.random() * 2.5,
   104→    };
   105→  });
   106→
   107→interface GameState {
   108→  players: PlayerState[];
   109→  bullets: Bullet[];
   110→  enemies: Enemy[];
   111→  particles: Particle[];
   112→  stars: Star[];
   113→  keys: Set<string>;
   114→  frame: number;
   115→  wave: number;
   116→  waveFrame: number;
   117→  running: boolean;
   118→  animId: number;
   119→}
   120→
   121→const SpaceShooter = ({ onScore, liveCode }: Props) => {
   122→  const canvasRef = useRef<HTMLCanvasElement>(null);
   123→  const gameRef = useRef<GameState | null>(null);
   124→  const touchRef = useRef<Record<string, boolean>>({});
   125→
   126→  const [gameStatus, setGameStatus] = useState<"idle" | "playing" | "over">("idle");
   127→  const [scores, setScores] = useState({ p1: 0, p2: 0 });
   128→  const [lives, setLives] = useState({ p1: 3, p2: 3 });
   129→  const [wave, setWave] = useState(1);
   130→  const [winner, setWinner] = useState<string | null>(null);
   131→  const [forceTick, setForceTick] = useState(0);
   132→
   133→  const triggerUI = () => setForceTick((t) => t + 1);
   134→
   135→  const endGame = useCallback(
   136→    (p1Score: number, p2Score: number) => {
   137→      setGameStatus("over");
   138→      setScores({ p1: p1Score, p2: p2Score });
   139→      const w = p1Score > p2Score ? "Jogador 1" : p2Score > p1Score ? "Jogador 2" : null;
   140→      setWinner(w);
   141→      onScore?.("Jogador 1", p1Score);
   142→      onScore?.("Jogador 2", p2Score);
   143→    },
   144→    [onScore]
   145→  );
   146→
   147→  const spawnEnemy = useCallback((g: GameState) => {
   148→    const waveMultiplier = 1 + (g.wave - 1) * 0.15;
   149→    const speed = Math.min(INITIAL_ENEMY_SPEED * waveMultiplier + Math.random() * 0.8, MAX_ENEMY_SPEED);
   150→    const isBonus = Math.random() < BONUS_CHANCE + g.wave * 0.005;
   151→    const type = Math.random() < 0.5 ? ("triangle" as const) : ("diamond" as const);
   152→
   153→    g.enemies.push({
   154→      x: 30 + Math.random() * (CANVAS_W - 60),
   155→      y: -20,
   156→      speed,
   157→      size: isBonus ? 16 : 12 + Math.random() * 6,
   158→      type,
   159→      color: isBonus ? "#fbbf24" : Math.random() < 0.5 ? "#ef4444" : "#f97316",
   160→      hp: 1,
   161→      bonus: isBonus,
   162→      rotation: 0,
   163→      rotSpeed: (Math.random() - 0.5) * 0.06,
   164→      wobble: Math.random() * Math.PI * 2,
   165→      wobbleSpeed: 0.02 + Math.random() * 0.03,
   166→    });
   167→  }, []);
   168→
   169→  const gameLoop = useCallback(
   170→    (g: GameState) => {
   171→      const canvas = canvasRef.current;
   172→      if (!canvas || !g.running) return;
   173→      const ctx = canvas.getContext("2d");
   174→      if (!ctx) return;
   175→
   176→      g.frame++;
   177→      g.waveFrame++;
   178→
   179→      if (g.waveFrame >= WAVE_DURATION) {
   180→        g.wave++;
   181→        g.waveFrame = 0;
   182→        setWave(g.wave);
   183→      }
   184→
   185→      const spawnInterval = Math.max(MIN_ENEMY_INTERVAL, INITIAL_ENEMY_INTERVAL - (g.wave - 1) * 5);
   186→      if (g.frame % Math.round(spawnInterval) === 0) {
   187→        const count = g.wave >= 5 ? 2 : 1;
   188→        for (let i = 0; i < count; i++) spawnEnemy(g);
   189→      }
   190→
   191→      const keys = new Set(g.keys);
   192→      for (const [k, v] of Object.entries(touchRef.current)) {
   193→        if (v) keys.add(k);
   194→      }
   195→
   196→      const p1 = g.players[0];
   197→      const p2 = g.players[1];
   198→
   199→      if (keys.has("a") || keys.has("A")) p1.x = Math.max(SHIP_SIZE, p1.x - PLAYER_SPEED);
   200→      if (keys.has("d") || keys.has("D")) p1.x = Math.min(CANVAS_W - SHIP_SIZE, p1.x + PLAYER_SPEED);
   201→      if (p1.shootCooldown > 0) p1.shootCooldown--;
   202→      if (
   203→        (keys.has("w") || keys.has("W") || keys.has("p1shoot")) &&
   204→        p1.shootCooldown <= 0 &&
   205→        p1.lives > 0
   206→      ) {
   207→        g.bullets.push({ x: p1.x, y: p1.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 1 });
   208→        p1.shootCooldown = BULLET_COOLDOWN;
   209→      }
   210→      if (p1.invincible > 0) p1.invincible--;
   211→
   212→      if (keys.has("ArrowLeft")) p2.x = Math.max(SHIP_SIZE, p2.x - PLAYER_SPEED);
   213→      if (keys.has("ArrowRight")) p2.x = Math.min(CANVAS_W - SHIP_SIZE, p2.x + PLAYER_SPEED);
   214→      if (p2.shootCooldown > 0) p2.shootCooldown--;
   215→      if (
   216→        (keys.has("ArrowUp") || keys.has("p2shoot")) &&
   217→        p2.shootCooldown <= 0 &&
   218→        p2.lives > 0
   219→      ) {
   220→        g.bullets.push({ x: p2.x, y: p2.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 2 });
   221→        p2.shootCooldown = BULLET_COOLDOWN;
   222→      }
   223→      if (p2.invincible > 0) p2.invincible--;
   224→
   225→      g.bullets = g.bullets.filter((b) => {
   226→        b.y -= b.speed;
   227→        return b.y > -10;
   228→      });
   229→
   230→      g.enemies = g.enemies.filter((e) => {
   231→        e.y += e.speed;
   232→        e.rotation += e.rotSpeed;
   233→        e.wobble += e.wobbleSpeed;
   234→        e.x += Math.sin(e.wobble) * 0.5;
   235→        e.x = Math.max(e.size, Math.min(CANVAS_W - e.size, e.x));
   236→
   237→        if (e.y > CANVAS_H + 10) {
   238→          const midX = CANVAS_W / 2;
   239→          if (e.x < midX) {
   240→            if (p1.lives > 0 && p1.invincible <= 0) {
   241→              p1.lives--;
   242→              p1.invincible = 60;
   243→              setLives({ p1: p1.lives, p2: p2.lives });
   244→            }
   245→          } else {
   246→            if (p2.lives > 0 && p2.invincible <= 0) {
   247→              p2.lives--;
   248→              p2.invincible = 60;
   249→              setLives({ p1: p1.lives, p2: p2.lives });
   250→            }
   251→          }
   252→          return false;
   253→        }
   254→        return true;
   255→      });
   256→
   257→      const bulletsToRemove = new Set<number>();
   258→      const enemiesToRemove = new Set<number>();
   259→      let scoreChanged = false;
   260→
   261→      for (let bi = 0; bi < g.bullets.length; bi++) {
   262→        const b = g.bullets[bi];
   263→        for (let ei = 0; ei < g.enemies.length; ei++) {
   264→          if (enemiesToRemove.has(ei)) continue;
   265→          const e = g.enemies[ei];
   266→          const dx = b.x - e.x;
   267→          const dy = b.y - e.y;
   268→          const dist = Math.sqrt(dx * dx + dy * dy);
   269→          if (dist < e.size + 4) {
   270→            e.hp--;
   271→            bulletsToRemove.add(bi);
   272→            if (e.hp <= 0) {
   273→              enemiesToRemove.add(ei);
   274→              const pts = e.bonus ? POINTS_BONUS : POINTS_NORMAL;
   275→              if (b.owner === 1) p1.score += pts;
   276→              else p2.score += pts;
   277→              g.particles.push(...createExplosion(e.x, e.y, e.color));
   278→              scoreChanged = true;
   279→            }
   280→            break;
   281→          }
   282→        }
   283→      }
   284→
   285→      if (scoreChanged) {
   286→        setScores({ p1: p1.score, p2: p2.score });
   287→      }
   288→
   289→      g.bullets = g.bullets.filter((_, i) => !bulletsToRemove.has(i));
   290→      g.enemies = g.enemies.filter((_, i) => !enemiesToRemove.has(i));
   291→
   292→      g.particles = g.particles.filter((p) => {
   293→        p.x += p.vx;
   294→        p.y += p.vy;
   295→        p.vy += 0.05;
   296→        p.life--;
   297→        return p.life > 0;
   298→      });
   299→
   300→      g.stars.forEach((s) => {
   301→        s.y += s.speed;
   302→        if (s.y > CANVAS_H) {
   303→          s.y = 0;
   304→          s.x = Math.random() * CANVAS_W;
   305→        }
   306→      });
   307→
   308→      if (p1.lives <= 0 && p2.lives <= 0) {
   309→        g.running = false;
   310→        endGame(p1.score, p2.score);
   311→        return;
   312→      }
   313→
   314→      ctx.fillStyle = "#05080f";
   315→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   316→
   317→      g.stars.forEach((s) => {
   318→        const twinkle = 0.6 + 0.4 * Math.sin(g.frame * 0.03 + s.x);
   319→        const alpha = s.brightness * twinkle;
   320→        ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   321→        ctx.beginPath();
   322→        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
   323→        ctx.fill();
   324→      });
   325→
   326→      ctx.strokeStyle = "rgba(255,255,255,0.06)";
   327→      ctx.setLineDash([4, 8]);
   328→      ctx.beginPath();
   329→      ctx.moveTo(CANVAS_W / 2, 0);
   330→      ctx.lineTo(CANVAS_W / 2, CANVAS_H);
   331→      ctx.stroke();
   332→      ctx.setLineDash([]);
   333→
   334→      g.enemies.forEach((e) => {
   335→        ctx.save();
   336→        ctx.translate(e.x, e.y);
   337→        ctx.rotate(e.rotation);
   338→
   339→        if (e.bonus) {
   340→          const glowSize = e.size + 6 + Math.sin(g.frame * 0.1) * 3;
   341→          const glow = ctx.createRadialGradient(0, 0, e.size * 0.3, 0, 0, glowSize);
   342→          glow.addColorStop(0, "rgba(251, 191, 36, 0.4)");
   343→          glow.addColorStop(1, "rgba(251, 191, 36, 0)");
   344→          ctx.fillStyle = glow;
   345→          ctx.beginPath();
   346→          ctx.arc(0, 0, glowSize, 0, Math.PI * 2);
   347→          ctx.fill();
   348→
   349→          const sparkleAngle = g.frame * 0.15;
   350→          for (let i = 0; i < 4; i++) {
   351→            const a = sparkleAngle + (i * Math.PI) / 2;
   352→            const sx = Math.cos(a) * (e.size + 8);
   353→            const sy = Math.sin(a) * (e.size + 8);
   354→            ctx.fillStyle = "rgba(255, 255, 200, 0.8)";
   355→            ctx.beginPath();
   356→            ctx.arc(sx, sy, 1.5, 0, Math.PI * 2);
   357→            ctx.fill();
   358→          }
   359→        }
   360→
   361→        ctx.fillStyle = e.color;
   362→        ctx.strokeStyle = e.bonus ? "#fef08a" : "#fca5a5";
   363→        ctx.lineWidth = 1.5;
   364→
   365→        if (e.type === "triangle") {
   366→          ctx.beginPath();
   367→          ctx.moveTo(0, -e.size);
   368→          ctx.lineTo(-e.size * 0.8, e.size * 0.7);
   369→          ctx.lineTo(e.size * 0.8, e.size * 0.7);
   370→          ctx.closePath();
   371→          ctx.fill();
   372→          ctx.stroke();
   373→        } else {
   374→          ctx.beginPath();
   375→          ctx.moveTo(0, -e.size);
   376→          ctx.lineTo(e.size * 0.7, 0);
   377→          ctx.lineTo(0, e.size);
   378→          ctx.lineTo(-e.size * 0.7, 0);
   379→          ctx.closePath();
   380→          ctx.fill();
   381→          ctx.stroke();
   382→        }
   383→
   384→        ctx.restore();
   385→      });
   386→
   387→      g.bullets.forEach((b) => {
   388→        const player = b.owner === 1 ? p1 : p2;
   389→        const bGlow = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, 6);
   390→        bGlow.addColorStop(0, player.bulletColor);
   391→        bGlow.addColorStop(1, "rgba(0,0,0,0)");
   392→        ctx.fillStyle = bGlow;
   393→        ctx.beginPath();
   394→        ctx.arc(b.x, b.y, 6, 0, Math.PI * 2);
   395→        ctx.fill();
   396→
   397→        ctx.fillStyle = "#fff";
   398→        ctx.beginPath();
   399→        ctx.arc(b.x, b.y, 2, 0, Math.PI * 2);
   400→        ctx.fill();
   401→
   402→        ctx.fillStyle = player.bulletColor;
   403→        ctx.globalAlpha = 0.3;
   404→        ctx.beginPath();
   405→        ctx.arc(b.x, b.y + 4, 1.5, 0, Math.PI * 2);
   406→        ctx.fill();
   407→        ctx.globalAlpha = 1;
   408→      });
   409→
   410→      const drawShip = (p: PlayerState) => {
   411→        if (p.lives <= 0) return;
   412→        if (p.invincible > 0 && Math.floor(p.invincible / 4) % 2 === 0) return;
   413→
   414→        ctx.save();
   415→        ctx.translate(p.x, p.y);
   416→
   417→        const engineGlow = ctx.createRadialGradient(0, SHIP_SIZE * 0.6, 0, 0, SHIP_SIZE * 0.6, SHIP_SIZE);
   418→        engineGlow.addColorStop(0, p.color + "60");
   419→        engineGlow.addColorStop(1, "rgba(0,0,0,0)");
   420→        ctx.fillStyle = engineGlow;
   421→        ctx.beginPath();
   422→        ctx.arc(0, SHIP_SIZE * 0.6, SHIP_SIZE, 0, Math.PI * 2);
   423→        ctx.fill();
   424→
   425→        const flameH = 8 + Math.sin(g.frame * 0.4) * 4;
   426→        ctx.fillStyle = p.color;
   427→        ctx.globalAlpha = 0.6;
   428→        ctx.beginPath();
   429→        ctx.moveTo(-5, SHIP_SIZE * 0.4);
   430→        ctx.lineTo(0, SHIP_SIZE * 0.4 + flameH);
   431→        ctx.lineTo(5, SHIP_SIZE * 0.4);
   432→        ctx.closePath();
   433→        ctx.fill();
   434→        ctx.globalAlpha = 1;
   435→
   436→        ctx.fillStyle = p.color;
   437→        ctx.strokeStyle = "#fff";
   438→        ctx.lineWidth = 1;
   439→        ctx.beginPath();
   440→        ctx.moveTo(0, -SHIP_SIZE);
   441→        ctx.lineTo(-SHIP_SIZE * 0.8, SHIP_SIZE * 0.5);
   442→        ctx.lineTo(-SHIP_SIZE * 0.3, SHIP_SIZE * 0.3);
   443→        ctx.lineTo(0, SHIP_SIZE * 0.45);
   444→        ctx.lineTo(SHIP_SIZE * 0.3, SHIP_SIZE * 0.3);
   445→        ctx.lineTo(SHIP_SIZE * 0.8, SHIP_SIZE * 0.5);
   446→        ctx.closePath();
   447→        ctx.fill();
   448→        ctx.stroke();
   449→
   450→        ctx.fillStyle = "rgba(255,255,255,0.5)";
   451→        ctx.beginPath();
   452→        ctx.arc(0, -SHIP_SIZE * 0.15, 3, 0, Math.PI * 2);
   453→        ctx.fill();
   454→
   455→        ctx.restore();
   456→      };
   457→
   458→      drawShip(p1);
   459→      drawShip(p2);
   460→
   461→      g.particles.forEach((pt) => {
   462→        const alpha = pt.life / pt.maxLife;
   463→        ctx.globalAlpha = alpha;
   464→        ctx.fillStyle = pt.color;
   465→        ctx.beginPath();
   466→        ctx.arc(pt.x, pt.y, pt.size * alpha, 0, Math.PI * 2);
   467→        ctx.fill();
   468→      });
   469→      ctx.globalAlpha = 1;
   470→
   471→      const drawDangerZone = (leftX: number, rightX: number, color: string) => {
   472→        const grad = ctx.createLinearGradient(leftX, CANVAS_H - 30, leftX, CANVAS_H);
   473→        grad.addColorStop(0, "rgba(0,0,0,0)");
   474→        grad.addColorStop(1, color + "15");
   475→        ctx.fillStyle = grad;
   476→        ctx.fillRect(leftX, CANVAS_H - 30, rightX - leftX, 30);
   477→      };
   478→      drawDangerZone(0, CANVAS_W / 2, "#22d3ee");
   479→      drawDangerZone(CANVAS_W / 2, CANVAS_W, "#f472b6");
   480→
   481→      g.animId = requestAnimationFrame(() => gameLoop(g));
   482→    },
   483→    [endGame, spawnEnemy]
   484→  );
   485→
   486→  const initGame = useCallback(
   487→    (carryScores: { p1: number; p2: number }, startWave: number) => {
   488→      const g: GameState = {
   489→        players: [
   490→          {
   491→            x: CANVAS_W * 0.3,
   492→            y: CANVAS_H - 40,
   493→            score: carryScores.p1,
   494→            lives: 3,
   495→            color: "#22d3ee",
   496→            bulletColor: "#67e8f9",
   497→            shootCooldown: 0,
   498→            invincible: 0,
   499→            name: "Jogador 1",
   500→          },
   501→          {
   502→            x: CANVAS_W * 0.7,
   503→            y: CANVAS_H - 40,
   504→            score: carryScores.p2,
   505→            lives: 3,
   506→            color: "#f472b6",
   507→            bulletColor: "#f9a8d4",
   508→            shootCooldown: 0,
   509→            invincible: 0,
   510→            name: "Jogador 2",
   511→          },
   512→        ],
   513→        bullets: [],
   514→        enemies: [],
   515→        particles: [],
   516→        stars: createStars(),
   517→        keys: new Set<string>(),
   518→        frame: 0,
   519→        wave: startWave,
   520→        waveFrame: 0,
   521→        running: true,
   522→        animId: 0,
   523→      };
   524→      gameRef.current = g;
   525→      setLives({ p1: 3, p2: 3 });
   526→      setWave(startWave);
   527→      setGameStatus("playing");
   528→      return g;
   529→    },
   530→    []
   531→  );
   532→
   533→  const startGame = useCallback(() => {
   534→    if (gameRef.current) cancelAnimationFrame(gameRef.current.animId);
   535→    const g = initGame({ p1: 0, p2: 0 }, 1);
   536→    g.animId = requestAnimationFrame(() => gameLoop(g));
   537→  }, [initGame, gameLoop]);
   538→
   539→  const nextRound = useCallback(() => {
   540→    if (gameRef.current) cancelAnimationFrame(gameRef.current.animId);
