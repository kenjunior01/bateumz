     1→import { useState, useEffect, useCallback, useRef } from "react";
     2→import { motion, AnimatePresence } from "framer-motion";
     3→import { Button } from "@/components/ui/button";

     8→interface Props {
     9→  onScore?: (name: string, score: number) => void;

    12→
    13→type TargetOwner = "p1" | "p2" | "any";
    14→type TargetKind = "normal" | "fast" | "golden" | "bomb";
    15→type GamePhase = "idle" | "playing" | "done";
    16→
    17→interface TargetData {
    18→  id: number;
    19→  kind: TargetKind;
    20→  owner: TargetOwner;
    21→  row: number;
    22→  col: number;
    23→  createdAt: number;
    24→  duration: number;
    25→  points: number;
    26→}
    27→
    28→interface FloatingScore {
    29→  id: number;
    30→  value: number;
    31→  row: number;
    32→  col: number;
    33→  player: 1 | 2;
    34→  isPenalty: boolean;
    35→}
    36→
    37→interface HitRipple {
    38→  id: number;
    39→  row: number;
    40→  col: number;
    41→  color: string;
    42→}
    43→
    44→const GRID_SIZE = 4;
    45→
    46→const TARGET_CONFIGS: Record<TargetKind, { points: number; duration: number; weight: number; size: "normal" | "small" }> = {
    47→  normal: { points: 1, duration: 1500, weight: 55, size: "normal" },
    48→  fast: { points: 2, duration: 800, weight: 15, size: "small" },
    49→  golden: { points: 3, duration: 1000, weight: 10, size: "normal" },
    50→  bomb: { points: -2, duration: 2000, weight: 20, size: "normal" },
    51→};
    52→
    53→function pickTargetKind(): TargetKind {
    54→  const kinds = Object.keys(TARGET_CONFIGS) as TargetKind[];
    55→  const totalWeight = kinds.reduce((s, k) => s + TARGET_CONFIGS[k].weight, 0);
    56→  let r = Math.random() * totalWeight;
    57→  for (const k of kinds) {
    58→    r -= TARGET_CONFIGS[k].weight;
    59→    if (r <= 0) return k;
    60→  }
    61→  return "normal";
    62→}
    63→
    64→function getTargetOwner(kind: TargetKind): TargetOwner {
    65→  if (kind === "golden" || kind === "bomb") return "any";
    66→  return Math.random() < 0.5 ? "p1" : "p2";
    67→}
    68→
    69→let nextTargetId = 0;
    70→let nextFloatingId = 0;
    71→let nextRippleId = 0;

   120→      glow: "shadow-red-500/40",

   135→      text: "text-cyan-400",

   146→
   147→export default function TargetTap({ onScore, liveCode }: Props) {
   148→  const [phase, setPhase] = useState<GamePhase>("idle");
   149→  const [duration, setDuration] = useState(60);
   150→  const [timeLeft, setTimeLeft] = useState(60);
   151→  const [p1Score, setP1Score] = useState(0);
   152→  const [p2Score, setP2Score] = useState(0);
   153→  const [targets, setTargets] = useState<TargetData[]>([]);
   154→  const [floats, setFloats] = useState<FloatingScore[]>([]);
   155→  const [ripples, setRipples] = useState<HitRipple[]>([]);
   156→  const [shakingCell, setShakingCell] = useState<string | null>(null);
   157→  const [currentKindLabel, setCurrentKindLabel] = useState<string>("");
   158→
   159→  const timerRef = useRef<number | null>(null);
   160→  const spawnRef = useRef<number | null>(null);
   161→  const cleanupRef = useRef<number | null>(null);
   162→  const p1ScoreRef = useRef(0);
   163→  const p2ScoreRef = useRef(0);
   164→
   165→  const clearAllTimers = useCallback(() => {
   166→    if (timerRef.current) { clearInterval(timerRef.current); timerRef.current = null; }
   167→    if (spawnRef.current) { clearInterval(spawnRef.current); spawnRef.current = null; }
   168→    if (cleanupRef.current) { clearInterval(cleanupRef.current); cleanupRef.current = null; }
   169→  }, []);
   170→
   171→  const finishGame = useCallback(() => {
   172→    clearAllTimers();
   173→    setPhase("done");
   174→    setTargets([]);
   175→    onScore?.("Jogador 1", p1ScoreRef.current);
   176→    onScore?.("Jogador 2", p2ScoreRef.current);
   177→  }, [onScore, clearAllTimers]);
   178→
   179→  const spawnTarget = useCallback(() => {
   180→    setTargets((prev) => {
   181→      const occupied = new Set(prev.map((t) => `${t.row}-${t.col}`));
   182→      const empty: [number, number][] = [];
   183→      for (let r = 0; r < GRID_SIZE; r++) {
   184→        for (let c = 0; c < GRID_SIZE; c++) {
   185→          if (!occupied.has(`${r}-${c}`)) empty.push([r, c]);
   186→        }
   187→      }
   188→      if (empty.length === 0) return prev;
   189→
   190→      const [row, col] = empty[Math.floor(Math.random() * empty.length)];
   191→      const kind = pickTargetKind();
   192→      const config = TARGET_CONFIGS[kind];
   193→      const owner = getTargetOwner(kind);
   194→
   195→      const labels: Record<TargetKind, string> = {
   196→        normal: "",
   197→        fast: "Alvo Rápido",
   198→        golden: "Bônus",
   199→        bomb: "Bomba",
   200→      };
   201→      if (labels[kind]) setCurrentKindLabel(labels[kind]);
   202→
   203→      const target: TargetData = {
   204→        id: nextTargetId++,
   205→        kind,
   206→        owner,
   207→        row,
   208→        col,
   209→        createdAt: Date.now(),
   210→        duration: config.duration,
   211→        points: config.points,
   212→      };
   213→
   214→      const newTargets = [...prev, target];
   215→      if (newTargets.length > 6) return newTargets.slice(-6);
   216→      return newTargets;
   217→    });
   218→  }, []);
   219→
   220→  const cleanupTargets = useCallback(() => {
   221→    const now = Date.now();
   222→    setTargets((prev) => prev.filter((t) => now - t.createdAt < t.duration));
   223→  }, []);
   224→
   225→  const startGame = useCallback(() => {
   226→    clearAllTimers();
   227→    setP1Score(0);
   228→    setP2Score(0);
   229→    p1ScoreRef.current = 0;
   230→    p2ScoreRef.current = 0;
   231→    setTargets([]);
   232→    setFloats([]);
   233→    setRipples([]);
   234→    setShakingCell(null);
   235→    setCurrentKindLabel("");
   236→    setTimeLeft(duration);
   237→    setPhase("playing");
   238→
   239→    const startTime = Date.now();
   240→    timerRef.current = window.setInterval(() => {
   241→      const elapsed = (Date.now() - startTime) / 1000;
   242→      const remaining = Math.max(0, duration - elapsed);
   243→      setTimeLeft(Math.ceil(remaining));
   244→      if (remaining <= 0) finishGame();
   245→    }, 100);
   246→
   247→    spawnRef.current = window.setInterval(spawnTarget, 700);
   248→    cleanupRef.current = window.setInterval(cleanupTargets, 100);
   249→  }, [duration, clearAllTimers, spawnTarget, cleanupTargets, finishGame]);
   250→
   251→  const restartAll = useCallback(() => {
   252→    clearAllTimers();
   253→    setPhase("idle");
   254→    setP1Score(0);
   255→    setP2Score(0);
   256→    p1ScoreRef.current = 0;
   257→    p2ScoreRef.current = 0;
   258→    setTargets([]);
   259→    setFloats([]);
   260→    setRipples([]);
   261→    setShakingCell(null);
   262→    setCurrentKindLabel("");
   263→    setTimeLeft(duration);
   264→  }, [clearAllTimers, duration]);
   265→
   266→  const nextRound = useCallback(() => {
   267→    clearAllTimers();
   268→    setP1Score(0);
   269→    setP2Score(0);
   270→    p1ScoreRef.current = 0;
   271→    p2ScoreRef.current = 0;
   272→    setTargets([]);
   273→    setFloats([]);
   274→    setRipples([]);
   275→    setShakingCell(null);
   276→    setCurrentKindLabel("");
   277→    setTimeLeft(duration);
   278→    startGame();
   279→  }, [clearAllTimers, duration, startGame]);
   280→
   281→  useEffect(() => {
   282→    return () => clearAllTimers();
   283→  }, [clearAllTimers]);
   284→
   285→  const handleTap = useCallback(
   286→    (row: number, col: number) => {
   287→      if (phase !== "playing") return;
   288→
   289→      setTargets((prev) => {
   290→        const target = prev.find((t) => t.row === row && t.col === col);
   291→        if (!target) return prev;
   292→
   293→        const isP1Target = target.owner === "p1";
   294→        const isP2Target = target.owner === "p2";
   295→        const isAnyTarget = target.owner === "any";
   296→        const isBomb = target.kind === "bomb";
   297→
   298→        let p1Delta = 0;
   299→        let p2Delta = 0;
   300→        let p1Float: FloatingScore | null = null;
   301→        let p2Float: FloatingScore | null = null;
   302→        let isPenalty = false;
   303→
   304→        if (isBomb) {
   305→          // Either player hitting a bomb gets -2. Since both players tap the same board,
   306→          // the bomb penalizes the player who tapped it.
   307→          // We assign penalty to whichever player is more likely the one tapping.
   308→          // For simplicity, the bomb penalizes the player whose color zone it's NOT in.
   309→          // Actually, bombs penalize whoever taps. Since we can't distinguish on a shared board,
   310→          // we penalize both players.
   311→          p1Delta = target.points;
   312→          p2Delta = target.points;
   313→          isPenalty = true;
   314→          p1Float = {
   315→            id: nextFloatingId++,
   316→            value: target.points,
   317→            row, col,
   318→            player: 1,
   319→            isPenalty: true,
   320→          };
   321→          p2Float = {
   322→            id: nextFloatingId++,
   323→            value: target.points,
   324→            row: row + 0.1,
   325→            col: col + 0.1,
   326→            player: 2,
   327→            isPenalty: true,
   328→          };
   329→        } else if (isAnyTarget) {
   330→          // Golden: both players can score. We give points to both.
   331→          p1Delta = target.points;
   332→          p2Delta = target.points;
   333→          p1Float = {
   334→            id: nextFloatingId++,
   335→            value: target.points,
   336→            row, col,
   337→            player: 1,
   338→            isPenalty: false,
   339→          };
   340→          p2Float = {
   341→            id: nextFloatingId++,
   342→            value: target.points,
   343→            row: row + 0.1,
   344→            col: col + 0.1,
   345→            player: 2,
   346→            isPenalty: false,
   347→          };
   348→        } else if (isP1Target) {
   349→          // P1 scores for hitting their own color
   350→          p1Delta = target.points;
   351→          p1Float = {
   352→            id: nextFloatingId++,
   353→            value: target.points,
   354→            row, col,
   355→            player: 1,
   356→            isPenalty: false,
   357→          };
   358→        } else if (isP2Target) {
   359→          // P2 scores for hitting their own color
   360→          p2Delta = target.points;
   361→          p2Float = {
   362→            id: nextFloatingId++,
   363→            value: target.points,
   364→            row, col,
   365→            player: 2,
   366→            isPenalty: false,
   367→          };
   368→        }
   369→
   370→        // Remove the tapped target
   371→        const newTargets = prev.filter((t) => t.id !== target.id);
   372→
   373→        // Apply score changes
   374→        if (p1Delta !== 0) {
   375→          setP1Score((s) => {
   376→            const ns = Math.max(0, s + p1Delta);
   377→            p1ScoreRef.current = ns;
   378→            return ns;
   379→          });
   380→        }
   381→        if (p2Delta !== 0) {
   382→          setP2Score((s) => {
   383→            const ns = Math.max(0, s + p2Delta);
   384→            p2ScoreRef.current = ns;
   385→            return ns;
   386→          });
   387→        }
   388→
   389→        // Add floating scores
   390→        if (p1Float) setFloats((f) => [...f, p1Float!]);
   391→        if (p2Float) setFloats((f) => [...f, p2Float!]);
   392→
   393→        // Add ripple
   394→        const colors = getTargetColors(target.kind, target.owner);
   395→        const rippleColor =
   396→          target.kind === "bomb"
   397→            ? "rgba(248,113,113,0.6)"
   398→            : target.kind === "golden"
   399→            ? "rgba(251,191,36,0.6)"
   400→            : isP1Target
   401→            ? "rgba(34,211,238,0.6)"
   402→            : "rgba(244,114,182,0.6)";
   403→        setRipples((r) => [
   404→          ...r,
   405→          { id: nextRippleId++, row, col, color: rippleColor },
   406→        ]);
   407→
   408→        // Shake on penalty
   409→        if (isPenalty) {
   410→          setShakingCell(`${row}-${col}`);
   411→          setTimeout(() => setShakingCell(null), 400);
   412→        }
   413→
   414→        // Clean up floats
   415→        setTimeout(() => {
   416→          if (p1Float) setFloats((f) => f.filter((x) => x.id !== p1Float!.id));
   417→          if (p2Float) setFloats((f) => f.filter((x) => x.id !== p2Float!.id));
   418→        }, 800);
   419→        setTimeout(() => {
   420→          setRipples((r) => r.filter((x) => x.row !== row || x.col !== col));
   421→        }, 500);
   422→
   423→        return newTargets;
   424→      });
   425→    },
   426→    [phase]
   427→  );
   428→
   429→  const grid: (TargetData | null)[][] = Array.from({ length: GRID_SIZE }, () =>
   430→    Array.from({ length: GRID_SIZE }, () => null)
   431→  );
   432→  for (const t of targets) {
   433→    grid[t.row][t.col] = t;
   434→  }
   435→
   436→  const timerColor =
   437→    timeLeft <= 10 ? "text-red-400" : timeLeft <= 20 ? "text-amber-400" : "text-slate-200";
   438→
   439→  return (
   440→    <div className="flex flex-col items-center gap-4 w-full max-w-md mx-auto select-none">
   441→      <div className="w-full rounded-2xl bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 p-4">
   442→        <div className="flex items-center justify-between">
   443→          <div className="flex flex-col items-center gap-1 min-w-[80px]">
   444→            <span className="text-xs font-medium text-cyan-400 tracking-wider uppercase">
   445→              Jogador 1
   446→            </span>
   447→            <motion.span
   448→              key={p1Score}
   449→              initial={{ scale: 1.3, y: -4 }}
   450→              animate={{ scale: 1, y: 0 }}
   451→              className="text-3xl font-bold text-cyan-300 tabular-nums"
   452→            >
   453→              {p1Score}
   454→            </motion.span>
   455→          </div>
   456→
   457→          <div className="flex flex-col items-center gap-1">
   458→            <AnimatePresence mode="wait">
   459→              {currentKindLabel && phase === "playing" && (
   460→                <motion.div
   461→                  key={currentKindLabel}
   462→                  initial={{ opacity: 0, y: -10 }}
   463→                  animate={{ opacity: 1, y: 0 }}
   464→                  exit={{ opacity: 0, y: 10 }}
   465→                  transition={{ duration: 0.2 }}
   466→                >
   467→                  <Badge
   468→                    variant="outline"
   469→                    className={cn(
   470→                      "text-[10px] sm:text-xs font-bold tracking-widest uppercase",
   471→                      currentKindLabel === "Alvo Rápido" && "border-amber-400/50 text-amber-400",
   472→                      currentKindLabel === "Bônus" && "border-amber-400/50 text-amber-400",
   473→                      currentKindLabel === "Bomba" && "border-red-400/50 text-red-400"
   474→                    )}
   475→                  >
   476→                    <Target className="w-3 h-3 mr-1" />
   477→                    {currentKindLabel}
   478→                  </Badge>
   479→                </motion.div>
   480→              )}
   481→            </AnimatePresence>
   482→
   483→            <motion.div
   484→              className={cn("text-4xl sm:text-5xl font-black tabular-nums", timerColor)}
   485→              animate={timeLeft <= 10 && phase === "playing" ? { scale: [1, 1.1, 1] } : { scale: 1 }}
   486→              transition={{ repeat: Infinity, duration: 0.5 }}
   487→            >
   488→              {timeLeft}s
   489→            </motion.div>
   490→          </div>
   491→
   492→          <div className="flex flex-col items-center gap-1 min-w-[80px]">
   493→            <span className="text-xs font-medium text-pink-400 tracking-wider uppercase">
   494→              Jogador 2
   495→            </span>
   496→            <motion.span
   497→              key={p2Score}
   498→              initial={{ scale: 1.3, y: -4 }}
   499→              animate={{ scale: 1, y: 0 }}
   500→              className="text-3xl font-bold text-pink-300 tabular-nums"
   501→            >
   502→              {p2Score}
   503→            </motion.span>
   504→          </div>
   505→        </div>
   506→      </div>
   507→
   508→      {phase === "idle" && (
   509→        <div className="flex items-center gap-2">
   510→          <Timer className="w-4 h-4 text-slate-400" />
   511→          <span className="text-xs text-slate-400 font-medium">Duração</span>
   512→          <div className="flex gap-1">
   513→            {[30, 60, 90].map((d) => (
   514→              <Button
   515→                key={d}
   516→                size="sm"
   517→                variant={duration === d ? "default" : "outline"}
   518→                className={cn(
   519→                  "h-7 px-3 text-xs",
   520→                  duration === d && "bg-slate-700 hover:bg-slate-600"
   521→                )}
   522→                onClick={() => setDuration(d)}
   523→              >
   524→                {d}s
   525→              </Button>
   526→            ))}
   527→          </div>
   528→        </div>
   529→      )}
   530→
   531→      <div className="relative w-full bg-slate-900/50 border border-slate-800 rounded-2xl p-3">
   532→        <div className="grid grid-cols-4 gap-2 justify-items-center">
   533→          {grid.map((row, r) =>
   534→            row.map((cell, c) => {
   535→              const target = cell;
   536→              const cellKey = `${r}-${c}`;
   537→              const isShaking = shakingCell === cellKey;
   538→              const colors = target ? getTargetColors(target.kind, target.owner) : null;
   539→              const isSmall = target?.kind === "fast";
   540→
   541→              return (
   542→                <div key={cellKey} className="relative">
   543→                  <AnimatePresence>
   544→                    {ripples.some((rp) => rp.row === r && rp.col === c) && (
   545→                      <motion.div
   546→                        key={`ripple-${cellKey}`}
   547→                        initial={{ scale: 0.3, opacity: 1 }}
   548→                        animate={{ scale: 2.5, opacity: 0 }}
   549→                        exit={{ opacity: 0 }}
   550→                        transition={{ duration: 0.4, ease: "easeOut" }}
   551→                        className="absolute inset-0 rounded-xl pointer-events-none"
   552→                        style={{
   553→                          backgroundColor: ripples.find((rp) => rp.row === r && rp.col === c)?.color || "transparent",
   554→                          zIndex: 5,
   555→                        }}
   556→                      />
   557→                    )}
   558→                  </AnimatePresence>
   559→
   560→                  <AnimatePresence>
   561→                    {floats

   788→
   789→function TargetTimerBar({ target, colorClass }: { target: TargetData; colorClass: string }) {
   790→  const [pct, setPct] = useState(100);
   791→  const rafRef = useRef<number>(0);
   792→
   793→  useEffect(() => {
   794→    const tick = () => {
   795→      const elapsed = Date.now() - target.createdAt;
   796→      const remaining = Math.max(0, 1 - elapsed / target.duration);
   797→      setPct(remaining * 100);
   798→      if (remaining > 0) {
   799→        rafRef.current = requestAnimationFrame(tick);
   800→      }
   801→    };
   802→    rafRef.current = requestAnimationFrame(tick);
   803→    return () => cancelAnimationFrame(rafRef.current);
   804→  }, [target.createdAt, target.duration]);
   805→
   806→  return (
   807→    <div className="absolute bottom-1 left-1/2 -translate-x-1/2 w-10 h-0.5 bg-slate-700 rounded-full overflow-hidden">
   808→      <motion.div
   809→        className={cn("h-full rounded-full", colorClass)}
   810→        style={{ width: `${pct}%` }}
   811→        transition={{ duration: 0.05 }}
   812→      />
   813→    </div>
   814→  );
   815→}
