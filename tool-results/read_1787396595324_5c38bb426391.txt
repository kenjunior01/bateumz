   153→const SAVE_KEY = "bateu_rpg_save";

   155→function loadSave(): PlayerState | null {
   156→  try { const r = localStorage.getItem(SAVE_KEY); return r ? JSON.parse(r) : null; } catch { return null; }
   157→}

   159→function persistSave(s: PlayerState) {
   160→  try { localStorage.setItem(SAVE_KEY, JSON.stringify(s)); } catch {}
   161→}

   163→function calcStats(p: PlayerState): Stats {
   164→  const c = CLASSES[p.classId];
   165→  const lvl = p.level;
   166→  let hp = c.baseHp + lvl * 12;
   167→  let maxHp = hp;
   168→  let atk = c.baseAtk + lvl * 2;
   169→  let def = c.baseDef + lvl * 1;
   170→  let spd = c.baseSpd + lvl * 1;
   171→  let mp = c.baseMp + lvl * 5;
   172→  let maxMp = mp;
   173→  for (const eq of p.equipment) {
   174→    if (!eq) continue;
   175→    hp += eq.hp || 0; maxHp += eq.hp || 0;
   176→    atk += eq.atk || 0;
   177→    def += eq.def || 0;
   178→    spd += eq.spd || 0;
   179→    mp += eq.mp || 0; maxMp += eq.mp || 0;
   180→  }
   181→  return { hp, maxHp, atk, def, spd, mp, maxMp };
   182→}

   275→        ctx.fillText(cl.emoji, px, py);

   284→        ctx.fillStyle = "#e5e7eb";
   285→        ctx.font = "bold 14px sans-serif";
   286→        ctx.fillText(`${Math.ceil(stats.hp)}/${stats.maxHp}`, px, py + 36);
   287→
   288→        // Player MP bar
   289→        const mpPct = stats.mp / stats.maxMp;
   290→        ctx.fillStyle = "#1e1b4b";
   291→        ctx.fillRect(px - 50, py + 40, 100, 6);
   292→        ctx.fillStyle = "#6366f1";
   293→        ctx.fillRect(px - 50, py + 40, 100 * mpPct, 6);
   294→

   296→        battleEnemies.forEach((e, i) => {
   297→          if (e.hp <= 0) return;
   298→          const hit = animatingHit === i + 1;
   299→          const ex = cx + 120 + (i > 0 ? i * 60 : 0) + (hit ? (Math.random()-0.5)*16 : 0);
   300→          const ey = cy - 20 + (i > 0 ? i * -30 : 0);

   302→          if (hit) ctx.globalAlpha = 0.4 + Math.random() * 0.6;

   304→          ctx.textAlign = "center";
   305→          ctx.fillText(e.emoji, ex, ey);
   306→          ctx.restore();
   307→
   308→          // Enemy HP bar
   309→          const ehp = e.hp / e.maxHp;
   310→          ctx.fillStyle = "#1f2937";
   311→          ctx.fillRect(ex - 40, ey + 8, 80, 6);
   312→          ctx.fillStyle = e.color || "#ef4444";
   313→          ctx.fillRect(ex - 40, ey + 8, 80 * ehp, 6);
   314→          ctx.fillStyle = "#9ca3af";
   315→          ctx.font = "12px sans-serif";
   316→          ctx.fillText(`${Math.ceil(e.hp)}/${e.maxHp}`, ex, ey + 28);
   317→        });
   318→      }
   319→
   320→      // Draw particles
   321→      particlesRef.current = particlesRef.current.filter(p => {
   322→        p.x += p.vx;
   323→        p.y += p.vy;
   324→        p.vy += 0.15;
   325→        p.life -= 0.02;
   326→        if (p.life <= 0) return false;
