    90→    90→  if (occupied.size >= totalCells) {
    91→    91→    // Board is full — return a fallback position (will cause game over naturally)
    92→    92→    return { x: 0, y: 0 };
    93→    93→  }
    94→    94→  let pt: Point;
    95→    95→  let attempts = 0;
    96→    96→  const maxAttempts = totalCells * 2;
    97→    97→  do {
    98→    98→    pt = {
    99→    99→      x: Math.floor(Math.random() * GRID_SIZE),
   100→   100→      y: Math.floor(Math.random() * GRID_SIZE),
   101→   101→    };
   102→   102→    attempts++;
   103→   103→  } while (occupied.has(`${pt.x},${pt.y}`) && attempts < maxAttempts);
   104→   104→  return pt;

   126→   126→/** BFS from start to target, returning the first step direction (or null). */
   127→   127→function bfsDirection(
   128→   128→  start: Point,
   129→   129→  target: Point,
   130→   130→  obstacles: Set<string>
   131→   131→): Direction | null {
   132→   132→  const queue: { pos: Point; firstDir: Direction | null }[] = [
   133→   133→    { pos: start, firstDir: null },
   134→   134→  ];
   135→   135→  const visited = new Set<string>();
   136→   136→  visited.add(key(start));
   137→   137→
   138→   138→  while (queue.length > 0) {
   139→   139→    const { pos, firstDir } = queue.shift()!;
   140→   140→
   141→   141→    for (const dir of ALL_DIRECTIONS) {
   142→   142→      const delta = DIRECTION_DELTA[dir];
   143→   143→      const next: Point = { x: pos.x + delta.x, y: pos.y + delta.y };
   144→   144→
   145→   145→      if (next.x < 0 || next.x >= GRID_SIZE || next.y < 0 || next.y >= GRID_SIZE) continue;
   146→   146→      if (obstacles.has(key(next))) continue;
   147→   147→      if (visited.has(key(next))) continue;
   148→   148→
   149→   149→      const newFirstDir = firstDir ?? dir;
   150→   150→
   151→   151→      if (pointsEqual(next, target)) {
   152→   152→        return newFirstDir;
   153→   153→      }
   154→   154→
   155→   155→      visited.add(key(next));
   156→   156→      queue.push({ pos: next, firstDir: newFirstDir });
   157→   157→    }
   158→   158→  }
   159→   159→  return null;
   160→   160→}

   162→   162→/** BFS that avoids obstacles but also avoids cells adjacent to the player snake's head (cut-off avoidance). */
   163→   163→function bfsSmartDirection(
   164→   164→  aiSnake: Snake,
   165→   165→  target: Point,
   166→   166→  obstacles: Set<string>,
   167→   167→  playerSnake: Snake
   168→   168→): Direction | null {
   169→   169→  const playerHead = playerSnake.body[0];
   170→   170→  const dangerZone = new Set<string>();
   171→   171→  // Mark cells adjacent to player head as dangerous (avoid getting cut off)
   172→   172→  for (const dir of ALL_DIRECTIONS) {
   173→   173→    const delta = DIRECTION_DELTA[dir];
   174→   174→    dangerZone.add(`${playerHead.x + delta.x},${playerHead.y + delta.y}`);
   175→   175→  }
   176→   176→
   177→   177→  const queue: { pos: Point; firstDir: Direction | null }[] = [
   178→   178→    { pos: aiSnake.body[0], firstDir: null },
   179→   179→  ];
   180→   180→  const visited = new Set<string>();
   181→   181→  visited.add(key(aiSnake.body[0]));
   182→   182→
   183→   183→  while (queue.length > 0) {
   184→   184→    const { pos, firstDir } = queue.shift()!;
   185→   185→
   186→   186→    for (const dir of ALL_DIRECTIONS) {
   187→   187→      const delta = DIRECTION_DELTA[dir];
   188→   188→      const next: Point = { x: pos.x + delta.x, y: pos.y + delta.y };
   189→   189→
   190→   190→      if (next.x < 0 || next.x >= GRID_SIZE || next.y < 0 || next.y >= GRID_SIZE) continue;
   191→   191→      if (obstacles.has(key(next))) continue;
   192→   192→      if (visited.has(key(next))) continue;
   193→   193→
   194→   194→      // For smart difficulty, deprioritize but don't completely block danger zones
   195→   195→      // We handle this by preferring non-dangerous paths through ordering
   196→   196→
   197→   197→      const newFirstDir = firstDir ?? dir;
   198→   198→
   199→   199→      if (pointsEqual(next, target)) {
   200→   200→        return newFirstDir;
   201→   201→      }
   202→   202→
   203→   203→      visited.add(key(next));
   204→   204→
   205→   205→      // Prioritize safe cells (push dangerous ones to back of queue)
   206→   206→      const entry = { pos: next, firstDir: newFirstDir };
   207→   207→      if (dangerZone.has(key(next))) {
   208→   208→        queue.push(entry); // lower priority
   209→   209→      } else {
   210→   210→        queue.unshift(entry); // higher priority
   211→   211→      }
   212→   212→    }
   213→   213→  }
   214→   214→
   215→   215→  // Fallback: try to find any safe direction (survival mode)
   216→   216→  return bfsSurvivalDirection(aiSnake, obstacles);
   217→   217→}

   219→   219→/** Fallback: find any direction that doesn't immediately kill the snake. */
   220→   220→function bfsSurvivalDirection(
   221→   221→  snake: Snake,
   222→   222→  obstacles: Set<string>
   223→   223→): Direction | null {
   224→   224→  const head = snake.body[0];
   225→   225→  for (const dir of ALL_DIRECTIONS) {
   226→   226→    if (dir === OPPOSITE[snake.direction]) continue;
   227→   227→    const delta = DIRECTION_DELTA[dir];
   228→   228→    const next: Point = { x: head.x + delta.x, y: head.y + delta.y };
   229→   229→    if (
   230→   230→      next.x >= 0 && next.x < GRID_SIZE &&
   231→   231→      next.y >= 0 && next.y < GRID_SIZE &&
   232→   232→      !obstacles.has(key(next))
   233→   233→    ) {
   234→   234→      return dir;
   235→   235→    }
   236→   236→  }
   237→   237→  return null;
   238→   238→}

   240→   240→/** Compute the best AI direction based on difficulty. */
   241→   241→function getAIDirection(
   242→   242→  aiSnake: Snake,
   243→   243→  playerSnake: Snake,
   244→   244→  food: Point,
   245→   245→  difficulty: Difficulty,
   246→   246→  tickCount: number
   247→   247→): Direction {
   248→   248→  const config = AI_CONFIG[difficulty];
   249→   249→
   250→   250→  // Reaction delay: AI doesn't react for the first N ticks
   251→   251→  if (tickCount > 0 && tickCount <= config.reactionDelay) {
   252→   252→    return aiSnake.direction;
   253→   253→  }
   254→   254→
   255→   255→  // Build obstacle set (all snake bodies)
   256→   256→  const obstacles = new Set<string>();
   257→   257→  for (const p of aiSnake.body) obstacles.add(key(p));
   258→   258→  for (const p of playerSnake.body) obstacles.add(key(p));
   259→   259→  // Remove tail tip since it will move (unless growing)
   260→   260→  // Simple: keep full body as obstacle for safety
   261→   261→
   262→   262→  // Random chance to make a random move
   263→   263→  if (Math.random() < config.randomChance) {
   264→   264→    const safeDirs = ALL_DIRECTIONS.filter((dir) => {
   265→   265→      if (dir === OPPOSITE[aiSnake.direction]) return false;
   266→   266→      const delta = DIRECTION_DELTA[dir];
   267→   267→      const next: Point = { x: aiSnake.body[0].x + delta.x, y: aiSnake.body[0].y + delta.y };
   268→   268→      return (
   269→   269→        next.x >= 0 && next.x < GRID_SIZE &&
   270→   270→        next.y >= 0 && next.y < GRID_SIZE &&
   271→   271→        !obstacles.has(key(next))
   272→   272→      );
   273→   273→    });
   274→   274→    if (safeDirs.length > 0) {
   275→   275→      return safeDirs[Math.floor(Math.random() * safeDirs.length)];
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  let chosenDir: Direction | null;
   280→   280→
   281→   281→  if (difficulty === "facil") {
   282→   282→    // Easy: simple greedy — just move towards food if possible
   283→   283→    chosenDir = bfsDirection(aiSnake.body[0], food, obstacles);
   284→   284→  } else if (difficulty === "medio") {
   285→   285→    // Medium: BFS to food
   286→   286→    chosenDir = bfsDirection(aiSnake.body[0], food, obstacles);
   287→   287→  } else {
   288→   288→    // Hard: smart BFS with avoidance and cut-off potential
   289→   289→    chosenDir = bfsSmartDirection(aiSnake, food, obstacles, playerSnake);
   290→   290→  }
   291→   291→
   292→   292→  if (chosenDir && chosenDir !== OPPOSITE[aiSnake.direction]) {
   293→   293→    return chosenDir;
   294→   294→  }
   295→   295→
   296→   296→  // Fallback survival
   297→   297→  return bfsSurvivalDirection(aiSnake, obstacles) ?? aiSnake.direction;
   298→   298→}
