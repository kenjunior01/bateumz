   161→   161→    const isBonus = Math.random() < BONUS_CHANCE + g.wave * 0.005;
   162→   162→    const type = Math.random() < 0.5 ? ("triangle" as const) : ("diamond" as const);
   163→   163→
   164→   164→    g.enemies.push({
   165→   165→      x: 30 + Math.random() * (CANVAS_W - 60),
   166→   166→      y: -20,
   167→   167→      speed,
   168→   168→      size: isBonus ? 16 : 12 + Math.random() * 6,
   169→   169→      type,
   170→   170→      color: isBonus ? "#fbbf24" : Math.random() < 0.5 ? "#ef4444" : "#f97316",
   171→   171→      hp: 1,
   172→   172→      bonus: isBonus,
   173→   173→      rotation: 0,
   174→   174→      rotSpeed: (Math.random() - 0.5) * 0.06,
   175→   175→      wobble: Math.random() * Math.PI * 2,
   176→   176→      wobbleSpeed: 0.02 + Math.random() * 0.03,

   179→   179→
   180→   180→  const gameLoop = useCallback(
   181→   181→    (g: GameState) => {
   182→   182→      const canvas = canvasRef.current;
   183→   183→      if (!canvas || !g.running) return;
   184→   184→      const ctx = canvas.getContext("2d");

   188→   188→      g.waveFrame++;

   191→   191→        g.wave++;
   192→   192→        g.waveFrame = 0;
   193→   193→        setWave(g.wave);
   194→   194→      }
   195→   195→
   196→   196→      const spawnInterval = Math.max(MIN_ENEMY_INTERVAL, INITIAL_ENEMY_INTERVAL - (g.wave - 1) * 5);
   197→   197→      if (g.frame % Math.round(spawnInterval) === 0) {
   198→   198→        const count = g.wave >= 5 ? 2 : 1;
   199→   199→        for (let i = 0; i < count; i++) spawnEnemy(g);
   200→   200→      }
   201→   201→
   202→   202→      const keys = new Set(g.keys);
   203→   203→      for (const [k, v] of Object.entries(touchRef.current)) {
   204→   204→        if (v) keys.add(k);
   205→   205→      }
   206→   206→
   207→   207→      const p1 = g.players[0];
   208→   208→      const p2 = g.players[1];
   209→   209→
   210→   210→      if (keys.has("a") || keys.has("A")) p1.x = Math.max(SHIP_SIZE, p1.x - PLAYER_SPEED);
   211→   211→      if (keys.has("d") || keys.has("D")) p1.x = Math.min(CANVAS_W - SHIP_SIZE, p1.x + PLAYER_SPEED);
   212→   212→      if (p1.shootCooldown > 0) p1.shootCooldown--;
   213→   213→      if (
   214→   214→        (keys.has("w") || keys.has("W") || keys.has("p1shoot")) &&
   215→   215→        p1.shootCooldown <= 0 &&
   216→   216→        p1.lives > 0
   217→   217→      ) {
   218→   218→        g.bullets.push({ x: p1.x, y: p1.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 1 });
   219→   219→        p1.shootCooldown = BULLET_COOLDOWN;
   220→   220→      }
   221→   221→      if (p1.invincible > 0) p1.invincible--;
   222→   222→
   223→   223→      if (keys.has("ArrowLeft")) p2.x = Math.max(SHIP_SIZE, p2.x - PLAYER_SPEED);
   224→   224→      if (keys.has("ArrowRight")) p2.x = Math.min(CANVAS_W - SHIP_SIZE, p2.x + PLAYER_SPEED);
   225→   225→      if (p2.shootCooldown > 0) p2.shootCooldown--;
   226→   226→      if (
   227→   227→        (keys.has("ArrowUp") || keys.has("p2shoot")) &&
   228→   228→        p2.shootCooldown <= 0 &&
   229→   229→        p2.lives > 0
   230→   230→      ) {
   231→   231→        g.bullets.push({ x: p2.x, y: p2.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 2 });
   232→   232→        p2.shootCooldown = BULLET_COOLDOWN;
   233→   233→      }
   234→   234→      if (p2.invincible > 0) p2.invincible--;
   235→   235→
   236→   236→      g.bullets = g.bullets.filter((b) => {
   237→   237→        b.y -= b.speed;
   238→   238→        return b.y > -10;
   239→   239→      });
   240→   240→
   241→   241→      g.enemies = g.enemies.filter((e) => {
   242→   242→        e.y += e.speed;
   243→   243→        e.rotation += e.rotSpeed;
   244→   244→        e.wobble += e.wobbleSpeed;
   245→   245→        e.x += Math.sin(e.wobble) * 0.5;
   246→   246→        e.x = Math.max(e.size, Math.min(CANVAS_W - e.size, e.x));
   247→   247→
   248→   248→        if (e.y > CANVAS_H + 10) {
   249→   249→          const midX = CANVAS_W / 2;
   250→   250→          if (e.x < midX) {
   251→   251→            if (p1.lives > 0 && p1.invincible <= 0) {
   252→   252→              p1.lives--;
   253→   253→              p1.invincible = 60;
   254→   254→              setLives({ p1: p1.lives, p2: p2.lives });
   255→   255→            }
   256→   256→          } else {
   257→   257→            if (p2.lives > 0 && p2.invincible <= 0) {
   258→   258→              p2.lives--;
   259→   259→              p2.invincible = 60;
   260→   260→              setLives({ p1: p1.lives, p2: p2.lives });
   261→   261→            }
   262→   262→          }
   263→   263→          return false;
   264→   264→        }
   265→   265→        return true;
   266→   266→      });
   267→   267→
   268→   268→      const bulletsToRemove = new Set<number>();
   269→   269→      const enemiesToRemove = new Set<number>();
   270→   270→      let scoreChanged = false;
   271→   271→
   272→   272→      for (let bi = 0; bi < g.bullets.length; bi++) {
   273→   273→        const b = g.bullets[bi];
   274→   274→        for (let ei = 0; ei < g.enemies.length; ei++) {
   275→   275→          if (enemiesToRemove.has(ei)) continue;
   276→   276→          const e = g.enemies[ei];
   277→   277→          const dx = b.x - e.x;
   278→   278→          const dy = b.y - e.y;
   279→   279→          const dist = Math.sqrt(dx * dx + dy * dy);
   280→   280→          if (dist < e.size + 4) {
   281→   281→            e.hp--;
   282→   282→            bulletsToRemove.add(bi);
   283→   283→            if (e.hp <= 0) {
   284→   284→              enemiesToRemove.add(ei);
   285→   285→              const isBonus = e.bonus;
   286→   286→              const pts = isBonus ? POINTS_BONUS : POINTS_NORMAL;
   287→   287→              const streakIdx = b.owner === 1 ? 0 : 1;
   288→   288→              g.killStreaks[streakIdx]++;
   289→   289→              const streak = g.killStreaks[streakIdx];
   290→   290→              const streakBonus = streak >= 10 ? 3 : streak >= 5 ? 2 : 0;
   291→   291→              const totalPts = pts + streakBonus * 5;
   292→   292→              if (b.owner === 1) p1.score += totalPts;
   293→   293→              else p2.score += totalPts;
   294→   294→              g.particles.push(...createExplosion(e.x, e.y, e.color, isBonus));
   295→   295→              g.floatingTexts.push(createScoreText(e.x, e.y - 10, totalPts, isBonus));
   296→   296→              if (isBonus) g.screenFlash = 8;
   297→   297→              scoreChanged = true;
   298→   298→            }
   299→   299→            break;
   300→   300→          }
   301→   301→        }
   302→   302→      }
   303→   303→
   304→   304→      if (scoreChanged) {
   305→   305→        setScores({ p1: p1.score, p2: p2.score });
   306→   306→      }
   307→   307→
   308→   308→      g.bullets = g.bullets.filter((_, i) => !bulletsToRemove.has(i));
   309→   309→      g.enemies = g.enemies.filter((_, i) => !enemiesToRemove.has(i));
   310→   310→
   311→   311→      g.particles = g.particles.filter((p) => {
   312→   312→        p.x += p.vx;
   313→   313→        p.y += p.vy;
   314→   314→        p.vy += 0.05;
   315→   315→        p.vx *= 0.98;
   316→   316→        p.life--;
   317→   317→        return p.life > 0;
   318→   318→      });
   319→   319→
   320→   320→      g.floatingTexts = g.floatingTexts.filter((ft) => {
   321→   321→        ft.y += ft.vy;
   322→   322→        ft.life--;
   323→   323→        return ft.life > 0;
   324→   324→      });
   325→   325→
   326→   326→      if (g.screenFlash > 0) g.screenFlash--;
   327→   327→
   328→   328→      g.stars.forEach((s) => {
   329→   329→        s.y += s.speed;
   330→   330→        if (s.y > CANVAS_H) {
   331→   331→          s.y = 0;
   332→   332→          s.x = Math.random() * CANVAS_W;
   333→   333→        }
   334→   334→      });
   335→   335→
   336→   336→      if (p1.lives <= 0 && p2.lives <= 0) {
   337→   337→        g.running = false;
   338→   338→        endGame(p1.score, p2.score);
   339→   339→        return;
   340→   340→      }
   341→   341→
   342→   342→      ctx.fillStyle = "#05080f";
   343→   343→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   344→   344→
   345→   345→      // Screen flash
   346→   346→      if (g.screenFlash > 0) {
   347→   347→        ctx.fillStyle = `rgba(251, 191, 36, ${g.screenFlash * 0.03})`;
   348→   348→        ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   349→   349→      }
   350→   350→
   351→   351→      g.stars.forEach((s) => {
   352→   352→        const twinkle = 0.6 + 0.4 * Math.sin(g.frame * 0.03 + s.x);
   353→   353→        const alpha = s.brightness * twinkle;
   354→   354→        ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   355→   355→        ctx.beginPath();
   356→   356→        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
   357→   357→        ctx.fill();
   358→   358→      });
   359→   359→
   360→   360→      ctx.strokeStyle = "rgba(255,255,255,0.06)";
   361→   361→      ctx.setLineDash([4, 8]);
   362→   362→      ctx.beginPath();
   363→   363→      ctx.moveTo(CANVAS_W / 2, 0);
   364→   364→      ctx.lineTo(CANVAS_W / 2, CANVAS_H);
   365→   365→      ctx.stroke();
   366→   366→      ctx.setLineDash([]);

   368→   368→      g.enemies.forEach((e) => {
   369→   369→        ctx.save();
   370→   370→        ctx.translate(e.x, e.y);
   371→   371→        ctx.rotate(e.rotation);

   373→   373→        if (e.bonus) {
   374→   374→          const glowSize = e.size + 6 + Math.sin(g.frame * 0.1) * 3;
   375→   375→          const glow = ctx.createRadialGradient(0, 0, e.size * 0.3, 0, 0, glowSize);
   376→   376→          glow.addColorStop(0, "rgba(251, 191, 36, 0.4)");
   377→   377→          glow.addColorStop(1, "rgba(251, 191, 36, 0)");
   378→   378→          ctx.fillStyle = glow;
   379→   379→          ctx.beginPath();
   380→   380→          ctx.arc(0, 0, glowSize, 0, Math.PI * 2);
   381→   381→          ctx.fill();
   382→   382→
   383→   383→          const sparkleAngle = g.frame * 0.15;
   384→   384→          for (let i = 0; i < 4; i++) {
   385→   385→            const a = sparkleAngle + (i * Math.PI) / 2;
   386→   386→            const sx = Math.cos(a) * (e.size + 8);
   387→   387→            const sy = Math.sin(a) * (e.size + 8);
   388→   388→            ctx.fillStyle = "rgba(255, 255, 200, 0.8)";
   389→   389→            ctx.beginPath();
   390→   390→            ctx.arc(sx, sy, 1.5, 0, Math.PI * 2);

   393→   393→        }
   394→   394→
   395→   395→        ctx.fillStyle = e.color;
   396→   396→        ctx.strokeStyle = e.bonus ? "#fef08a" : "#fca5a5";
   397→   397→        ctx.lineWidth = 1.5;
   398→   398→
   399→   399→        if (e.type === "triangle") {
   400→   400→          ctx.beginPath();
   401→   401→          ctx.moveTo(0, -e.size);
   402→   402→          ctx.lineTo(-e.size * 0.8, e.size * 0.7);

   406→   406→          ctx.stroke();
   407→   407→        } else {
   408→   408→          ctx.beginPath();
   409→   409→          ctx.moveTo(0, -e.size);
   410→   410→          ctx.lineTo(e.size * 0.7, 0);
   411→   411→          ctx.lineTo(0, e.size);
   412→   412→          ctx.lineTo(-e.size * 0.7, 0);
   413→   413→          ctx.closePath();
   414→   414→          ctx.fill();
   415→   415→          ctx.stroke();
   416→   416→        }
   417→   417→
   418→   418→        ctx.restore();
   419→   419→      });
   420→   420→
   421→   421→      g.bullets.forEach((b) => {
   422→   422→        const player = b.owner === 1 ? p1 : p2;
   423→   423→        const bGlow = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, 6);
   424→   424→        bGlow.addColorStop(0, player.bulletColor);

   426→   426→        ctx.fillStyle = bGlow;
   427→   427→        ctx.beginPath();
   428→   428→        ctx.arc(b.x, b.y, 6, 0, Math.PI * 2);
   429→   429→        ctx.fill();
   430→   430→
   431→   431→        ctx.fillStyle = "#fff";
   432→   432→        ctx.beginPath();
   433→   433→        ctx.arc(b.x, b.y, 2, 0, Math.PI * 2);
   434→   434→        ctx.fill();

   436→   436→        ctx.fillStyle = player.bulletColor;
   437→   437→        ctx.globalAlpha = 0.3;
   438→   438→        ctx.beginPath();
   439→   439→        ctx.arc(b.x, b.y + 4, 1.5, 0, Math.PI * 2);
   440→   440→        ctx.fill();
   441→   441→        ctx.globalAlpha = 1;
