    76→const MAX_ENEMY_SPEED = 4.5;
    77→const BONUS_CHANCE = 0.08;
    78→const WAVE_DURATION = 900;
    79→const POINTS_NORMAL = 10;
    80→const POINTS_BONUS = 30;
    81→
    82→const createStars = (): Star[] =>
    83→  Array.from({ length: STAR_COUNT }, () => ({
    84→    x: Math.random() * CANVAS_W,
    85→    y: Math.random() * CANVAS_H,
    86→    speed: 0.2 + Math.random() * 1.2,
    87→    size: 0.5 + Math.random() * 1.8,
    88→    brightness: 0.3 + Math.random() * 0.7,
    89→  }));
    90→
    91→interface FloatingText {
    92→  x: number; y: number; text: string; color: string; life: number; maxLife: number; vy: number; }
    93→
    94→const createExplosion = (x: number, y: number, color: string, isBonus: boolean = false): Particle[] => {
    95→  const count = isBonus ? 30 : 14;
    96→  return Array.from({ length: count }, () => {
    97→    const angle = Math.random() * Math.PI * 2;
    98→    const speed = (isBonus ? 2 : 1) + Math.random() * (isBonus ? 5 : 3.5);
    99→    return {
   100→      x, y,
   101→      vx: Math.cos(angle) * speed,
   102→      vy: Math.sin(angle) * speed,
   103→      life: (isBonus ? 30 : 20) + Math.random() * 20,
   104→      maxLife: isBonus ? 50 : 40,
   105→      color: isBonus ? (Math.random() < 0.5 ? '#fbbf24' : '#fef08a') : color,
   106→      size: (isBonus ? 2 : 1.5) + Math.random() * (isBonus ? 4 : 2.5),
   107→    };
   108→  });
   109→};
   110→
   111→const createScoreText = (x: number, y: number, points: number, isBonus: boolean): FloatingText => ({
   112→  x, y, text: `+${points}`, color: isBonus ? '#fbbf24' : '#ffffff', life: 40, maxLife: 40, vy: -1.5,
   113→});
   114→
   115→interface GameState {
   116→  players: PlayerState[];
   117→  bullets: Bullet[];
   118→  enemies: Enemy[];
   119→  particles: Particle[];
   120→  floatingTexts: FloatingText[];
   121→  stars: Star[];
   122→  keys: Set<string>;
   123→  frame: number;
   124→  wave: number;
   125→  waveFrame: number;
   126→  running: boolean;
   127→  screenFlash: number;
   128→  killStreaks: [number, number];
   129→  animId: number;
   130→}
   131→
   132→const SpaceShooter = ({ onScore, liveCode }: Props) => {
   133→  const canvasRef = useRef<HTMLCanvasElement>(null);
   134→  const gameRef = useRef<GameState | null>(null);
   135→  const touchRef = useRef<Record<string, boolean>>({});
   136→
   137→  const [gameStatus, setGameStatus] = useState<"idle" | "playing" | "over">("idle");
   138→  const [scores, setScores] = useState({ p1: 0, p2: 0 });
   139→  const [lives, setLives] = useState({ p1: 3, p2: 3 });
   140→  const [wave, setWave] = useState(1);
   141→  const [winner, setWinner] = useState<string | null>(null);
   142→  const [forceTick, setForceTick] = useState(0);
   143→
   144→  const triggerUI = () => setForceTick((t) => t + 1);
   145→
   146→  const endGame = useCallback(
   147→    (p1Score: number, p2Score: number) => {
   148→      setGameStatus("over");
   149→      setScores({ p1: p1Score, p2: p2Score });
   150→      const w = p1Score > p2Score ? "Jogador 1" : p2Score > p1Score ? "Jogador 2" : null;
   151→      setWinner(w);
   152→      onScore?.("Jogador 1", p1Score);
   153→      onScore?.("Jogador 2", p2Score);
   154→    },
   155→    [onScore]
   156→  );
   157→
   158→  const spawnEnemy = useCallback((g: GameState) => {
   159→    const waveMultiplier = 1 + (g.wave - 1) * 0.15;
   160→    const speed = Math.min(INITIAL_ENEMY_SPEED * waveMultiplier + Math.random() * 0.8, MAX_ENEMY_SPEED);
   161→    const isBonus = Math.random() < BONUS_CHANCE + g.wave * 0.005;
   162→    const type = Math.random() < 0.5 ? ("triangle" as const) : ("diamond" as const);
   163→
   164→    g.enemies.push({
   165→      x: 30 + Math.random() * (CANVAS_W - 60),
   166→      y: -20,
   167→      speed,
   168→      size: isBonus ? 16 : 12 + Math.random() * 6,
   169→      type,
   170→      color: isBonus ? "#fbbf24" : Math.random() < 0.5 ? "#ef4444" : "#f97316",
   171→      hp: 1,
   172→      bonus: isBonus,
   173→      rotation: 0,
   174→      rotSpeed: (Math.random() - 0.5) * 0.06,
   175→      wobble: Math.random() * Math.PI * 2,
   176→      wobbleSpeed: 0.02 + Math.random() * 0.03,
   177→    });
   178→  }, []);
   179→
   180→  const gameLoop = useCallback(
   181→    (g: GameState) => {
   182→      const canvas = canvasRef.current;
   183→      if (!canvas || !g.running) return;
   184→      const ctx = canvas.getContext("2d");
   185→      if (!ctx) return;
   186→
   187→      g.frame++;
   188→      g.waveFrame++;
   189→
   190→      if (g.waveFrame >= WAVE_DURATION) {
   191→        g.wave++;
   192→        g.waveFrame = 0;
   193→        setWave(g.wave);
   194→      }
   195→
   196→      const spawnInterval = Math.max(MIN_ENEMY_INTERVAL, INITIAL_ENEMY_INTERVAL - (g.wave - 1) * 5);
   197→      if (g.frame % Math.round(spawnInterval) === 0) {
   198→        const count = g.wave >= 5 ? 2 : 1;
   199→        for (let i = 0; i < count; i++) spawnEnemy(g);
   200→      }
   201→
   202→      const keys = new Set(g.keys);
   203→      for (const [k, v] of Object.entries(touchRef.current)) {
   204→        if (v) keys.add(k);
   205→      }
   206→
   207→      const p1 = g.players[0];
   208→      const p2 = g.players[1];
   209→
   210→      if (keys.has("a") || keys.has("A")) p1.x = Math.max(SHIP_SIZE, p1.x - PLAYER_SPEED);
   211→      if (keys.has("d") || keys.has("D")) p1.x = Math.min(CANVAS_W - SHIP_SIZE, p1.x + PLAYER_SPEED);
   212→      if (p1.shootCooldown > 0) p1.shootCooldown--;
   213→      if (
   214→        (keys.has("w") || keys.has("W") || keys.has("p1shoot")) &&
   215→        p1.shootCooldown <= 0 &&
   216→        p1.lives > 0
   217→      ) {
   218→        g.bullets.push({ x: p1.x, y: p1.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 1 });
   219→        p1.shootCooldown = BULLET_COOLDOWN;
   220→      }
   221→      if (p1.invincible > 0) p1.invincible--;
   222→
   223→      if (keys.has("ArrowLeft")) p2.x = Math.max(SHIP_SIZE, p2.x - PLAYER_SPEED);
   224→      if (keys.has("ArrowRight")) p2.x = Math.min(CANVAS_W - SHIP_SIZE, p2.x + PLAYER_SPEED);
   225→      if (p2.shootCooldown > 0) p2.shootCooldown--;
   226→      if (
   227→        (keys.has("ArrowUp") || keys.has("p2shoot")) &&
   228→        p2.shootCooldown <= 0 &&
   229→        p2.lives > 0
   230→      ) {
   231→        g.bullets.push({ x: p2.x, y: p2.y - SHIP_SIZE, speed: BULLET_SPEED, owner: 2 });
   232→        p2.shootCooldown = BULLET_COOLDOWN;
   233→      }
   234→      if (p2.invincible > 0) p2.invincible--;
   235→
   236→      g.bullets = g.bullets.filter((b) => {
   237→        b.y -= b.speed;
   238→        return b.y > -10;
   239→      });
   240→
   241→      g.enemies = g.enemies.filter((e) => {
   242→        e.y += e.speed;
   243→        e.rotation += e.rotSpeed;
   244→        e.wobble += e.wobbleSpeed;
   245→        e.x += Math.sin(e.wobble) * 0.5;
   246→        e.x = Math.max(e.size, Math.min(CANVAS_W - e.size, e.x));
   247→
   248→        if (e.y > CANVAS_H + 10) {
   249→          const midX = CANVAS_W / 2;
   250→          if (e.x < midX) {
   251→            if (p1.lives > 0 && p1.invincible <= 0) {
   252→              p1.lives--;
   253→              p1.invincible = 60;
   254→              setLives({ p1: p1.lives, p2: p2.lives });
   255→            }
   256→          } else {
   257→            if (p2.lives > 0 && p2.invincible <= 0) {
   258→              p2.lives--;
   259→              p2.invincible = 60;
   260→              setLives({ p1: p1.lives, p2: p2.lives });
   261→            }
   262→          }
   263→          return false;
   264→        }
   265→        return true;
   266→      });
   267→
   268→      const bulletsToRemove = new Set<number>();
   269→      const enemiesToRemove = new Set<number>();
   270→      let scoreChanged = false;
   271→
   272→      for (let bi = 0; bi < g.bullets.length; bi++) {
   273→        const b = g.bullets[bi];
   274→        for (let ei = 0; ei < g.enemies.length; ei++) {
   275→          if (enemiesToRemove.has(ei)) continue;
   276→          const e = g.enemies[ei];
   277→          const dx = b.x - e.x;
   278→          const dy = b.y - e.y;
   279→          const dist = Math.sqrt(dx * dx + dy * dy);
   280→          if (dist < e.size + 4) {
   281→            e.hp--;
   282→            bulletsToRemove.add(bi);
   283→            if (e.hp <= 0) {
   284→              enemiesToRemove.add(ei);
   285→              const isBonus = e.bonus;
   286→              const pts = isBonus ? POINTS_BONUS : POINTS_NORMAL;
   287→              const streakIdx = b.owner === 1 ? 0 : 1;
   288→              g.killStreaks[streakIdx]++;
   289→              const streak = g.killStreaks[streakIdx];
   290→              const streakBonus = streak >= 10 ? 3 : streak >= 5 ? 2 : 0;
   291→              const totalPts = pts + streakBonus * 5;
   292→              if (b.owner === 1) p1.score += totalPts;
   293→              else p2.score += totalPts;
   294→              g.particles.push(...createExplosion(e.x, e.y, e.color, isBonus));
   295→              g.floatingTexts.push(createScoreText(e.x, e.y - 10, totalPts, isBonus));
   296→              if (isBonus) g.screenFlash = 8;
   297→              scoreChanged = true;
   298→            }
   299→            break;
   300→          }
   301→        }
   302→      }
   303→
   304→      if (scoreChanged) {
   305→        setScores({ p1: p1.score, p2: p2.score });
   306→      }
   307→
   308→      g.bullets = g.bullets.filter((_, i) => !bulletsToRemove.has(i));
   309→      g.enemies = g.enemies.filter((_, i) => !enemiesToRemove.has(i));
   310→
   311→      g.particles = g.particles.filter((p) => {
   312→        p.x += p.vx;
   313→        p.y += p.vy;
   314→        p.vy += 0.05;
   315→        p.vx *= 0.98;
   316→        p.life--;
   317→        return p.life > 0;
   318→      });
   319→
   320→      g.floatingTexts = g.floatingTexts.filter((ft) => {
   321→        ft.y += ft.vy;
   322→        ft.life--;
   323→        return ft.life > 0;
   324→      });
   325→
   326→      if (g.screenFlash > 0) g.screenFlash--;
   327→
   328→      g.stars.forEach((s) => {
   329→        s.y += s.speed;
   330→        if (s.y > CANVAS_H) {
   331→          s.y = 0;
   332→          s.x = Math.random() * CANVAS_W;
   333→        }
   334→      });
   335→
   336→      if (p1.lives <= 0 && p2.lives <= 0) {
   337→        g.running = false;
   338→        endGame(p1.score, p2.score);
   339→        return;
   340→      }
   341→
   342→      ctx.fillStyle = "#05080f";
   343→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   344→
   345→      // Screen flash
   346→      if (g.screenFlash > 0) {
   347→        ctx.fillStyle = `rgba(251, 191, 36, ${g.screenFlash * 0.03})`;
   348→        ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   349→      }
   350→
   351→      g.stars.forEach((s) => {
   352→        const twinkle = 0.6 + 0.4 * Math.sin(g.frame * 0.03 + s.x);
   353→        const alpha = s.brightness * twinkle;
   354→        ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   355→        ctx.beginPath();
   356→        ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
   357→        ctx.fill();
   358→      });
   359→
   360→      ctx.strokeStyle = "rgba(255,255,255,0.06)";
   361→      ctx.setLineDash([4, 8]);
   362→      ctx.beginPath();
   363→      ctx.moveTo(CANVAS_W / 2, 0);
   364→      ctx.lineTo(CANVAS_W / 2, CANVAS_H);
   365→      ctx.stroke();
   366→      ctx.setLineDash([]);
   367→
   368→      g.enemies.forEach((e) => {
   369→        ctx.save();
   370→        ctx.translate(e.x, e.y);
   371→        ctx.rotate(e.rotation);
   372→
   373→        if (e.bonus) {
   374→          const glowSize = e.size + 6 + Math.sin(g.frame * 0.1) * 3;
   375→          const glow = ctx.createRadialGradient(0, 0, e.size * 0.3, 0, 0, glowSize);
   376→          glow.addColorStop(0, "rgba(251, 191, 36, 0.4)");
   377→          glow.addColorStop(1, "rgba(251, 191, 36, 0)");
   378→          ctx.fillStyle = glow;
   379→          ctx.beginPath();
   380→          ctx.arc(0, 0, glowSize, 0, Math.PI * 2);
   381→          ctx.fill();
   382→
   383→          const sparkleAngle = g.frame * 0.15;
   384→          for (let i = 0; i < 4; i++) {
   385→            const a = sparkleAngle + (i * Math.PI) / 2;
   386→            const sx = Math.cos(a) * (e.size + 8);
   387→            const sy = Math.sin(a) * (e.size + 8);
   388→            ctx.fillStyle = "rgba(255, 255, 200, 0.8)";
   389→            ctx.beginPath();
   390→            ctx.arc(sx, sy, 1.5, 0, Math.PI * 2);
   391→            ctx.fill();
   392→          }
   393→        }
   394→
   395→        ctx.fillStyle = e.color;
   396→        ctx.strokeStyle = e.bonus ? "#fef08a" : "#fca5a5";
   397→        ctx.lineWidth = 1.5;
   398→
   399→        if (e.type === "triangle") {
   400→          ctx.beginPath();
   401→          ctx.moveTo(0, -e.size);
   402→          ctx.lineTo(-e.size * 0.8, e.size * 0.7);
   403→          ctx.lineTo(e.size * 0.8, e.size * 0.7);
   404→          ctx.closePath();
   405→          ctx.fill();
   406→          ctx.stroke();
   407→        } else {
   408→          ctx.beginPath();
   409→          ctx.moveTo(0, -e.size);
   410→          ctx.lineTo(e.size * 0.7, 0);
   411→          ctx.lineTo(0, e.size);
   412→          ctx.lineTo(-e.size * 0.7, 0);
   413→          ctx.closePath();
   414→          ctx.fill();
   415→          ctx.stroke();
   416→        }
   417→
   418→        ctx.restore();
   419→      });
   420→
   421→      g.bullets.forEach((b) => {
   422→        const player = b.owner === 1 ? p1 : p2;
   423→        const bGlow = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, 6);
   424→        bGlow.addColorStop(0, player.bulletColor);
   425→        bGlow.addColorStop(1, "rgba(0,0,0,0)");
   426→        ctx.fillStyle = bGlow;
   427→        ctx.beginPath();
   428→        ctx.arc(b.x, b.y, 6, 0, Math.PI * 2);
   429→        ctx.fill();
   430→
   431→        ctx.fillStyle = "#fff";
   432→        ctx.beginPath();
   433→        ctx.arc(b.x, b.y, 2, 0, Math.PI * 2);
   434→        ctx.fill();
   435→
   436→        ctx.fillStyle = player.bulletColor;
   437→        ctx.globalAlpha = 0.3;
   438→        ctx.beginPath();
   439→        ctx.arc(b.x, b.y + 4, 1.5, 0, Math.PI * 2);
   440→        ctx.fill();
   441→        ctx.globalAlpha = 1;
   442→      });
   443→
   444→      const drawShip = (p: PlayerState) => {
   445→        if (p.lives <= 0) return;
   446→        if (p.invincible > 0 && Math.floor(p.invincible / 4) % 2 === 0) return;
   447→
   448→        ctx.save();
   449→        ctx.translate(p.x, p.y);
   450→
   451→        const engineGlow = ctx.createRadialGradient(0, SHIP_SIZE * 0.6, 0, 0, SHIP_SIZE * 0.6, SHIP_SIZE);
   452→        engineGlow.addColorStop(0, p.color + "60");
   453→        engineGlow.addColorStop(1, "rgba(0,0,0,0)");
   454→        ctx.fillStyle = engineGlow;
   455→        ctx.beginPath();
   456→        ctx.arc(0, SHIP_SIZE * 0.6, SHIP_SIZE, 0, Math.PI * 2);
   457→        ctx.fill();
   458→
   459→        const flameH = 8 + Math.sin(g.frame * 0.4) * 4;
   460→        ctx.fillStyle = p.color;
   461→        ctx.globalAlpha = 0.6;
   462→        ctx.beginPath();
   463→        ctx.moveTo(-5, SHIP_SIZE * 0.4);
   464→        ctx.lineTo(0, SHIP_SIZE * 0.4 + flameH);
   465→        ctx.lineTo(5, SHIP_SIZE * 0.4);
   466→        ctx.closePath();
   467→        ctx.fill();
   468→        ctx.globalAlpha = 1;
   469→
   470→        ctx.fillStyle = p.color;
   471→        ctx.strokeStyle = "#fff";
   472→        ctx.lineWidth = 1;
   473→        ctx.beginPath();
   474→        ctx.moveTo(0, -SHIP_SIZE);
   475→        ctx.lineTo(-SHIP_SIZE * 0.8, SHIP_SIZE * 0.5);
   476→        ctx.lineTo(-SHIP_SIZE * 0.3, SHIP_SIZE * 0.3);
   477→        ctx.lineTo(0, SHIP_SIZE * 0.45);
   478→        ctx.lineTo(SHIP_SIZE * 0.3, SHIP_SIZE * 0.3);
   479→        ctx.lineTo(SHIP_SIZE * 0.8, SHIP_SIZE * 0.5);
   480→        ctx.closePath();
   481→        ctx.fill();
   482→        ctx.stroke();
   483→
   484→        ctx.fillStyle = "rgba(255,255,255,0.5)";
   485→        ctx.beginPath();
   486→        ctx.arc(0, -SHIP_SIZE * 0.15, 3, 0, Math.PI * 2);
   487→        ctx.fill();
   488→
   489→        ctx.restore();
   490→      };
   491→
   492→      drawShip(p1);
   493→      drawShip(p2);
   494→
   495→      g.particles.forEach((pt) => {
   496→        const alpha = pt.life / pt.maxLife;
   497→        ctx.globalAlpha = alpha;
   498→        ctx.fillStyle = pt.color;
   499→        ctx.beginPath();
   500→        ctx.arc(pt.x, pt.y, pt.size * alpha, 0, Math.PI * 2);
   501→        ctx.fill();
   502→        // Glow for larger particles
   503→        if (pt.size > 3) {
   504→          ctx.globalAlpha = alpha * 0.3;
   505→          ctx.beginPath();
   506→          ctx.arc(pt.x, pt.y, pt.size * alpha * 2, 0, Math.PI * 2);
   507→          ctx.fill();
   508→        }
   509→      });
   510→      ctx.globalAlpha = 1;
   511→
   512→      // Floating score texts
   513→      g.floatingTexts.forEach((ft) => {
   514→        const alpha = ft.life / ft.maxLife;
   515→        const scale = 0.8 + (1 - alpha) * 0.5;
   516→        ctx.save();
   517→        ctx.globalAlpha = alpha;
   518→        ctx.font = `bold ${Math.round(14 * scale)}px sans-serif`;
   519→        ctx.textAlign = 'center';
   520→        ctx.strokeStyle = 'rgba(0,0,0,0.8)';
   521→        ctx.lineWidth = 3;
   522→        ctx.strokeText(ft.text, ft.x, ft.y);
   523→        ctx.fillStyle = ft.color;
   524→        ctx.fillText(ft.text, ft.x, ft.y);
   525→        ctx.restore();
   526→      });
   527→
   528→      const drawDangerZone = (leftX: number, rightX: number, color: string) => {
   529→        const grad = ctx.createLinearGradient(leftX, CANVAS_H - 30, leftX, CANVAS_H);
   530→        grad.addColorStop(0, "rgba(0,0,0,0)");
   531→        grad.addColorStop(1, color + "15");
   532→        ctx.fillStyle = grad;
   533→        ctx.fillRect(leftX, CANVAS_H - 30, rightX - leftX, 30);
   534→      };
   535→      drawDangerZone(0, CANVAS_W / 2, "#22d3ee");
   536→      drawDangerZone(CANVAS_W / 2, CANVAS_W, "#f472b6");
   537→
   538→      g.animId = requestAnimationFrame(() => gameLoop(g));
   539→    },
   540→    [endGame, spawnEnemy]
   541→  );
   542→
   543→  const initGame = useCallback(
   544→    (carryScores: { p1: number; p2: number }, startWave: number) => {
   545→      const g: GameState = {
   546→        players: [
   547→          {
   548→            x: CANVAS_W * 0.3,
   549→            y: CANVAS_H - 40,
   550→            score: carryScores.p1,
   551→            lives: 3,
   552→            color: "#22d3ee",
   553→            bulletColor: "#67e8f9",
   554→            shootCooldown: 0,
   555→            invincible: 0,
   556→            name: "Jogador 1",
   557→          },
   558→          {
   559→            x: CANVAS_W * 0.7,
   560→            y: CANVAS_H - 40,
   561→            score: carryScores.p2,
   562→            lives: 3,
   563→            color: "#f472b6",
   564→            bulletColor: "#f9a8d4",
   565→            shootCooldown: 0,
   566→            invincible: 0,
   567→            name: "Jogador 2",
   568→          },
   569→        ],
   570→        bullets: [],
   571→        enemies: [],
   572→        particles: [],
   573→        floatingTexts: [],
   574→        stars: createStars(),
   575→        keys: new Set<string>(),
   576→        frame: 0,
   577→        wave: startWave,
   578→        waveFrame: 0,
   579→        running: true,
   580→        screenFlash: 0,
   581→        killStreaks: [0, 0],
   582→        animId: 0,
   583→      };
   584→      gameRef.current = g;
