   105→   105→function makeBricks(offsetX: number): Brick[] {
   106→   106→  const bricks: Brick[] = [];
   107→   107→  const usable = HALF - 12;
   108→   108→  const brickW = (usable - BRICK_PAD * (BRICK_COLS - 1)) / BRICK_COLS;
   109→   109→  for (let r = 0; r < BRICK_ROWS; r++) {
   110→   110→    for (let c = 0; c < BRICK_COLS; c++) {
   111→   111→      bricks.push({
   112→   112→        x: offsetX + 6 + c * (brickW + BRICK_PAD),
   113→   113→        y: BRICK_TOP + r * (BRICK_H + BRICK_PAD),
   114→   114→        w: brickW,
   115→   115→        h: BRICK_H,
   116→   116→        color: ROW_COLORS[r],
   117→   117→        points: ROW_POINTS[r],
   118→   118→        alive: true,
   119→   119→      });
   120→   120→    }
   121→   121→  }
   122→   122→  return bricks;
   123→   123→}

   154→   154→function spawnParticles(particles: Particle[], cx: number, cy: number, color: string, count: number) {
   155→   155→  for (let i = 0; i < count; i++) {
   156→   156→    const angle = Math.random() * Math.PI * 2;
   157→   157→    const speed = 1 + Math.random() * 3;
   158→   158→    particles.push({
   159→   159→      x: cx,
   160→   160→      y: cy,
   161→   161→      vx: Math.cos(angle) * speed,
   162→   162→      vy: Math.sin(angle) * speed,
   163→   163→      life: 1,
   164→   164→      maxLife: 0.4 + Math.random() * 0.4,
   165→   165→      color,
   166→   166→      size: 2 + Math.random() * 3,
   167→   167→    });
   168→   168→  }
   169→   169→}

   285→   285→  /* ── update one field ── */
   286→   286→  function updateField(field: PlayerField, offset: number, dt: number, moveLeft: boolean, moveRight: boolean) {
   287→   287→    if (field.gameOver || field.won) return;
   288→   288→
   289→   289→    // paddle movement
   290→   290→    const pSpeed = 6;
   291→   291→    if (moveLeft) field.paddleX -= pSpeed;
   292→   292→    if (moveRight) field.paddleX += pSpeed;
   293→   293→    field.paddleX = clamp(field.paddleX, offset + 2, offset + HALF - field.paddleW - 2);
   294→   294→
   295→   295→    // timers
   296→   296→    if (field.wideTimer > 0) {
   297→   297→      field.wideTimer -= dt;
   298→   298→      if (field.wideTimer <= 0) {
   299→   299→        field.paddleW = PADDLE_BASE_W;
   300→   300→        field.wideTimer = 0;
   301→   301→      }
   302→   302→    }
   303→   303→    if (field.slowTimer > 0) {
   304→   304→      field.slowTimer -= dt;
   305→   305→      if (field.slowTimer <= 0) {
   306→   306→        field.slowTimer = 0;
   307→   307→        // restore speed
   308→   308→        field.balls.forEach((b) => {
   309→   309→          const spd = Math.sqrt(b.vx * b.vx + b.vy * b.vy);
   310→   310→          if (spd > 0.1) {
   311→   311→            const ratio = BALL_BASE_SPEED / spd;
   312→   312→            b.vx *= ratio;
   313→   313→            b.vy *= ratio;
   314→   314→          }
   315→   315→        });
   316→   316→      }
   317→   317→    }
   318→   318→
   319→   319→    // stick ball to paddle if not launched
   320→   320→    if (!field.launched && field.balls.length > 0) {
   321→   321→      field.balls[0].x = field.paddleX + field.paddleW / 2;
   322→   322→      field.balls[0].y = CH - 30 - PADDLE_H - BALL_R - 2;
   323→   323→    }
   324→   324→
   325→   325→    const speed = field.slowTimer > 0 ? BALL_SLOW_SPEED : BALL_BASE_SPEED;
   326→   326→
   327→   327→    // update balls
   328→   328→    const newBalls: Ball[] = [];
   329→   329→    for (const ball of field.balls) {
   330→   330→      if (!field.launched) {
   331→   331→        newBalls.push(ball);
   332→   332→        continue;
   333→   333→      }
   334→   334→
   335→   335→      // trail
   336→   336→      ball.trail.push({ x: ball.x, y: ball.y, alpha: 1 });
   337→   337→      if (ball.trail.length > 12) ball.trail.shift();
   338→   338→      ball.trail.forEach((t) => (t.alpha *= 0.85));
   339→   339→
   340→   340→      ball.x += ball.vx;
   341→   341→      ball.y += ball.vy;
   342→   342→
   343→   343→      // wall bounces
   344→   344→      if (ball.x - BALL_R < offset) {
   345→   345→        ball.x = offset + BALL_R;
   346→   346→        ball.vx = Math.abs(ball.vx);
   347→   347→      }
   348→   348→      if (ball.x + BALL_R > offset + HALF) {
   349→   349→        ball.x = offset + HALF - BALL_R;
   350→   350→        ball.vx = -Math.abs(ball.vx);
   351→   351→      }
   352→   352→      if (ball.y - BALL_R < 0) {
   353→   353→        ball.y = BALL_R;
   354→   354→        ball.vy = Math.abs(ball.vy);
   355→   355→      }
   356→   356→
   357→   357→      // paddle bounce
   358→   358→      const paddleY = CH - 30;
   359→   359→      if (
   360→   360→        ball.vy > 0 &&
   361→   361→        ball.y + BALL_R >= paddleY &&
   362→   362→        ball.y + BALL_R <= paddleY + PADDLE_H + 4 &&
   363→   363→        ball.x >= field.paddleX - BALL_R &&
   364→   364→        ball.x <= field.paddleX + field.paddleW + BALL_R
   365→   365→      ) {
   366→   366→        ball.y = paddleY - BALL_R;
   367→   367→        const hitPos = (ball.x - field.paddleX) / field.paddleW; // 0..1
   368→   368→        const angle = -Math.PI / 6 - hitPos * (2 * Math.PI / 3); // -30 to -150 deg roughly
   369→   369→        ball.vx = Math.cos(angle) * speed;
   370→   370→        ball.vy = Math.sin(angle) * speed;
   371→   371→        // ensure going up
   372→   372→        if (ball.vy > -1) ball.vy = -speed * 0.7;
   373→   373→      }
   374→   374→
   375→   375→      // brick collisions
   376→   376→      for (const brick of field.bricks) {
   377→   377→        if (!brick.alive) continue;
   378→   378→        if (
   379→   379→          ball.x + BALL_R > brick.x &&
   380→   380→          ball.x - BALL_R < brick.x + brick.w &&
   381→   381→          ball.y + BALL_R > brick.y &&
   382→   382→          ball.y - BALL_R < brick.y + brick.h
   383→   383→        ) {
   384→   384→          brick.alive = false;
   385→   385→          field.score += brick.points;
   386→   386→          spawnParticles(field.particles, brick.x + brick.w / 2, brick.y + brick.h / 2, brick.color, 8);
   387→   387→
   388→   388→          // determine bounce direction
   389→   389→          const overlapL = ball.x + BALL_R - brick.x;
   390→   390→          const overlapR = brick.x + brick.w - (ball.x - BALL_R);
   391→   391→          const overlapT = ball.y + BALL_R - brick.y;
   392→   392→          const overlapB = brick.y + brick.h - (ball.y - BALL_R);
   393→   393→          const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);
   394→   394→          if (minOverlap === overlapT || minOverlap === overlapB) {
   395→   395→            ball.vy = -ball.vy;
   396→   396→          } else {
   397→   397→            ball.vx = -ball.vx;
   398→   398→          }
   399→   399→
   400→   400→          // maybe spawn power-up (15% chance)
   401→   401→          if (Math.random() < 0.15) {
   402→   402→            const types: PowerUpType[] = ["wide", "multi", "slow"];
   403→   403→            const puType = types[Math.floor(Math.random() * types.length)];
   404→   404→            field.powerUps.push({
   405→   405→              x: brick.x + brick.w / 2,
   406→   406→              y: brick.y + brick.h / 2,
   407→   407→              type: puType,
   408→   408→              color: PU_COLORS[puType],
   409→   409→              vy: 1.5,
   410→   410→            });
   411→   411→          }
   412→   412→
   413→   413→          break; // one brick per ball per frame
   414→   414→        }
   415→   415→      }
   416→   416→
   417→   417→      // ball fell below
   418→   418→      if (ball.y - BALL_R > CH) {
   419→   419→        continue; // remove ball
   420→   420→      }
   421→   421→
   422→   422→      newBalls.push(ball);
   423→   423→    }
   424→   424→
   425→   425→    field.balls = newBalls;
   426→   426→
   427→   427→    // lost all balls
   428→   428→    if (field.launched && field.balls.length === 0) {
   429→   429→      field.lives--;
   430→   430→      field.launched = false;
   431→   431→      if (field.lives <= 0) {
   432→   432→        field.gameOver = true;
   433→   433→      } else {
   434→   434→        field.balls = [makeBall(field.paddleX + field.paddleW / 2, offset)];
   435→   435→        field.paddleW = PADDLE_BASE_W;
   436→   436→        field.wideTimer = 0;
   437→   437→        field.slowTimer = 0;
   438→   438→      }
   439→   439→    }
   440→   440→
   441→   441→    // check win
   442→   442→    if (field.bricks.every((b) => !b.alive) && !field.won) {
   443→   443→      field.won = true;
   444→   444→      field.score += field.lives * 100;
   445→   445→    }
   446→   446→
   447→   447→    // update power-ups
   448→   448→    const newPU: PowerUp[] = [];
   449→   449→    for (const pu of field.powerUps) {
   450→   450→      pu.y += pu.vy;
   451→   451→      const paddleY = CH - 30;
   452→   452→      if (
   453→   453→        pu.y + 6 >= paddleY &&
   454→   454→        pu.y - 6 <= paddleY + PADDLE_H &&
   455→   455→        pu.x >= field.paddleX &&
   456→   456→        pu.x <= field.paddleX + field.paddleW
   457→   457→      ) {
   458→   458→        // collected
   459→   459→        applyPowerUp(field, pu.type, offset);
   460→   460→        spawnParticles(field.particles, pu.x, pu.y, pu.color, 10);
   461→   461→      } else if (pu.y < CH + 10) {
   462→   462→        newPU.push(pu);
   463→   463→      }
   464→   464→    }
   465→   465→    field.powerUps = newPU;
   466→   466→
   467→   467→    // update particles
   468→   468→    for (let i = field.particles.length - 1; i >= 0; i--) {
   469→   469→      const p = field.particles[i];
   470→   470→      p.x += p.vx;
   471→   471→      p.y += p.vy;
   472→   472→      p.vy += 0.05; // gravity
   473→   473→      p.life -= dt / p.maxLife;
   474→   474→      if (p.life <= 0) {
   475→   475→        field.particles.splice(i, 1);
   476→   476→      }
   477→   477→    }
   478→   478→  }
   479→   479→
   480→   480→  function applyPowerUp(field: PlayerField, type: PowerUpType, offset: number) {
   481→   481→    switch (type) {
   482→   482→      case "wide":
   483→   483→        field.paddleW = PADDLE_WIDE_W;
   484→   484→        field.wideTimer = 8;
   485→   485→        break;
   486→   486→      case "multi": {
   487→   487→        const currentBalls = [...field.balls];
   488→   488→        const extras: Ball[] = [];
   489→   489→        for (const b of currentBalls) {
   490→   490→          for (let i = 0; i < 2; i++) {
   491→   491→            const nb: Ball = {
   492→   492→              x: b.x,
   493→   493→              y: b.y,
   494→   494→              vx: b.vx * Math.cos(0.5 * (i === 0 ? 1 : -1)) - b.vy * Math.sin(0.5 * (i === 0 ? 1 : -1)),
   495→   495→              vy: b.vx * Math.sin(0.5 * (i === 0 ? 1 : -1)) + b.vy * Math.cos(0.5 * (i === 0 ? 1 : -1)),
   496→   496→              trail: [],
   497→   497→            };
   498→   498→            extras.push(nb);
   499→   499→          }
   500→   500→        }
   501→   501→        field.balls.push(...extras);
   502→   502→        break;
   503→   503→      }
   504→   504→      case "slow":
   505→   505→        field.slowTimer = 6;
   506→   506→        field.balls.forEach((b) => {
   507→   507→          const spd = Math.sqrt(b.vx * b.vx + b.vy * b.vy);
   508→   508→          if (spd > 0.1) {
   509→   509→            const ratio = BALL_SLOW_SPEED / spd;
   510→   510→            b.vx *= ratio;
   511→   511→            b.vy *= ratio;
   512→   512→          }
   513→   513→        });
   514→   514→        break;
   515→   515→    }
   516→   516→  }
