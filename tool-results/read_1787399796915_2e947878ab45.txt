   135→function initField(offset: number): PlayerField {
   136→  const paddleX = offset + HALF / 2 - PADDLE_BASE_W / 2;
   137→  return {
   138→    score: 0,
   139→    lives: 3,
   140→    paddleX,
   141→    paddleW: PADDLE_BASE_W,
   142→    balls: [makeBall(paddleX + PADDLE_BASE_W / 2, offset)],
   143→    bricks: makeBricks(offset),
   144→    particles: [],
   145→    powerUps: [],
   146→    wideTimer: 0,
   147→    slowTimer: 0,
   148→    launched: false,
   149→    gameOver: false,
   150→    won: false,
   151→  };
   152→}
   153→
   154→function spawnParticles(particles: Particle[], cx: number, cy: number, color: string, count: number) {
   155→  for (let i = 0; i < count; i++) {
   156→    const angle = Math.random() * Math.PI * 2;
   157→    const speed = 1 + Math.random() * 3;
   158→    particles.push({
   159→      x: cx,
   160→      y: cy,
   161→      vx: Math.cos(angle) * speed,
   162→      vy: Math.sin(angle) * speed,
   163→      life: 1,
   164→      maxLife: 0.4 + Math.random() * 0.4,
   165→      color,
   166→      size: 2 + Math.random() * 3,
   167→    });
   168→  }
   169→}
   170→
   171→function clamp(v: number, min: number, max: number) {
   172→  return Math.max(min, Math.min(max, v));
   173→}

   271→  /* ── launch balls ── */
   272→  function launchBalls(field: PlayerField, _offset: number) {
   273→    if (field.gameOver || field.won) return;
   274→    if (!field.launched) {
   275→      field.launched = true;
   276→      field.balls.forEach((b) => {
   277→        const angle = -Math.PI / 4 - Math.random() * Math.PI / 2;
   278→        const speed = field.slowTimer > 0 ? BALL_SLOW_SPEED : BALL_BASE_SPEED;
   279→        b.vx = Math.cos(angle) * speed;
   280→        b.vy = Math.sin(angle) * speed;
   281→      });
   282→    }
   283→  }

   285→  /* ── update one field ── */
   286→  function updateField(field: PlayerField, offset: number, dt: number, moveLeft: boolean, moveRight: boolean) {
   287→    if (field.gameOver || field.won) return;
   288→
   289→    // paddle movement
   290→    const pSpeed = 6;
   291→    if (moveLeft) field.paddleX -= pSpeed;
   292→    if (moveRight) field.paddleX += pSpeed;
   293→    field.paddleX = clamp(field.paddleX, offset + 2, offset + HALF - field.paddleW - 2);
   294→
   295→    // timers
   296→    if (field.wideTimer > 0) {
   297→      field.wideTimer -= dt;
   298→      if (field.wideTimer <= 0) {
   299→        field.paddleW = PADDLE_BASE_W;
   300→        field.wideTimer = 0;
   301→      }
   302→    }
   303→    if (field.slowTimer > 0) {
   304→      field.slowTimer -= dt;
   305→      if (field.slowTimer <= 0) {
   306→        field.slowTimer = 0;
   307→        // restore speed
   308→        field.balls.forEach((b) => {
   309→          const spd = Math.sqrt(b.vx * b.vx + b.vy * b.vy);
   310→          if (spd > 0.1) {
   311→            const ratio = BALL_BASE_SPEED / spd;
   312→            b.vx *= ratio;
   313→            b.vy *= ratio;
   314→          }
   315→        });
   316→      }
   317→    }
   318→
   319→    // stick ball to paddle if not launched
   320→    if (!field.launched && field.balls.length > 0) {
   321→      field.balls[0].x = field.paddleX + field.paddleW / 2;
   322→      field.balls[0].y = CH - 30 - PADDLE_H - BALL_R - 2;
   323→    }
   324→
   325→    const speed = field.slowTimer > 0 ? BALL_SLOW_SPEED : BALL_BASE_SPEED;
   326→
   327→    // update balls
   328→    const newBalls: Ball[] = [];
   329→    for (const ball of field.balls) {
   330→      if (!field.launched) {
   331→        newBalls.push(ball);
   332→        continue;
   333→      }
   334→
   335→      // trail
   336→      ball.trail.push({ x: ball.x, y: ball.y, alpha: 1 });
   337→      if (ball.trail.length > 12) ball.trail.shift();
   338→      ball.trail.forEach((t) => (t.alpha *= 0.85));
   339→
   340→      ball.x += ball.vx;
   341→      ball.y += ball.vy;
   342→
   343→      // wall bounces
   344→      if (ball.x - BALL_R < offset) {
   345→        ball.x = offset + BALL_R;
   346→        ball.vx = Math.abs(ball.vx);
   347→      }
   348→      if (ball.x + BALL_R > offset + HALF) {
   349→        ball.x = offset + HALF - BALL_R;
   350→        ball.vx = -Math.abs(ball.vx);
   351→      }
   352→      if (ball.y - BALL_R < 0) {
   353→        ball.y = BALL_R;
   354→        ball.vy = Math.abs(ball.vy);
   355→      }
   356→
   357→      // paddle bounce
   358→      const paddleY = CH - 30;
   359→      if (
   360→        ball.vy > 0 &&
   361→        ball.y + BALL_R >= paddleY &&
   362→        ball.y + BALL_R <= paddleY + PADDLE_H + 4 &&
   363→        ball.x >= field.paddleX - BALL_R &&
   364→        ball.x <= field.paddleX + field.paddleW + BALL_R
   365→      ) {
   366→        ball.y = paddleY - BALL_R;
   367→        const hitPos = (ball.x - field.paddleX) / field.paddleW; // 0..1
   368→        const angle = -Math.PI / 6 - hitPos * (2 * Math.PI / 3); // -30 to -150 deg roughly
   369→        ball.vx = Math.cos(angle) * speed;
   370→        ball.vy = Math.sin(angle) * speed;
   371→        // ensure going up
   372→        if (ball.vy > -1) ball.vy = -speed * 0.7;
   373→      }
   374→
   375→      // brick collisions
   376→      for (const brick of field.bricks) {
   377→        if (!brick.alive) continue;
   378→        if (
   379→          ball.x + BALL_R > brick.x &&
   380→          ball.x - BALL_R < brick.x + brick.w &&
   381→          ball.y + BALL_R > brick.y &&
   382→          ball.y - BALL_R < brick.y + brick.h
   383→        ) {
   384→          brick.alive = false;
   385→          field.score += brick.points;
   386→          spawnParticles(field.particles, brick.x + brick.w / 2, brick.y + brick.h / 2, brick.color, 8);
   387→
   388→          // determine bounce direction
   389→          const overlapL = ball.x + BALL_R - brick.x;
   390→          const overlapR = brick.x + brick.w - (ball.x - BALL_R);
   391→          const overlapT = ball.y + BALL_R - brick.y;
   392→          const overlapB = brick.y + brick.h - (ball.y - BALL_R);
   393→          const minOverlap = Math.min(overlapL, overlapR, overlapT, overlapB);
   394→          if (minOverlap === overlapT || minOverlap === overlapB) {
   395→            ball.vy = -ball.vy;
   396→          } else {
   397→            ball.vx = -ball.vx;
   398→          }
   399→
   400→          // maybe spawn power-up (15% chance)
   401→          if (Math.random() < 0.15) {
   402→            const types: PowerUpType[] = ["wide", "multi", "slow"];
   403→            const puType = types[Math.floor(Math.random() * types.length)];
   404→            field.powerUps.push({
   405→              x: brick.x + brick.w / 2,
   406→              y: brick.y + brick.h / 2,
   407→              type: puType,
   408→              color: PU_COLORS[puType],
   409→              vy: 1.5,
   410→            });
   411→          }
   412→
   413→          break; // one brick per ball per frame
   414→        }
   415→      }
   416→
   417→      // ball fell below
   418→      if (ball.y - BALL_R > CH) {
   419→        continue; // remove ball
   420→      }
   421→
   422→      newBalls.push(ball);
   423→    }
   424→
   425→    field.balls = newBalls;
   426→
   427→    // lost all balls
   428→    if (field.launched && field.balls.length === 0) {
   429→      field.lives--;
   430→      field.launched = false;
   431→      if (field.lives <= 0) {
   432→        field.gameOver = true;
   433→      } else {
   434→        field.balls = [makeBall(field.paddleX + field.paddleW / 2, offset)];
   435→        field.paddleW = PADDLE_BASE_W;
   436→        field.wideTimer = 0;
   437→        field.slowTimer = 0;
   438→      }
   439→    }
   440→
   441→    // check win
   442→    if (field.bricks.every((b) => !b.alive) && !field.won) {
   443→      field.won = true;
   444→      field.score += field.lives * 100;
   445→    }
   446→
   447→    // update power-ups
   448→    const newPU: PowerUp[] = [];
   449→    for (const pu of field.powerUps) {
   450→      pu.y += pu.vy;
   451→      const paddleY = CH - 30;
   452→      if (
   453→        pu.y + 6 >= paddleY &&
   454→        pu.y - 6 <= paddleY + PADDLE_H &&
   455→        pu.x >= field.paddleX &&
   456→        pu.x <= field.paddleX + field.paddleW
   457→      ) {
   458→        // collected
   459→        applyPowerUp(field, pu.type, offset);
   460→        spawnParticles(field.particles, pu.x, pu.y, pu.color, 10);
   461→      } else if (pu.y < CH + 10) {
   462→        newPU.push(pu);
   463→      }
   464→    }
   465→    field.powerUps = newPU;
   466→
   467→    // update particles
   468→    for (let i = field.particles.length - 1; i >= 0; i--) {
   469→      const p = field.particles[i];
   470→      p.x += p.vx;
   471→      p.y += p.vy;
   472→      p.vy += 0.05; // gravity
   473→      p.life -= dt / p.maxLife;
   474→      if (p.life <= 0) {
   475→        field.particles.splice(i, 1);
   476→      }
   477→    }
   478→  }
   479→
   480→  function applyPowerUp(field: PlayerField, type: PowerUpType, offset: number) {
   481→    switch (type) {
   482→      case "wide":
   483→        field.paddleW = PADDLE_WIDE_W;
   484→        field.wideTimer = 8;
   485→        break;
   486→      case "multi": {
   487→        const currentBalls = [...field.balls];
   488→        const extras: Ball[] = [];
   489→        for (const b of currentBalls) {
   490→          for (let i = 0; i < 2; i++) {
   491→            const nb: Ball = {
   492→              x: b.x,
   493→              y: b.y,
   494→              vx: b.vx * Math.cos(0.5 * (i === 0 ? 1 : -1)) - b.vy * Math.sin(0.5 * (i === 0 ? 1 : -1)),
   495→              vy: b.vx * Math.sin(0.5 * (i === 0 ? 1 : -1)) + b.vy * Math.cos(0.5 * (i === 0 ? 1 : -1)),
   496→              trail: [],
   497→            };
   498→            extras.push(nb);
   499→          }
   500→        }
   501→        field.balls.push(...extras);
   502→        break;
   503→      }
   504→      case "slow":
   505→        field.slowTimer = 6;
   506→        field.balls.forEach((b) => {
   507→          const spd = Math.sqrt(b.vx * b.vx + b.vy * b.vy);
   508→          if (spd > 0.1) {
   509→            const ratio = BALL_SLOW_SPEED / spd;
   510→            b.vx *= ratio;
   511→            b.vy *= ratio;
   512→          }
   513→        });
   514→        break;
   515→    }
   516→  }

   694→      ctx.restore();

   736→      ctx.moveTo(0, gy);
   737→      ctx.lineTo(CW, gy);

   769→    const p2Done = p2Ref.current.gameOver || p2Ref.current.won;

   771→      phaseRef.current = "done";
   772→      setPhase("done");
   773→
   774→      const p1 = p1Ref.current;
   775→      const p2 = p2Ref.current;
   776→
   777→      // both won → higher score wins; if one won and other didn't, winner
   778→      let winText: string;
   779→      if (p1.won && p2.won) {
   780→        if (p1.score >= p2.score) {
   781→          winText = P1_NAME;
   782→          onScore?.(P1_NAME, p1.score);
   783→        } else {
   784→          winText = P2_NAME;
   785→          onScore?.(P2_NAME, p2.score);
   786→        }
   787→      } else if (p1.won) {
   788→        winText = P1_NAME;
   789→        onScore?.(P1_NAME, p1.score);
   790→      } else if (p2.won) {
   791→        winText = P2_NAME;
   792→        onScore?.(P2_NAME, p2.score);
   793→      } else {
   794→        // both lost
   795→        if (p1.score >= p2.score) {
   796→          winText = P1_NAME;
   797→          onScore?.(P1_NAME, p1.score);
   798→        } else {
   799→          winText = P2_NAME;
   800→          onScore?.(P2_NAME, p2.score);
   801→        }
   802→      }
   803→      setWinner(winText);
   804→
   805→      // also report second place
   806→      const otherName = winText === P1_NAME ? P2_NAME : P1_NAME;
   807→      const otherScore = winText === P1_NAME ? p2.score : p1.score;
   808→      onScore?.(otherName, otherScore);
   809→    }
   810→
   811→    if (phaseRef.current === "playing") {
   812→      rafRef.current = requestAnimationFrame(gameLoop);
   813→    }
   814→  }, [onScore]);

   822→    setPhase("playing");
   823→    phaseRef.current = "playing";
   824→    rafRef.current = requestAnimationFrame(gameLoop);
   825→  }, [gameLoop]);

   837→      if (ctx) {
   838→        ctx.fillStyle = "#0a0a0f";
   839→        ctx.fillRect(0, 0, CW, CH);
   840→        ctx.setLineDash([6, 6]);
   841→        ctx.strokeStyle = "rgba(255,255,255,0.1)";
   842→        ctx.lineWidth = 2;
   843→        ctx.beginPath();
   844→        ctx.moveTo(HALF, 0);
   845→        ctx.lineTo(HALF, CH);
   846→        ctx.stroke();
   847→        ctx.setLineDash([]);
   848→      }
   849→    }
   850→  }, []);
   851→
   852→  useEffect(() => {
   853→    return () => cancelAnimationFrame(rafRef.current);
   854→  }, []);
   855→
   856→  // draw initial idle canvas once
   857→  useEffect(() => {
   858→    if (phase === "idle") {
   859→      resetGame();
   860→    }
   861→  // eslint-disable-next-line react-hooks/exhaustive-deps
   862→  }, []);
   863→
   864→  /* ──────────────────────── JSX ──────────────────────── */
   865→  return (
   866→    <div className="flex flex-col items-center gap-3 w-full max-w-[720px] mx-auto select-none">
   867→      <motion.div
   868→        initial={{ opacity: 0, y: -20 }}
   869→        animate={{ opacity: 1, y: 0 }}
   870→        className="flex items-center gap-3 w-full"
   871→      >
   872→        <div className="flex-1 text-right">
   873→          <Badge className="text-xs font-bold" style={{ backgroundColor: P1_COLOR + "22", color: P1_COLOR, borderColor: P1_COLOR }}>
   874→            <Gamepad2 className="w-3 h-3 mr-1" />
   875→            {P1_NAME}
   876→          </Badge>
   877→        </div>
   878→        <motion.div
   879→          initial={{ scale: 0 }}
   880→          animate={{ scale: 1 }}
   881→          className="text-center"
   882→        >
   883→          <h2 className="text-lg font-bold text-white tracking-tight">
   884→            ⚡ Quebra-Tijolos
   885→          </h2>
   886→          <p className="text-[10px] text-zinc-500">Destrua todos os tijolos!</p>
   887→        </motion.div>
   888→        <div className="flex-1 text-left">
   889→          <Badge className="text-xs font-bold" style={{ backgroundColor: P2_COLOR + "22", color: P2_COLOR, borderColor: P2_COLOR }}>
   890→            <Gamepad2 className="w-3 h-3 mr-1" />
   891→            {P2_NAME}
   892→          </Badge>
   893→        </div>
   894→      </motion.div>
   895→
   896→      <div className="flex items-center justify-between w-full max-w-[700px] px-1">
   897→        <motion.div
   898→          animate={{ scale: p1Ref.current.score > 0 ? [1, 1.05, 1] : 1 }}
   899→          className="flex items-center gap-2"
   900→        >
   901→          <span className="text-cyan-400 font-bold text-sm">{p1Ref.current.score}</span>
   902→          <div className="flex gap-0.5">
   903→            {Array.from({ length: p1Ref.current.lives }).map((_, i) => (
   904→              <Heart key={i} className="w-3 h-3 text-red-500 fill-red-500" />
   905→            ))}
   906→          </div>
   907→          <span className="text-[10px] text-zinc-500 ml-1">A / D</span>
   908→        </motion.div>
   909→
   910→        <div className="text-[10px] text-zinc-600 font-mono">
   911→          {phase === "playing" && `VS ${liveCode || ""}`}
   912→        </div>
   913→
   914→        <motion.div
   915→          animate={{ scale: p2Ref.current.score > 0 ? [1, 1.05, 1] : 1 }}
   916→          className="flex items-center gap-2"
   917→        >
   918→          <span className="text-[10px] text-zinc-500 mr-1">← →</span>
   919→          <div className="flex gap-0.5">
   920→            {Array.from({ length: p2Ref.current.lives }).map((_, i) => (
   921→              <Heart key={i} className="w-3 h-3 text-red-500 fill-red-500" />
   922→            ))}
   923→          </div>
   924→          <span className="text-pink-400 font-bold text-sm">{p2Ref.current.score}</span>

   928→      <motion.div
   929→        initial={{ opacity: 0, scale: 0.95 }}
   930→        animate={{ opacity: 1, scale: 1 }}
   931→        transition={{ type: "spring", damping: 20 }}

   933→      >
   934→        <canvas
   935→          ref={canvasRef}
   936→          width={CW}
   937→          height={CH}

   939→          style={{ imageRendering: "auto" }}
   940→          onClick={() => {
   941→            if (phase === "playing") {

   943→              launchBalls(p2Ref.current, HALF);

   952→              initial={{ opacity: 0 }}
