   184→const BattleRoyaleGame = ({ onScore, liveCode }: Props) => {
   185→  const canvasRef = useRef<HTMLCanvasElement>(null);
   186→  const gameRef = useRef<GameState | null>(null);
   187→
   188→  const [gameStatus, setGameStatus] = useState<GameStatus>("idle");
   189→  const [difficulty, setDifficulty] = useState<Difficulty>("normal");
   190→  const [hudData, setHudData] = useState({ hp: 100, shield: 0, weapon: "pistol" as WeaponType, ammo: 30, kills: 0, alive: 40, placement: 1 });
   191→  const [killFeed, setKillFeed] = useState<KillFeedEntry[]>([]);
   192→  const [resultData, setResultData] = useState({ placement: 1, kills: 0, damage: 0, time: 0 });
   193→
   194→  const updateHUD = useCallback((g: GameState) => {
   195→    const p = g.entities[g.playerIdx];
   196→    if (!p) return;
   197→    setHudData({ hp: p.hp, shield: p.shield, weapon: p.weapon, ammo: p.ammo, kills: p.kills, alive: g.aliveCount, placement: p.alive ? g.aliveCount : 0 });
   198→    setKillFeed(g.killFeed.slice(-5));
   199→  }, []);
   200→
   201→  const endGame = useCallback((g: GameState) => {
   202→    g.running = false;
   203→    const p = g.entities[g.playerIdx];
   204→    const placement = p.alive ? 1 : g.aliveCount + 1;
   205→    const survivalTime = Math.floor((Date.now() - g.startTime) / 1000);
   206→    setResultData({ placement, kills: p.kills, damage: p.damageDealt, time: survivalTime });
   207→    setGameStatus("over");
   208→    const score = p.kills * 100 + p.damageDealt + (placement === 1 ? 500 : Math.max(0, (50 - placement) * 10));
   209→    onScore?.("Jogador", score);
   210→  }, [onScore]);
   211→
   212→  const shoot = useCallback((shooter: Entity, sIdx: number, g: GameState, angle: number) => {
   213→    if (shooter.fireCooldown > 0 || shooter.ammo <= 0) return;
   214→    const w = WEAPONS[shooter.weapon];
   215→    shooter.fireCooldown = w.fireRate;
   216→    shooter.ammo -= 1;
   217→    for (let i = 0; i < w.bulletsPerShot; i++) {
   218→      const spread = (Math.random() - 0.5) * w.spread;
   219→      const a = angle + spread;
   220→      g.bullets.push({
   221→        x: shooter.x + Math.cos(a) * (shooter.radius + 4),
   222→        y: shooter.y + Math.sin(a) * (shooter.radius + 4),
   223→        vx: Math.cos(a) * w.bulletSpeed, vy: Math.sin(a) * w.bulletSpeed,
   224→        damage: w.damage, ownerIdx: sIdx, life: w.range / w.bulletSpeed, color: w.color,
   225→      });
   226→    }
   227→  }, []);
   228→
   229→  const updateBots = useCallback((g: GameState) => {
   230→    const diff = DIFF_STATS[g.difficulty];
   231→    for (let i = 0; i < g.entities.length; i++) {
   232→      const bot = g.entities[i];
   233→      if (!bot.alive || bot.isPlayer) continue;
   234→      if (bot.fireCooldown > 0) bot.fireCooldown--;
   235→
   236→      let closestDist = Infinity;
   237→      let closestIdx = -1;
   238→      for (let j = 0; j < g.entities.length; j++) {
   239→        if (i === j || !g.entities[j].alive) continue;
   240→        const d = dist(bot, g.entities[j]);
   241→        if (d < closestDist) { closestDist = d; closestIdx = j; }
   242→      }
   243→
   244→      if (closestIdx >= 0 && closestDist < 350) {
   245→        const target = g.entities[closestIdx];
   246→        bot.targetAngle = angleTo(bot, target);
   247→        bot.angle += (bot.targetAngle - bot.angle) * 0.15;
   248→
   249→        if (closestDist > 120) {
   250→          bot.vx = Math.cos(bot.angle) * 2.2;
   251→          bot.vy = Math.sin(bot.angle) * 2.2;
   252→        } else if (closestDist < 70) {
   253→          bot.vx = -Math.cos(bot.angle) * 1.5;
   254→          bot.vy = -Math.sin(bot.angle) * 1.5;
   255→        } else {
   256→          const strafe = bot.angle + Math.PI / 2 * (Math.sin(g.frame * 0.03 + i) > 0 ? 1 : -1);
   257→          bot.vx = Math.cos(strafe) * 1.2;
   258→          bot.vy = Math.sin(strafe) * 1.2;
   259→        }
   260→
   261→        if (Math.random() < diff.botAggro * 0.5 && closestDist < 300) {
   262→          const aimAngle = bot.targetAngle + (Math.random() - 0.5) * (1 - diff.botAcc) * 0.8;
   263→          shoot(bot, i, g, aimAngle);
   264→        }
   265→      } else {
   266→        const zAngle = angleTo(bot, { x: g.zone.cx, y: g.zone.cy });
   267→        const zDist = dist(bot, { x: g.zone.cx, y: g.zone.cy });
   268→        if (zDist > g.zone.radius * 0.7) {
   269→          bot.vx = Math.cos(zAngle) * 2;
   270→          bot.vy = Math.sin(zAngle) * 2;
   271→        } else {
   272→          if (g.frame % 120 === i % 120) {
   273→            bot.targetAngle = Math.random() * Math.PI * 2;
   274→          }
   275→          bot.vx = Math.cos(bot.targetAngle) * 1.5;
   276→          bot.vy = Math.sin(bot.targetAngle) * 1.5;
   277→        }
   278→        bot.angle += (bot.targetAngle - bot.angle) * 0.08;
   279→
   280→        for (const l of g.loot) {
   281→          if (!l.alive) continue;
   282→          if (dist(bot, l) < 200 && Math.random() < 0.02) {
   283→            bot.targetAngle = angleTo(bot, l);
   284→            break;
   285→          }
   286→        }
   287→      }
   288→
   289→      bot.x += bot.vx;
   290→      bot.y += bot.vy;
   291→      bot.x = clamp(bot.x, bot.radius, WORLD - bot.radius);
   292→      bot.y = clamp(bot.y, bot.radius, WORLD - bot.radius);
   293→      bot.vx *= 0.92;
   294→      bot.vy *= 0.92;
   295→    }
   296→  }, [shoot]);
   297→
   298→  const gameLoop = useCallback((g: GameState) => {
   299→    const canvas = canvasRef.current;
   300→    if (!canvas || !g.running) return;
   301→    const ctx = canvas.getContext("2d");
   302→    if (!ctx) return;
   303→    g.frame++;
   304→
   305→    const player = g.entities[g.playerIdx];
   306→
   307→    // Zone shrinking
   308→    g.zone.phaseTimer++;
   309→    const phaseDuration = 600 - g.zone.phase * 60;
   310→    if (g.zone.phaseTimer > phaseDuration && g.zone.phase < 6) {
   311→      g.zone.phase++;
   312→      g.zone.phaseTimer = 0;
   313→      g.zone.targetRadius = Math.max(100, g.zone.radius * 0.55);
   314→      const offset = g.zone.targetRadius * 0.3;
   315→      g.zone.targetCx = clamp(g.zone.cx + (Math.random() - 0.5) * offset, g.zone.targetRadius + 50, WORLD - g.zone.targetRadius - 50);
   316→      g.zone.targetCy = clamp(g.zone.cy + (Math.random() - 0.5) * offset, g.zone.targetRadius + 50, WORLD - g.zone.targetRadius - 50);
   317→      g.zone.damage = 1 + g.zone.phase * 0.5;
   318→    }
   319→    g.zone.cx += (g.zone.targetCx - g.zone.cx) * 0.003;
   320→    g.zone.cy += (g.zone.targetCy - g.zone.cy) * 0.003;
   321→    g.zone.radius += (g.zone.targetRadius - g.zone.radius) * 0.005;
   322→
   323→    // Supply drops
   324→    if (g.frame % 900 === 450 && g.zone.phase < 5) {
   325→      const angle = Math.random() * Math.PI * 2;
   326→      const r = Math.random() * g.zone.radius * 0.6;
   327→      g.supplyDrops.push({
   328→        x: clamp(g.zone.cx + Math.cos(angle) * r, 100, WORLD - 100),
   329→        y: clamp(g.zone.cy + Math.sin(angle) * r, 100, WORLD - 100),
   330→        loot: [(Math.random() < 0.5 ? "rifle" : "shotgun") as LootType, "health" as LootType, "shield" as LootType],
   331→        landed: false, fallY: -100, timer: 600,
   332→      });
   333→    }
   334→    g.supplyDrops = g.supplyDrops.filter((sd) => {
   335→      if (!sd.landed) { sd.fallY += 6; if (sd.fallY >= sd.y) { sd.landed = true; sd.y = sd.fallY; } sd.timer--; return sd.timer > 0; }
   336→      sd.timer--;
   337→      return sd.timer > 0;
   338→    });
   339→
   340→    // Player input
   341→    if (player.alive) {
   342→      const speed = 3.2;
   343→      if (player.fireCooldown > 0) player.fireCooldown--;
   344→      if (g.keys.has("w") || g.keys.has("W")) player.y -= speed;
   345→      if (g.keys.has("s") || g.keys.has("S")) player.y += speed;
   346→      if (g.keys.has("a") || g.keys.has("A")) player.x -= speed;
   347→      if (g.keys.has("d") || g.keys.has("D")) player.x += speed;
   348→      player.x = clamp(player.x, player.radius, WORLD - player.radius);
   349→      player.y = clamp(player.y, player.radius, WORLD - player.radius);
   350→
   351→      const rect = canvas.getBoundingClientRect();
   352→      const scaleX = CANVAS_W / rect.width;
   353→      const scaleY = CANVAS_H / rect.height;
   354→      const worldMouseX = g.mouseX * scaleX + (player.x - CANVAS_W / 2);
   355→      const worldMouseY = g.mouseY * scaleY + (player.y - CANVAS_H / 2);
   356→      player.angle = angleTo(player, { x: worldMouseX, y: worldMouseY });
   357→
   358→      if (g.mouseDown) shoot(player, g.playerIdx, g, player.angle);
   359→    }
   360→
   361→    updateBots(g);
   362→
   363→    // Bullets update
   364→    g.bullets = g.bullets.filter((b) => {
   365→      b.x += b.vx;
   366→      b.y += b.vy;
   367→      b.life--;
   368→      if (b.life <= 0 || b.x < 0 || b.x > WORLD || b.y < 0 || b.y > WORLD) return false;
   369→      for (let i = 0; i < g.entities.length; i++) {
   370→        const e = g.entities[i];
   371→        if (!e.alive || i === b.ownerIdx) continue;
   372→        if (dist(b, e) < e.radius + 3) {
   373→          let dmg = b.damage;
   374→          if (e.shield > 0) { const absorbed = Math.min(e.shield, dmg); e.shield -= absorbed; dmg -= absorbed; }
   375→          e.hp -= dmg;
   376→          g.entities[b.ownerIdx].damageDealt += b.damage;
   377→          g.particles.push(...createExplosion(b.x, b.y, e.color, 4));
   378→          if (e.hp <= 0) {
   379→            e.alive = false;
   380→            g.aliveCount--;
   381→            g.entities[b.ownerIdx].kills++;
   382→            g.particles.push(...createExplosion(e.x, e.y, e.color, 12));
   383→            g.killFeed.push({ killer: g.entities[b.ownerIdx].name, victim: e.name, time: g.frame });
   384→            if (g.killFeed.length > 20) g.killFeed.shift();
   385→            if (e.isPlayer) { endGame(g); return false; }
   386→            if (g.aliveCount <= 1 && g.entities[g.playerIdx].alive) { endGame(g); return false; }
   387→          }
   388→          return false;
   389→        }
   390→      }
   391→      return true;
   392→    });
   393→
   394→    // Zone damage
   395→    for (const e of g.entities) {
   396→      if (!e.alive) continue;
   397→      const d = dist(e, { x: g.zone.cx, y: g.zone.cy });
   398→      if (d > g.zone.radius && g.frame % 30 === 0) {
   399→        e.hp -= g.zone.damage;
   400→        if (e.hp <= 0) {
   401→          e.alive = false;
   402→          g.aliveCount--;
   403→          g.particles.push(...createExplosion(e.x, e.y, "#ef4444", 8));
   404→          g.killFeed.push({ killer: "Zona", victim: e.name, time: g.frame });
   405→          if (e.isPlayer) { endGame(g); }
   406→        }
   407→      }
   408→    }
   409→
   410→    // Loot pickup
   411→    for (const l of g.loot) {
   412→      if (!l.alive) continue;
   413→      for (const e of g.entities) {
   414→        if (!e.alive) continue;
   415→        if (dist(e, l) < e.radius + 12) {
   416→          if (l.type === "health") { e.hp = Math.min(e.maxHp, e.hp + 30); }
   417→          else if (l.type === "shield") { e.shield = Math.min(50, e.shield + 25); }
   418→          else if (l.type === "ammo") { e.ammo = Math.min(99, e.ammo + 15); }
   419→          else { e.weapon = l.type as WeaponType; e.ammo = Math.min(99, e.ammo + 20); }
   420→          l.alive = false;
   421→          g.particles.push(...createExplosion(l.x, l.y, LOOT_COLORS[l.type], 5));
   422→        }
   423→      }
   424→    }
   425→
   426→    // Supply drop loot pickup
   427→    for (const sd of g.supplyDrops) {
   428→      if (!sd.landed) continue;
   429→      for (let li = sd.loot.length - 1; li >= 0; li--) {
   430→        for (const e of g.entities) {
   431→          if (!e.alive) continue;
   432→          if (dist(e, sd) < e.radius + 20) {
   433→            const lt = sd.loot[li];
   434→            if (lt === "health") e.hp = Math.min(e.maxHp, e.hp + 50);
   435→            else if (lt === "shield") e.shield = Math.min(50, e.shield + 50);
   436→            else { e.weapon = lt as WeaponType; e.ammo = Math.min(99, e.ammo + 30); }
   437→            sd.loot.splice(li, 1);
   438→            g.particles.push(...createExplosion(sd.x, sd.y, "#fbbf24", 8));
   439→            break;
   440→          }
   441→        }
   442→      }
   443→    }
   444→
   445→    // Particles
   446→    g.particles = g.particles.filter((p) => { p.x += p.vx; p.y += p.vy; p.vy += 0.05; p.life--; return p.life > 0; });
   447→
   448→    if (g.frame % 10 === 0) updateHUD(g);
   449→
   450→    // ---- RENDER ----
   451→    const camX = player.x - CANVAS_W / 2;
   452→    const camY = player.y - CANVAS_H / 2;
   453→
   454→    ctx.fillStyle = "#0a0e14";
   455→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   456→
   457→    // Grid
   458→    ctx.strokeStyle = "rgba(255,255,255,0.03)";
   459→    ctx.lineWidth = 1;
   460→    const gridSize = 80;
