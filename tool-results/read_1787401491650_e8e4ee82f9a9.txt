   184→   184→const BattleRoyaleGame = ({ onScore, liveCode }: Props) => {
   185→   185→  const canvasRef = useRef<HTMLCanvasElement>(null);
   186→   186→  const gameRef = useRef<GameState | null>(null);
   187→   187→
   188→   188→  const [gameStatus, setGameStatus] = useState<GameStatus>("idle");
   189→   189→  const [difficulty, setDifficulty] = useState<Difficulty>("normal");
   190→   190→  const [hudData, setHudData] = useState({ hp: 100, shield: 0, weapon: "pistol" as WeaponType, ammo: 30, kills: 0, alive: 40, placement: 1 });
   191→   191→  const [killFeed, setKillFeed] = useState<KillFeedEntry[]>([]);
   192→   192→  const [resultData, setResultData] = useState({ placement: 1, kills: 0, damage: 0, time: 0 });
   193→   193→
   194→   194→  const updateHUD = useCallback((g: GameState) => {
   195→   195→    const p = g.entities[g.playerIdx];
   196→   196→    if (!p) return;
   197→   197→    setHudData({ hp: p.hp, shield: p.shield, weapon: p.weapon, ammo: p.ammo, kills: p.kills, alive: g.aliveCount, placement: p.alive ? g.aliveCount : 0 });
   198→   198→    setKillFeed(g.killFeed.slice(-5));
   199→   199→  }, []);
   200→   200→
   201→   201→  const endGame = useCallback((g: GameState) => {
   202→   202→    g.running = false;
   203→   203→    const p = g.entities[g.playerIdx];
   204→   204→    const placement = p.alive ? 1 : g.aliveCount + 1;
   205→   205→    const survivalTime = Math.floor((Date.now() - g.startTime) / 1000);
   206→   206→    setResultData({ placement, kills: p.kills, damage: p.damageDealt, time: survivalTime });
   207→   207→    setGameStatus("over");
   208→   208→    const score = p.kills * 100 + p.damageDealt + (placement === 1 ? 500 : Math.max(0, (50 - placement) * 10));
   209→   209→    onScore?.("Jogador", score);
   210→   210→  }, [onScore]);
   211→   211→
   212→   212→  const shoot = useCallback((shooter: Entity, sIdx: number, g: GameState, angle: number) => {
   213→   213→    if (shooter.fireCooldown > 0 || shooter.ammo <= 0) return;
   214→   214→    const w = WEAPONS[shooter.weapon];
   215→   215→    shooter.fireCooldown = w.fireRate;
   216→   216→    shooter.ammo -= 1;
   217→   217→    for (let i = 0; i < w.bulletsPerShot; i++) {
   218→   218→      const spread = (Math.random() - 0.5) * w.spread;
   219→   219→      const a = angle + spread;
   220→   220→      g.bullets.push({
   221→   221→        x: shooter.x + Math.cos(a) * (shooter.radius + 4),
   222→   222→        y: shooter.y + Math.sin(a) * (shooter.radius + 4),
   223→   223→        vx: Math.cos(a) * w.bulletSpeed, vy: Math.sin(a) * w.bulletSpeed,
   224→   224→        damage: w.damage, ownerIdx: sIdx, life: w.range / w.bulletSpeed, color: w.color,
   225→   225→      });
   226→   226→    }
   227→   227→  }, []);
   228→   228→
   229→   229→  const updateBots = useCallback((g: GameState) => {
   230→   230→    const diff = DIFF_STATS[g.difficulty];
   231→   231→    for (let i = 0; i < g.entities.length; i++) {
   232→   232→      const bot = g.entities[i];
   233→   233→      if (!bot.alive || bot.isPlayer) continue;
   234→   234→      if (bot.fireCooldown > 0) bot.fireCooldown--;
   235→   235→
   236→   236→      let closestDist = Infinity;
   237→   237→      let closestIdx = -1;
   238→   238→      for (let j = 0; j < g.entities.length; j++) {
   239→   239→        if (i === j || !g.entities[j].alive) continue;
   240→   240→        const d = dist(bot, g.entities[j]);
   241→   241→        if (d < closestDist) { closestDist = d; closestIdx = j; }
   242→   242→      }
   243→   243→
   244→   244→      if (closestIdx >= 0 && closestDist < 350) {
   245→   245→        const target = g.entities[closestIdx];
   246→   246→        bot.targetAngle = angleTo(bot, target);
   247→   247→        bot.angle += (bot.targetAngle - bot.angle) * 0.15;
   248→   248→
   249→   249→        if (closestDist > 120) {
   250→   250→          bot.vx = Math.cos(bot.angle) * 2.2;
   251→   251→          bot.vy = Math.sin(bot.angle) * 2.2;
   252→   252→        } else if (closestDist < 70) {
   253→   253→          bot.vx = -Math.cos(bot.angle) * 1.5;
   254→   254→          bot.vy = -Math.sin(bot.angle) * 1.5;
   255→   255→        } else {
   256→   256→          const strafe = bot.angle + Math.PI / 2 * (Math.sin(g.frame * 0.03 + i) > 0 ? 1 : -1);
   257→   257→          bot.vx = Math.cos(strafe) * 1.2;
   258→   258→          bot.vy = Math.sin(strafe) * 1.2;
   259→   259→        }
   260→   260→
   261→   261→        if (Math.random() < diff.botAggro * 0.5 && closestDist < 300) {
   262→   262→          const aimAngle = bot.targetAngle + (Math.random() - 0.5) * (1 - diff.botAcc) * 0.8;
   263→   263→          shoot(bot, i, g, aimAngle);
   264→   264→        }
   265→   265→      } else {
   266→   266→        const zAngle = angleTo(bot, { x: g.zone.cx, y: g.zone.cy });
   267→   267→        const zDist = dist(bot, { x: g.zone.cx, y: g.zone.cy });
   268→   268→        if (zDist > g.zone.radius * 0.7) {
   269→   269→          bot.vx = Math.cos(zAngle) * 2;
   270→   270→          bot.vy = Math.sin(zAngle) * 2;
   271→   271→        } else {
   272→   272→          if (g.frame % 120 === i % 120) {
   273→   273→            bot.targetAngle = Math.random() * Math.PI * 2;
   274→   274→          }
   275→   275→          bot.vx = Math.cos(bot.targetAngle) * 1.5;
   276→   276→          bot.vy = Math.sin(bot.targetAngle) * 1.5;
   277→   277→        }
   278→   278→        bot.angle += (bot.targetAngle - bot.angle) * 0.08;
   279→   279→
   280→   280→        for (const l of g.loot) {
   281→   281→          if (!l.alive) continue;
   282→   282→          if (dist(bot, l) < 200 && Math.random() < 0.02) {
   283→   283→            bot.targetAngle = angleTo(bot, l);
   284→   284→            break;
   285→   285→          }
   286→   286→        }
   287→   287→      }
   288→   288→
   289→   289→      bot.x += bot.vx;
   290→   290→      bot.y += bot.vy;
   291→   291→      bot.x = clamp(bot.x, bot.radius, WORLD - bot.radius);
   292→   292→      bot.y = clamp(bot.y, bot.radius, WORLD - bot.radius);
   293→   293→      bot.vx *= 0.92;
   294→   294→      bot.vy *= 0.92;
   295→   295→    }
   296→   296→  }, [shoot]);
   297→   297→
   298→   298→  const gameLoop = useCallback((g: GameState) => {
   299→   299→    const canvas = canvasRef.current;
   300→   300→    if (!canvas || !g.running) return;
   301→   301→    const ctx = canvas.getContext("2d");
   302→   302→    if (!ctx) return;
   303→   303→    g.frame++;
   304→   304→
   305→   305→    const player = g.entities[g.playerIdx];
   306→   306→
   307→   307→    // Zone shrinking
   308→   308→    g.zone.phaseTimer++;
   309→   309→    const phaseDuration = 600 - g.zone.phase * 60;
   310→   310→    if (g.zone.phaseTimer > phaseDuration && g.zone.phase < 6) {
   311→   311→      g.zone.phase++;
   312→   312→      g.zone.phaseTimer = 0;
   313→   313→      g.zone.targetRadius = Math.max(100, g.zone.radius * 0.55);
   314→   314→      const offset = g.zone.targetRadius * 0.3;
   315→   315→      g.zone.targetCx = clamp(g.zone.cx + (Math.random() - 0.5) * offset, g.zone.targetRadius + 50, WORLD - g.zone.targetRadius - 50);
   316→   316→      g.zone.targetCy = clamp(g.zone.cy + (Math.random() - 0.5) * offset, g.zone.targetRadius + 50, WORLD - g.zone.targetRadius - 50);
   317→   317→      g.zone.damage = 1 + g.zone.phase * 0.5;
   318→   318→    }
   319→   319→    g.zone.cx += (g.zone.targetCx - g.zone.cx) * 0.003;
   320→   320→    g.zone.cy += (g.zone.targetCy - g.zone.cy) * 0.003;
   321→   321→    g.zone.radius += (g.zone.targetRadius - g.zone.radius) * 0.005;
   322→   322→
   323→   323→    // Supply drops
   324→   324→    if (g.frame % 900 === 450 && g.zone.phase < 5) {
   325→   325→      const angle = Math.random() * Math.PI * 2;
   326→   326→      const r = Math.random() * g.zone.radius * 0.6;
   327→   327→      g.supplyDrops.push({
   328→   328→        x: clamp(g.zone.cx + Math.cos(angle) * r, 100, WORLD - 100),
   329→   329→        y: clamp(g.zone.cy + Math.sin(angle) * r, 100, WORLD - 100),
   330→   330→        loot: [(Math.random() < 0.5 ? "rifle" : "shotgun") as LootType, "health" as LootType, "shield" as LootType],
   331→   331→        landed: false, fallY: -100, timer: 600,
   332→   332→      });
   333→   333→    }
   334→   334→    g.supplyDrops = g.supplyDrops.filter((sd) => {
   335→   335→      if (!sd.landed) { sd.fallY += 6; if (sd.fallY >= sd.y) { sd.landed = true; sd.y = sd.fallY; } sd.timer--; return sd.timer > 0; }
   336→   336→      sd.timer--;
   337→   337→      return sd.timer > 0;
   338→   338→    });
   339→   339→
   340→   340→    // Player input
   341→   341→    if (player.alive) {
   342→   342→      const speed = 3.2;
   343→   343→      if (player.fireCooldown > 0) player.fireCooldown--;
   344→   344→      if (g.keys.has("w") || g.keys.has("W")) player.y -= speed;
   345→   345→      if (g.keys.has("s") || g.keys.has("S")) player.y += speed;
   346→   346→      if (g.keys.has("a") || g.keys.has("A")) player.x -= speed;
   347→   347→      if (g.keys.has("d") || g.keys.has("D")) player.x += speed;
   348→   348→      player.x = clamp(player.x, player.radius, WORLD - player.radius);
   349→   349→      player.y = clamp(player.y, player.radius, WORLD - player.radius);
   350→   350→
   351→   351→      const rect = canvas.getBoundingClientRect();
   352→   352→      const scaleX = CANVAS_W / rect.width;
   353→   353→      const scaleY = CANVAS_H / rect.height;
   354→   354→      const worldMouseX = g.mouseX * scaleX + (player.x - CANVAS_W / 2);
   355→   355→      const worldMouseY = g.mouseY * scaleY + (player.y - CANVAS_H / 2);
   356→   356→      player.angle = angleTo(player, { x: worldMouseX, y: worldMouseY });
   357→   357→
   358→   358→      if (g.mouseDown) shoot(player, g.playerIdx, g, player.angle);
   359→   359→    }
   360→   360→
   361→   361→    updateBots(g);
   362→   362→
   363→   363→    // Bullets update
   364→   364→    g.bullets = g.bullets.filter((b) => {
   365→   365→      b.x += b.vx;
   366→   366→      b.y += b.vy;
   367→   367→      b.life--;
   368→   368→      if (b.life <= 0 || b.x < 0 || b.x > WORLD || b.y < 0 || b.y > WORLD) return false;
   369→   369→      for (let i = 0; i < g.entities.length; i++) {
   370→   370→        const e = g.entities[i];
   371→   371→        if (!e.alive || i === b.ownerIdx) continue;
   372→   372→        if (dist(b, e) < e.radius + 3) {
   373→   373→          let dmg = b.damage;
   374→   374→          if (e.shield > 0) { const absorbed = Math.min(e.shield, dmg); e.shield -= absorbed; dmg -= absorbed; }
   375→   375→          e.hp -= dmg;
   376→   376→          g.entities[b.ownerIdx].damageDealt += b.damage;
   377→   377→          g.particles.push(...createExplosion(b.x, b.y, e.color, 4));
   378→   378→          if (e.hp <= 0) {
   379→   379→            e.alive = false;
   380→   380→            g.aliveCount--;
   381→   381→            g.entities[b.ownerIdx].kills++;
   382→   382→            g.particles.push(...createExplosion(e.x, e.y, e.color, 12));
   383→   383→            g.killFeed.push({ killer: g.entities[b.ownerIdx].name, victim: e.name, time: g.frame });
   384→   384→            if (g.killFeed.length > 20) g.killFeed.shift();
   385→   385→            if (e.isPlayer) { endGame(g); return false; }
   386→   386→            if (g.aliveCount <= 1 && g.entities[g.playerIdx].alive) { endGame(g); return false; }
   387→   387→          }
   388→   388→          return false;
   389→   389→        }
   390→   390→      }
   391→   391→      return true;
   392→   392→    });
   393→   393→
   394→   394→    // Zone damage
