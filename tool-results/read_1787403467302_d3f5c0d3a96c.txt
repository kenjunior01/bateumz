   177→/* ─── Floating Particles Component ──────────────────── */
   178→const FloatingParticles = ({ color, count = 30 }: { color: string; count?: number }) => {
   179→  const particles = useMemo(
   180→    () =>
   181→      Array.from({ length: count }, (_, i) => ({
   182→        id: i,
   183→        x: Math.random() * 100,
   184→        y: Math.random() * 100,
   185→        size: 2 + Math.random() * 4,
   186→        duration: 6 + Math.random() * 8,
   187→        delay: Math.random() * 4,
   188→        opacity: 0.15 + Math.random() * 0.35,
   189→      })),
   190→    [count],
   191→  );
   192→  return (
   193→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
   194→      {particles.map((p) => (
   195→        <motion.div
   196→          key={p.id}
   197→          className="absolute rounded-full"
   198→          style={{
   199→            left: `${p.x}%`,
   200→            width: p.size,
   201→            height: p.size,
   202→            backgroundColor: color,
   203→          }}
   204→          animate={{
   205→            y: [0, -30, 0, 20, 0],
   206→            x: [0, 10, -10, 5, 0],
   207→            opacity: [p.opacity, p.opacity * 1.5, p.opacity * 0.5, p.opacity],
   208→          }}
   209→          transition={{
   210→            duration: p.duration,
   211→            repeat: Infinity,
   212→            delay: p.delay,
   213→            ease: "easeInOut",
   214→          }}
   215→        />
   216→      ))}
   217→    </div>
   218→  );

   724→

   794→                </div>
   795→              </div>

   797→          </TabsContent>
   798→
   799→          <TabsContent value="contests" className="mt-6">
   800→            {contests.length > 0 ? (

   802→                {contests.map((c, i) => (
   803→                  <motion.div key={c.id} initial={{ y: 15, opacity: 0 }} whileInView={{ y: 0, opacity: 1 }} viewport={{ once: true }} transition={{ ...springUp, delay: i * 0.04 }}>

   822→                            <Calendar className="h-3 w-3" /> Termina {new Date(c.end_date).toLocaleDateString("pt-PT")}

   824→                        )}

   834→              </div>
   835→            ) : (

   837→                <Megaphone className="h-10 w-10 text-muted-foreground/20 mx-auto mb-4" />

   840→              </div>

   842→          </TabsContent>
   843→
   844→          <TabsContent value="activity" className="mt-6">
   845→            {resolvedId ? <BusinessTimeline businessUserId={resolvedId} /> : (

   847→                <Clock className="h-10 w-10 text-muted-foreground/20 mx-auto mb-4" />

   849→              </div>
   850→            )}
   851→          </TabsContent>
   852→        </Tabs>
   853→      </div>

   855→      <Footer />
   856→    </div>
