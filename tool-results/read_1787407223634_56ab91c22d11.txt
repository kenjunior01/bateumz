   199→function createFighter(classDef: CharacterClass, level: number, name: string): Fighter {
   200→  const hpBonus = (level - 1) * 5;
   201→  const atkBonus = (level - 1) * 2;
   202→  const defBonus = (level - 1) * 1;
   203→  const spdBonus = (level - 1) * 1;
   204→  return {
   205→    className: classDef.id,
   206→    name,
   207→    hp: classDef.hp + hpBonus,
   208→    maxHp: classDef.hp + hpBonus,
   209→    atk: classDef.atk + atkBonus,
   210→    def: classDef.def + defBonus,
   211→    spd: classDef.spd + spdBonus,
   212→    mana: classDef.mana,
   213→    maxMana: classDef.mana,
   214→    level,
   215→    xp: 0,
   216→    xpToNext: level * 100,
   217→  };
   218→}
   219→
   220→function getClassDef(id: ClassName): CharacterClass {
   221→  return CLASS_DEFS.find((c) => c.id === id)!;
   222→}

   224→function calcDamage(
   225→  attacker: Fighter,
   226→  defender: Fighter,
   227→  ability: Ability,
   228→  isPlayer: boolean,
   229→  difficulty: Difficulty,
   230→): { damage: number; isCrit: boolean; isDodge: boolean } {
   231→  const dodgeChance = (defender.spd * 0.8) / 100;
   232→  const isDodge = Math.random() < dodgeChance;
   233→  if (isDodge) return { damage: 0, isCrit: false, isDodge: true };
   234→
   235→  const critChance = (attacker.spd * 0.6) / 100 + (ability.name.includes("Cr\u00EDtic") ? 0.25 : 0);
   236→  const isCrit = Math.random() < critChance;
   237→
   238→  const baseDmg = (attacker.atk * (ability.power / 100)) - (defender.def * 0.3);
   239→  const diffMult = isPlayer ? 1 : DIFFICULTY_MULTIPLIER[difficulty];
   240→  const damage = Math.max(1, Math.floor(baseDmg * diffMult * (isCrit ? 1.75 : 1) * (0.9 + Math.random() * 0.2)));
   241→
   242→  return { damage, isCrit, isDodge: false };
   243→}

   275→    roundsWon: 0,

   283→

   290→
   291→  const resetStats = useCallback(() => {

   310→      const bName = BOT_NAMES[Math.floor(Math.random() * BOT_NAMES.length)];
   311→      const bLevel = diff === "facil" ? Math.max(1, pLevel - 1) : diff === "medio" ? pLevel : pLevel + 1;
   312→
   313→      setBotClassId(bDef.id);
   314→      setBotName(bName);
   315→      setPlayer(createFighter(pDef, pLevel, "Jogador"));
   316→      setBot(createFighter(bDef, bLevel, bName));
   317→      setBattleLog([{ text: `\u2694\uFE0F ${bName} (${bDef.name} Nv.${bLevel}) apareceu!`, type: "system" }]);

   319→      setRound(1);
   320→      setPlayerRounds(0);
   321→      setBotRounds(0);
   322→      setXpGained(0);
   323→      setLeveledUp(false);
   324→      setPlayerBuff({ atk: 0, def: 0 });
   325→      setBotBuff({ atk: 0, def: 0 });
   326→      resetStats();
   327→      setPhase("battle");
   328→    },
   329→    [resetStats],
   330→  );

   336→
   337→  const endRound = useCallback(
   338→    (winner: "player" | "bot") => {
   339→      const newPR = winner === "player" ? playerRounds + 1 : playerRounds;
   340→      const newBR = winner === "bot" ? botRounds + 1 : botRounds;
   341→      setPlayerRounds(newPR);
   342→      setBotRounds(newBR);
   343→
   344→      if (newPR >= ROUNDS_TO_WIN) {
   345→        const xp = Math.floor(BASE_XP * DIFFICULTY_MULTIPLIER[difficulty] * round);
   346→        setXpGained(xp);
   347→        const newTotalXp = (playerLevel - 1) * 100 + xp;
   348→        const newLvl = Math.floor(newTotalXp / 100) + 1;
   349→        if (newLvl > playerLevel) {
   350→          setLeveledUp(true);
   351→          setPlayerLevel(newLvl);
   352→        }
   353→        setStats((s) => ({ ...s, roundsWon: s.roundsWon + 1 }));
   354→        setPhase("matchEnd");
   355→        onScore?.("Jogador", 100 * round);
   356→        return;
   357→      }
   358→      if (newBR >= ROUNDS_TO_WIN) {
   359→        setStats((s) => ({ ...s, roundsLost: s.roundsLost + 1 }));
   360→        setPhase("matchEnd");
   361→        onScore?.("Jogador", 20 * round);
   362→        return;
   363→      }
   364→
   365→      setPhase("roundEnd");
   366→      setTimeout(() => {
   367→        if (player && bot) {
   368→          const pDef = getClassDef(player.className);
   369→          const bDef = getClassDef(bot.className);
   370→          setPlayer((p) => (p ? { ...p, hp: p.maxHp, mana: p.maxMana } : p));
   371→          setBot((b) => (b ? { ...b, hp: b.maxHp, mana: b.maxMana } : b));
   372→          setPlayerBuff({ atk: 0, def: 0 });
   373→          setBotBuff({ atk: 0, def: 0 });
   374→          setBattleLog([
   375→            { text: `\uD83D\uDCAA Rodada ${round + 1} come\u00E7ou!`, type: "system" },
   376→          ]);
   377→          setRound((r) => r + 1);
   378→          setPlayerTurn(true);
   379→          setPhase("battle");
   380→        }
   381→      }, 2500);
   382→    },
   383→    [playerRounds, botRounds, player, bot, playerLevel, difficulty, round, onScore],
   384→  );
   385→
   386→  const botActRef = useRef<(b: Fighter, p: Fighter) => void>(() => {});
   387→
   388→  const executeAbility = useCallback(
   389→    (attacker: Fighter, defender: Fighter, ability: Ability, isPlayer: boolean) => {
   390→      setTurnAnim(true);
   391→      setTimeout(() => setTurnAnim(false), 500);
   392→
   393→      if (isPlayer) {
   394→        setPlayer((p) => (p ? { ...p, mana: Math.max(0, p.mana - ability.manaCost) } : p));
   395→        setStats((s) => ({ ...s, abilitiesUsed: s.abilitiesUsed + 1 }));
   396→      } else {
   397→        setBot((b) => (b ? { ...b, mana: Math.max(0, b.mana - ability.manaCost) } : b));
   398→      }
   399→

   404→          setPlayer((p) => (p ? { ...p, hp: Math.min(p.maxHp, p.hp + actualHeal) } : p));
   405→          setStats((s) => ({ ...s, heals: s.heals + 1 }));
   406→        } else {
   407→          setBot((b) => (b ? { ...b, hp: Math.min(b.maxHp, b.hp + actualHeal) } : b));
   408→        }
   409→        setBattleLog((log) => [
   410→          ...log,
   411→          { text: `${attacker.name} usou ${ability.icon} ${ability.name} e recuperou ${actualHeal} HP!`, type: isPlayer ? "player" : "bot" },
   412→        ]);

   414→        if (isPlayer) {
   415→          if (ability.name.includes("Escudo")) {
   416→            setPlayerBuff((buf) => ({ ...buf, def: buf.def + ability.power }));
   417→            setBattleLog((log) => [...log, { text: `${attacker.name} usou ${ability.icon} ${ability.name}! DEF +${ability.power}`, type: "player" }]);
   418→          } else {
   419→            setPlayerBuff((buf) => ({ ...buf, atk: buf.atk + ability.power }));
   420→            setBattleLog((log) => [...log, { text: `${attacker.name} usou ${ability.icon} ${ability.name}! ATK +${ability.power}`, type: "player" }]);
   421→          }
   422→        } else {
   423→          if (ability.name.includes("Escudo")) {
   424→            setBotBuff((buf) => ({ ...buf, def: buf.def + ability.power }));
   425→            setBattleLog((log) => [...log, { text: `${attacker.name} usou ${ability.icon} ${ability.name}! DEF +${ability.power}`, type: "bot" }]);
   426→          } else {
   427→            setBotBuff((buf) => ({ ...buf, atk: buf.atk + ability.power }));
   428→            setBattleLog((log) => [...log, { text: `${attacker.name} usou ${ability.icon} ${ability.name}! ATK +${ability.power}`, type: "bot" }]);
   429→          }
   430→        }
   431→      } else {
   432→        const effectiveAtk = isPlayer
   433→          ? { ...attacker, atk: attacker.atk + playerBuff.atk }
   434→          : { ...attacker, atk: attacker.atk + botBuff.atk };
   435→        const effectiveDef = isPlayer
   436→          ? { ...defender, def: defender.def + botBuff.def }
   437→          : { ...defender, def: defender.def + playerBuff.def };
   438→
   439→        const { damage, isCrit, isDodge } = calcDamage(effectiveAtk, effectiveDef, ability, isPlayer, difficulty);
   440→
   441→        if (isDodge) {
   442→          setBattleLog((log) => [
   443→            ...log,
   444→            { text: `${defender.name} esquivou do ${ability.name}!`, type: isPlayer ? "player" : "bot", isDodge: true },
   445→          ]);
   446→          if (isPlayer) {
   447→            setStats((s) => ({ ...s, dodges: s.dodges + 1 }));
   448→          }
   449→        } else {
   450→          const critText = isCrit ? " \u2B50 CR\u00CDTICO!" : "";
   451→          if (isPlayer) {
   452→            setBot((b) => (b ? { ...b, hp: Math.max(0, b.hp - damage) } : b));
   453→            setStats((s) => ({ ...s, damageDealt: s.damageDealt + damage, critsLanded: s.critsLanded + (isCrit ? 1 : 0) }));
   454→          } else {
   455→            setPlayer((p) => (p ? { ...p, hp: Math.max(0, p.hp - damage) } : p));
   456→            setStats((s) => ({ ...s, damageTaken: s.damageTaken + damage, critsReceived: s.critsReceived + (isCrit ? 1 : 0) }));
   457→          }
   458→          setBattleLog((log) => [
   459→            ...log,
   460→            {
   461→              text: `${attacker.name} usou ${ability.icon} ${ability.name} e causou ${damage} de dano!${critText}`,
   462→              type: isPlayer ? "player" : "bot",
   463→              isCrit,
   464→            },
   465→          ]);
   466→        }
   467→      }
   468→
   469→      setTimeout(() => {
   470→        setBot((currentBot) => {
   471→          setPlayer((currentPlayer) => {
   472→            const defHp = isPlayer ? currentBot?.hp ?? 0 : currentPlayer?.hp ?? 0;
   473→            if (defHp <= 0) {
   474→              setBattleLog((log) => [...log, { text: `${isPlayer ? currentBot?.name : currentPlayer?.name} foi derrotado!`, type: "system" }]);
   475→              endRound(isPlayer ? "player" : "bot");
   476→            } else if (isPlayer) {
   477→              setPlayerTurn(false);
   478→              if (currentBot) {
   479→                setBot((b) => (b ? { ...b, mana: Math.min(b.maxMana, b.mana + MANA_REGEN) } : b));
   480→                botActRef.current({ ...currentBot, mana: Math.min(currentBot.maxMana, currentBot.mana + MANA_REGEN) }, currentPlayer!);
   481→              }
   482→            } else {
   483→              setPlayerTurn(true);
   484→              if (currentPlayer) {
   485→                setPlayer((p) => (p ? { ...p, mana: Math.min(p.maxMana, p.mana + MANA_REGEN) } : p));
   486→              }
   487→            }

   965→  // ===== RENDER: MATCH END =====
