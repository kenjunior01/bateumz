   199→   199→function createFighter(classDef: CharacterClass, level: number, name: string): Fighter {
   200→   200→  const hpBonus = (level - 1) * 5;
   201→   201→  const atkBonus = (level - 1) * 2;
   202→   202→  const defBonus = (level - 1) * 1;
   203→   203→  const spdBonus = (level - 1) * 1;
   204→   204→  return {
   205→   205→    className: classDef.id,
   206→   206→    name,
   207→   207→    hp: classDef.hp + hpBonus,
   208→   208→    maxHp: classDef.hp + hpBonus,
   209→   209→    atk: classDef.atk + atkBonus,
   210→   210→    def: classDef.def + defBonus,
   211→   211→    spd: classDef.spd + spdBonus,
   212→   212→    mana: classDef.mana,
   213→   213→    maxMana: classDef.mana,
   214→   214→    level,
   215→   215→    xp: 0,
   216→   216→    xpToNext: level * 100,
   217→   217→  };
   218→   218→}

   224→   224→function calcDamage(
   225→   225→  attacker: Fighter,
   226→   226→  defender: Fighter,
   227→   227→  ability: Ability,
   228→   228→  isPlayer: boolean,
   229→   229→  difficulty: Difficulty,
   230→   230→): { damage: number; isCrit: boolean; isDodge: boolean } {
   231→   231→  const dodgeChance = (defender.spd * 0.8) / 100;
   232→   232→  const isDodge = Math.random() < dodgeChance;
   233→   233→  if (isDodge) return { damage: 0, isCrit: false, isDodge: true };
   234→   234→
   235→   235→  const critChance = (attacker.spd * 0.6) / 100 + (ability.name.includes("Cr\u00EDtic") ? 0.25 : 0);
   236→   236→  const isCrit = Math.random() < critChance;
   237→   237→
   238→   238→  const baseDmg = (attacker.atk * (ability.power / 100)) - (defender.def * 0.3);
   239→   239→  const diffMult = isPlayer ? 1 : DIFFICULTY_MULTIPLIER[difficulty];
   240→   240→  const damage = Math.max(1, Math.floor(baseDmg * diffMult * (isCrit ? 1.75 : 1) * (0.9 + Math.random() * 0.2)));
   241→   241→
   242→   242→  return { damage, isCrit, isDodge: false };
   243→   243→}
