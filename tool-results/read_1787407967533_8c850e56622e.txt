     1→     1→import { useState, useEffect, useRef, useCallback } from "react";
     2→     2→import { motion, AnimatePresence } from "framer-motion";
     3→     3→import { Play, RotateCcw, Trophy, User, Bot } from "lucide-react";
     4→     4→import confetti from "canvas-confetti";
     5→     5→
     6→     6→interface Props {
     7→     7→  onScore?: (name: string, score: number) => void;
     8→     8→  liveCode?: string;
     9→     9→}
    10→    10→
    11→    11→type Phase = "idle" | "countdown" | "playing" | "done";
    12→    12→type Mode = "bot" | "vs";
    13→    13→
    14→    14→type MoleState = "hidden" | "showing" | "hit" | "missed";
    15→    15→
    16→    16→interface FloatingScore {
    17→    17→  id: number;
    18→    18→  value: number;
    19→    19→  row: number;
    20→    20→  col: number;
    21→    21→  player: 1 | 2;
    22→    22→  combo: number;
    23→    23→}
    24→    24→
    25→    25→interface HitParticle {
    26→    26→  id: number;
    27→    27→  x: number;
    28→    28→  y: number;
    29→    29→  emoji: string;
    30→    30→}
    31→    31→
    32→    32→const GRID_SIZE = 9; // 3x3
    33→    33→const GAME_DURATION = 30;
    34→    34→const MOLE_SHOW_MIN = 600;
    35→    35→const MOLE_SHOW_MAX = 1400;
    36→    36→const MOLE_INTERVAL_MIN = 400;
    37→    37→const MOLE_INTERVAL_MAX = 900;
    38→    38→const HIT_POINTS = 1;
    39→    39→
    40→    40→const MOLE_EMOJIS = ["🐹", "🦁", "🐾", "🦇", "🐯", "🐰"];
    41→    41→const HIT_EMOJIS = ["💥","💥","💥","⚡","🔥","✨"];
    42→    42→
    43→    43→const WhackAMole = ({ onScore, liveCode }: Props) => {
    44→    44→  const [phase, setPhase] = useState<Phase>("idle");
    45→    45→  const [mode, setMode] = useState<Mode>("bot");
    46→    46→  const [p1Score, setP1Score] = useState(0);
    47→    47→  const [p2Score, setP2Score] = useState(0);
    48→    48→  const [timeLeft, setTimeLeft] = useState(GAME_DURATION);
    49→    49→  const [countdown, setCountdown] = useState(3);
    50→    50→  const [winner, setWinner] = useState("");
    51→    51→  const [p1Name, setP1Name] = useState("Jogador 1");
    52→    52→  const [p2Name, setP2Name] = useState("Jogador 2");
    53→    53→  const [p1Combo, setP1Combo] = useState(0);
    54→    54→  const [p2Combo, setP2Combo] = useState(0);
    55→    55→  const [lastHit, setLastHit] = useState<"p1" | "p2" | null>(null);
    56→    56→  const [screenShake, setScreenShake] = useState(false);
    57→    57→  const [floatingScores, setFloatingScores] = useState<FloatingScore[]>([]);
    58→    58→  const [hitParticles, setHitParticles] = useState<HitParticle[]>([]);
    59→    59→  const [p1BestCombo, setP1BestCombo] = useState(0);
    60→    60→  const [p2BestCombo, setP2BestCombo] = useState(0);
    61→    61→  const [totalHits, setTotalHits] = useState(0);
    62→    62→
    63→    63→  // Each cell: which player's mole is showing (null = empty)
    64→    64→  const [cells, setCells] = useState<(null | { player: 1 | 2; state: MoleState; emoji: string; id: number })[]>(
    65→    65→    Array(GRID_SIZE).fill(null)
    66→    66→  );
    67→    67→
    68→    68→  const moleIdRef = useRef(0);
    69→    69→  const timerRef = useRef<number>(0);
    70→    70→  const spawnRef = useRef<number>(0);
    71→    71→  const p2ScoreRef = useRef(0);
    72→    72→  const phaseRef = useRef<Phase>("idle");
    73→    73→  const modeRef = useRef<Mode>("bot");
    74→    74→  const pendingTimeoutsRef = useRef<number[]>([]);
    75→    75→  const floatIdRef = useRef(0);
    76→    76→  const particleIdRef = useRef(0);
    77→    77→  const p1ComboRef = useRef(0);
    78→    78→  const p2ComboRef = useRef(0);
    79→    79→  const totalHitsRef = useRef(0);
    80→    80→
    81→    81→  const trackTimeout = useCallback((id: number) => {
    82→    82→    pendingTimeoutsRef.current.push(id);
    83→    83→  }, []);
    84→    84→
    85→    85→  const clearAllPendingTimeouts = useCallback(() => {
    86→    86→    pendingTimeoutsRef.current.forEach(clearTimeout);
    87→    87→    pendingTimeoutsRef.current = [];
    88→    88→    clearTimeout(spawnRef.current);
    89→    89→    clearInterval(timerRef.current);
    90→    90→  }, []);
    91→    91→
    92→    92→  useEffect(() => { modeRef.current = mode; }, [mode]);
    93→    93→  useEffect(() => { phaseRef.current = phase; }, [phase]);
    94→    94→  useEffect(() => { p2ScoreRef.current = p2Score; }, [p2Score]);
    95→    95→
    96→    96→  const triggerShake = useCallback(() => {
    97→    97→    setScreenShake(true);
    98→    98→    setTimeout(() => setScreenShake(false), 150);
    99→    99→  }, []);
   100→   100→
   101→   101→  const spawnFloatingScore = useCallback((row: number, col: number, player: 1 | 2, value: number, combo: number) => {
   102→   102→    const id = ++floatIdRef.current;
   103→   103→    setFloatingScores(prev => [...prev, { id, value, row, col, player, combo }]);
   104→   104→    setTimeout(() => {
   105→   105→      setFloatingScores(prev => prev.filter(f => f.id !== id));
   106→   106→    }, 800);
   107→   107→  }, []);
   108→   108→
   109→   109→  const spawnHitParticles = useCallback((row: number, col: number) => {
   110→   110→    const newParticles: HitParticle[] = [];
   111→   111→    for (let i = 0; i < 4; i++) {
   112→   112→      newParticles.push({
   113→   113→        id: ++particleIdRef.current,
   114→   114→        x: col * 33.3 + Math.random() * 33.3,
   115→   115→        y: row * 33.3 + Math.random() * 33.3,
   116→   116→        emoji: HIT_EMOJIS[Math.floor(Math.random() * HIT_EMOJIS.length)],
   117→   117→      });
   118→   118→    }
   119→   119→    setHitParticles(prev => [...prev, ...newParticles]);
   120→   120→    setTimeout(() => {
   121→   121→      setHitParticles(prev => prev.filter(p => !newParticles.find(np => np.id === p.id)));
   122→   122→    }, 700);
   123→   123→  }, []);
   124→   124→
   125→   125→  const playTapSound = useCallback((combo: number) => {
   126→   126→    try {
   127→   127→      const ctx = new AudioContext();
   128→   128→      const osc = ctx.createOscillator();
   129→   129→      const gain = ctx.createGain();
   130→   130→      osc.connect(gain);
   131→   131→      gain.connect(ctx.destination);
   132→   132→      osc.frequency.value = 300 + Math.min(combo, 10) * 60;
   133→   133→      osc.type = 'sine';
   134→   134→      gain.gain.value = 0.08;
   135→   135→      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
   136→   136→      osc.start();
   137→   137→      osc.stop(ctx.currentTime + 0.1);
   138→   138→    } catch {}
   139→   139→  }, []);
   140→   140→
   141→   141→  const hideMole = useCallback((index: number) => {
   142→   142→    setCells((prev) => {
   143→   143→      const next = [...prev];
   144→   144→      if (next[index] && next[index]!.state === "showing") {
   145→   145→        next[index] = { ...next[index]!, state: "missed" };
   146→   146→        const tid = window.setTimeout(() => {
   147→   147→          setCells((p) => {
   148→   148→            const n = [...p];
   149→   149→            if (n[index] && n[index]!.state === "missed") n[index] = null;
   150→   150→            return n;
   151→   151→          });
   152→   152→        }, 300);
   153→   153→        trackTimeout(tid);
   154→   154→      }
   155→   155→      return next;
   156→   156→    });
   157→   157→  }, [trackTimeout]);
   158→   158→
   159→   159→  const spawnMole = useCallback(() => {
   160→   160→    if (phaseRef.current !== "playing") return;
   161→   161→
   162→   162→    setCells((prev) => {
   163→   163→      const emptyIndices = prev.reduce<number[]>((acc, c, i) => { if (!c) acc.push(i); return acc; }, []);
   164→   164→      if (emptyIndices.length === 0) return prev;
   165→   165→
   166→   166→      const next = [...prev];
   167→   167→      const idx = emptyIndices[Math.floor(Math.random() * emptyIndices.length)];
   168→   168→      const emoji = MOLE_EMOJIS[Math.floor(Math.random() * MOLE_EMOJIS.length)];
   169→   169→      const id = ++moleIdRef.current;
   170→   170→
   171→   171→      next[idx] = { player: 1, state: "showing", emoji, id };
   172→   172→
   173→   173→      const showTime = MOLE_SHOW_MIN + Math.random() * (MOLE_SHOW_MAX - MOLE_SHOW_MIN);
   174→   174→      const tid = window.setTimeout(() => hideMole(idx), showTime);
   175→   175→      trackTimeout(tid);
   176→   176→
   177→   177→      return next;
   178→   178→    });
   179→   179→
   180→   180→    const nextInterval = MOLE_INTERVAL_MIN + Math.random() * (MOLE_INTERVAL_MAX - MOLE_INTERVAL_MIN);
   181→   181→    spawnRef.current = window.setTimeout(spawnMole, nextInterval);
   182→   182→  }, [hideMole, trackTimeout]);
   183→   183→
   184→   184→  const botTick = useCallback(() => {
   185→   185→    if (phaseRef.current !== "playing" || modeRef.current !== "bot") return;
   186→   186→
   187→   187→    setCells((prev) => {
   188→   188→      const showingIdx = prev.findIndex((c) => c && c.state === "showing" && c.player === 1);
   189→   189→      if (showingIdx === -1) return prev;
   190→   190→
   191→   191→      if (Math.random() < 0.6) {
   192→   192→        const next = [...prev];
   193→   193→        next[showingIdx] = { ...next[showingIdx]!, state: "hit" };
   194→   194→        p2ScoreRef.current += 1;
   195→   195→        const row = Math.floor(showingIdx / 3);
   196→   196→        const col = showingIdx % 3;
   197→   197→        spawnHitParticles(row, col);
   198→   198→        spawnFloatingScore(row, col, 2, 1, p2ComboRef.current + 1);
   199→   199→        p2ComboRef.current++;
   200→   200→        if (p2ComboRef.current > 2) setP2BestCombo(prev2 => Math.max(prev2, p2ComboRef.current));
   201→   201→        const tid = window.setTimeout(() => {
   202→   202→          setP2Score(p2ScoreRef.current);
   203→   203→          setP2Combo(p2ComboRef.current);
   204→   204→          setCells((p) => {
   205→   205→            const n = [...p];
   206→   206→            if (n[showingIdx] && n[showingIdx]!.state === "hit") n[showingIdx] = null;
   207→   207→            return n;
   208→   208→          });
   209→   209→        }, 300);
   210→   210→        trackTimeout(tid);
   211→   211→        return next;
   212→   212→      }
   213→   213→      return prev;
   214→   214→    });
   215→   215→  }, [trackTimeout, spawnHitParticles, spawnFloatingScore]);
   216→   216→
   217→   217→  useEffect(() => {
   218→   218→    if (phase !== "playing") return;
   219→   219→    const botInterval = window.setInterval(botTick, 700 + Math.random() * 500);
   220→   220→    return () => clearInterval(botInterval);
   221→   221→  }, [phase, botTick]);
   222→   222→
   223→   223→  const hitCell = useCallback((index: number, player: 1 | 2) => {
   224→   224→    setCells((prev) => {
   225→   225→      const cell = prev[index];
   226→   226→      if (!cell || cell.state !== "showing") return prev;
   227→   227→
   228→   228→      const next = [...prev];
   229→   229→      next[index] = { ...cell, state: "hit" };
   230→   230→
   231→   231→      const row = Math.floor(index / 3);
   232→   232→      const col = index % 3;
   233→   233→
   234→   234→      if (player === 1) {
   235→   235→        p1ComboRef.current++;
   236→   236→        p2ComboRef.current = 0;
   237→   237→        setP2Combo(0);
   238→   238→        const currentCombo = p1ComboRef.current;
   239→   239→        if (currentCombo > 2) setP1BestCombo(prev1 => Math.max(prev1, currentCombo));
   240→   240→        setP1Combo(currentCombo);
   241→   241→        const bonus = Math.min(Math.floor(currentCombo / 3), 3);
   242→   242→        const points = HIT_POINTS + bonus;
   243→   243→        setP1Score((s) => s + points);
   244→   244→        setLastHit("p1");
   245→   245→        playTapSound(currentCombo);
   246→   246→        spawnFloatingScore(row, col, 1, points, currentCombo);
   247→   247→        if (currentCombo >= 3) triggerShake();
   248→   248→        if (currentCombo >= 5) {
   249→   249→          confetti({ particleCount: 15, spread: 30, origin: { x: 0.2 + col * 0.3, y: 0.4 + row * 0.2 }, colors: ['#3b82f6', '#8b5cf6', '#06b6d4'] });
   250→   250→        }
   251→   251→      } else {
   252→   252→        p2ComboRef.current++;
   253→   253→        p1ComboRef.current = 0;
   254→   254→        setP1Combo(0);
   255→   255→        const currentCombo = p2ComboRef.current;
   256→   256→        if (currentCombo > 2) setP2BestCombo(prev2 => Math.max(prev2, currentCombo));
   257→   257→        setP2Combo(currentCombo);
   258→   258→        setP2Score((s) => s + HIT_POINTS);
   259→   259→        setLastHit("p2");
   260→   260→        spawnFloatingScore(row, col, 2, 1, currentCombo);
   261→   261→      }
   262→   262→
   263→   263→      spawnHitParticles(row, col);
   264→   264→      totalHitsRef.current++;
   265→   265→      setTotalHits(totalHitsRef.current);
   266→   266→
   267→   267→      const tid = window.setTimeout(() => {
   268→   268→        setCells((p) => {
   269→   269→          const n = [...p];
   270→   270→          if (n[index] && n[index]!.state === "hit") n[index] = null;
   271→   271→          return n;
   272→   272→        });
   273→   273→      }, 300);
   274→   274→      trackTimeout(tid);
   275→   275→
   276→   276→      return next;
   277→   277→    });
   278→   278→  }, [trackTimeout, triggerShake, playTapSound, spawnFloatingScore, spawnHitParticles]);
   279→   279→
   280→   280→  const startGame = () => {
   281→   281→    clearAllPendingTimeouts();
   282→   282→    setP1Score(0); setP2Score(0);
   283→   283→    setTimeLeft(GAME_DURATION);
   284→   284→    setP1Combo(0); setP2Combo(0);
   285→   285→    p1ComboRef.current = 0; p2ComboRef.current = 0;
   286→   286→    totalHitsRef.current = 0;
   287→   287→    setLastHit(null);
   288→   288→    setWinner("");
   289→   289→    setP1BestCombo(0); setP2BestCombo(0);
   290→   290→    setTotalHits(0);
   291→   291→    setCells(Array(GRID_SIZE).fill(null));
   292→   292→    setFloatingScores([]);
   293→   293→    setHitParticles([]);
   294→   294→    setPhase("countdown");
   295→   295→    setCountdown(3);
   296→   296→  };
   297→   297→
   298→   298→  // Countdown
   299→   299→  useEffect(() => {
   300→   300→    if (phase !== "countdown") return;
   301→   301→    if (countdown <= 0) {
   302→   302→      setPhase("playing");
   303→   303→      return;
   304→   304→    }
   305→   305→    const t = setTimeout(() => setCountdown((c) => c - 1), 800);
   306→   306→    return () => clearTimeout(t);
   307→   307→  }, [phase, countdown]);
   308→   308→
   309→   309→  // Game timer
   310→   310→  useEffect(() => {
   311→   311→    if (phase !== "playing") return;
   312→   312→    const start = Date.now();
   313→   313→    timerRef.current = window.setInterval(() => {
   314→   314→      const elapsed = (Date.now() - start) / 1000;
   315→   315→      const remaining = Math.max(0, GAME_DURATION - elapsed);
   316→   316→      setTimeLeft(Number(remaining.toFixed(1)));
   317→   317→      if (remaining <= 0) {
   318→   318→        clearInterval(timerRef.current);
   319→   319→        clearTimeout(spawnRef.current);
   320→   320→        setPhase("done");
   321→   321→      }
   322→   322→    }, 50);
   323→   323→
   324→   324→    const firstSpawn = window.setTimeout(spawnMole, 500);
   325→   325→
   326→   326→    return () => {
   327→   327→      clearInterval(timerRef.current);
   328→   328→      clearTimeout(spawnRef.current);
   329→   329→      clearTimeout(firstSpawn);
   330→   330→    };
   331→   331→  }, [phase, spawnMole]);
   332→   332→
   333→   333→  // Determine winner
   334→   334→  useEffect(() => {
   335→   335→    if (phase !== "done") return;
   336→   336→    if (p1Score > p2Score) {
   337→   337→      const w = p1Name;
   338→   338→      setWinner(w);
   339→   339→      onScore?.(w, p1Score);
   340→   340→      confetti({ particleCount: 150, spread: 100, origin: { x: 0.3, y: 0.6 } });
   341→   341→    } else if (p2Score > p1Score) {
   342→   342→      const w = mode === "bot" ? "Bot" : p2Name;
   343→   343→      setWinner(w);
   344→   344→      onScore?.(w, p2Score);
   345→   345→      confetti({ particleCount: 150, spread: 100, origin: { x: 0.7, y: 0.6 } });
   346→   346→    }
   347→   347→  }, [phase]); // eslint-disable-line react-hooks/exhaustive-deps
   348→   348→
   349→   349→  const getCellVisual = (cell: typeof cells[0]) => {
   350→   350→    if (!cell) return null;
   351→   351→    const colors = cell.player === 1
   352→   352→      ? { bg: "from-blue-500 to-indigo-600", border: "border-blue-400" }
   353→   353→      : { bg: "from-amber-500 to-orange-600", border: "border-amber-400" };
   354→   354→    return { ...cell, colors };
   355→   355→  };
   356→   356→
   357→   357→  return (
   358→   358→    <div className={`max-w-lg mx-auto flex flex-col items-center gap-4 transition-transform ${screenShake ? "scale-[0.98]" : ""}`}>
   359→   359→      {phase === "idle" && (
   360→   360→        <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="w-full flex flex-col gap-4">
   361→   361→          {/* Epic Header */}
   362→   362→          <div className="text-center relative">
   363→   363→            <div className="absolute inset-0 bg-gradient-to-b from-amber-500/10 via-transparent to-transparent rounded-2xl pointer-events-none" />
   364→   364→            <motion.div animate={{ y: [0, -6, 0] }} transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}>
   365→   365→              <span className="text-6xl block mb-2">🐹</span>
   366→   366→            </motion.div>
   367→   367→            <h2 className="text-2xl font-black bg-gradient-to-r from-amber-300 via-yellow-400 to-orange-500 bg-clip-text text-transparent">
   368→   368→              Whack-a-Mole
   369→   369→            </h2>
   370→   370→            <p className="text-[11px] text-muted-foreground mt-1">Bate nas criaturas o mais rapido que puderes!</p>
   371→   371→            <div className="flex justify-center gap-3 mt-2 text-[10px] text-muted-foreground">
   372→   372→              <span className="flex items-center gap-1">🔥 Combos</span>
   373→   373→              <span className="flex items-center gap-1">⚡ Bonus</span>
   374→   374→              <span className="flex items-center gap-1">🏆 {GAME_DURATION}s</span>
   375→   375→            </div>
   376→   376→          </div>
   377→   377→
   378→   378→          <div className="flex gap-2 w-full">
   379→   379→            <button
   380→   380→              onClick={() => setMode("bot")}
   381→   381→              className={`flex-1 flex items-center justify-center gap-2 py-2.5 rounded-xl text-xs font-bold transition-all ${mode === "bot" ? "bg-primary text-primary-foreground shadow-lg shadow-primary/20" : "bg-card border border-border hover:border-primary/40"}`}
   382→   382→            >
   383→   383→              <Bot className="h-4 w-4" /> vs Bot
   384→   384→            </button>
   385→   385→            <button
   386→   386→              onClick={() => setMode("vs")}
   387→   387→              className={`flex-1 flex items-center justify-center gap-2 py-2.5 rounded-xl text-xs font-bold transition-all ${mode === "vs" ? "bg-primary text-primary-foreground shadow-lg shadow-primary/20" : "bg-card border border-border hover:border-primary/40"}`}
   388→   388→            >
   389→   389→              <User className="h-4 w-4" /> 2 Jogadores
   390→   390→            </button>
   391→   391→          </div>
   392→   392→
   393→   393→          {mode === "vs" && (
   394→   394→            <div className="grid grid-cols-2 gap-2 w-full">
   395→   395→              <input value={p1Name} onChange={(e) => setP1Name(e.target.value)} placeholder="Jogador 1" className="px-3 py-2 rounded-xl bg-card border border-border text-sm text-center" />
   396→   396→              <input value={p2Name} onChange={(e) => setP2Name(e.target.value)} placeholder="Jogador 2" className="px-3 py-2 rounded-xl bg-card border border-border text-sm text-center" />
   397→   397→            </div>
   398→   398→          )}
   399→   399→
   400→   400→          <motion.button
   401→   401→            whileHover={{ scale: 1.02 }}
   402→   402→            whileTap={{ scale: 0.97 }}
   403→   403→            onClick={startGame}
   404→   404→            className="w-full py-3.5 rounded-xl bg-gradient-to-r from-emerald-500 to-green-600 text-white font-bold text-sm shadow-lg shadow-emerald-500/20 flex items-center justify-center gap-2 hover:shadow-xl hover:shadow-emerald-500/30 transition-all"
   405→   405→          >
   406→   406→            <Play className="h-4 w-4 fill-current" /> Começar Jogo
   407→   407→          </motion.button>
