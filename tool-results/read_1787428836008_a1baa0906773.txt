    43→    43→function generateProblem(round: number): MathProblem {
    44→    44→  const roundIndex = round - 1;
    45→    45→  let difficulty: Difficulty;
    46→    46→  let expression: string;
    47→    47→  let answer: number;
    48→    48→
    49→    49→  if (roundIndex < 3) {
    50→    50→    difficulty = "Fácil";
    51→    51→    const ops = ["+", "-"];
    52→    52→    const op = ops[Math.floor(Math.random() * ops.length)];
    53→    53→    const a = Math.floor(Math.random() * 20) + 1;
    54→    54→    const b = Math.floor(Math.random() * 20) + 1;
    55→    55→    if (op === "+") {
    56→    56→      expression = `${a} + ${b}`;
    57→    57→      answer = a + b;
    58→    58→    } else {
    59→    59→      const big = Math.max(a, b);
    60→    60→      const small = Math.min(a, b);
    61→    61→      expression = `${big} - ${small}`;
    62→    62→      answer = big - small;
    63→    63→    }
    64→    64→  } else if (roundIndex < 6) {
    65→    65→    difficulty = "Médio";
    66→    66→    if (Math.random() < 0.5) {
    67→    67→      const a = Math.floor(Math.random() * 12) + 1;
    68→    68→      const b = Math.floor(Math.random() * 12) + 1;
    69→    69→      expression = `${a} × ${b}`;
    70→    70→      answer = a * b;
    71→    71→    } else {
    72→    72→      const a = Math.floor(Math.random() * 41) + 10;
    73→    73→      const b = Math.floor(Math.random() * 41) + 10;
    74→    74→      expression = `${a} + ${b}`;
    75→    75→      answer = a + b;
    76→    76→    }
    77→    77→  } else {
    78→    78→    difficulty = "Difícil";
    79→    79→    const twoStepType = Math.floor(Math.random() * 4);
    80→    80→    if (twoStepType === 0) {
    81→    81→      const a = Math.floor(Math.random() * 8) + 2;
    82→    82→      const b = Math.floor(Math.random() * 8) + 2;
    83→    83→      const c = Math.floor(Math.random() * 15) + 1;
    84→    84→      expression = `${a} × ${b} + ${c}`;
    85→    85→      answer = a * b + c;
    86→    86→    } else if (twoStepType === 1) {
    87→    87→      const a = Math.floor(Math.random() * 8) + 2;
    88→    88→      const b = Math.floor(Math.random() * 8) + 2;
    89→    89→      const c = Math.floor(Math.random() * (a * b - 1)) + 1;
    90→    90→      expression = `${a * b} - ${c}`;
    91→    91→      answer = a * b - c;
    92→    92→    } else if (twoStepType === 2) {
    93→    93→      const a = Math.floor(Math.random() * 10) + 2;
    94→    94→      const b = Math.floor(Math.random() * 10) + 2;
    95→    95→      const c = Math.floor(Math.random() * 5) + 1;
    96→    96→      const sum = a + b;
    97→    97→      expression = `(${a} + ${b}) × ${c}`;
    98→    98→      answer = sum * c;
    99→    99→    } else {
   100→   100→      const a = Math.floor(Math.random() * 50) + 20;
   101→   101→      const b = Math.floor(Math.random() * 30) + 10;
   102→   102→      const c = Math.floor(Math.random() * 8) + 2;
   103→   103→      expression = `${a} + ${b} × ${c}`;
   104→   104→      answer = a + b * c;
   105→   105→    }
   106→   106→  }
   107→   107→
   108→   108→  return { expression, answer, difficulty };
   109→   109→}
