    64→const checkWin = (b: (Cell | MiniResult)[]): { winner: Player | null; line: number[] | null } => {
    65→  for (const line of WIN_LINES) {
    66→    const [a, bb, c] = line;
    67→    if (b[a] && b[a] !== "draw" && b[a] === b[bb] && b[a] === b[c]) {
    68→      return { winner: b[a] as Player, line };
    69→    }
    70→  }
    71→  return { winner: null, line: null };
    72→};

    95→/** Return every legal (board, cell) pair for the given game state. */
    96→const getValidMoves = (cells: Cell[], miniResults: MiniResult[], targetBoard: number | null): AIMove[] => {
    97→  const moves: AIMove[] = [];
    98→
    99→  // If target board is playable, restrict to it
   100→  const candidateBoards =
   101→    targetBoard !== null &&
   102→    miniResults[targetBoard] === null &&
   103→    !isBoardFull(cells, targetBoard)
   104→      ? [targetBoard]
   105→      : [0, 1, 2, 3, 4, 5, 6, 7, 8];
   106→
   107→  for (const bi of candidateBoards) {
   108→    if (miniResults[bi] !== null) continue;
   109→    const small = getSmallBoardCells(cells, bi);
   110→    for (let ci = 0; ci < 9; ci++) {
   111→      if (small[ci] === null) moves.push({ boardIdx: bi, cellIdx: ci });
   112→    }
   113→  }
   114→  return moves;
   115→};
   116→
   117→/** After a move at `cellIdx`, compute the next target board. */
   118→const getNextTarget = (miniResults: MiniResult[], cells: Cell[], cellIdx: number): number | null => {
   119→  if (miniResults[cellIdx] !== null) return null;
   120→  if (isBoardFull(cells, cellIdx)) return null;
   121→  return cellIdx;
   122→};
   123→
   124→/** Heuristic evaluation of a single unresolved mini-board for a given player. */
   125→const evaluateOpenBoard = (smallBoard: Cell[], player: Player): number => {
   126→  let score = 0;
   127→  for (const line of WIN_LINES) {
   128→    const [a, b, c] = line;
   129→    const vals = [smallBoard[a], smallBoard[b], smallBoard[c]];
   130→    const pCount = vals.filter((v) => v === player).length;
   131→    const eCount = vals.filter((v) => v === null).length;
   132→    const oppCount = vals.filter((v) => v !== null && v !== player).length;
   133→
   134→    if (oppCount === 0) {
   135→      // line is still alive for `player`
   136→      if (pCount === 3) score += 100;
   137→      else if (pCount === 2 && eCount === 1) score += 10;
   138→      else if (pCount === 1 && eCount === 2) score += 1;
   139→    } else {
   140→      // line is partially blocked
   141→      if (pCount === 2 && oppCount === 1) score += 3; // already blocked, low value
   142→    }
   143→  }
   144→  return score;
   145→};
   146→
   147→/** Heuristic evaluation of the overall position (from Bot's "O" perspective). */
   148→const evaluatePosition = (cells: Cell[], miniResults: MiniResult[]): number => {
   149→  // Terminal checks
   150→  const bigResult = checkWin(miniResults);
   151→  if (bigResult.winner === "O") return 100_000;
   152→  if (bigResult.winner === "X") return -100_000;
   153→  if (miniResults.every((r) => r !== null)) return 0; // full draw
   154→
   155→  let score = 0;
   156→
   157→  // 1. Resolved mini-boards
   158→  for (let i = 0; i < 9; i++) {
   159→    const w = BOARD_WEIGHTS[i];
   160→    if (miniResults[i] === "O") score += 1000 * w;
   161→    else if (miniResults[i] === "X") score -= 1000 * w;
   162→    else if (miniResults[i] === "draw") score += 5 * w;
   163→  }
   164→
   165→  // 2. Open mini-boards — evaluate potential
   166→  for (let i = 0; i < 9; i++) {
   167→    if (miniResults[i] !== null) continue;
   168→    const w = BOARD_WEIGHTS[i];
   169→    const small = getSmallBoardCells(cells, i);
   170→    score += evaluateOpenBoard(small, "O") * w;
   171→    score -= evaluateOpenBoard(small, "X") * w;
   172→  }
   173→
   174→  // 3. Big-board line potential
   175→  for (const line of WIN_LINES) {
   176→    const vals = line.map((i) => miniResults[i]);
   177→    const oWon = vals.filter((v) => v === "O").length;
   178→    const xWon = vals.filter((v) => v === "X").length;
   179→    const nulls = vals.filter((v) => v === null).length;
   180→    const draws = vals.filter((v) => v === "draw").length;
   181→
   182→    if (oWon === 2 && draws === 1) score += 200; // X can't block anymore
   183→    if (xWon === 2 && draws === 1) score -= 200;
   184→    if (xWon === 0) score += nulls * 15; // line still open for O
   185→    if (oWon === 0) score -= nulls * 15;
   186→  }
   187→
   188→  return score;
   189→};
   190→
   191→/**
   192→ * Minimax with alpha-beta pruning.
   193→ * `maximizing = true` → it's Bot's turn (O).
   194→ */
   195→const minimaxMove = (
   196→  cells: Cell[],
   197→  miniResults: MiniResult[],
   198→  targetBoard: number | null,
   199→  depth: number,
   200→  alpha: number,
   201→  beta: number,
   202→  maximizing: boolean
   203→): number => {
   204→  const moves = getValidMoves(cells, miniResults, targetBoard);
   205→
   206→  if (moves.length === 0 || depth <= 0) {
   207→    return evaluatePosition(cells, miniResults);
   208→  }
   209→
   210→  const mark: Player = maximizing ? "O" : "X";
   211→
   212→  if (maximizing) {
   213→    let best = -Infinity;
   214→    for (const move of moves) {
   215→      const nc = [...cells];
   216→      const nm = [...miniResults];
   217→      nc[move.boardIdx * 9 + move.cellIdx] = mark;
   218→
   219→      // resolve mini-board
   220→      const sb = getSmallBoardCells(nc, move.boardIdx);
   221→      const sr = checkWin(sb);
   222→      if (sr.winner) nm[move.boardIdx] = sr.winner;
   223→      else if (isBoardFull(nc, move.boardIdx)) nm[move.boardIdx] = "draw";
   224→
   225→      // resolve big board
   226→      const br = checkWin(nm);
   227→      if (br.winner === "O") return 100_000 + depth;
   228→      if (br.winner === "X") return -100_000 - depth;
   229→      if (nm.every((r) => r !== null)) return 0;
   230→
   231→      const nt = getNextTarget(nm, nc, move.cellIdx);
   232→      const val = minimaxMove(nc, nm, nt, depth - 1, alpha, beta, false);
   233→      best = Math.max(best, val);
   234→      alpha = Math.max(alpha, val);
   235→      if (beta <= alpha) break;
   236→    }
   237→    return best;
   238→  } else {
   239→    let best = Infinity;
   240→    for (const move of moves) {
   241→      const nc = [...cells];
   242→      const nm = [...miniResults];
   243→      nc[move.boardIdx * 9 + move.cellIdx] = mark;
   244→
   245→      const sb = getSmallBoardCells(nc, move.boardIdx);
   246→      const sr = checkWin(sb);
   247→      if (sr.winner) nm[move.boardIdx] = sr.winner;
   248→      else if (isBoardFull(nc, move.boardIdx)) nm[move.boardIdx] = "draw";
   249→
   250→      const br = checkWin(nm);
   251→      if (br.winner === "O") return 100_000 + depth;
   252→      if (br.winner === "X") return -100_000 - depth;
   253→      if (nm.every((r) => r !== null)) return 0;
   254→
   255→      const nt = getNextTarget(nm, nc, move.cellIdx);
   256→      const val = minimaxMove(nc, nm, nt, depth - 1, alpha, beta, true);
   257→      best = Math.min(best, val);
   258→      beta = Math.min(beta, val);
   259→      if (beta <= alpha) break;
   260→    }
   261→    return best;
   262→  }
   263→};
   264→
   265→/** Top-level entry: pick the best move for the Bot according to difficulty. */
   266→const getBotMove = (
   267→  cells: Cell[],
   268→  miniResults: MiniResult[],
   269→  targetBoard: number | null,
   270→  difficulty: Difficulty
   271→): AIMove | null => {
   272→  const moves = getValidMoves(cells, miniResults, targetBoard);
   273→  if (moves.length === 0) return null;
   274→  if (moves.length === 1) return moves[0];
   275→
   276→  const cfg = DIFFICULTY_CONFIG[difficulty];
   277→
   278→  // Random move based on difficulty
   279→  if (Math.random() < cfg.randomChance) {
   280→    return moves[Math.floor(Math.random() * moves.length)];
   281→  }
   282→
   283→  let bestMove = moves[0];
   284→  let bestScore = -Infinity;
   285→
   286→  for (const move of moves) {
   287→    const nc = [...cells];
   288→    const nm = [...miniResults];
   289→    nc[move.boardIdx * 9 + move.cellIdx] = "O";
   290→
   291→    const sb = getSmallBoardCells(nc, move.boardIdx);
   292→    const sr = checkWin(sb);
   293→    if (sr.winner) nm[move.boardIdx] = sr.winner;
   294→    else if (isBoardFull(nc, move.boardIdx)) nm[move.boardIdx] = "draw";
   295→
   296→    // Immediate big-board win → take it
   297→    const br = checkWin(nm);
   298→    if (br.winner === "O") return move;
   299→
   300→    const nt = getNextTarget(nm, nc, move.cellIdx);
   301→    const score = minimaxMove(nc, nm, nt, cfg.depth - 1, -Infinity, Infinity, false);
   302→
   303→    if (score > bestScore) {
   304→      bestScore = score;
   305→      bestMove = move;
   306→    }
   307→  }
   308→
   309→  return bestMove;
   310→};

   329→  const [bigWinLine, setBigWinLine] = useState<number[] | null>(null);
   330→  const [isDraw, setIsDraw] = useState(false);

   334→  const [showHelp, setShowHelp] = useState(false);
   335→  const [moveCount, setMoveCount] = useState(0);

   341→    (p: Player): string => {

   358→    (boardIdx: number, cellIdx: number): boolean => {

   360→      if (gameMode === "bot" && currentPlayer === "O") return false; // block during bot turn
   361→      const globalIdx = boardIdx * 9 + cellIdx;
   362→      if (cells[globalIdx] !== null) return false;
   363→      if (miniResults[boardIdx] !== null) return false;
   364→      if (targetBoard !== null && targetBoard !== boardIdx) return false;
   365→      return true;
   366→    },
   367→    [cells, gameOver, miniResults, targetBoard, gameMode, currentPlayer]
   368→  );
   369→
   370→  const activeTarget = useMemo(() => {
   371→    if (targetBoard !== null && isValidTarget(targetBoard)) return targetBoard;
   372→    return null;
   373→  }, [isValidTarget, targetBoard]);
   374→
   375→  /* ---- Core play logic ---- */
   376→  const playCell = useCallback(
   377→    (boardIdx: number, cellIdx: number) => {
   378→      if (gameOver) return;
   379→      const globalIdx = boardIdx * 9 + cellIdx;
   380→      if (cells[globalIdx] !== null) return;
   381→      if (miniResults[boardIdx] !== null) return;
   382→      if (targetBoard !== null && targetBoard !== boardIdx) return;
   383→      if (gameMode === "bot" && currentPlayer === "O") return;
   384→
   385→      const newCells = [...cells];
   386→      newCells[globalIdx] = currentPlayer;
   387→      setCells(newCells);
   388→      const newMoveCount = moveCount + 1;

   399→        setLastMiniWinBoard(boardIdx);
   400→

   407→          const timeBonus = Math.max(0, 50 - elapsed);
   408→          const moveBonus = Math.max(0, 200 - newMoveCount * 2);
   409→          const winScore = 100 + timeBonus + moveBonus;
   410→          const winnerName =
   411→            gameMode === "bot" && bigResult.winner === "O"
   412→              ? "🤖 Bot"
   413→              : playerNameBase(bigResult.winner);

   416→            bigResult.winner === "X"

   420→          return;
   421→        }

   423→        const allDone = newMiniResults.every((r) => r !== null);

   437→          setIsDraw(true);

   439→        }

   441→        setLastMiniWinBoard(null);
   442→      }

   444→      const nextTarget = cellIdx;

   446→        setTargetBoard(null);
   447→      } else {
   448→        setTargetBoard(nextTarget);
   449→      }
   450→
   451→      setCurrentPlayer(currentPlayer === "X" ? "O" : "X");
   452→    },

   470→    }, 450);

   474→      setIsBotThinking(false);
   475→    };
   476→    // eslint-disable-next-line react-hooks/exhaustive-deps

   478→
   479→  /* ---- Reset helpers ---- */
   480→  const initGame = useCallback(() => {
   481→    setCells(createEmptyCells());
   482→    setMiniResults(createEmptyMiniResults());
   483→    setCurrentPlayer("X");
   484→    setTargetBoard(null);
   485→    setGameOver(false);
   486→    setBigWinner(null);
   487→    setBigWinLine(null);
   488→    setIsDraw(false);
   489→    setLastMiniWinBoard(null);
   490→    setMoveCount(0);
   491→    setIsBotThinking(false);
   492→    startTimeRef.current = Date.now();
   493→  }, []);
   494→
   495→  const startGame = useCallback(
   496→    (mode: "pvp" | "bot") => {
   497→      setGameMode(mode);
   498→      setScores({ x: 0, o: 0 });
   499→      setRound(1);
   500→      initGame();
   501→    },
   502→    [initGame]
   503→  );

   513→
   514→  const fullReset = useCallback(() => {
   515→    setScores({ x: 0, o: 0 });
   516→    setRound(1);
   517→    initGame();
   518→  }, [initGame]);
   519→
   520→  const goToMenu = useCallback(() => {

   522→  }, []);
   523→

   529→    }

   531→      return `— Tabuleiro ${BOARD_NAMES[activeTarget]}`;

   535→
   536→  /* ================================================================ */
   537→  /*  START SCREEN                                                      */

   539→  if (gameMode === "menu") {
