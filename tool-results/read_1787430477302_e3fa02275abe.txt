   113→/** Predict where the ball will be when it reaches targetX */
   114→function predictBallY(ball: Ball, targetX: number): number {
   115→  if (
   116→    (targetX > ball.x && ball.vx <= 0) ||
   117→    (targetX < ball.x && ball.vx >= 0)
   118→  ) {
   119→    return ball.y;
   120→  }
   121→
   122→  let x = ball.x;
   123→  let y = ball.y;
   124→  let vy = ball.vy;
   125→  const step = Math.abs(ball.vx) || 1;
   126→  const dir = ball.vx > 0 ? 1 : -1;
   127→
   128→  for (let i = 0; i < 2000; i++) {
   129→    x += dir * step;
   130→    y += vy;
   131→
   132→    if (y <= BALL_R) {
   133→      y = BALL_R;
   134→      vy = Math.abs(vy);
   135→    }
   136→    if (y >= CANVAS_H - BALL_R) {
   137→      y = CANVAS_H - BALL_R;
   138→      vy = -Math.abs(vy);
   139→    }
   140→
   141→    if ((dir > 0 && x >= targetX) || (dir < 0 && x <= targetX)) {
   142→      return y;
   143→    }
   144→  }
   145→
   146→  return ball.y;
   147→}

   631→                  layoutId="mode-indicator"

   641→          <AnimatePresence mode="wait">
   642→            {mode === "bot" && (
   643→              <motion.div
   644→                key="difficulty-panel"
   645→                initial={{ opacity: 0, height: 0, y: -8 }}
   646→                animate={{ opacity: 1, height: "auto", y: 0 }}
   647→                exit={{ opacity: 0, height: 0, y: -8 }}

   651→                <div className="flex items-center gap-2 mb-2">

   658→                  {(["facil", "medio", "dificil"] as BotDifficulty[]).map(

   661→                      const cfg = BOT_CONFIGS[d];

   664→                          key={d}
   665→                          whileHover={{ scale: 1.03 }}

   670→                              ? `border-current bg-gradient-to-br ${style.bg} shadow-md`

   675→                            className={`text-sm font-bold ${
   676→                              difficulty === d

   678→                                : "text-muted-foreground"
   679→                            }`}
   680→                          >
   681→                            {cfg.label}
   682→                          </span>

   684→                            {style.desc}
   685→                          </span>
   686→                        </motion.button>
   687→                      );
   688→                    }
   689→                  )}
   690→                </div>
   691→              </motion.div>
   692→            )}
   693→          </AnimatePresence>
   694→
   695→          <motion.div
   696→            initial={{ opacity: 0 }}
   697→            animate={{ opacity: 1 }}
   698→            transition={{ delay: 0.1 }}
   699→            className="grid grid-cols-2 gap-2 w-full"
   700→          >
   701→            <input
   702→              value={p1Name}
   703→              onChange={(e) => setP1Name(e.target.value)}
   704→              placeholder="Jogador 1"
   705→              className="px-3 py-2.5 rounded-xl bg-card border border-border text-sm text-center focus:border-primary/50 focus:outline-none transition-colors"
   706→            />
   707→            {mode === "vs" ? (
   708→              <input
   709→                value={p2Name}
   710→                onChange={(e) => setP2Name(e.target.value)}
   711→                placeholder="Jogador 2"
   712→                className="px-3 py-2.5 rounded-xl bg-card border border-border text-sm text-center focus:border-primary/50 focus:outline-none transition-colors"
   713→              />
   714→            ) : (
   715→              <div className="px-3 py-2.5 rounded-xl bg-muted/50 border border-border text-sm text-center text-muted-foreground flex items-center justify-center gap-1.5">
   716→                <Bot className="h-3.5 w-3.5" />
   717→                <span>
   718→                  IA — {BOT_CONFIGS[difficulty].label}
   719→                </span>
   720→              </div>
   721→            )}
   722→          </motion.div>
   723→
   724→          <motion.button
   725→            whileHover={{ scale: 1.01 }}
   726→            whileTap={{ scale: 0.99 }}
   727→            onClick={startGame}
   728→            className="w-full py-3.5 rounded-xl bg-gradient-to-r from-blue-500 to-indigo-600 text-white font-bold text-sm shadow-lg flex items-center justify-center gap-2 hover:shadow-xl transition-shadow cursor-pointer"
   729→          >
   730→            <Play className="h-4 w-4 fill-current" /> Começar Jogo
   731→          </motion.button>
   732→
   733→          <div className="rounded-xl bg-muted/40 border border-border p-3 text-center">
   734→            <p className="text-[11px] text-muted-foreground font-bold uppercase tracking-wider mb-1">
   735→              Controles
   736→            </p>
   737→            <p className="text-xs text-muted-foreground">
   738→              {mode === "vs"
   739→                ? "Jogador 1: W/S  •  Jogador 2: ↑/↓"
   740→                : "W/S ou ↑/↓ para mover a raquete"}
   741→            </p>
   742→            <p className="text-xs text-muted-foreground mt-1">
   743→              Primeiro a {WIN_SCORE} pontos vence!
   744→            </p>
   745→          </div>
   746→        </motion.div>
   747→      )}
   748→
   749→      {phase !== "idle" && (
   750→        <div className="w-full flex items-center justify-between px-2">
   751→          <div className="flex items-center gap-2">
   752→            <div className="h-3 w-3 rounded-full bg-gradient-to-r from-blue-500 to-indigo-500" />
   753→            <span className="text-xs font-bold text-muted-foreground">
   754→              {p1Name}
   755→            </span>
   756→            <span className="font-display text-xl font-bold text-foreground">
   757→              {p1Score}
   758→            </span>
   759→          </div>
   760→          <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-bold">
   761→            vs
   762→          </span>
   763→          <div className="flex items-center gap-2">
   764→            <span className="font-display text-xl font-bold text-foreground">
   765→              {p2Score}
   766→            </span>
   767→            <span className="text-xs font-bold text-muted-foreground">
   768→              {mode === "bot" ? `IA (${BOT_CONFIGS[difficulty].label})` : p2Name}
   769→            </span>
   770→            <div className="h-3 w-3 rounded-full bg-gradient-to-r from-amber-500 to-red-500" />
   771→          </div>
   772→        </div>
   773→      )}
   774→
   775→      <div className="relative w-full" style={{ maxWidth: CANVAS_W }}>
   776→        <canvas
   777→          ref={canvasRef}
   778→          width={CANVAS_W}
   779→          height={CANVAS_H}
   780→          className="w-full rounded-2xl border-2 border-border shadow-xl"
   781→          tabIndex={0}
   782→        />
   783→
   784→        <AnimatePresence>
   785→          {phase === "countdown" && (
   786→            <motion.div
   787→              key={countdown}
   788→              initial={{ scale: 2, opacity: 0 }}
   789→              animate={{ scale: 1, opacity: 1 }}
   790→              exit={{ scale: 0.5, opacity: 0 }}
   791→              className="absolute inset-0 flex items-center justify-center bg-black/50 rounded-2xl"
   792→            >
   793→              <span className="font-display text-7xl font-bold text-white">
   794→                {countdown > 0 ? countdown : "GO!"}
   795→              </span>
   796→            </motion.div>
   797→          )}
   798→        </AnimatePresence>
   799→
   800→        <AnimatePresence>
   801→          {phase === "done" && (
   802→            <motion.div
   803→              initial={{ opacity: 0 }}
   804→              animate={{ opacity: 1 }}
   805→              className="absolute inset-0 flex flex-col items-center justify-center bg-black/60 rounded-2xl gap-3"
   806→            >
   807→              <Trophy className="h-10 w-10 text-yellow-400" />
   808→              <p className="font-display text-2xl font-bold text-white">
   809→                {winner} venceu!
   810→              </p>
   811→              <p className="text-sm text-white/70">
   812→                {p1Score} - {p2Score}
   813→              </p>
   814→              <button
   815→                onClick={startGame}
   816→                className="mt-2 px-6 py-2.5 rounded-full bg-gradient-to-r from-blue-500 to-indigo-600 text-white font-bold text-sm flex items-center gap-2 shadow-lg hover:shadow-xl transition-shadow cursor-pointer"
   817→              >
   818→                <RotateCcw className="h-4 w-4" /> Jogar Novamente
   819→              </button>
   820→              <button
   821→                onClick={() => setPhase("idle")}
   822→                className="px-4 py-2 rounded-full bg-white/10 text-white/80 text-xs font-medium hover:bg-white/20 transition cursor-pointer"
   823→              >
   824→                Voltar ao Menu
   825→              </button>
   826→            </motion.div>
   827→          )}
   828→        </AnimatePresence>
   829→      </div>
   830→
   831→      {phase === "playing" && (
   832→        <p className="text-[10px] text-muted-foreground text-center">
   833→          {mode === "vs"
   834→            ? "P1: W/S  •  P2: ↑/↓"
   835→            : "W/S ou ↑/↓ para mover"}{" "}
   836→          | Clique no canvas para focar
   837→        </p>
   838→      )}
   839→    </div>
   840→  );
   841→};
