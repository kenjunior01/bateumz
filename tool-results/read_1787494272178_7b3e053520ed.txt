   157→function shuffle<T>(arr: T[]): T[] {
   158→  const a = [...arr];
   159→  for (let i = a.length - 1; i > 0; i--) {
   160→    const j = Math.floor(Math.random() * (i + 1));
   161→    [a[i], a[j]] = [a[j], a[i]];
   162→  }
   163→  return a;
   164→}

   224→function GameCard({ card, index, disabled, onClick }: CardProps) {
   225→  const isFlipped = card.status === 'flipping' || card.status === 'matched';
   226→  const isShaking = card.status === 'shaking';
   227→  const isMatched = card.status === 'matched';
   228→  const shapeDef = SHAPE_DEFS[card.shape];
   229→
   230→  return (
   231→    <motion.div
   232→      className="relative"
   233→      style={{ perspective: 600 }}
   234→      layout
   235→    >
   236→      <AnimatePresence>
   237→        {isMatched && (
   238→          <motion.div
   239→            key={`glow-${card.id}`}
   240→            initial={{ opacity: 0, scale: 0.8 }}
   241→            animate={{ opacity: [0, 1, 0.6], scale: [0.8, 1.15, 1] }}
   242→            transition={{ duration: 0.6, ease: 'easeOut' }}
   243→            className="absolute inset-0 rounded-xl z-10 pointer-events-none"
   244→            style={{
   245→              boxShadow: `0 0 20px 4px ${shapeDef.color}66, 0 0 40px 8px ${shapeDef.color}33`,
   246→            }}
   247→          />
   248→        )}
   249→      </AnimatePresence>
   250→
   251→      <motion.button
   252→        type="button"
   253→        disabled={disabled || isFlipped || isMatched}
   254→        onClick={() => onClick(index)}
   255→        animate={
   256→          isShaking

   261→        }
   262→        transition={
   263→          isShaking
   264→            ? { duration: 0.5, ease: 'easeInOut' }

   268→        }
   269→        className={cn(
   270→          'relative w-16 h-20 sm:w-20 sm:h-24 rounded-xl cursor-pointer focus:outline-none transition-shadow',
   271→          isMatched && 'cursor-default',
   272→          disabled && !isFlipped && !isMatched && 'cursor-not-allowed opacity-60',
   273→        )}
   274→        style={{
   275→          transformStyle: 'preserve-3d',
   276→        }}
   277→        aria-label={`Carta ${index + 1}`}
   278→      >
   279→        <motion.div
   280→          className={cn(
   281→            'absolute inset-0 rounded-xl border bg-slate-800 border-slate-700 flex items-center justify-center',

   299→        <motion.div
   300→          className={cn(
   301→            'absolute inset-0 rounded-xl border border-slate-300 dark:border-slate-400 bg-white dark:bg-slate-100 flex items-center justify-center',

   307→          }}

   309→          transition={{ duration: 0.45, ease: [0.4, 0, 0.2, 1] }}
   310→        >
   311→          <ShapeSVG shape={card.shape} color={shapeDef.color} />
   312→          {isMatched && (
   313→            <motion.span
   314→              initial={{ opacity: 0, scale: 0 }}
   315→              animate={{ opacity: 1, scale: 1 }}
   316→              transition={{ delay: 0.3, duration: 0.25 }}
   317→              className="absolute top-0.5 right-0.5 text-green-500 text-xs"
   318→            >

   320→            </motion.span>
   321→          )}
   322→        </motion.div>
   323→      </motion.button>
   324→    </motion.div>
   325→  );
   326→}

   331→
   332→export default function MemoryCardsVS({ onScore, liveCode: _liveCode }: Props) {
   333→  /* ---- State ---- */
   334→  const [gameState, setGameState] = useState<GameState>('idle');
   335→  const [mode, setMode] = useState<GameMode>('player');
   336→  const [botDifficulty, setBotDifficulty] = useState<BotDifficulty>('Médio');
   337→  const [gridSize, setGridSize] = useState<GridSize>('4x4');
   338→  const [timerOption, setTimerOption] = useState<TimerOption>(90);
   339→  const [cards, setCards] = useState<CardData[]>([]);
   340→  const [flippedIndices, setFlippedIndices] = useState<number[]>([]);
   341→  const [currentPlayer, setCurrentPlayer] = useState<1 | 2>(1);
   342→  const [scores, setScores] = useState<[number, number]>([0, 0]);
   343→  const [timeRemaining, setTimeRemaining] = useState<number>(0);
   344→  const [canFlip, setCanFlip] = useState(false);
   345→  const [feedback, setFeedback] = useState<{ message: string; color: string; id: number } | null>(null);
   346→  const [winner, setWinner] = useState<1 | 2 | 'draw' | null>(null);
   347→  const [botThinking, setBotThinking] = useState(false);
   348→
   349→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   350→  const feedbackIdRef = useRef(0);
   351→  const botTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   352→  const botMemoryRef = useRef<Map<number, number>>(new Map()); // index -> pairId
   353→  const isProcessingRef = useRef(false);
   354→
   355→  const config = GRID_CONFIG[gridSize];
   356→  const totalPairs = config.pairs;
   357→  const matchedCount = cards.filter(c => c.status === 'matched').length / 2;
   358→  const pairsLeft = totalPairs - matchedCount;
   359→
   360→  const getP2Name = () => mode === 'bot' ? 'Computador' : 'Jogador 2';
   361→
   362→  /* ---- Derived ---- */
   363→  const activeGlow =
   364→    currentPlayer === 1
   365→      ? 'shadow-[0_0_20px_4px_rgba(6,182,212,0.35)] border-cyan-500/50'
   366→      : 'shadow-[0_0_20px_4px_rgba(236,72,153,0.35)] border-pink-500/50';
   367→
   368→  /* ---- Init game ---- */
   369→  const initGame = useCallback(() => {
   370→    if (botTimeoutRef.current) clearTimeout(botTimeoutRef.current);
   371→    botMemoryRef.current = new Map();
   372→    const shapes = shuffle(ALL_SHAPES).slice(0, totalPairs);
   373→    const deck: CardData[] = [];
   374→    shapes.forEach((shape, pairIdx) => {
   375→      const color = SHAPE_DEFS[shape].color;
   376→      deck.push({ id: deck.length, pairId: pairIdx, shape, color, status: 'hidden' });
   377→      deck.push({ id: deck.length, pairId: pairIdx, shape, color, status: 'hidden' });
   378→    });
   379→    const shuffled = shuffle(deck);
   380→    setCards(shuffled);
   381→    setFlippedIndices([]);
   382→    setCurrentPlayer(1);
   383→    setScores([0, 0]);
   384→    setCanFlip(true);
   385→    setFeedback(null);
   386→    setWinner(null);
   387→    setTimeRemaining(timerOption);
   388→    setBotThinking(false);
   389→    isProcessingRef.current = false;
   390→    setGameState('playing');
   391→  }, [totalPairs, timerOption]);
   392→
   393→  /* ---- Timer effect ---- */
   394→  useEffect(() => {
   395→    if (gameState !== 'playing' || timerOption === 0) return;
   396→    timerRef.current = setInterval(() => {
   397→      setTimeRemaining(prev => {
   398→        if (prev <= 1) {
   399→          clearInterval(timerRef.current!);
   400→          return 0;
   401→        }
   402→        return prev - 1;
   403→      });
   404→    }, 1000);
   405→    return () => {
   406→      if (timerRef.current) clearInterval(timerRef.current);
   407→    };
   408→  }, [gameState, timerOption]);
   409→
   410→  /* ---- End game when timer hits 0 ---- */
   411→  const endGame = useCallback(() => {
   412→    setGameState('ended');
   413→    setCanFlip(false);
   414→    setBotThinking(false);
   415→    if (botTimeoutRef.current) clearTimeout(botTimeoutRef.current);
   416→    if (timerRef.current) clearInterval(timerRef.current);
   417→
   418→    const timeBonus = timerOption > 0 ? timeRemaining : 0;
   419→
   420→    if (scores[0] > scores[1]) {
   421→      setWinner(1);
   422→      onScore?.('Jogador 1', scores[0] * 10 + timeBonus);
   423→    } else if (scores[1] > scores[0]) {
   424→      setWinner(2);
   425→      onScore?.(getP2Name(), scores[1] * 10 + timeBonus);
   426→    } else {
   427→      setWinner('draw');
   428→    }
   429→  }, [scores, timerOption, timeRemaining, onScore, mode]);
   430→
   431→  useEffect(() => {
   432→    if (gameState === 'playing' && timerOption > 0 && timeRemaining === 0) {
   433→      endGame();
   434→    }
   435→  }, [timeRemaining, gameState, timerOption, endGame]);
   436→
   437→  /* ---- Show feedback toast ---- */
   438→  const showFeedback = useCallback((message: string, color: string) => {
   439→    const id = ++feedbackIdRef.current;
   440→    setFeedback({ message, color, id });
   441→    setTimeout(() => setFeedback(prev => (prev && prev.id === id ? null : prev)), 900);
   442→  }, []);
   443→
   444→  /* ---- Core card click handler (shared by player and bot) ---- */
   445→  const processFlip = useCallback((
   446→    index: number,
   447→    player: 1 | 2,
   448→    currentCards: CardData[],
   449→    currentFlipped: number[],
   450→    currentCanFlip: boolean,
   451→  ) => {
   452→    if (!currentCanFlip || gameState !== 'playing' || isProcessingRef.current) return false;
   453→    const card = currentCards[index];
   454→    if (card.status !== 'hidden') return false;
   455→    if (currentFlipped.includes(index)) return false;
   456→
   457→    isProcessingRef.current = true;
   458→    const newFlipped = [...currentFlipped, index];
   459→    const updated = currentCards.map((c, i) => (i === index ? { ...c, status: 'flipping' as CardStatus } : c));
   460→    setCards(updated);
   461→    setFlippedIndices(newFlipped);
   462→
   463→    // Remember revealed cards for bot
   464→    if (player === 1 || mode === 'bot') {
   465→      botMemoryRef.current.set(index, updated[index].pairId);
   466→    }
   467→
   468→    if (newFlipped.length === 2) {
   469→      setCanFlip(false);
   470→      const [a, b] = newFlipped;
   471→      if (updated[a].pairId === updated[b].pairId) {
   472→        setTimeout(() => {
   473→          setCards(prev =>
   474→            prev.map((c, i) => (i === a || i === b ? { ...c, status: 'matched' as CardStatus } : c)),
   475→          );
   476→          setScores(prev => {
   477→            const next = [...prev] as [number, number];
   478→            next[player - 1] += 1;
   479→            return next;
   480→          });
   481→          showFeedback('Par!', 'bg-green-600 text-white');
   482→          setFlippedIndices([]);
   483→          setCanFlip(true);
   484→          isProcessingRef.current = false;
   485→        }, 500);
   486→        return true;
   487→      } else {
   488→        showFeedback('Não é par', 'bg-red-600 text-white');
   489→        setTimeout(() => {
   490→          setCards(prev =>
   491→            prev.map((c, i) =>
   492→              i === a || i === b ? { ...c, status: 'shaking' as CardStatus } : c,
   493→            ),
   494→          );
   495→        }, 400);
   496→        setTimeout(() => {
   497→          setCards(prev =>
   498→            prev.map((c, i) =>
   499→              i === a || i === b ? { ...c, status: 'hidden' as CardStatus } : c,
   500→            ),
   501→          );
   502→          setFlippedIndices([]);
   503→          setCurrentPlayer(prev => (prev === 1 ? 2 : 1));
   504→          setCanFlip(true);
   505→          isProcessingRef.current = false;
   506→        }, 1000);
   507→        return true;
   508→      }
   509→    }
   510→    isProcessingRef.current = false;
   511→    return true;
   512→  }, [gameState, mode, showFeedback]);
   513→
   514→  /* ---- Player card click ---- */
   515→  const handleCardClick = useCallback(
   516→    (index: number) => {
   517→      if (mode === 'bot' && currentPlayer === 2) return;
   518→      processFlip(index, 1, cards, flippedIndices, canFlip);
   519→    },
   520→    [mode, currentPlayer, cards, flippedIndices, canFlip, processFlip],
   521→  );
   522→
   523→  /* ---- Keyboard support ---- */
   524→  useEffect(() => {
   525→    if (gameState !== 'playing' || !canFlip) return;
   526→    function handleKey(e: KeyboardEvent) {
   527→      if (mode === 'bot' && currentPlayer === 2) return;
   528→      let idx = -1;
   529→      if (e.key >= '1' && e.key <= '9') idx = parseInt(e.key) - 1;
   530→      else if (e.key === '0' && cards.length >= 10) idx = 9;
   531→      else return;
   532→      if (idx >= 0 && idx < cards.length) {
   533→        handleCardClick(idx);
   534→      }
   535→    }
   536→    window.addEventListener('keydown', handleKey);
   537→    return () => window.removeEventListener('keydown', handleKey);
   538→  }, [gameState, canFlip, cards, handleCardClick, mode, currentPlayer]);
   539→
   540→  /* ---- Check if all matched ---- */

   557→      'Difícil': 0.9,

   564→        .map((c, i) => ({ index: i, pairId: c.pairId, status: c.status }))

   571→        return;

   576→        setBotThinking(false);
   577→        return;

   580→      // First flip
   581→      let firstPick: number;
   582→      let secondPick: number | null = null;

   584→      // Check if bot remembers a matching pair
   585→      const rememberedPairs = new Map<number, number[]>();
   586→      for (const [idx, pairId] of memory) {
   587→        if (cards[idx]?.status === 'hidden') {
   588→          if (!rememberedPairs.has(pairId)) rememberedPairs.set(pairId, []);
   589→          rememberedPairs.get(pairId)!.push(idx);
   590→        }
   591→      }
   592→
   593→      let foundMatch = false;
   594→      for (const [, indices] of rememberedPairs) {
   595→        if (indices.length >= 2) {
   596→          if (Math.random() < memoryChance) {
   597→            firstPick = indices[0];
   598→            secondPick = indices[1];
   599→            foundMatch = true;
   600→            break;
   601→          }
   602→        }
   603→      }
   604→
   605→      // Check if bot remembers one card of a pair and another is available
   606→      if (!foundMatch) {
   607→        for (const [pairId, indices] of rememberedPairs) {
   608→          if (indices.length === 1 && Math.random() < memoryChance) {
   609→            firstPick = indices[0];
   610→            // Find the other card of this pair
   611→            const otherIdx = available.find(
   612→              c => c.pairId === pairId && c.index !== firstPick

   614→            if (otherIdx) {
   615→              secondPick = otherIdx.index;
   616→              foundMatch = true;
   617→              break;
   618→            }
   619→          }
   620→        }
   621→      }
   622→
   623→      if (!foundMatch) {
   624→        firstPick = available[Math.floor(Math.random() * available.length)].index;
   625→      }
   626→
   627→      // Flip first card
   628→      const currentCards = cards;
   629→      const currentFlipped = flippedIndices;
   630→
   631→      if (currentCards[firstPick]?.status !== 'hidden') {
   632→        setBotThinking(false);
   633→        return;
   634→      }
   635→
   636→      const newFlipped = [...currentFlipped, firstPick];
   637→      const updated = currentCards.map((c, i) =>
   638→        i === firstPick ? { ...c, status: 'flipping' as CardStatus } : c
   639→      );
   640→      setCards(updated);
   641→      setFlippedIndices(newFlipped);
   642→      memory.set(firstPick, updated[firstPick].pairId);
   643→
   644→      // After 800ms, flip second card
   645→      botTimeoutRef.current = setTimeout(() => {
   646→        if (gameState !== 'playing' || currentPlayer !== 2) {
   647→          setBotThinking(false);
   648→          return;
   649→        }
   650→
   651→        const freshCards = cards; // re-read latest
   652→        const freshAvailable = freshCards
   653→          .map((c, i) => ({ index: i, pairId: c.pairId, status: c.status }))
   654→          .filter(c => c.status === 'hidden');
   655→
   656→        let actualSecond: number;
   657→
   658→        if (secondPick !== null && freshCards[secondPick]?.status === 'hidden') {
   659→          actualSecond = secondPick;
   660→        } else {
   661→          // Try to find a match with the first card
   662→          const firstPairId = updated[firstPick].pairId;
   663→          const matchIdx = freshAvailable.find(c => c.pairId === firstPairId && c.index !== firstPick);
   664→          if (matchIdx && Math.random() < memoryChance) {
   665→            actualSecond = matchIdx.index;
   666→          } else {
   667→            actualSecond = freshAvailable[Math.floor(Math.random() * freshAvailable.length)].index;
   668→          }
   669→        }
   670→
   671→        const finalFlipped = [firstPick, actualSecond];
   672→        const finalUpdated = freshCards.map((c, i) =>
   673→          i === actualSecond ? { ...c, status: 'flipping' as CardStatus } : c
   674→        );
   675→        setCards(finalUpdated);
   676→        setFlippedIndices(finalFlipped);
   677→        memory.set(actualSecond, finalUpdated[actualSecond].pairId);
   678→
   679→        setCanFlip(false);
   680→        const [a, b] = finalFlipped;
   681→        if (finalUpdated[a].pairId === finalUpdated[b].pairId) {
   682→          setTimeout(() => {
   683→            setCards(prev =>
   684→              prev.map((c, i) => (i === a || i === b ? { ...c, status: 'matched' as CardStatus } : c)),
   685→            );
   686→            setScores(prev => {
   687→              const next = [...prev] as [number, number];
   688→              next[1] += 1;
   689→              return next;
   690→            });
   691→            showFeedback('Par!', 'bg-green-600 text-white');
   692→            setFlippedIndices([]);
   693→            setCanFlip(true);
   694→            setBotThinking(false);
   695→          }, 500);
   696→        } else {
   697→          showFeedback('Não é par', 'bg-red-600 text-white');
   698→          setTimeout(() => {
   699→            setCards(prev =>
   700→              prev.map((c, i) =>
   701→                i === a || i === b ? { ...c, status: 'shaking' as CardStatus } : c,
   702→              ),
   703→            );
   704→          }, 400);
   705→          setTimeout(() => {
   706→            setCards(prev =>
   707→              prev.map((c, i) =>
   708→                i === a || i === b ? { ...c, status: 'hidden' as CardStatus } : c,
   709→              ),
   710→            );
   711→            setFlippedIndices([]);
   712→            setCurrentPlayer(1);
   713→            setCanFlip(true);
   714→            setBotThinking(false);
   715→          }, 1000);
   716→        }
   717→      }, 800);
   718→    };
   719→
   720→    // Bot thinks for 800ms before first flip
   721→    botTimeoutRef.current = setTimeout(doBotTurn, 800);
   722→
   723→    return () => {
   724→      if (botTimeoutRef.current) {
   725→        clearTimeout(botTimeoutRef.current);
   726→        botTimeoutRef.current = null;
   727→      }
   728→      setBotThinking(false);
   729→    };
   730→  }, [gameState, mode, currentPlayer, canFlip, cards, flippedIndices, botDifficulty, showFeedback]);
   731→

   735→    if (botTimeoutRef.current) clearTimeout(botTimeoutRef.current);

   737→    setCards([]);
   738→    setFlippedIndices([]);
   739→    setScores([0, 0]);
   740→    setCurrentPlayer(1);
   741→    setTimeRemaining(0);
   742→    setCanFlip(false);
   743→    setFeedback(null);
   744→    setWinner(null);
   745→    setBotThinking(false);
   746→    botMemoryRef.current = new Map();
   747→    isProcessingRef.current = false;
   748→  };

   755→  };

   767→      <div className="w-full rounded-xl bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 p-3 flex items-center justify-between">
   768→        <div
   769→          className={cn(
   770→            'flex items-center gap-2 px-3 py-1.5 rounded-lg transition-all',

   776→          <div className="w-3 h-3 rounded-full bg-cyan-400" />

   789→        <div className="flex flex-col items-center gap-1">
   790→          {gameState === 'playing' && (
   791→            <AnimatePresence mode="wait">
   792→              <motion.span

   794→                initial={{ opacity: 0, x: currentPlayer === 1 ? -10 : 10 }}

   798→                  'text-xs font-semibold px-2 py-0.5 rounded-full',
   799→                  currentPlayer === 1
   800→                    ? 'text-cyan-300 bg-cyan-500/10'
   801→                    : 'text-pink-300 bg-pink-500/10',
   802→                )}
   803→              >
   804→                {currentPlayer === 2 && mode === 'bot' && botThinking ? 'Computador pensando...' : 'Sua vez'}
   805→              </motion.span>
