    68→    68→function generateTerrain(width: number, height: number): number[] {
    69→    69→  const terrain: number[] = [];
    70→    70→  const freq1 = 0.008 + Math.random() * 0.012;
    71→    71→  const freq2 = 0.02 + Math.random() * 0.03;
    72→    72→  const freq3 = 0.003 + Math.random() * 0.005;
    73→    73→  const amp1 = 30 + Math.random() * 40;
    74→    74→  const amp2 = 10 + Math.random() * 20;
    75→    75→  const amp3 = 20 + Math.random() * 30;
    76→    76→  const baseY = height * 0.65;
    77→    77→  const phase1 = Math.random() * Math.PI * 2;
    78→    78→  const phase2 = Math.random() * Math.PI * 2;
    79→    79→  const phase3 = Math.random() * Math.PI * 2;
    80→    80→
    81→    81→  for (let x = 0; x < width; x++) {
    82→    82→    const y =
    83→    83→      baseY +
    84→    84→      Math.sin(x * freq1 + phase1) * amp1 +
    85→    85→      Math.sin(x * freq2 + phase2) * amp2 +
    86→    86→      Math.sin(x * freq3 + phase3) * amp3;
    87→    87→    terrain.push(Math.min(Math.max(y, height * 0.35), height * 0.9));
    88→    88→  }
    89→    89→  return terrain;
    90→    90→}

    96→    96→function createExplosion(x: number, y: number): Explosion {
    97→    97→  const particles: ExplosionParticle[] = [];
    98→    98→  const colors = ["#ff6600", "#ff9900", "#ffcc00", "#ff3300", "#ffffff"];
    99→    99→  for (let i = 0; i < 20; i++) {
   100→   100→    const ang = Math.random() * Math.PI * 2;
   101→   101→    const speed = 1 + Math.random() * 4;
   102→   102→    particles.push({
   103→   103→      x,
   104→   104→      y,
   105→   105→      vx: Math.cos(ang) * speed,
   106→   106→      vy: Math.sin(ang) * speed - 2,
   107→   107→      size: 2 + Math.random() * 4,
   108→   108→      alpha: 1,
   109→   109→      color: colors[Math.floor(Math.random() * colors.length)],
   110→   110→    });
   111→   111→  }
   112→   112→  return { x, y, frame: 0, maxFrames: 40, particles };
   113→   113→}
