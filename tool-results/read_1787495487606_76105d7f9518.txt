    68→function generateTerrain(width: number, height: number): number[] {
    69→  const terrain: number[] = [];
    70→  const freq1 = 0.008 + Math.random() * 0.012;
    71→  const freq2 = 0.02 + Math.random() * 0.03;
    72→  const freq3 = 0.003 + Math.random() * 0.005;
    73→  const amp1 = 30 + Math.random() * 40;
    74→  const amp2 = 10 + Math.random() * 20;
    75→  const amp3 = 20 + Math.random() * 30;
    76→  const baseY = height * 0.65;
    77→  const phase1 = Math.random() * Math.PI * 2;
    78→  const phase2 = Math.random() * Math.PI * 2;
    79→  const phase3 = Math.random() * Math.PI * 2;
    80→
    81→  for (let x = 0; x < width; x++) {
    82→    const y =
    83→      baseY +
    84→      Math.sin(x * freq1 + phase1) * amp1 +
    85→      Math.sin(x * freq2 + phase2) * amp2 +
    86→      Math.sin(x * freq3 + phase3) * amp3;
    87→    terrain.push(Math.min(Math.max(y, height * 0.35), height * 0.9));
    88→  }
    89→  return terrain;
    90→}
    91→
    92→function generateWind(): number {
    93→  return (Math.random() - 0.5) * 6;
    94→}

    96→function createExplosion(x: number, y: number): Explosion {
    97→  const particles: ExplosionParticle[] = [];
    98→  const colors = ["#ff6600", "#ff9900", "#ffcc00", "#ff3300", "#ffffff"];
    99→  for (let i = 0; i < 20; i++) {
   100→    const ang = Math.random() * Math.PI * 2;
   101→    const speed = 1 + Math.random() * 4;
   102→    particles.push({
   103→      x,
   104→      y,
   105→      vx: Math.cos(ang) * speed,
   106→      vy: Math.sin(ang) * speed - 2,
   107→      size: 2 + Math.random() * 4,
   108→      alpha: 1,
   109→      color: colors[Math.floor(Math.random() * colors.length)],
   110→    });
   111→  }
   112→  return { x, y, frame: 0, maxFrames: 40, particles };
   113→}

   115→function createProjectile(
   116→  x: number,
   117→  y: number,
   118→  angleDeg: number,
   119→  power: number
   120→): Projectile {
   121→  const rad = (angleDeg * Math.PI) / 180;
   122→  return {
   123→    x,
   124→    y,
   125→    vx: power * Math.cos(rad) * 0.15,
   126→    vy: -power * Math.sin(rad) * 0.15,
   127→    trail: [],
   128→    active: true,
   129→  };
   130→}

   680→    animFrameRef.current = requestAnimationFrame(gameLoop);
