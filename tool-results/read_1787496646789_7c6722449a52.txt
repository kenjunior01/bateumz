    36→// ---- Tap sound via AudioContext ----
    37→function playTapSound(pitch = 600) {
    38→  try {
    39→    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    40→    const osc = ctx.createOscillator();
    41→    const gain = ctx.createGain();
    42→    osc.type = "sine";
    43→    osc.frequency.setValueAtTime(pitch, ctx.currentTime);
    44→    osc.frequency.exponentialRampToValueAtTime(pitch * 1.5, ctx.currentTime + 0.05);
    45→    gain.gain.setValueAtTime(0.12, ctx.currentTime);
    46→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
    47→    osc.connect(gain);
    48→    gain.connect(ctx.destination);
    49→    osc.start(ctx.currentTime);
    50→    osc.stop(ctx.currentTime + 0.1);
    51→    setTimeout(() => ctx.close(), 150);
    52→  } catch { /* silent fail */ }
    53→}
    54→
    55→function playCountdownBeep(final = false) {
    56→  try {
    57→    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    58→    const osc = ctx.createOscillator();
    59→    const gain = ctx.createGain();
    60→    osc.type = "square";
    61→    osc.frequency.setValueAtTime(final ? 880 : 440, ctx.currentTime);
    62→    gain.gain.setValueAtTime(0.08, ctx.currentTime);
    63→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + (final ? 0.3 : 0.15));
    64→    osc.connect(gain);
    65→    gain.connect(ctx.destination);
    66→    osc.start(ctx.currentTime);
    67→    osc.stop(ctx.currentTime + (final ? 0.3 : 0.15));
    68→    setTimeout(() => ctx.close(), 400);
    69→  } catch { /* silent fail */ }
    70→}
