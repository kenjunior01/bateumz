    37→// ---- Tap sound via AudioContext ----
    38→function playTapSound(pitch = 600) {
    39→  try {
    40→    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    41→    const osc = ctx.createOscillator();
    42→    const gain = ctx.createGain();
    43→    osc.type = "sine";
    44→    osc.frequency.setValueAtTime(pitch, ctx.currentTime);
    45→    osc.frequency.exponentialRampToValueAtTime(pitch * 1.5, ctx.currentTime + 0.05);
    46→    gain.gain.setValueAtTime(0.12, ctx.currentTime);
    47→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
    48→    osc.connect(gain);
    49→    gain.connect(ctx.destination);
    50→    osc.start(ctx.currentTime);
    51→    osc.stop(ctx.currentTime + 0.1);
    52→    setTimeout(() => ctx.close(), 150);
    53→  } catch { /* silent fail */ }
    54→}

    56→function playCountdownBeep(final = false) {
    57→  try {
    58→    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
    59→    const osc = ctx.createOscillator();
    60→    const gain = ctx.createGain();
    61→    osc.type = "square";
    62→    osc.frequency.setValueAtTime(final ? 880 : 440, ctx.currentTime);
    63→    gain.gain.setValueAtTime(0.08, ctx.currentTime);
    64→    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + (final ? 0.3 : 0.15));
    65→    osc.connect(gain);
    66→    gain.connect(ctx.destination);
    67→    osc.start(ctx.currentTime);
    68→    osc.stop(ctx.currentTime + (final ? 0.3 : 0.15));
    69→    setTimeout(() => ctx.close(), 400);
    70→  } catch { /* silent fail */ }
    71→}
