    54→    54→function pickPattern(total: number, count: number): Set<number> {
    55→    55→  const set = new Set<number>();
    56→    56→  while (set.size < count) set.add(Math.floor(Math.random() * total));
    57→    57→  return set;
    58→    58→}

    60→    60→function calcRoundScore(taps: Set<number>, pattern: Set<number>): number {
    61→    61→  let correct = 0;
    62→    62→  let wrong = 0;
    63→    63→  taps.forEach((c) => (pattern.has(c) ? correct++ : wrong++));
    64→    64→  return Math.max(0, correct * 10 - wrong * 5);
    65→    65→}

   150→   150→function Cell({
   151→   151→  index,
   152→   152→  inPattern,
   153→   153→  patternStaggerIndex,
   154→   154→  isTapped,
   155→   155→  isWrongTap,
   156→   156→  isCorrectTap,
   157→   157→  isMissed,
   158→   158→  showPattern,
   159→   159→  interactive,
   160→   160→  tapColor,
   161→   161→  phase,
   162→   162→  onTap,
   163→   163→}: {
   164→   164→  index: number;
   165→   165→  inPattern: boolean;
   166→   166→  patternStaggerIndex: number;
   167→   167→  isTapped: boolean;
   168→   168→  isWrongTap: boolean;
   169→   169→  isCorrectTap: boolean;
   170→   170→  isMissed: boolean;
   171→   171→  showPattern: boolean;
   172→   172→  interactive: boolean;
   173→   173→  tapColor: "cyan" | "pink" | null;
   174→   174→  phase: Phase;
   175→   175→  onTap: (i: number) => void;
   176→   176→}) {
   177→   177→  /* bg class */
   178→   178→  let bg = "bg-zinc-800/70 border-zinc-700/40";
   179→   179→  let glow = "";
   180→   180→  let shouldShake = false;
   181→   181→  let shouldPulse = false;
   182→   182→  let animDelay = 0;
   183→   183→
   184→   184→  if (phase === "memorize") {
   185→   185→    if (inPattern) {
   186→   186→      bg = "bg-violet-500 border-violet-400/60";
   187→   187→      glow = "shadow-[0_0_22px_4px_rgba(139,92,246,0.55)]";
   188→   188→      shouldPulse = true;
   189→   189→      animDelay = patternStaggerIndex * 0.07;
   190→   190→    } else {
   191→   191→      bg = "bg-zinc-800/50 border-zinc-700/30";
   192→   192→    }
   193→   193→  } else if (phase === "recall_p1" || phase === "recall_p2") {
   194→   194→    if (isTapped && tapColor === "cyan") {
   195→   195→      bg = "bg-cyan-500 border-cyan-400/60";
   196→   196→      glow = "shadow-[0_0_16px_3px_rgba(6,182,212,0.45)]";
   197→   197→    } else if (isTapped && tapColor === "pink") {
   198→   198→      bg = "bg-pink-500 border-pink-400/60";
   199→   199→      glow = "shadow-[0_0_16px_3px_rgba(236,72,153,0.45)]";
   200→   200→    } else {
   201→   201→      bg = "bg-zinc-800/70 border-zinc-700/40";
   202→   202→    }
   203→   203→  } else if (phase === "feedback_p1" || phase === "feedback_p2") {
   204→   204→    if (isCorrectTap) {
   205→   205→      bg = "bg-emerald-500 border-emerald-400/60";
   206→   206→      glow = "shadow-[0_0_16px_3px_rgba(16,185,129,0.5)]";
   207→   207→    } else if (isWrongTap) {
   208→   208→      bg = "bg-red-500 border-red-400/60";
   209→   209→      shouldShake = true;
   210→   210→    } else if (isMissed) {
   211→   211→      bg = "bg-violet-700/35 border-violet-600/30";
   212→   212→    } else {
   213→   213→      bg = "bg-zinc-800/40 border-zinc-700/20";
   214→   214→    }
   215→   215→  } else if (phase === "round_result" || phase === "game_over") {
   216→   216→    if (showPattern && inPattern) {
   217→   217→      bg = "bg-violet-600/40 border-violet-500/30";
   218→   218→    } else {
   219→   219→      bg = "bg-zinc-800/30 border-zinc-700/20";
   220→   220→    }
   221→   221→  } else {
   222→   222→    bg = "bg-zinc-800/70 border-zinc-700/40";
   223→   223→  }
   224→   224→
   225→   225→  return (
   226→   226→    <motion.button
   227→   227→      type="button"
   228→   228→      onClick={() => interactive && onTap(index)}
   229→   229→      className={cn(
   230→   230→        "w-14 h-14 sm:w-18 sm:h-18 rounded-xl border-2 transition-[background-color,border-color,box-shadow] duration-150",
   231→   231→        bg,
   232→   232→        glow,
   233→   233→        interactive && "cursor-pointer hover:brightness-125",
   234→   234→        !interactive && "cursor-default",
   235→   235→        shouldPulse && phase === "memorize" && "animate-pulse"
   236→   236→      )}
   237→   237→      animate={{
   238→   238→        ...(shouldShake ? shakeKeyframes : { x: 0 }),
   239→   239→        scale: phase === "memorize" && inPattern
   240→   240→          ? [0.6, 1.08, 1]
   241→   241→          : shouldShake
   242→   242→            ? [1, 0.95, 1.02, 0.98, 1]
   243→   243→            : 1,
   244→   244→      }}
   245→   245→      transition={{
   246→   246→        x: shouldShake ? { duration: 0.5, ease: "easeOut" } : { duration: 0.1 },
   247→   247→        scale: { duration: 0.35, ease: "easeOut" },
   248→   248→        delay: animDelay,
   249→   249→      }}
   250→   250→      whileTap={interactive ? { scale: 0.82 } : undefined}
   251→   251→      whileHover={interactive ? { scale: 1.06 } : undefined}
   252→   252→    />
   253→   253→  );
   254→   254→}

   259→   259→
   260→   260→export default function PatternMemory({ onScore, liveCode }: Props) {
   261→   261→  /* ── state ── */
   262→   262→  const [phase, setPhase] = useState<Phase>("idle");
   263→   263→  const [round, setRound] = useState(0);
   264→   264→  const [pattern, setPattern] = useState<Set<number>>(new Set());
   265→   265→  const [p1Taps, setP1Taps] = useState<Set<number>>(new Set());
   266→   266→  const [p2Taps, setP2Taps] = useState<Set<number>>(new Set());
   267→   267→  const [p1Total, setP1Total] = useState(0);
   268→   268→  const [p2Total, setP2Total] = useState(0);
   269→   269→  const [p1RoundPts, setP1RoundPts] = useState(0);
   270→   270→  const [p2RoundPts, setP2RoundPts] = useState(0);
   271→   271→  const [memProgress, setMemProgress] = useState(100);
   272→   272→
   273→   273→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   274→   274→  const onScoreFired = useRef(false);
   275→   275→  const scoresRef = useRef({ p1: 0, p2: 0 });
   276→   276→
   277→   277→  const cfg = ROUNDS[round] ?? ROUNDS[0];
   278→   278→  const totalCells = cfg.gridSize * cfg.gridSize;
   279→   279→  const patternArr = Array.from(pattern).sort((a, b) => a - b);
   280→   280→
   281→   281→  const recalling = phase === "recall_p1" || phase === "recall_p2";
   282→   282→  const isP1 = phase === "recall_p1";
   283→   283→
   284→   284→  /* ── timer cleanup ── */
   285→   285→  const clearTimer = useCallback(() => {
   286→   286→    if (timerRef.current) {
   287→   287→      clearInterval(timerRef.current);
   288→   288→      timerRef.current = null;
   289→   289→    }
   290→   290→  }, []);
   291→   291→
   292→   292→  useEffect(() => clearTimer, [clearTimer]);
   293→   293→
   294→   294→  /* ── memorize timer ── */
   295→   295→  useEffect(() => {
   296→   296→    if (phase !== "memorize") return;
   297→   297→    clearTimer();
   298→   298→    const start = Date.now();
   299→   299→    const durationMs = cfg.memorizeTime * 1000;
   300→   300→    timerRef.current = setInterval(() => {
   301→   301→      const elapsed = Date.now() - start;
   302→   302→      const pct = Math.max(0, 100 - (elapsed / durationMs) * 100);
   303→   303→      setMemProgress(pct);
   304→   304→      if (elapsed >= durationMs) {
   305→   305→        clearTimer();
   306→   306→        setPhase("recall_p1");
   307→   307→      }
   308→   308→    }, 40);
   309→   309→    return clearTimer;
   310→   310→  }, [phase, cfg.memorizeTime, clearTimer]);
   311→   311→
   312→   312→  /* ── feedback auto-advance timers ── */
   313→   313→  useEffect(() => {
   314→   314→    if (phase === "feedback_p1") {
   315→   315→      const id = setTimeout(() => setPhase("recall_p2"), 1800);
   316→   316→      return () => clearTimeout(id);
   317→   317→    }
   318→   318→    if (phase === "feedback_p2") {
   319→   319→      const id = setTimeout(() => setPhase("round_result"), 1800);
   320→   320→      return () => clearTimeout(id);
   321→   321→    }
   322→   322→  }, [phase]);
   323→   323→
   324→   324→  /* ── start round ── */
   325→   325→  const startRound = useCallback(() => {
   326→   326→    const pat = pickPattern(totalCells, cfg.patternSize);
   327→   327→    setPattern(pat);
   328→   328→    setP1Taps(new Set());
   329→   329→    setP2Taps(new Set());
   330→   330→    setP1RoundPts(0);
   331→   331→    setP2RoundPts(0);
   332→   332→    setMemProgress(100);
   333→   333→    setPhase("memorize");
   334→   334→  }, [totalCells, cfg.patternSize]);
   335→   335→
   336→   336→  /* ── begin full game ── */
   337→   337→  const handleStart = useCallback(() => {
   338→   338→    scoresRef.current = { p1: 0, p2: 0 };
   339→   339→    onScoreFired.current = false;
   340→   340→    setRound(0);
   341→   341→    setP1Total(0);
   342→   342→    setP2Total(0);
   343→   343→    setP1Taps(new Set());
   344→   344→    setP2Taps(new Set());
   345→   345→    setP1RoundPts(0);
   346→   346→    setP2RoundPts(0);
   347→   347→    setPattern(new Set());
   348→   348→    setPhase("idle");
   349→   349→    // Use round 0 config values directly instead of stale closure
   350→   350→    const r0Cells = 3;
   351→   351→    const r0Size = 2;
   352→   352→    const pat = pickPattern(r0Cells * r0Cells, r0Size);
   353→   353→    setPattern(pat);
   354→   354→    setP1Taps(new Set());
   355→   355→    setP2Taps(new Set());
   356→   356→    setP1RoundPts(0);
   357→   357→    setP2RoundPts(0);
   358→   358→    setMemProgress(100);
   359→   359→    setTimeout(() => setPhase("memorize"), 50);
   360→   360→  }, []);
   361→   361→
   362→   362→  /* ── tap cell ── */
   363→   363→  const handleTap = useCallback(
   364→   364→    (idx: number) => {
   365→   365→      if (phase === "recall_p1") {
   366→   366→        setP1Taps((prev) => {
   367→   367→          const next = new Set(prev);
   368→   368→          next.has(idx) ? next.delete(idx) : next.add(idx);
   369→   369→          return next;
   370→   370→        });
   371→   371→      } else if (phase === "recall_p2") {
   372→   372→        setP2Taps((prev) => {
   373→   373→          const next = new Set(prev);
   374→   374→          next.has(idx) ? next.delete(idx) : next.add(idx);
   375→   375→          return next;
   376→   376→        });
   377→   377→      }
   378→   378→    },
   379→   379→    [phase]
   380→   380→  );
   381→   381→
   382→   382→  /* ── confirm recall ── */
   383→   383→  const handleConfirm = useCallback(() => {
   384→   384→    if (phase === "recall_p1") {
   385→   385→      const pts = calcRoundScore(p1Taps, pattern);
   386→   386→      setP1RoundPts(pts);
   387→   387→      scoresRef.current.p1 += pts;
   388→   388→      setP1Total(scoresRef.current.p1);
   389→   389→      setPhase("feedback_p1");
   390→   390→    } else if (phase === "recall_p2") {
   391→   391→      const pts = calcRoundScore(p2Taps, pattern);
   392→   392→      setP2RoundPts(pts);
   393→   393→      scoresRef.current.p2 += pts;
   394→   394→      setP2Total(scoresRef.current.p2);
   395→   395→      setPhase("feedback_p2");
   396→   396→    }
   397→   397→  }, [phase, p1Taps, p2Taps, pattern]);
   398→   398→
   399→   399→  /* ── next round / game over ── */
   400→   400→  const handleNextRound = useCallback(() => {
   401→   401→    const nextRound = round + 1;
   402→   402→    if (nextRound >= TOTAL_ROUNDS) {
   403→   403→      setPhase("game_over");
   404→   404→      if (onScore && !onScoreFired.current) {
   405→   405→        onScoreFired.current = true;
   406→   406→        const { p1, p2 } = scoresRef.current;
   407→   407→        if (p1 > p2) {
   408→   408→          onScore("Jogador 1", p1);
   409→   409→        } else if (p2 > p1) {
   410→   410→          onScore("Jogador 2", p2);
   411→   411→        } else {
   412→   412→          onScore("Jogador 1", p1);
   413→   413→          onScore("Jogador 2", p2);
   414→   414→        }
   415→   415→      }
   416→   416→    } else {
   417→   417→      setRound(nextRound);
   418→   418→      setPhase("idle");
   419→   419→      setTimeout(() => startRound(), 0);
   420→   420→    }
   421→   421→  }, [round, onScore, startRound]);
   422→   422→
   423→   423→  /* ── restart ── */
   424→   424→  const handleRestart = useCallback(() => {
   425→   425→    clearTimer();
   426→   426→    handleStart();
   427→   427→  }, [clearTimer, handleStart]);
   428→   428→
   429→   429→  /* ── derived sets for feedback ── */
   430→   430→  const fbTaps =
   431→   431→    phase === "feedback_p1" ? p1Taps : phase === "feedback_p2" ? p2Taps : null;
   432→   432→  const correctSet = new Set<number>();
   433→   433→  const wrongSet = new Set<number>();
   434→   434→  const missedSet = new Set<number>();
   435→   435→  if (fbTaps) {
   436→   436→    fbTaps.forEach((c) => (pattern.has(c) ? correctSet.add(c) : wrongSet.add(c)));
   437→   437→    pattern.forEach((c) => {
   438→   438→      if (!fbTaps.has(c)) missedSet.add(c);
   439→   439→    });
   440→   440→  }
   441→   441→
   442→   442→  const currentTaps =
   443→   443→    phase === "recall_p1" ? p1Taps : phase === "recall_p2" ? p2Taps : new Set<number>();
   444→   444→  const activePlayer: 1 | 2 = phase === "recall_p1" ? 1 : 2;
   445→   445→
   446→   446→  /* ================================================================ */
   447→   447→  /*  RENDER                                                           */
   448→   448→  /* ================================================================ */
   449→   449→
   450→   450→  return (
   451→   451→    <div className="min-h-[540px] bg-gradient-to-b from-zinc-900 to-zinc-950 text-white flex flex-col items-center justify-center p-4 select-none overflow-hidden rounded-2xl border border-zinc-800/60 shadow-2xl">
   452→   452→      <AnimatePresence>
   453→   453→        {phase !== "idle" && (
   454→   454→          <motion.div
   455→   455→            initial={{ opacity: 0, y: -12 }}
   456→   456→            animate={{ opacity: 1, y: 0 }}
   457→   457→            exit={{ opacity: 0 }}
   458→   458→            className="w-full flex justify-center mb-4"
   459→   459→          >
   460→   460→            <ScoreBar
   461→   461→              p1={p1Total}
   462→   462→              p2={p2Total}
   463→   463→              round={round}
   464→   464→              highlight={phase === "recall_p1" ? 1 : phase === "recall_p2" ? 2 : undefined}
   465→   465→            />
   466→   466→          </motion.div>
   467→   467→        )}
   468→   468→      </AnimatePresence>
