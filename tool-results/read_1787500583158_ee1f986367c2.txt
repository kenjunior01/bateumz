    66→    66→function shuffle<T>(arr: T[]): T[] {
    67→    67→  const a = [...arr];
    68→    68→  for (let i = a.length - 1; i > 0; i--) {
    69→    69→    const j = Math.floor(Math.random() * (i + 1));
    70→    70→    [a[i], a[j]] = [a[j], a[i]];
    71→    71→  }
    72→    72→  return a;
    73→    73→}

    75→    75→function makeRound(): RoundData {
    76→    76→  const word = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    77→    77→  let ink: ColorDef;
    78→    78→  do {
    79→    79→    ink = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    80→    80→  } while (ink.id === word.id);
    81→    81→  const others = shuffle(ALL_COLORS.filter((c) => c.id !== ink.id)).slice(0, 3);
    82→    82→  return { word, inkColor: ink, choices: shuffle([ink, ...others]) };
    83→    83→}

    86→    86→  const t = (roundNum - 1) / (TOTAL_ROUNDS - 1);
    87→    87→  return Math.round(INITIAL_TIME_MS - t * (INITIAL_TIME_MS - MIN_TIME_MS));
    88→    88→}
