    59→
    60→function generateTiles(): Tile[] {
    61→  const tiles: Tile[] = [];
    62→  for (let i = 0; i <= 6; i++) {
    63→    for (let j = i; j <= 6; j++) {
    64→      tiles.push({ left: i, right: j, id: `${i}-${j}` });
    65→    }
    66→  }
    67→  return tiles;
    68→}
    69→
    70→function shuffle<T>(arr: T[]): T[] {
    71→  const a = [...arr];
    72→  for (let i = a.length - 1; i > 0; i--) {
    73→    const j = Math.floor(Math.random() * (i + 1));
    74→    [a[i], a[j]] = [a[j], a[i]];
    75→  }
    76→  return a;
    77→}
    78→
    79→function tilePips(t: Tile | ChainTile): number {
    80→  return t.left + t.right;
    81→}
    82→
    83→function handPips(hand: (Tile | ChainTile)[]): number {
    84→  return hand.reduce((s, t) => s + tilePips(t), 0);
    85→}
    86→
    87→function canMatch(value: number, tile: Tile): boolean {
    88→  return tile.left === value || tile.right === value;
    89→}
    90→
    91→function validEndsForTile(
    92→  tile: Tile,
    93→  leftEnd: number | null,
    94→  rightEnd: number | null,
    95→  chainLen: number
    96→): End[] {
    97→  if (chainLen === 0) return ['left'];
    98→  if (leftEnd === null || rightEnd === null) return [];
    99→  const ends: End[] = [];
   100→  if (canMatch(leftEnd, tile)) ends.push('left');
   101→  if (canMatch(rightEnd, tile)) ends.push('right');
   102→  return [...new Set(ends)];
   103→}

   107→function DotHalf({ value }: { value: number }) {

   114→          <div
   115→            key={i}
   116→            className="absolute w-1.5 h-1.5 rounded-full bg-slate-900 dark:bg-slate-800"
   117→            style={{
   118→              top: p.top,
   119→              left: p.left,
   120→              transform: 'translate(-50%, -50%)',
   121→            }}
   122→          />
   123→        );
   124→      })}
   125→    </div>
   126→  );
   127→}

   129→function DominoTile({
   130→  tile,
   131→  faceDown = false,
   132→  selected = false,
   133→  playable = false,
   134→  highlightEnd = false,
   135→  onClick,
   136→}: {
   137→  tile?: Tile | ChainTile;
   138→  faceDown?: boolean;
   139→  selected?: boolean;
   140→  playable?: boolean;
   141→  highlightEnd?: boolean;

   147→        <div className="w-3/4 h-3/4 rounded-sm bg-slate-600/40 border border-slate-500/30">
   148→          <div className="w-full h-full rounded-sm bg-gradient-to-br from-slate-500/20 to-slate-700/20" />
   149→        </div>
   150→      </div>
   151→    );
   152→  }
   153→
   154→  if (!tile) return null;
   155→
   156→  return (
   157→    <motion.div
   158→      onClick={onClick}
   159→      whileHover={onClick ? { scale: 1.08, y: -3 } : undefined}
   160→      whileTap={onClick ? { scale: 0.93 } : undefined}
   161→      className={cn(
   162→        'w-16 h-8 sm:w-20 sm:h-10 rounded-lg bg-white dark:bg-slate-100 border border-slate-300 dark:border-slate-500 flex overflow-hidden shadow-sm shrink-0',
   163→        selected && 'ring-2 ring-cyan-400 shadow-cyan-400/30 shadow-lg z-10',
   164→        playable && !selected && 'ring-2 ring-amber-400/80 shadow-amber-400/20 shadow-md',
   165→        highlightEnd && 'ring-2 ring-amber-400 shadow-amber-400/30 shadow-lg',
   166→        onClick && 'cursor-pointer'
   167→      )}
   168→    >
   169→      <div className="w-1/2 h-full relative border-r border-slate-300 dark:border-slate-400">
   170→        <DotHalf value={tile.left} />
   171→      </div>
   172→      <div className="w-1/2 h-full relative">
   173→        <DotHalf value={tile.right} />
   174→      </div>
   175→    </motion.div>
   176→  );
   177→}

   179→/* ═══════════════════════ Main Component ═══════════════════════ */
   180→
   181→export default function Dominoes({ onScore, liveCode: _liveCode }: Props) {
   182→  /* ─── state ─── */
   183→  const [phase, setPhase] = useState<Phase>('dealing');
   184→  const [hands, setHands] = useState<Tile[][]>([[], []]);
   185→  const [chain, setChain] = useState<ChainTile[]>([]);
   186→  const [boneyard, setBoneyard] = useState<Tile[]>([]);
   187→  const [currentPlayer, setCurrentPlayer] = useState<0 | 1>(0);
   188→  const [selectedTileId, setSelectedTileId] = useState<string | null>(null);
   189→  const [scores, setScores] = useState<[number, number]>([0, 0]);
   190→  const [vsBot, setVsBot] = useState(false);
   191→  const [roundResult, setRoundResult] = useState('');
   192→  const [message, setMessage] = useState('');
   193→  const [consecutivePasses, setConsecutivePasses] = useState(0);
   194→  const [roundNum, setRoundNum] = useState(1);
   195→
   196→  const chainRef = useRef<HTMLDivElement>(null);
   197→  const botTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   198→
   199→  /* ─── derived ─── */
   200→  const leftEnd = chain.length > 0 ? chain[0].left : null;
   201→  const rightEnd = chain.length > 0 ? chain[chain.length - 1].right : null;
   202→  const oppIdx = (1 - currentPlayer) as 0 | 1;
   203→
   204→  const currentHand = hands[currentPlayer];
   205→
   206→  const hasValidMove = useMemo(() => {
   207→    if (chain.length === 0) return currentHand.length > 0;
   208→    if (leftEnd === null || rightEnd === null) return false;
   209→    return currentHand.some((t) => canMatch(leftEnd, t) || canMatch(rightEnd, t));
   210→  }, [currentHand, chain.length, leftEnd, rightEnd]);
   211→
   212→  const canDraw = boneyard.length > 0 && !hasValidMove;
   213→  const canPass = boneyard.length === 0 && !hasValidMove && chain.length > 0;
   214→
   215→  const selectedTile = selectedTileId
   216→    ? currentHand.find((t) => t.id === selectedTileId) ?? null
   217→    : null;
   218→
   219→  const selectedValidEnds: End[] = useMemo(() => {
   220→    if (!selectedTile) return [];
   221→    return validEndsForTile(selectedTile, leftEnd, rightEnd, chain.length);
   222→  }, [selectedTile, leftEnd, rightEnd, chain.length]);
   223→
   224→  const bottomIdx = vsBot ? 0 : currentPlayer;
   225→  const topIdx = (1 - bottomIdx) as 0 | 1;
   226→  const isHumanTurn = phase === 'playing' && bottomIdx === currentPlayer;
   227→
   228→  /* ─── callbacks ─── */
   229→
   230→  const endRoundBlocked = useCallback(() => {
   231→    const p1 = handPips(hands[0]);
   232→    const p2 = handPips(hands[1]);
   233→    if (p1 < p2) {
   234→      setScores((s) => [s[0] + p2, s[1]]);
   235→      setRoundResult(`Venceu! ${PLAYER_NAMES[0]} (${p1} vs ${p2})`);
   236→      onScore?.(PLAYER_NAMES[0], p2);
   237→    } else if (p2 < p1) {
   238→      setScores((s) => [s[0], s[1] + p1]);
   239→      setRoundResult(`Venceu! ${PLAYER_NAMES[1]} (${p2} vs ${p1})`);
   240→      onScore?.(PLAYER_NAMES[1], p1);
   241→    } else {
   242→      setRoundResult('Empate!');
   243→    }
   244→    setPhase('roundOver');
   245→  }, [hands, onScore]);
   246→
   247→  const startRound = useCallback(() => {
   248→    const all = shuffle(generateTiles());
   249→    setHands([all.slice(0, HAND_SIZE), all.slice(HAND_SIZE, HAND_SIZE * 2)]);
   250→    setChain([]);
   251→    setBoneyard(all.slice(HAND_SIZE * 2));
   252→    setCurrentPlayer(0);
   253→    setSelectedTileId(null);
   254→    setRoundResult('');
   255→    setMessage('');
   256→    setConsecutivePasses(0);
   257→    setPhase('dealing');
   258→  }, []);
   259→
   260→  const restartAll = useCallback(() => {
   261→    setScores([0, 0]);
   262→    setRoundNum(1);
   263→    startRound();
   264→  }, [startRound]);
   265→
   266→  const nextRound = useCallback(() => {
   267→    setRoundNum((n) => n + 1);
   268→    startRound();
   269→  }, [startRound]);
   270→
   271→  const placeTile = useCallback(
   272→    (tileId: string, end: End) => {
   273→      const hand = [...hands[currentPlayer]];
   274→      const idx = hand.findIndex((t) => t.id === tileId);
   275→      if (idx === -1) return;
   276→
   277→      const tile = hand.splice(idx, 1)[0];
   278→      let ct: ChainTile;
   279→
   280→      if (chain.length === 0) {
   281→        ct = { left: tile.left, right: tile.right, id: `${tile.id}-c0` };
   282→        setChain([ct]);
   283→      } else if (end === 'left') {
   284→        ct =
   285→          tile.right === leftEnd
   286→            ? { left: tile.left, right: tile.right, id: `${tile.id}-L` }
   287→            : { left: tile.right, right: tile.left, id: `${tile.id}-L` };
   288→        setChain((prev) => [ct, ...prev]);
   289→      } else {
   290→        ct =
   291→          tile.left === rightEnd
   292→            ? { left: tile.left, right: tile.right, id: `${tile.id}-R` }
   293→            : { left: tile.right, right: tile.left, id: `${tile.id}-R` };
   294→        setChain((prev) => [...prev, ct]);
   295→      }
   296→
   297→      setHands((prev) => {
   298→        const h: Tile[][] = [prev[0].slice(), prev[1].slice()];
   299→        h[currentPlayer] = hand;
   300→        return h;
   301→      });
   302→      setSelectedTileId(null);
   303→      setConsecutivePasses(0);
   304→
   305→      if (hand.length === 0) {
   306→        const oppPips = handPips(hands[oppIdx]);
   307→        setScores((s) => {
   308→          const n: [number, number] = [s[0], s[1]];
   309→          n[currentPlayer] += oppPips;
   310→          return n;
   311→        });
   312→        setRoundResult(`Venceu! ${PLAYER_NAMES[currentPlayer]}`);
   313→        onScore?.(PLAYER_NAMES[currentPlayer], oppPips);
   314→        setPhase('roundOver');
   315→        return;
   316→      }
   317→
   318→      setCurrentPlayer(oppIdx);
   319→    },
   320→    [hands, currentPlayer, chain, leftEnd, rightEnd, oppIdx, onScore]
   321→  );
   322→
   323→  const handleTileClick = useCallback(
   324→    (tileId: string) => {
   325→      if (phase !== 'playing' || !isHumanTurn) return;
   326→      if (selectedTileId === tileId) {
   327→        setSelectedTileId(null);
   328→        return;
   329→      }
   330→
   331→      const tile = currentHand.find((t) => t.id === tileId);
   332→      if (!tile) return;
   333→
   334→      const ends = validEndsForTile(tile, leftEnd, rightEnd, chain.length);
   335→      if (ends.length === 0) return;
   336→
   337→      setSelectedTileId(tileId);
   338→
   339→      if (ends.length === 1) {
   340→        placeTile(tileId, ends[0]);
   341→      }
   342→    },
   343→    [phase, isHumanTurn, selectedTileId, currentHand, leftEnd, rightEnd, chain.length, placeTile]
   344→  );
   345→
   346→  const handleEndClick = useCallback(
   347→    (end: End) => {
   348→      if (!selectedTileId || phase !== 'playing' || !isHumanTurn) return;
   349→      if (selectedValidEnds.includes(end)) {
   350→        placeTile(selectedTileId, end);
   351→      }
   352→    },
   353→    [selectedTileId, phase, isHumanTurn, selectedValidEnds, placeTile]
   354→  );
   355→
   356→  const drawFromBoneyard = useCallback(() => {
   357→    if (boneyard.length === 0 || hasValidMove) return;
   358→    const newBy = [...boneyard];
   359→    const drawn = newBy.pop()!;
   360→    setBoneyard(newBy);
   361→    setHands((prev) => {
   362→      const h: Tile[][] = [prev[0].slice(), prev[1].slice()];
   363→      h[currentPlayer] = [...h[currentPlayer], drawn];
   364→      return h;
   365→    });
   366→    setMessage('Comprou uma pe\u00e7a');
   367→  }, [boneyard, hasValidMove, currentPlayer]);
   368→
   369→  const passTurn = useCallback(() => {
   370→    if (!canPass) return;
   371→    const np = consecutivePasses + 1;
   372→    if (np >= 2) {
   373→      endRoundBlocked();
   374→      return;
   375→    }
   376→    setConsecutivePasses(np);
   377→    setSelectedTileId(null);
   378→    setCurrentPlayer(oppIdx);
   379→    setMessage(`${PLAYER_NAMES[currentPlayer]} passou`);
   380→  }, [canPass, consecutivePasses, oppIdx, endRoundBlocked, currentPlayer]);
   381→
   382→  const toggleMode = useCallback(() => {
   383→    setVsBot((v) => !v);
   384→    startRound();
   385→  }, [startRound]);
   386→
   387→  /* ─── effects ─── */
   388→
   389→  useEffect(() => {
   390→    startRound();
   391→  }, [startRound]);
   392→
   393→  useEffect(() => {
   394→    if (phase === 'dealing') {
   395→      const t = setTimeout(() => setPhase('playing'), 1200);
   396→      return () => clearTimeout(t);
   397→    }
   398→  }, [phase]);
   399→
   400→  useEffect(() => {
   401→    if (phase !== 'playing') return;
   402→    if (vsBot && currentPlayer === 1) {
   403→      setMessage(`${PLAYER_NAMES[1]} est\u00e1 pensando...`);
   404→    } else if (chain.length === 0) {
   405→      setMessage('Coloque a primeira pe\u00e7a');
   406→    } else if (hasValidMove) {
   407→      setMessage(`Sua vez \u2014 ${PLAYER_NAMES[currentPlayer]}`);
   408→    } else if (canDraw) {
   409→      setMessage('Sem pe\u00e7as v\u00e1lidas \u2014 compre do osso');
   410→    } else {
   411→      setMessage('Bloqueado!');
   412→    }
   413→  }, [phase, vsBot, currentPlayer, chain.length, hasValidMove, canDraw]);
   414→
   415→  useEffect(() => {
   416→    if (phase !== 'playing' || !vsBot || currentPlayer !== 1) return;
   417→
   418→    botTimer.current = setTimeout(() => {
   419→      const botHand = hands[1];
   420→      if (botHand.length === 0) return;
   421→
   422→      if (chain.length === 0) {
   423→        const sorted = [...botHand].sort((a, b) => tilePips(b) - tilePips(a));
   424→        placeTile(sorted[0].id, 'left');
   425→        return;
   426→      }
   427→
   428→      const moves = botHand
   429→        .map((t) => ({
   430→          tile: t,
   431→          ends: validEndsForTile(t, leftEnd, rightEnd, chain.length),
   432→        }))
   433→        .filter((m) => m.ends.length > 0)
   434→        .sort((a, b) => tilePips(b.tile) - tilePips(a.tile));
   435→
   436→      if (moves.length > 0) {
   437→        placeTile(moves[0].tile.id, moves[0].ends[0]);
   438→      } else if (boneyard.length > 0) {
   439→        drawFromBoneyard();
   440→      } else {
   441→        const np = consecutivePasses + 1;
   442→        if (np >= 2) {
   443→          endRoundBlocked();
   444→        } else {
   445→          setConsecutivePasses(np);
   446→          setCurrentPlayer(0);
   447→          setMessage(`${PLAYER_NAMES[1]} passou`);
   448→        }
   449→      }
   450→    }, 700 + Math.random() * 500);
   451→
   452→    return () => {
   453→      if (botTimer.current) clearTimeout(botTimer.current);
   454→    };
   455→  }, [
   456→    phase, vsBot, currentPlayer, hands, chain, leftEnd, rightEnd,
   457→    boneyard, placeTile, drawFromBoneyard, consecutivePasses, endRoundBlocked,
   458→  ]);
   459→
   460→  /* ─── render helpers ─── */
   461→
   462→  const colorClass = (idx: 0 | 1) => (idx === 0 ? 'text-cyan-400' : 'text-pink-400');
   463→  const dotColor = (idx: 0 | 1) => (idx === 0 ? 'bg-cyan-400' : 'bg-pink-400');
   464→
   465→  const isPlayableInHand = (tile: Tile): boolean => {
   466→    if (phase !== 'playing' || !isHumanTurn) return false;
   467→    return validEndsForTile(tile, leftEnd, rightEnd, chain.length).length > 0;
   468→  };
   469→
   470→  /* ═══════════════════════ JSX ═══════════════════════ */
   471→
   472→  return (
   473→    <div className="relative w-full max-w-4xl mx-auto flex flex-col gap-3 p-2 sm:p-4 select-none">
   474→      <motion.div
   475→        initial={{ opacity: 0, y: -20 }}
   476→        animate={{ opacity: 1, y: 0 }}
   477→        className="bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 rounded-xl p-3 flex items-center justify-between"
   478→      >
   479→        <div className="flex items-center gap-2">
   480→          <div className="w-3 h-3 rounded-full bg-cyan-400" />
   481→          <span className="text-cyan-400 font-bold text-xs sm:text-sm">{PLAYER_NAMES[0]}</span>
   482→          <motion.span
   483→            key={scores[0]}
   484→            initial={{ scale: 1.6, color: '#22d3ee' }}
   485→            animate={{ scale: 1, color: '#ffffff' }}
   486→            transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   487→            className="text-white font-mono text-lg sm:text-xl font-bold inline-block"
   488→          >
   489→            {scores[0]}
   490→          </motion.span>
   491→        </div>
   492→
   493→        <div className="text-center">
   494→          <h2 className="text-white font-bold text-base sm:text-2xl tracking-widest">DOMIN\u00d3</h2>
   495→          <span className="text-slate-500 text-[10px]">Rodada {roundNum}</span>
   496→        </div>
   497→
   498→        <div className="flex items-center gap-2">
   499→          <motion.span
   500→            key={scores[1]}
   501→            initial={{ scale: 1.6, color: '#f472b6' }}
   502→            animate={{ scale: 1, color: '#ffffff' }}
   503→            transition={{ type: 'spring', stiffness: 400, damping: 15 }}
   504→            className="text-white font-mono text-lg sm:text-xl font-bold inline-block"
   505→          >
   506→            {scores[1]}
   507→          </motion.span>
   508→          <span className="text-pink-400 font-bold text-xs sm:text-sm">{PLAYER_NAMES[1]}</span>
   509→          <div className="w-3 h-3 rounded-full bg-pink-400" />
   510→        </div>
   511→      </motion.div>
   512→
   513→      <div className="flex items-center justify-between gap-2 flex-wrap">
   514→        <div className="flex items-center gap-2">
   515→          <Button
   516→            size="sm"
   517→            variant={vsBot ? 'default' : 'outline'}
   518→            onClick={toggleMode}
   519→            className={cn('text-xs gap-1', vsBot ? 'bg-pink-600 hover:bg-pink-700 text-white' : '')}
   520→          >
   521→            <Bot className="w-3 h-3" />
   522→            VS Bot
   523→          </Button>
   524→          <Badge variant="outline" className="text-[10px] sm:text-xs text-slate-400 gap-1">
   525→            <ArrowDown className="w-3 h-3" />
   526→            Pecas no osso: {boneyard.length}
   527→          </Badge>
   528→        </div>
   529→        <Button size="sm" variant="ghost" onClick={restartAll} className="text-xs text-slate-400 gap-1">
   530→          <RotateCcw className="w-3 h-3" />
   531→          Reiniciar Tudo
   532→        </Button>
   533→      </div>
   534→
