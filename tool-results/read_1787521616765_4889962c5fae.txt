     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import { Button } from "@/components/ui/button";
     6→import { Badge } from "@/components/ui/badge";
     7→import { cn } from "@/lib/utils";
     8→import {
     9→  RotateCcw,
    10→  Timer,
    11→  Zap,
    12→  Trophy,
    13→  Crown,
    14→  Target,
    15→  AlertTriangle,
    16→  X,
    17→  Swords,
    18→  Flame,
    19→  Sparkles,
    20→} from "lucide-react";
    21→
    22→interface Props {
    23→  onScore?: (name: string, score: number) => void;
    24→  liveCode?: string;
    25→}
    26→
    27→type GameState = "waiting" | "countdown" | "ready" | "result" | "gameOver";
    28→
    29→type RoundResult = {
    30→  p1Time: number | null;
    31→  p2Time: number | null;
    32→  p1FalseStart: boolean;
    33→  p2FalseStart: boolean;
    34→  winner: 0 | 1 | 2;
    35→};
    36→
    37→const ROUND_OPTIONS = [3, 5, 7] as const;
    38→
    39→export default function ReactionRace({ onScore, liveCode }: Props) {
    40→  const [gameState, setGameState] = useState<GameState>("waiting");
    41→  const [totalRounds, setTotalRounds] = useState<3 | 5 | 7>(5);
    42→  const [currentRound, setCurrentRound] = useState(1);
    43→  const [roundResults, setRoundResults] = useState<RoundResult[]>([]);
    44→  const [p1Time, setP1Time] = useState<number | null>(null);
    45→  const [p2Time, setP2Time] = useState<number | null>(null);
    46→  const [p1FalseStart, setP1FalseStart] = useState(false);
    47→  const [p2FalseStart, setP2FalseStart] = useState(false);
    48→  const [roundWinner, setRoundWinner] = useState<0 | 1 | 2>(0);
    49→  const [goTimestamp, setGoTimestamp] = useState(0);
    50→  const [p1Pressed, setP1Pressed] = useState(false);
    51→  const [p2Pressed, setP2Pressed] = useState(false);
    52→  const countdownRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    53→  const autoAdvanceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    54→  const gameActiveRef = useRef(false);
    55→  const [elapsedMs, setElapsedMs] = useState(0);
    56→  const elapsedRafRef = useRef<number>(0);
    57→  const [prevRound, setPrevRound] = useState(0);
    58→  const [showRoundTransition, setShowRoundTransition] = useState(false);
    59→
    60→  const clearTimers = useCallback(() => {
    61→    if (countdownRef.current) {
    62→      clearTimeout(countdownRef.current);
    63→      countdownRef.current = null;
    64→    }
    65→    if (autoAdvanceRef.current) {
    66→      clearTimeout(autoAdvanceRef.current);
    67→      autoAdvanceRef.current = null;
    68→    }
    69→  }, []);
    70→
    71→  const resetGame = useCallback(() => {
    72→    clearTimers();
    73→    gameActiveRef.current = false;
    74→    setGameState("waiting");
    75→    setCurrentRound(1);
    76→    setRoundResults([]);
    77→    setP1Time(null);
    78→    setP2Time(null);
    79→    setP1FalseStart(false);
    80→    setP2FalseStart(false);
    81→    setRoundWinner(0);
    82→    setP1Pressed(false);
    83→    setP2Pressed(false);
    84→    setGoTimestamp(0);
    85→    setElapsedMs(0);
    86→    setPrevRound(0);
    87→    setShowRoundTransition(false);
    88→  }, [clearTimers]);
    89→
    90→  const startRound = useCallback(() => {
    91→    clearTimers();
    92→    gameActiveRef.current = false;
    93→    setP1Time(null);
    94→    setP2Time(null);
    95→    setP1FalseStart(false);
    96→    setP2FalseStart(false);
    97→    setRoundWinner(0);
    98→    setP1Pressed(false);
    99→    setP2Pressed(false);
   100→    setGoTimestamp(0);
   101→    setElapsedMs(0);
   102→    setGameState("countdown");
   103→
   104→    const delay = 2000 + Math.random() * 3000;
   105→    countdownRef.current = setTimeout(() => {
   106→      const now = performance.now();
   107→      setGoTimestamp(now);
   108→      gameActiveRef.current = true;
   109→      setGameState("ready");
   110→    }, delay);
   111→  }, [clearTimers]);
   112→
   113→  const handleReaction = useCallback(
   114→    (player: 1 | 2) => {
   115→      if (gameState === "waiting" || gameState === "result" || gameState === "gameOver") {
   116→        return;
   117→      }
   118→
   119→      if (player === 1 && p1Pressed) return;
   120→      if (player === 2 && p2Pressed) return;
   121→
   122→      if (gameState === "countdown") {
   123→        if (player === 1) {
   124→          setP1FalseStart(true);
   125→          setP1Pressed(true);
   126→        } else {
   127→          setP2FalseStart(true);
   128→          setP2Pressed(true);
   129→        }
   130→
   131→        if (
   132→          (player === 1 && p2Pressed && p2FalseStart) ||
   133→          (player === 2 && p1Pressed && p1FalseStart)
   134→        ) {
   135→          gameActiveRef.current = false;
   136→          clearTimers();
   137→          setRoundWinner(0);
   138→          autoAdvanceRef.current = setTimeout(() => {
   139→            setGameState("result");
   140→          }, 1500);
   141→        } else if (
   142→          (player === 1 && p2Pressed && !p2FalseStart) ||
   143→          (player === 2 && p1Pressed && !p1FalseStart)
   144→        ) {
   145→          const nonFalsePlayer = player === 1 ? 2 : 1;
   146→          gameActiveRef.current = false;
   147→          clearTimers();
   148→          setRoundWinner(nonFalsePlayer as 0 | 1 | 2);
   149→          autoAdvanceRef.current = setTimeout(() => {
   150→            setGameState("result");
   151→          }, 1500);
   152→        } else {
   153→          const otherFalseStart = player === 1 ? p2FalseStart : p1FalseStart;
   154→          const otherPressed = player === 1 ? p2Pressed : p1Pressed;
   155→          if (!otherPressed && !otherFalseStart) {
   156→            autoAdvanceRef.current = setTimeout(() => {
   157→              gameActiveRef.current = false;
   158→              clearTimers();
   159→              const winner = player === 1 ? 2 : 1;
   160→              setRoundWinner(winner as 0 | 1 | 2);
   161→              setGameState("result");
   162→            }, 2000);
   163→          }
   164→        }
   165→        return;
   166→      }
   167→
   168→      if (gameState === "ready") {
   169→        const reactionTime = Math.round(performance.now() - goTimestamp);
   170→        if (player === 1) {
   171→          setP1Time(reactionTime);
   172→          setP1Pressed(true);
   173→        } else {
   174→          setP2Time(reactionTime);
   175→          setP2Pressed(true);
   176→        }
   177→
   178→        const otherPressed = player === 1 ? p2Pressed : p1Pressed;
   179→        const otherFalseStart = player === 1 ? p2FalseStart : p1FalseStart;
   180→
   181→        if (otherFalseStart) {
   182→          gameActiveRef.current = false;
   183→          clearTimers();
   184→          setRoundWinner(player);
   185→          autoAdvanceRef.current = setTimeout(() => {
   186→            setGameState("result");
   187→          }, 1500);
   188→          return;
   189→        }
   190→
   191→        if (otherPressed) {
   192→          const otherTime = player === 1 ? p2Time : p1Time;
   193→          if (otherTime !== null) {
   194→            gameActiveRef.current = false;
   195→            clearTimers();
   196→            if (reactionTime < otherTime) {
   197→              setRoundWinner(player);
   198→            } else if (otherTime < reactionTime) {
   199→              setRoundWinner(player === 1 ? 2 : 1);
   200→            } else {
   201→              setRoundWinner(0);
   202→            }
   203→            autoAdvanceRef.current = setTimeout(() => {
   204→              setGameState("result");
   205→            }, 1500);
   206→          }
   207→        } else {
   208→          autoAdvanceRef.current = setTimeout(() => {
   209→            gameActiveRef.current = false;
   210→            clearTimers();
   211→            setRoundWinner(player);
   212→            setGameState("result");
   213→          }, 5000);
   214→        }
   215→      }
   216→    },
   217→    [gameState, p1Pressed, p2Pressed, p1FalseStart, p2FalseStart, p1Time, p2Time, goTimestamp, clearTimers]
   218→  );
   219→
   220→  const finishRound = useCallback(() => {
   221→    const result: RoundResult = {
   222→      p1Time,
   223→      p2Time,
   224→      p1FalseStart,
   225→      p2FalseStart,
   226→      winner: roundWinner,
   227→    };
   228→    const newResults = [...roundResults, result];
   229→    setRoundResults(newResults);
   230→
   231→    if (currentRound >= totalRounds) {
   232→      setGameState("gameOver");
   233→
   234→      const p1Wins = newResults.filter((r) => r.winner === 1).length;
   235→      const p2Wins = newResults.filter((r) => r.winner === 2).length;
   236→
   237→      const p1ValidTimes = newResults
   238→        .filter((r) => !r.p1FalseStart && r.p1Time !== null)
   239→        .map((r) => r.p1Time!);
   240→      const p2ValidTimes = newResults
   241→        .filter((r) => !r.p2FalseStart && r.p2Time !== null)
   242→        .map((r) => r.p2Time!);
   243→
   244→      const p1Avg = p1ValidTimes.length > 0 ? p1ValidTimes.reduce((a, b) => a + b, 0) / p1ValidTimes.length : 9999;
   245→      const p2Avg = p2ValidTimes.length > 0 ? p2ValidTimes.reduce((a, b) => a + b, 0) / p2ValidTimes.length : 9999;
   246→
   247→      const p1Score = Math.max(0, Math.round(1000 - p1Avg));
   248→      const p2Score = Math.max(0, Math.round(1000 - p2Avg));
   249→
   250→      if (p1Wins > p2Wins) {
   251→        onScore?.("Jogador 1", p1Score);
   252→      } else if (p2Wins > p1Wins) {
   253→        onScore?.("Jogador 2", p2Score);
   254→      } else {
   255→        if (p1Avg < p2Avg) {
   256→          onScore?.("Jogador 1", p1Score);
   257→        } else if (p2Avg < p1Avg) {
   258→          onScore?.("Jogador 2", p2Score);
   259→        } else {
   260→          onScore?.("Jogador 1", p1Score);
   261→        }
   262→      }
   263→    } else {
   264→      setPrevRound(currentRound);
   265→      setShowRoundTransition(true);
   266→      setTimeout(() => {
   267→        setCurrentRound((prev) => prev + 1);
   268→        setGameState("waiting");
   269→        setP1Time(null);
   270→        setP2Time(null);
   271→        setP1FalseStart(false);
   272→        setP2FalseStart(false);
   273→        setRoundWinner(0);
   274→        setP1Pressed(false);
   275→        setP2Pressed(false);
   276→        setGoTimestamp(0);
   277→        setElapsedMs(0);
   278→        setShowRoundTransition(false);
   279→      }, 900);
   280→    }
   281→  }, [p1Time, p2Time, p1FalseStart, p2FalseStart, roundWinner, roundResults, currentRound, totalRounds, onScore]);
   282→
   283→  useEffect(() => {
   284→    const handleKeyDown = (e: KeyboardEvent) => {
   285→      if (e.repeat) return;
   286→      if (e.code === "Space") {
   287→        e.preventDefault();
   288→        if (gameState === "waiting") {
   289→          startRound();
   290→        } else {
   291→          handleReaction(1);
   292→        }
   293→      }
   294→      if (e.code === "Enter") {
   295→        e.preventDefault();
   296→        if (gameState === "waiting") {
   297→          startRound();
   298→        } else {
   299→          handleReaction(2);
   300→        }
   301→      }
   302→    };
   303→
   304→    window.addEventListener("keydown", handleKeyDown);
   305→    return () => window.removeEventListener("keydown", handleKeyDown);
   306→  }, [gameState, handleReaction, startRound]);
   307→
   308→  useEffect(() => {
   309→    return () => {
   310→      clearTimers();
   311→      if (elapsedRafRef.current) cancelAnimationFrame(elapsedRafRef.current);
   312→    };
   313→  }, [clearTimers]);
   314→
   315→  // Elapsed time tracker for ready state
   316→  useEffect(() => {
   317→    if (gameState === "ready" && goTimestamp > 0) {
   318→      const tick = () => {
   319→        setElapsedMs(Math.round(performance.now() - goTimestamp));
   320→        elapsedRafRef.current = requestAnimationFrame(tick);
   321→      };
   322→      elapsedRafRef.current = requestAnimationFrame(tick);
   323→      return () => {
   324→        if (elapsedRafRef.current) cancelAnimationFrame(elapsedRafRef.current);
   325→      };
   326→    } else {
   327→      setElapsedMs(0);
   328→    }
   329→  }, [gameState, goTimestamp]);
   330→
   331→  const p1Wins = roundResults.filter((r) => r.winner === 1).length;
   332→  const p2Wins = roundResults.filter((r) => r.winner === 2).length;
   333→
   334→  const p1ValidTimes = roundResults
   335→    .filter((r) => !r.p1FalseStart && r.p1Time !== null)
   336→    .map((r) => r.p1Time!);
   337→  const p2ValidTimes = roundResults
   338→    .filter((r) => !r.p2FalseStart && r.p2Time !== null)
   339→    .map((r) => r.p2Time!);
   340→
   341→  const p1Avg =
   342→    p1ValidTimes.length > 0
   343→      ? Math.round(p1ValidTimes.reduce((a, b) => a + b, 0) / p1ValidTimes.length)
   344→      : null;
   345→  const p2Avg =
   346→    p2ValidTimes.length > 0
   347→      ? Math.round(p2ValidTimes.reduce((a, b) => a + b, 0) / p2ValidTimes.length)
   348→      : null;
   349→
   350→  const overallWinner = (() => {
   351→    if (roundResults.length === 0) return 0;
   352→    if (p1Wins > p2Wins) return 1;
   353→    if (p2Wins > p1Wins) return 2;
   354→    if (p1Avg !== null && p2Avg !== null) {
   355→      if (p1Avg < p2Avg) return 1;
   356→      if (p2Avg < p1Avg) return 2;
   357→    }
   358→    return 0;
   359→  })();
   360→
   361→  const centerBg =
   362→    gameState === "countdown"
   363→      ? "bg-red-500/10"
   364→      : gameState === "ready"
   365→      ? "bg-emerald-500/20"
   366→      : "bg-slate-800";
   367→
   368→  // Urgency color for elapsed timer
   369→  const urgencyColor =
   370→    elapsedMs < 400
   371→      ? "text-emerald-400"
   372→      : elapsedMs < 800
   373→      ? "text-yellow-400"
   374→      : "text-red-400";
   375→
   376→  const urgencyGlow =
   377→    elapsedMs < 400
   378→      ? "drop-shadow-[0_0_10px_rgba(52,211,153,0.6)]"
   379→      : elapsedMs < 800
   380→      ? "drop-shadow-[0_0_10px_rgba(250,204,21,0.6)]"
   381→      : "drop-shadow-[0_0_12px_rgba(248,113,113,0.7)]";
   382→
   383→  const roundDots = Array.from({ length: totalRounds }, (_, i) => {
   384→    const roundNum = i + 1;
   385→    const isCompleted =
   386→      roundNum < currentRound ||
   387→      (roundNum === currentRound && (gameState === "result" || gameState === "gameOver"));
   388→    const isCurrent =
   389→      roundNum === currentRound &&
   390→      gameState !== "result" &&
   391→      gameState !== "gameOver";
   392→    const result = roundResults[i];
   393→
   394→    return (
   395→      <motion.div
   396→        key={i}
   397→        initial={{ scale: 0 }}
   398→        animate={{ scale: 1 }}
   399→        transition={{ delay: i * 0.05, type: "spring", stiffness: 400, damping: 20 }}
   400→        className={cn(
   401→          "h-3 w-3 rounded-full transition-colors duration-300",
   402→          isCompleted && result
   403→            ? result.winner === 1
   404→              ? "bg-cyan-400"
   405→              : result.winner === 2
   406→              ? "bg-pink-400"
   407→              : "bg-yellow-400"
   408→            : isCurrent
   409→            ? "bg-white"
   410→            : "bg-slate-600"
   411→        )}
   412→      >
   413→        {isCurrent && (
   414→          <motion.div
   415→            className="h-3 w-3 rounded-full bg-white"
   416→            animate={{ scale: [1, 1.5, 1], opacity: [0.6, 0, 0.6] }}
   417→            transition={{ duration: 1.2, repeat: Infinity, ease: "easeInOut" }}
   418→          />
   419→        )}
   420→      </motion.div>
   421→    );
   422→  });
   423→
   424→  // Concentric ring burst for the GO! target
   425→  const rings = [
   426→    { size: 60, delay: 0, duration: 1.0, opacity: 0.5 },
   427→    { size: 90, delay: 0.08, duration: 1.1, opacity: 0.35 },
   428→    { size: 120, delay: 0.16, duration: 1.2, opacity: 0.25 },
   429→    { size: 160, delay: 0.24, duration: 1.3, opacity: 0.15 },
   430→  { size: 200, delay: 0.32, duration: 1.4, opacity: 0.08 },
   431→  ];
   432→
   433→  return (
   434→    <div className="mx-auto flex w-full max-w-2xl flex-col gap-4">
   435→      {/* Round transition overlay */}
   436→      <AnimatePresence>
   437→        {showRoundTransition && prevRound > 0 && (
   438→          <motion.div
   439→            initial={{ opacity: 0, scale: 0.8 }}
   440→            animate={{ opacity: 1, scale: 1 }}
   441→            exit={{ opacity: 0, scale: 1.2, y: -20 }}
   442→            transition={{ duration: 0.4, ease: "easeOut" }}
   443→            className="pointer-events-none absolute inset-0 z-50 flex items-center justify-center"
   444→          >
   445→            <div className="flex flex-col items-center gap-2">
   446→              <motion.div
   447→                initial={{ width: 0, opacity: 0 }}
   448→                animate={{ width: 200, opacity: 1 }}
   449→                exit={{ width: 300, opacity: 0 }}
   450→                transition={{ duration: 0.6 }}
   451→                className="h-0.5 bg-gradient-to-r from-cyan-500 via-white to-pink-500"
   452→              />
   453→              <motion.span
   454→                initial={{ y: 10, opacity: 0 }}
   455→                animate={{ y: 0, opacity: 1 }}
   456→                exit={{ y: -20, opacity: 0 }}
   457→                transition={{ delay: 0.15, duration: 0.3 }}
   458→                className="text-3xl font-black tracking-widest text-white"
   459→              >
   460→                Round {prevRound + 1}
   461→              </motion.span>
   462→              <motion.div
   463→                initial={{ width: 0, opacity: 0 }}
   464→                animate={{ width: 200, opacity: 1 }}
   465→                exit={{ width: 300, opacity: 0 }}
   466→                transition={{ duration: 0.6 }}
   467→                className="h-0.5 bg-gradient-to-r from-pink-500 via-white to-cyan-500"
   468→              />
   469→            </div>
   470→          </motion.div>
   471→        )}
   472→      </AnimatePresence>
   473→
   474→      <div className="rounded-xl bg-gradient-to-r from-cyan-900/30 to-pink-900/30 border border-cyan-500/20 p-4">
   475→        <div className="flex items-center justify-between">
   476→          <div className="flex flex-col items-start gap-1">
   477→            <span className="text-sm font-medium text-cyan-400">Jogador 1</span>
   478→            <div className="flex items-center gap-2">
   479→              <Badge className="bg-cyan-500/20 text-cyan-300 border-cyan-500/30 text-xs">
   480→                {p1Wins}V
   481→              </Badge>
   482→              {p1Avg !== null && (
   483→                <span className="text-xs text-slate-400">média: {p1Avg}ms</span>
   484→              )}
   485→            </div>
   486→          </div>
   487→
   488→          <div className="flex flex-col items-center gap-1">
   489→            <div className="flex items-center gap-1.5">
   490→              <Swords className="h-4 w-4 text-yellow-400" />
   491→              <span className="text-sm font-bold tracking-wider text-white">
   492→                CORRIDA DE REAÇÃO
   493→              </span>
   494→              <Swords className="h-4 w-4 text-yellow-400" />
   495→            </div>
   496→            <span className="text-xs text-slate-500">
   497→              Round {currentRound}/{totalRounds}
   498→            </span>
   499→          </div>
   500→
   501→          <div className="flex flex-col items-end gap-1">
   502→            <span className="text-sm font-medium text-pink-400">Jogador 2</span>
   503→            <div className="flex items-center gap-2">
   504→              {p2Avg !== null && (
   505→                <span className="text-xs text-slate-400">média: {p2Avg}ms</span>
   506→              )}
   507→              <Badge className="bg-pink-500/20 text-pink-300 border-pink-500/30 text-xs">
   508→                {p2Wins}V
   509→              </Badge>
   510→            </div>
   511→          </div>
   512→        </div>
   513→      </div>
   514→
   515→      <div className="flex items-center justify-center gap-2">{roundDots}</div>
   516→
   517→      <AnimatePresence mode="wait">
   518→        <motion.div
   519→          key={`${gameState}-${currentRound}`}
   520→          initial={{ opacity: 0, scale: 0.95 }}
   521→          animate={{ opacity: 1, scale: 1 }}
   522→          exit={{ opacity: 0, scale: 0.95 }}
   523→          transition={{ duration: 0.2 }}
   524→          className={cn(
   525→            "relative flex min-h-[300px] flex-col items-center justify-center rounded-2xl border transition-colors duration-300 overflow-hidden",
   526→            centerBg,
   527→            gameState === "countdown"
   528→              ? "border-red-500/30"
   529→              : gameState === "ready"
   530→              ? "border-emerald-500/30"
   531→              : "border-slate-700"
   532→          )}
   533→        >
   534→          {/* Waiting state */}
   535→          {gameState === "waiting" && (
   536→            <motion.div
   537→              initial={{ opacity: 0, y: 20 }}
   538→              animate={{ opacity: 1, y: 0 }}
   539→              className="flex flex-col items-center gap-4 px-6 text-center"
   540→            >
   541→              <motion.div
   542→                animate={{ rotate: [0, 10, -10, 0] }}
   543→                transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   544→              >
   545→                <Target className="h-16 w-16 text-slate-500" />
   546→              </motion.div>
   547→              <div className="space-y-2">
   548→                <p className="text-2xl font-bold text-white">Pronto?</p>
   549→                <p className="text-sm text-slate-400">
   550→                  Pressione{" "}
   551→                  <kbd className="rounded bg-slate-700 px-2 py-0.5 text-xs font-mono text-cyan-300">
   552→                    Space
   553→                  </kbd>{" "}
