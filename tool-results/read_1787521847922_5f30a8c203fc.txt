    70→function pickPattern(total: number, count: number): Set<number> {
    71→  const set = new Set<number>();
    72→  while (set.size < count) set.add(Math.floor(Math.random() * total));
    73→  return set;
    74→}

    76→function calcRoundScore(taps: Set<number>, pattern: Set<number>): number {
    77→  let correct = 0;
    78→  let wrong = 0;
    79→  taps.forEach((c) => (pattern.has(c) ? correct++ : wrong++));
    80→  return Math.max(0, correct * 10 - wrong * 5);
    81→}

   277→function Cell({
   278→  index,
   279→  inPattern,
   280→  patternStaggerIndex,
   281→  isTapped,
   282→  isWrongTap,
   283→  isCorrectTap,
   284→  isMissed,
   285→  showPattern,
   286→  interactive,
   287→  tapColor,
   288→  phase,
   289→  onTap,
   290→}: {
   291→  index: number;
   292→  inPattern: boolean;
   293→  patternStaggerIndex: number;
   294→  isTapped: boolean;
   295→  isWrongTap: boolean;
   296→  isCorrectTap: boolean;
   297→  isMissed: boolean;
   298→  showPattern: boolean;
   299→  interactive: boolean;
   300→  tapColor: "cyan" | "pink" | null;
   301→  phase: Phase;
   302→  onTap: (i: number) => void;
   303→}) {
   304→  /* ---- visual state ---- */
   305→  let bg = "bg-zinc-800/70 border-zinc-700/40";
   306→  let glow = "";
   307→  let showFlash = false;
   308→  let flashColor = "bg-white";
   309→
   310→  /* ---- per-phase animation config ---- */
   311→  const anim: Record<string, unknown> = {};
   312→  const trans: Record<string, unknown> = {};
   313→
   314→  if (phase === "memorize") {
   315→    if (inPattern) {
   316→      bg = "bg-violet-500 border-violet-400/60";
   317→      const d = patternStaggerIndex * 0.07;
   318→      anim.scale = [0.5, 1.06, 0.97, 1.04, 1];
   319→      anim.boxShadow = [
   320→        "0 0 6px 1px rgba(139,92,246,0.3)",
   321→        "0 0 30px 8px rgba(139,92,246,0.6)",
   322→        "0 0 18px 4px rgba(139,92,246,0.4)",
   323→        "0 0 36px 10px rgba(139,92,246,0.65)",
   324→        "0 0 22px 6px rgba(139,92,246,0.5)",
   325→      ];
   326→      trans.duration = 2.5;
   327→      trans.repeat = Infinity;
   328→      trans.ease = "easeInOut";
   329→      trans.delay = d;
   330→    } else {
   331→      bg = "bg-zinc-800/50 border-zinc-700/30";
   332→    }
   333→  } else if (phase === "recall_p1" || phase === "recall_p2") {
   334→    if (isTapped && tapColor === "cyan") {
   335→      bg = "bg-cyan-500 border-cyan-400/60";
   336→      glow = "shadow-[0_0_16px_3px_rgba(6,182,212,0.45)]";
   337→      anim.scale = [0.85, 1.06, 1];
   338→      trans.duration = 0.2;
   339→      trans.ease = "easeOut";
   340→    } else if (isTapped && tapColor === "pink") {
   341→      bg = "bg-pink-500 border-pink-400/60";
   342→      glow = "shadow-[0_0_16px_3px_rgba(236,72,153,0.45)]";
   343→      anim.scale = [0.85, 1.06, 1];
   344→      trans.duration = 0.2;
   345→      trans.ease = "easeOut";
   346→    } else {
   347→      bg = "bg-zinc-800/70 border-zinc-700/40";
   348→    }
   349→  } else if (phase === "feedback_p1" || phase === "feedback_p2") {
   350→    if (isCorrectTap) {
   351→      bg = "bg-emerald-500 border-emerald-400/60";
   352→      showFlash = true;
   353→      flashColor = "bg-emerald-200";
   354→      anim.scale = [1, 1.22, 1];
   355→      anim.boxShadow = [
   356→        "0 0 0 0 rgba(16,185,129,0)",
   357→        "0 0 28px 10px rgba(16,185,129,0.65)",
   358→        "0 0 16px 4px rgba(16,185,129,0.4)",
   359→      ];
   360→      trans.scale = { duration: 0.6, ease: "easeOut" };
   361→      trans.boxShadow = { duration: 0.6, ease: "easeOut" };
   362→    } else if (isWrongTap) {
   363→      bg = "bg-red-500 border-red-400/60";
   364→      showFlash = true;
   365→      flashColor = "bg-red-300";
   366→      anim.x = [0, -8, 8, -6, 6, -3, 3, 0];
   367→      anim.scale = [1, 0.88, 1.06, 0.94, 1];
   368→      anim.boxShadow = [
   369→        "0 0 0 0 rgba(239,68,68,0)",
   370→        "0 0 24px 8px rgba(239,68,68,0.65)",
   371→        "0 0 12px 3px rgba(239,68,68,0.35)",
   372→      ];
   373→      trans.x = { duration: 0.5, ease: "easeOut" };
   374→      trans.scale = { duration: 0.5, ease: "easeOut" };
   375→      trans.boxShadow = { duration: 0.5, ease: "easeOut" };
   376→    } else if (isMissed) {
   377→      bg = "bg-violet-700/40 border-violet-500/40";
   378→      anim.scale = [1, 1.1, 0.96, 1.04, 1];
   379→      anim.boxShadow = [
   380→        "0 0 0 0 rgba(139,92,246,0)",
   381→        "0 0 18px 5px rgba(139,92,246,0.45)",
   382→        "0 0 8px 2px rgba(139,92,246,0.25)",
   383→      ];
   384→      trans.duration = 0.9;
   385→      trans.ease = "easeOut";
   386→    } else {
   387→      bg = "bg-zinc-800/40 border-zinc-700/20";
   388→    }
   389→  } else if (phase === "round_result" || phase === "game_over") {
   390→    if (showPattern && inPattern) {
   391→      bg = "bg-violet-600/40 border-violet-500/30";
   392→    } else {
   393→      bg = "bg-zinc-800/30 border-zinc-700/20";
   394→    }
   395→  } else {
   396→    bg = "bg-zinc-800/70 border-zinc-700/40";
   397→  }
   398→
   399→  return (
   400→    <motion.button
   401→      type="button"
   402→      onClick={() => interactive && onTap(index)}
   403→      className={cn(
   404→        "w-14 h-14 sm:w-18 sm:h-18 rounded-xl border-2 relative overflow-hidden transition-[background-color,border-color] duration-150",
   405→        bg,
   406→        glow,
   407→        interactive && "cursor-pointer hover:brightness-125",
   408→        !interactive && "cursor-default",
   409→      )}
   410→      animate={anim}
   411→      transition={trans}
   412→      whileTap={interactive ? { scale: 0.82 } : undefined}
   413→      whileHover={interactive ? { scale: 1.06 } : undefined}
   414→    >
   415→      {/* Flash overlay for correct / wrong feedback */}
   416→      <AnimatePresence>
   417→        {showFlash && (
   418→          <motion.div
   419→            key="flash"
   420→            className={cn("absolute inset-0 rounded-[10px]", flashColor)}
   421→            initial={{ opacity: 0.9 }}
   422→            animate={{ opacity: 0 }}
   423→            exit={{ opacity: 0 }}
   424→            transition={{ duration: 0.45, ease: "easeOut" }}
   425→          />
   426→        )}
   427→      </AnimatePresence>
   428→    </motion.button>
   429→  );
   430→}

   435→
   436→export default function PatternMemory({ onScore, liveCode }: Props) {
   437→  /* ---- state ---- */
   438→  const [phase, setPhase] = useState<Phase>("idle");
   439→  const [round, setRound] = useState(0);
   440→  const [pattern, setPattern] = useState<Set<number>>(new Set());
   441→  const [p1Taps, setP1Taps] = useState<Set<number>>(new Set());
   442→  const [p2Taps, setP2Taps] = useState<Set<number>>(new Set());
   443→  const [p1Total, setP1Total] = useState(0);
   444→  const [p2Total, setP2Total] = useState(0);
   445→  const [p1RoundPts, setP1RoundPts] = useState(0);
   446→  const [p2RoundPts, setP2RoundPts] = useState(0);
   447→  const [memProgress, setMemProgress] = useState(100);
   448→  const [showLevelUp, setShowLevelUp] = useState(false);
   449→  const [levelUpText, setLevelUpText] = useState("");
   450→
   451→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   452→  const onScoreFired = useRef(false);
   453→  const scoresRef = useRef({ p1: 0, p2: 0 });
   454→
   455→  const cfg = ROUNDS[round] ?? ROUNDS[0];
   456→  const totalCells = cfg.gridSize * cfg.gridSize;
   457→  const patternArr = Array.from(pattern).sort((a, b) => a - b);
   458→
   459→  const recalling = phase === "recall_p1" || phase === "recall_p2";
   460→  const isP1 = phase === "recall_p1";
   461→
   462→  /* ---- timer cleanup ---- */
   463→  const clearTimer = useCallback(() => {
   464→    if (timerRef.current) {
   465→      clearInterval(timerRef.current);
   466→      timerRef.current = null;
   467→    }
   468→  }, []);
   469→
   470→  useEffect(() => clearTimer, [clearTimer]);
   471→
   472→  /* ---- memorize timer ---- */
   473→  useEffect(() => {
   474→    if (phase !== "memorize") return;
   475→    clearTimer();
   476→    const start = Date.now();
   477→    const durationMs = cfg.memorizeTime * 1000;
   478→    timerRef.current = setInterval(() => {
   479→      const elapsed = Date.now() - start;
   480→      const pct = Math.max(0, 100 - (elapsed / durationMs) * 100);
   481→      setMemProgress(pct);
   482→      if (elapsed >= durationMs) {
   483→        clearTimer();
   484→        setPhase("recall_p1");
   485→      }
   486→    }, 40);
   487→    return clearTimer;
   488→  }, [phase, cfg.memorizeTime, clearTimer]);
   489→
   490→  /* ---- feedback auto-advance timers ---- */
   491→  useEffect(() => {
   492→    if (phase === "feedback_p1") {
   493→      const id = setTimeout(() => setPhase("recall_p2"), 1800);
   494→      return () => clearTimeout(id);
   495→    }
   496→    if (phase === "feedback_p2") {
   497→      const id = setTimeout(() => setPhase("round_result"), 1800);
   498→      return () => clearTimeout(id);
   499→    }
   500→  }, [phase]);
   501→
   502→  /* ---- level-up detection ---- */
   503→  useEffect(() => {
   504→    if (round > 0 && phase === "memorize") {
   505→      const prevGrid = ROUNDS[round - 1]?.gridSize;
   506→      const currGrid = ROUNDS[round].gridSize;
   507→      if (prevGrid !== undefined && prevGrid !== currGrid) {
   508→        const txt = `${currGrid}\u00D7${currGrid}`;
   509→        setLevelUpText(txt);
   510→        setShowLevelUp(true);
   511→        const t = setTimeout(() => setShowLevelUp(false), 1300);
   512→        return () => clearTimeout(t);
   513→      }
   514→    }
   515→  }, [round, phase]);
   516→
   517→  /* ---- start round ---- */
   518→  const startRound = useCallback(() => {
   519→    const pat = pickPattern(totalCells, cfg.patternSize);
   520→    setPattern(pat);
   521→    setP1Taps(new Set());
   522→    setP2Taps(new Set());
   523→    setP1RoundPts(0);
   524→    setP2RoundPts(0);
   525→    setMemProgress(100);
   526→    setPhase("memorize");
   527→  }, [totalCells, cfg.patternSize]);
   528→
   529→  /* ---- begin full game ---- */
   530→  const handleStart = useCallback(() => {
   531→    scoresRef.current = { p1: 0, p2: 0 };
   532→    onScoreFired.current = false;
   533→    setRound(0);
   534→    setP1Total(0);
   535→    setP2Total(0);
   536→    setP1Taps(new Set());
   537→    setP2Taps(new Set());
   538→    setP1RoundPts(0);
   539→    setP2RoundPts(0);
   540→    setPattern(new Set());
   541→    setPhase("idle");
   542→    setShowLevelUp(false);
   543→    // Use round 0 config values directly instead of stale closure
   544→    const r0Cells = 3;
   545→    const r0Size = 2;
   546→    const pat = pickPattern(r0Cells * r0Cells, r0Size);
   547→    setPattern(pat);
   548→    setP1Taps(new Set());
   549→    setP2Taps(new Set());
   550→    setP1RoundPts(0);
   551→    setP2RoundPts(0);
   552→    setMemProgress(100);
   553→    setTimeout(() => setPhase("memorize"), 50);
   554→  }, []);
   555→
   556→  /* ---- tap cell ---- */
   557→  const handleTap = useCallback(
   558→    (idx: number) => {
   559→      if (phase === "recall_p1") {
   560→        setP1Taps((prev) => {
   561→          const next = new Set(prev);
   562→          next.has(idx) ? next.delete(idx) : next.add(idx);
