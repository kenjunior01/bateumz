    83→function shuffle<T>(arr: T[]): T[] {
    84→  const a = [...arr];
    85→  for (let i = a.length - 1; i > 0; i--) {
    86→    const j = Math.floor(Math.random() * (i + 1));
    87→    [a[i], a[j]] = [a[j], a[i]];
    88→  }
    89→  return a;
    90→}
    91→
    92→function makeRound(): RoundData {
    93→  const word = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    94→  let ink: ColorDef;
    95→  do {
    96→    ink = ALL_COLORS[Math.floor(Math.random() * ALL_COLORS.length)];
    97→  } while (ink.id === word.id);
    98→  const others = shuffle(ALL_COLORS.filter((c) => c.id !== ink.id)).slice(0, 3);
    99→  return { word, inkColor: ink, choices: shuffle([ink, ...others]) };
   100→}

   103→  const t = (roundNum - 1) / (TOTAL_ROUNDS - 1);
   104→  return Math.round(INITIAL_TIME_MS - t * (INITIAL_TIME_MS - MIN_TIME_MS));
   105→}

   307→          setP2Streak(newStreak);
   308→          setP2MaxStreak((m) => Math.max(m, newStreak));

   310→          setP2Bounce((b) => b + 1);
   311→          setP2Feedback({ colorId, type: "correct" });
   312→        }

   318→          setScorePopups((prev) => prev.filter((p) => p.id !== pid));

   320→
   321→        setFeedbackTick((t) => t + 1);
   322→        setLastWinner(player);
   323→        setLastCorrect(roundData.inkColor.name);
   324→        setPhase("roundResult");

   326→        const isP1 = player === 1;
   327→        if (isP1) {
   328→          setP1Score((s) => Math.max(0, s - 1));
   329→          setP1Streak(0);
   330→          setP1Bounce((b) => b + 1);
   331→          setP1Feedback({ colorId, type: "wrong" });
   332→        } else {
   333→          setP2Score((s) => Math.max(0, s - 1));
   334→          setP2Streak(0);
   335→          setP2Bounce((b) => b + 1);
   336→          setP2Feedback({ colorId, type: "wrong" });
   337→        }
   338→        setScreenShake((s) => s + 1);
   339→        setWrongFlash((f) => f + 1);
   340→        setFeedbackTick((t) => t + 1);
   341→      }
   342→    },
   343→    [phase, roundData, p1Streak, p2Streak, clearTimers],
   344→  );

   351→        const p2Name = botMode ? "Computador" : "Jogador 2";

   353→        onScore?.(winner, Math.max(p1Score, p2Score));
   354→      }
   355→    } else {
   356→      setRound((r) => r + 1);
   357→      setCountdownVal(3);
   358→      setP1Feedback(null); setP2Feedback(null);
   359→      setFeedbackTick((t) => t + 1);
   360→      setPhase("countdown");
   361→    }
   362→  }, [round, p1Score, p2Score, onScore]);

   367→    setPhase("idle");
