     1→import { useState, useEffect, useCallback, useRef } from "react";
     2→import { motion, AnimatePresence } from "framer-motion";
     3→
     4→interface UriProps {
     5→  onScore?: (name: string, score: number) => void;
     6→  liveCode?: string;
     7→}
     8→
     9→type Phase = "menu" | "countdown" | "playing" | "result";
    10→type Mode = "bot" | "pvp";
    11→type Difficulty = "Facil" | "Medio" | "Dificil";
    12→type Temperature = "Quente" | "Morno" | "Frio" | "Congelado" | "Acertou";
    13→
    14→interface GuessEntry {
    15→  value: number;
    16→  temperature: Temperature;
    17→  direction: "maior" | "menor" | "";
    18→}
    19→
    20→const GAME_DURATION = 60;
    21→
    22→const TEMP_COLORS: Record<Temperature, string> = {
    23→  Acertou: "#009140",
    24→  Quente: "#FF0000",
    25→  Morno: "#FFD700",
    26→  Frio: "#6B7280",
    27→  Congelado: "#9CA3AF",
    28→};
    29→
    30→const TEMP_BG: Record<Temperature, string> = {
    31→  Acertou: "rgba(0,145,64,0.15)",
    32→  Quente: "rgba(255,0,0,0.12)",
    33→  Morno: "rgba(255,215,0,0.12)",
    34→  Frio: "rgba(107,114,128,0.10)",
    35→  Congelado: "rgba(156,163,175,0.08)",
    36→};
    37→
    38→const BOT_NOISE: Record<Difficulty, number> = { Facil: 20, Medio: 8, Dificil: 2 };
    39→const BOT_SPEED: Record<Difficulty, [number, number]> = {
    40→  Facil: [2500, 4500],
    41→  Medio: [1500, 3000],
    42→  Dificil: [1000, 2000],
    43→};
    44→
    45→function getTemperature(guess: number, secret: number): Temperature {
    46→  const diff = Math.abs(guess - secret);
    47→  if (diff === 0) return "Acertou";
    48→  if (diff <= 10) return "Quente";
    49→  if (diff <= 25) return "Morno";
    50→  if (diff <= 50) return "Frio";
    51→  return "Congelado";
    52→}
    53→
    54→function getDirection(guess: number, secret: number): "maior" | "menor" | "" {
    55→  if (guess === secret) return "";
    56→  return guess < secret ? "maior" : "menor";
    57→}
    58→
    59→function generateSecret(): number {
    60→  return Math.floor(Math.random() * 100) + 1;
    61→}
    62→
    63→const CapulanaPattern = () => (
    64→  <svg
    65→    className="absolute inset-0 w-full h-full opacity-[0.04] pointer-events-none"
    66→    xmlns="http://www.w3.org/2000/svg"
    67→  >
    68→    <defs>
    69→      <pattern
    70→        id="uri-capulana-pattern"
    71→        x="0"
    72→        y="0"
    73→        width="28"
    74→        height="28"
    75→        patternUnits="userSpaceOnUse"
    76→      >
    77→        <path d="M14 0L28 14L14 28L0 14Z" fill="none" stroke="#FFD700" strokeWidth="0.6" />
    78→        <circle cx="14" cy="14" r="4" fill="none" stroke="#009140" strokeWidth="0.4" />
    79→        <path d="M14 2L26 14L14 26L2 14Z" fill="none" stroke="#FF6B35" strokeWidth="0.3" />
    80→      </pattern>
    81→    </defs>
    82→    <rect width="100%" height="100%" fill="url(#uri-capulana-pattern)" />
    83→  </svg>
    84→);
    85→
    86→export default function UriGame({ onScore, liveCode }: UriProps) {
    87→  const [mode, setMode] = useState<Mode>("bot");
    88→  const [difficulty, setDifficulty] = useState<Difficulty>("Medio");
    89→  const [phase, setPhase] = useState<Phase>("menu");
    90→  const [countdownNum, setCountdownNum] = useState(3);
    91→  const [timeLeft, setTimeLeft] = useState(GAME_DURATION);
    92→  const [secret, setSecret] = useState(0);
    93→  const [round, setRound] = useState(1);
    94→  const [score, setScore] = useState({ p1: 0, p2: 0 });
    95→  const [p1History, setP1History] = useState<GuessEntry[]>([]);
    96→  const [p2History, setP2History] = useState<GuessEntry[]>([]);
    97→  const [p1Tries, setP1Tries] = useState(0);
    98→  const [p2Tries, setP2Tries] = useState(0);
    99→  const [p1Input, setP1Input] = useState("");
   100→  const [p2Input, setP2Input] = useState("");
   101→  const [roundWinner, setRoundWinner] = useState<"p1" | "p2" | null>(null);
   102→  const [roundMsg, setRoundMsg] = useState("");
   103→  const [roundsWon, setRoundsWon] = useState({ p1: 0, p2: 0 });
   104→
   105→  const roundEndedRef = useRef(false);
   106→  const botStateRef = useRef({ low: 1, high: 100 });
   107→  const botTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   108→  const roundTransRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   109→  const gameTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   110→  const secretRef = useRef(0);
   111→  const p1InputRef = useRef<HTMLInputElement>(null);
   112→  const p2InputRef = useRef<HTMLInputElement>(null);
   113→
   114→  const secretSnapshot = useRef(0);
   115→  const p1TriesRef = useRef(0);
   116→  const p2TriesRef = useRef(0);
   117→
   118→  // Keep refs in sync
   119→  useEffect(() => {
   120→    secretSnapshot.current = secret;
   121→  }, [secret]);
   122→  useEffect(() => {
   123→    p1TriesRef.current = p1Tries;
   124→  }, [p1Tries]);
   125→  useEffect(() => {
   126→    p2TriesRef.current = p2Tries;
   127→  }, [p2Tries]);
   128→
   129→  const clearAllTimers = useCallback(() => {
   130→    if (botTimerRef.current) { clearTimeout(botTimerRef.current); botTimerRef.current = null; }
   131→    if (roundTransRef.current) { clearTimeout(roundTransRef.current); roundTransRef.current = null; }
   132→    if (gameTimerRef.current) { clearInterval(gameTimerRef.current); gameTimerRef.current = null; }
   133→  }, []);
   134→
   135→  const startNewRound = useCallback(() => {
   136→    const newSecret = generateSecret();
   137→    setSecret(newSecret);
   138→    secretRef.current = newSecret;
   139→    setP1History([]);
   140→    setP2History([]);
   141→    setP1Tries(0);
   142→    setP2Tries(0);
   143→    p1TriesRef.current = 0;
   144→    p2TriesRef.current = 0;
   145→    setRoundWinner(null);
   146→    setRoundMsg("");
   147→    roundEndedRef.current = false;
   148→    botStateRef.current = { low: 1, high: 100 };
   149→    setRound((r) => r + 1);
   150→  }, []);
   151→
   152→  const endGame = useCallback(() => {
   153→    clearAllTimers();
   154→    setPhase("result");
   155→    onScore?.("Uri", score.p1);
   156→  }, [clearAllTimers, onScore, score.p1]);
   157→
   158→  // Handle a guess from a player
   159→  const processGuess = useCallback(
   160→    (player: "p1" | "p2", value: number) => {
   161→      if (roundEndedRef.current) return;
   162→      const s = secretSnapshot.current;
   163→      const temp = getTemperature(value, s);
   164→      const dir = getDirection(value, s);
   165→      const entry: GuessEntry = { value, temperature: temp, direction: dir };
   166→
   167→      if (player === "p1") {
   168→        setP1History((h) => [...h, entry]);
   169→        setP1Tries((t) => t + 1);
   170→        p1TriesRef.current += 1;
   171→      } else {
   172→        setP2History((h) => [...h, entry]);
   173→        setP2Tries((t) => t + 1);
   174→        p2TriesRef.current += 1;
   175→      }
   176→
   177→      if (temp === "Acertou") {
   178→        roundEndedRef.current = true;
   179→        const tries = player === "p1" ? p1TriesRef.current : p2TriesRef.current;
   180→        const pts = Math.max(0, 100 - tries * 5);
   181→        setRoundWinner(player);
   182→        const label = player === "p1" ? "Jogador 1" : mode === "bot" ? "Bot" : "Jogador 2";
   183→        setRoundMsg(`${label} acertou! O numero era ${s}. +${pts} pontos!`);
   184→        setScore((prev) => ({ ...prev, [player]: prev[player as "p1" | "p2"] + pts }));
   185→        setRoundsWon((prev) => ({ ...prev, [player]: prev[player as "p1" | "p2"] + 1 }));
   186→
   187→        // Start new round after delay
   188→        roundTransRef.current = setTimeout(() => {
   189→          startNewRound();
   190→        }, 2000);
   191→      }
   192→    },
   193→    [mode, startNewRound]
   194→  );
   195→
   196→  // Bot AI
   197→  useEffect(() => {
   198→    if (phase !== "playing" || mode !== "bot" || roundEndedRef.current) return;
   199→
   200→    const [minMs, maxMs] = BOT_SPEED[difficulty];
   201→    const delay = minMs + Math.random() * (maxMs - minMs);
   202→
   203→    botTimerRef.current = setTimeout(() => {
   204→      if (roundEndedRef.current || phase !== "playing") return;
   205→
   206→      const bs = botStateRef.current;
   207→      const mid = Math.floor((bs.low + bs.high) / 2);
   208→      const noise = BOT_NOISE[difficulty];
   209→      const noisyGuess = mid + Math.floor(Math.random() * (noise * 2 + 1)) - noise;
   210→      const clamped = Math.max(1, Math.min(100, Math.max(bs.low, Math.min(bs.high, noisyGuess))));
   211→
   212→      const temp = getTemperature(clamped, secretSnapshot.current);
   213→      const dir = getDirection(clamped, secretSnapshot.current);
   214→
   215→      // Update bot's binary search bounds
   216→      if (dir === "maior") {
   217→        botStateRef.current.low = Math.min(100, clamped + 1);
   218→      } else if (dir === "menor") {
   219→        botStateRef.current.high = Math.max(1, clamped - 1);
   220→      }
   221→
   222→      processGuess("p2", clamped);
   223→    }, delay);
   224→
   225→    return () => {
   226→      if (botTimerRef.current) {
   227→        clearTimeout(botTimerRef.current);
   228→        botTimerRef.current = null;
   229→      }
   230→    };
   231→  }, [phase, mode, difficulty, round, roundWinner, processGuess]);
   232→
   233→  // Game timer
   234→  useEffect(() => {
   235→    if (phase !== "playing") return;
   236→    gameTimerRef.current = setInterval(() => {
   237→      setTimeLeft((prev) => {
   238→        if (prev <= 1) {
   239→          clearAllTimers();
   240→          setPhase("result");
   241→          onScore?.("Uri", score.p1);
   242→          return 0;
   243→        }
   244→        return prev - 1;
   245→      });
   246→    }, 1000);
   247→    return () => {
   248→      if (gameTimerRef.current) {
   249→        clearInterval(gameTimerRef.current);
   250→        gameTimerRef.current = null;
   251→      }
   252→    };
   253→  }, [phase, clearAllTimers, onScore, score.p1]);
   254→
   255→  // Countdown effect
   256→  useEffect(() => {
   257→    if (phase !== "countdown") return;
   258→    if (countdownNum <= 0) {
   259→      const s = generateSecret();
   260→      setSecret(s);
   261→      secretSnapshot.current = s;
   262→      setPhase("playing");
   263→      return;
   264→    }
   265→    const t = setTimeout(() => setCountdownNum((n) => n - 1), 800);
   266→    return () => clearTimeout(t);
   267→  }, [phase, countdownNum]);
   268→
   269→  const startGame = useCallback(() => {
   270→    setScore({ p1: 0, p2: 0 });
   271→    setRoundsWon({ p1: 0, p2: 0 });
   272→    setP1History([]);
   273→    setP2History([]);
   274→    setP1Tries(0);
   275→    setP2Tries(0);
   276→    p1TriesRef.current = 0;
   277→    p2TriesRef.current = 0;
   278→    setRound(1);
   279→    setTimeLeft(GAME_DURATION);
   280→    setRoundWinner(null);
   281→    setRoundMsg("");
   282→    setP1Input("");
   283→    setP2Input("");
   284→    roundEndedRef.current = false;
   285→    botStateRef.current = { low: 1, high: 100 };
   286→    setCountdownNum(3);
   287→    setPhase("countdown");
   288→  }, []);
   289→
   290→  const handleP1Submit = useCallback(() => {
   291→    const num = parseInt(p1Input);
   292→    if (isNaN(num) || num < 1 || num > 100) return;
   293→    processGuess("p1", num);
   294→    setP1Input("");
   295→    setTimeout(() => p1InputRef.current?.focus(), 50);
   296→  }, [p1Input, processGuess]);
   297→
   298→  const handleP2Submit = useCallback(() => {
   299→    const num = parseInt(p2Input);
   300→    if (isNaN(num) || num < 1 || num > 100) return;
   301→    processGuess("p2", num);
   302→    setP2Input("");
   303→    setTimeout(() => p2InputRef.current?.focus(), 50);
   304→  }, [p2Input, processGuess]);
   305→
   306→  const handleP1KeyDown = useCallback(
   307→    (e: React.KeyboardEvent) => {
   308→      if (e.key === "Enter") handleP1Submit();
   309→    },
   310→    [handleP1Submit]
   311→  );
   312→
   313→  const handleP2KeyDown = useCallback(
   314→    (e: React.KeyboardEvent) => {
   315→      if (e.key === "Enter") handleP2Submit();
   316→    },
   317→    [handleP2Submit]
   318→  );
   319→
   320→  const resetGame = useCallback(() => {
   321→    clearAllTimers();
   322→    setPhase("menu");
   323→  }, [clearAllTimers]);
   324→
   325→  // Cleanup on unmount
   326→  useEffect(() => {
   327→    return () => clearAllTimers();
   328→  }, [clearAllTimers]);
   329→
   330→  const p2Label = mode === "bot" ? "Bot" : "Jogador 2";
   331→  const timerPct = (timeLeft / GAME_DURATION) * 100;
   332→  const timerColor =
   333→    timeLeft > 30 ? "#009140" : timeLeft > 15 ? "#FFD700" : "#FF0000";
   334→
   335→  return (
   336→    <div className="relative w-full max-w-2xl mx-auto rounded-2xl border-2 border-amber-700/40 overflow-hidden bg-gradient-to-br from-stone-900 via-stone-800 to-stone-900 text-white">
   337→      <CapulanaPattern />
   338→
   339→      <div className="relative z-10">
   340→        <AnimatePresence mode="wait">
   341→
   342→          {phase === "menu" && (
   343→            <motion.div
   344→              key="menu"
   345→              initial={{ opacity: 0, y: 20 }}
   346→              animate={{ opacity: 1, y: 0 }}
   347→              exit={{ opacity: 0, y: -20 }}
   348→              className="p-6 space-y-6"
   349→            >
   350→
   351→              <div className="text-center space-y-2">
   352→                <motion.h2
   353→                  className="text-3xl font-black tracking-tight"
   354→                  style={{ color: "#FFD700" }}
   355→                  animate={{ scale: [1, 1.03, 1] }}
   356→                  transition={{ repeat: Infinity, duration: 2 }}
   357→                >
   358→                  Uri
   359→                </motion.h2>
   360→                <p className="text-stone-400 text-sm">
   361→                  Jogo tradicional mocambicano de adivinhacao
   362→                </p>
   363→                <div className="flex items-center justify-center gap-2 mt-2">
   364→                  <span className="text-xs px-2 py-0.5 rounded-full font-medium" style={{ backgroundColor: "rgba(0,145,64,0.2)", color: "#009140" }}>
   365→                    Quente
   366→                  </span>
   367→                  <span className="text-xs px-2 py-0.5 rounded-full font-medium" style={{ backgroundColor: "rgba(255,215,0,0.15)", color: "#FFD700" }}>
   368→                    Morno
   369→                  </span>
   370→                  <span className="text-xs px-2 py-0.5 rounded-full font-medium" style={{ backgroundColor: "rgba(107,114,128,0.15)", color: "#9CA3AF" }}>
   371→                    Frio
   372→                  </span>
   373→                  <span className="text-xs px-2 py-0.5 rounded-full font-medium" style={{ backgroundColor: "rgba(156,163,175,0.1)", color: "#6B7280" }}>
   374→                    Congelado
   375→                  </span>
   376→                </div>
   377→              </div>
   378→
   379→              <div className="space-y-2">
   380→                <label className="text-xs text-stone-500 uppercase tracking-wider font-semibold">
   381→                  Modo de Jogo
   382→                </label>
   383→                <div className="grid grid-cols-2 gap-3">
   384→                  {(["bot", "pvp"] as Mode[]).map((m) => (
   385→                    <motion.button
   386→                      key={m}
   387→                      whileHover={{ scale: 1.02 }}
   388→                      whileTap={{ scale: 0.98 }}
   389→                      onClick={() => setMode(m)}
   390→                      className="py-3 px-4 rounded-xl font-bold text-sm border-2 transition-all"
   391→                      style={{
   392→                        borderColor: mode === m ? "#FFD700" : "rgba(255,255,255,0.08)",
   393→                        backgroundColor: mode === m ? "rgba(255,215,0,0.12)" : "rgba(255,255,255,0.03)",
   394→                        color: mode === m ? "#FFD700" : "#9CA3AF",
   395→                      }}
   396→                    >
   397→                      {m === "bot" ? "Contra Bot" : "Jogador vs Jogador"}
   398→                    </motion.button>
   399→                  ))}
   400→                </div>
   401→              </div>
   402→
   403→              {mode === "bot" && (
   404→                <motion.div
   405→                  initial={{ opacity: 0, height: 0 }}
   406→                  animate={{ opacity: 1, height: "auto" }}
   407→                  exit={{ opacity: 0, height: 0 }}
   408→                  className="space-y-2"
   409→                >
   410→                  <label className="text-xs text-stone-500 uppercase tracking-wider font-semibold">
   411→                    Dificuldade
   412→                  </label>
   413→                  <div className="grid grid-cols-3 gap-2">
   414→                    {(["Facil", "Medio", "Dificil"] as Difficulty[]).map((d) => (
   415→                      <motion.button
   416→                        key={d}
   417→                        whileHover={{ scale: 1.02 }}
   418→                        whileTap={{ scale: 0.98 }}
   419→                        onClick={() => setDifficulty(d)}
   420→                        className="py-2.5 px-3 rounded-xl font-bold text-xs border-2 transition-all"
   421→                        style={{
   422→                          borderColor: difficulty === d ? "#FF6B35" : "rgba(255,255,255,0.08)",
   423→                          backgroundColor:
   424→                            difficulty === d ? "rgba(255,107,53,0.15)" : "rgba(255,255,255,0.03)",
   425→                          color: difficulty === d ? "#FF6B35" : "#9CA3AF",
   426→                        }}
   427→                      >
   428→                        {d}
   429→                      </motion.button>
   430→                    ))}
   431→                  </div>
   432→                  <p className="text-[11px] text-stone-500">
   433→                    {difficulty === "Facil" && "O bot e distraido — faz muitas tentativas aleatorias"}
   434→                    {difficulty === "Medio" && "O bot e esperto — comete alguns erros"}
   435→                    {difficulty === "Dificil" && "O bot e quase perfeito — busca binaria precisa"}
   436→                  </p>
   437→                </motion.div>
   438→              )}
   439→
   440→              <div className="rounded-xl p-4" style={{ backgroundColor: "rgba(255,215,0,0.05)", border: "1px solid rgba(255,215,0,0.1)" }}>
   441→                <h3 className="text-xs font-bold mb-2" style={{ color: "#FFD700" }}>
   442→                  Como Jogar
   443→                </h3>
   444→                <ul className="text-xs text-stone-400 space-y-1">
   445→                  <li>Ambos os jogadores tentam adivinhar o mesmo numero secreto (1-100)</li>
   446→                  <li>Apos cada tentativa, recebe uma dica de temperatura e direccao</li>
   447→                  <li>O primeiro a acertar ganha pontos: 100 - (tentativas x 5)</li>
   448→                  <li>Multiplas rodadas em 60 segundos. Acumule a maior pontuacao!</li>
   449→                </ul>
   450→              </div>
   451→
   452→              <motion.button
   453→                whileHover={{ scale: 1.03 }}
   454→                whileTap={{ scale: 0.97 }}
   455→                onClick={startGame}
   456→                className="w-full py-3.5 rounded-xl font-black text-base tracking-wide"
   457→                style={{
   458→                  background: "linear-gradient(135deg, #009140, #006B30)",
   459→                  color: "#FFD700",
   460→                  boxShadow: "0 4px 20px rgba(0,145,64,0.3)",
   461→                }}
   462→              >
   463→                COMECAR JOGO
   464→              </motion.button>
   465→            </motion.div>
   466→          )}
   467→
   468→          {phase === "countdown" && (
   469→            <motion.div
   470→              key="countdown"
   471→              initial={{ opacity: 0 }}
   472→              animate={{ opacity: 1 }}
   473→              exit={{ opacity: 0 }}
   474→              className="flex flex-col items-center justify-center py-20 space-y-4"
   475→            >
   476→              <AnimatePresence mode="wait">
   477→                <motion.div
   478→                  key={countdownNum}
   479→                  initial={{ scale: 0.5, opacity: 0 }}
   480→                  animate={{ scale: 1, opacity: 1 }}
   481→                  exit={{ scale: 1.5, opacity: 0 }}
   482→                  transition={{ duration: 0.6 }}
   483→                  className="text-8xl font-black"
   484→                  style={{ color: countdownNum > 0 ? "#FFD700" : "#009140" }}
   485→                >
   486→                  {countdownNum > 0 ? countdownNum : "VAI!"}
   487→                </motion.div>
   488→              </AnimatePresence>
   489→              <p className="text-stone-500 text-sm">Preparem-se...</p>
   490→            </motion.div>
   491→          )}
   492→
   493→          {phase === "playing" && (
   494→            <motion.div
   495→              key="playing"
   496→              initial={{ opacity: 0 }}
   497→              animate={{ opacity: 1 }}
   498→              exit={{ opacity: 0 }}
   499→              className="p-4 space-y-3"
   500→            >
   501→
   502→              <div className="space-y-1">
   503→                <div className="flex items-center justify-between">
   504→                  <span className="text-xs font-bold text-stone-400">
   505→                    Rodada {round}
   506→                  </span>
   507→                  <span
   508→                    className="text-sm font-black tabular-nums"
   509→                    style={{ color: timerColor }}
   510→                  >
   511→                    {timeLeft}s
   512→                  </span>
   513→                </div>
   514→                <div className="w-full h-2 rounded-full bg-stone-700/50 overflow-hidden">
   515→                  <motion.div
   516→                    className="h-full rounded-full"
   517→                    style={{ backgroundColor: timerColor }}
   518→                    animate={{ width: `${timerPct}%` }}
   519→                    transition={{ duration: 0.5 }}
   520→                  />
   521→                </div>
   522→              </div>
