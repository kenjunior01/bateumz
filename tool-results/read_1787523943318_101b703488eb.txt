    73→const rnd = () => Math.floor(Math.random() * TYPES);
    74→const CK = (r: number, c: number) => `${r},${c}`;
    75→
    76→function mkGrid(): { grid: Cell[][]; nid: number } {
    77→  let nid = 0;
    78→  const grid: Cell[][] = [];
    79→  for (let r = 0; r < ROWS; r++) {
    80→    const row: Cell[] = [];
    81→    for (let c = 0; c < COLS; c++) row.push({ type: rnd(), id: nid++ });
    82→    grid.push(row);
    83→  }
    84→  // Clear any pre-existing matches so both players start clean
    85→  let iter = 0;
    86→  let res = findMatches(grid);
    87→  while (res.matched.size > 0 && iter < 200) {
    88→    const arr = Array.from(res.matched);
    89→    for (let i = 0; i < arr.length; i++) {
    90→      const parts = arr[i].split(',');
    91→      const r = parseInt(parts[0], 10);
    92→      const c = parseInt(parts[1], 10);
    93→      grid[r][c].type = rnd();
    94→    }
    95→    res = findMatches(grid);
    96→    iter++;
    97→  }
    98→  return { grid, nid };
    99→}
   100→
   101→function findMatches(grid: Cell[][]): { matched: Set<string>; lines: [number, number][][] } {
   102→  const matched = new Set<string>();
   103→  const lines: [number, number][][] = [];
   104→
   105→  // Horizontal
   106→  for (let r = 0; r < ROWS; r++) {
   107→    let s = 0;
   108→    for (let c = 1; c <= COLS; c++) {
   109→      if (c < COLS && grid[r][c].type === grid[r][s].type) continue;
   110→      if (c - s >= 4) {
   111→        const ln: [number, number][] = [];
   112→        for (let k = s; k < c; k++) {
   113→          ln.push([r, k]);
   114→          matched.add(CK(r, k));
   115→        }
   116→        lines.push(ln);
   117→      }
   118→      s = c;
   119→    }
   120→  }
   121→
   122→  // Vertical
   123→  for (let c = 0; c < COLS; c++) {
   124→    let s = 0;
   125→    for (let r = 1; r <= ROWS; r++) {
   126→      if (r < ROWS && grid[r][c].type === grid[s][c].type) continue;
   127→      if (r - s >= 4) {
   128→        const ln: [number, number][] = [];
   129→        for (let k = s; k < r; k++) {
   130→          ln.push([k, c]);
   131→          matched.add(CK(k, c));
   132→        }
   133→        lines.push(ln);
   134→      }
   135→      s = r;
   136→    }
   137→  }
   138→
   139→  return { matched, lines };
   140→}
   141→
   142→function calcScore(lines: [number, number][][], cascade: number): number {
   143→  let t = 0;
   144→  for (const l of lines) {
   145→    t += l.length === 4 ? 40 : l.length === 5 ? 100 : 200;
   146→  }
   147→  return Math.floor(t * Math.pow(1.5, cascade));
   148→}
   149→
   150→function doGravity(
   151→  grid: Cell[][],
   152→  matched: Set<string>,
   153→  nid: number,
   154→): { g: Cell[][]; nid: number; ff: Map<string, number> } {
   155→  const g: Cell[][] = Array.from({ length: ROWS }, () =>
   156→    new Array<Cell>(COLS).fill(null!),
   157→  );
   158→  const ff = new Map<string, number>();
   159→
   160→  for (let c = 0; c < COLS; c++) {
   161→    // Collect surviving cells bottom-to-top
   162→    const surv: { cell: Cell; origR: number }[] = [];
   163→    for (let r = ROWS - 1; r >= 0; r--) {
   164→      if (!matched.has(CK(r, c))) {
   165→        surv.push({ cell: grid[r][c], origR: r });
   166→      }
   167→    }
   168→
   169→    // Place surviving cells from the bottom up
   170→    let wr = ROWS - 1;
   171→    for (const { cell, origR } of surv) {
   172→      g[wr][c] = cell;
   173→      ff.set(CK(wr, c), origR);
   174→      wr--;
   175→    }
   176→
   177→    // Fill remaining top rows with new cells
   178→    const numNew = wr + 1;
   179→    for (let r = wr; r >= 0; r--) {
   180→      g[r][c] = { type: rnd(), id: nid++ };
   181→      ff.set(CK(r, c), r - numNew);
   182→    }
   183→  }
   184→
   185→  return { g, nid, ff };
   186→}
   187→
   188→function mkPlayer(): PlayerState {
   189→  const { grid, nid } = mkGrid();
   190→  return {
   191→    grid,
   192→    score: 0,
   193→    nextId: nid,
   194→    selected: null,
   195→    phase: 'idle',
   196→    matchedCells: new Set(),
   197→    swapFrom: null,
   198→    swapTo: null,
   199→    fallFrom: new Map(),
   200→    gravVer: 0,
   201→    cascade: 0,
   202→    popups: [],
   203→    popId: 0,
   204→    comboShow: 0,
   205→    noMatch: false,
   206→  };
   207→}

   213→export default function Match4Grid({ onScore, liveCode }: Props) {
   214→  /* ---- state ---- */
   215→  const [p, setP] = useState<[PlayerState, PlayerState]>([mkPlayer(), mkPlayer()]);
   216→  const pref = useRef(p);
   217→  pref.current = p;
   218→
   219→  const [time, setTime] = useState(90);
   220→  const [dur, setDur] = useState(90);
   221→  const [active, setActive] = useState(false);
   222→  const [over, setOver] = useState(false);
   223→  const [gen, setGen] = useState(0);
   224→  const genRef = useRef(0);
   225→  const timers = useRef<number[]>([]);
   226→  const activeRef = useRef(false);
   227→  const overRef = useRef(false);
   228→  const onScoreRef = useRef(onScore);
   229→  onScoreRef.current = onScore;
   230→
   231→  activeRef.current = active;
   232→  overRef.current = over;
   233→
   234→  // Force re-render (shallow clone so React sees new ref)
   235→  const upd = useCallback(
   236→    () => setP((x) => [x[0], x[1]] as [PlayerState, PlayerState]),
   237→    [],
   238→  );
   239→
   240→  // Schedule a callback that is cancelled on restart / unmount
   241→  const sched = useCallback((fn: () => void, ms: number) => {
   242→    const g = genRef.current;
   243→    const id = window.setTimeout(() => {
   244→      if (genRef.current !== g) return; // stale generation
   245→      fn();
   246→    }, ms);
   247→    timers.current.push(id);
   248→  }, []);
   249→
   250→  const clearT = useCallback(() => {
   251→    timers.current.forEach(clearTimeout);
   252→    timers.current = [];
   253→  }, []);
   254→
   255→  /* ---- countdown timer ---- */
   256→  useEffect(() => {
   257→    if (!active || over) return;
   258→    const iv = setInterval(() => {
   259→      if (genRef.current !== gen) {
   260→        clearInterval(iv);
   261→        return;
   262→      }
   263→      setTime((t) => {
   264→        if (t <= 1) {
   265→          clearInterval(iv);
   266→          setActive(false);
   267→          setOver(true);
   268→          activeRef.current = false;
   269→          overRef.current = true;
   270→          return 0;
   271→        }
   272→        return t - 1;
   273→      });
   274→    }, 1000);
   275→    return () => clearInterval(iv);
   276→  }, [active, over, gen]);
   277→
   278→  /* ---- fire onScore when game ends ---- */
   279→  useEffect(() => {
   280→    if (!over) return;
   281→    const pp = pref.current;
   282→    onScoreRef.current?.('Jogador 1', pp[0].score);
   283→    onScoreRef.current?.('Jogador 2', pp[1].score);
   284→  }, [over]);
   285→
   286→  /* ---- cleanup on unmount ---- */
   287→  useEffect(() => () => clearT(), [clearT]);
   288→
   289→  /* ============================================================
   290→     Game logic (reads / mutates pref.current, then calls upd)
   291→     ============================================================ */
   292→
   293→  const processMatches = useCallback(
   294→    (
   295→      pi: number,
   296→      matched: Set<string>,
   297→      lines: [number, number][][],
   298→      cascade: number,
   299→    ) => {
   300→      const pl = pref.current[pi];
   301→      const pts = calcScore(lines, cascade);
   302→      pl.score += pts;
   303→      pl.cascade = cascade;
   304→
   305→      // Score popup at the centre of the first match line
   306→      if (lines.length > 0) {
   307→        const ln = lines[0];
   308→        const mid = ln[Math.floor(ln.length / 2)];
   309→        const pid = pl.popId++;
   310→        pl.popups.push({
   311→          id: pid,
   312→          text: `+${pts}`,
   313→          row: mid[0],
   314→          col: mid[1],
   315→        });
   316→        sched(() => {
   317→          pl.popups = pl.popups.filter((pp) => pp.id !== pid);
   318→          upd();
   319→        }, 1100);
   320→      }
   321→
   322→      if (cascade > 0) pl.comboShow = cascade + 1;
   323→
   324→      // Phase: removing
   325→      pl.phase = 'removing';
   326→      pl.matchedCells = matched;
   327→      upd();
   328→
   329→      sched(() => {
   330→        const pl2 = pref.current[pi];
   331→        if (overRef.current) {
   332→          pl2.phase = 'idle';
   333→          pl2.matchedCells = new Set();
   334→          upd();
   335→          return;
   336→        }
   337→
   338→        // Apply gravity
   339→        const { g, nid, ff } = doGravity(pl2.grid, matched, pl2.nextId);
   340→        pl2.grid = g;
   341→        pl2.nextId = nid;
   342→        pl2.fallFrom = ff;
   343→        pl2.gravVer++;
   344→        pl2.matchedCells = new Set();
   345→        pl2.phase = 'falling';
   346→        upd();
   347→
   348→        sched(() => {
   349→          const pl3 = pref.current[pi];
   350→          if (overRef.current) {
   351→            pl3.phase = 'idle';
   352→            pl3.fallFrom = new Map();
   353→            upd();
   354→            return;
   355→          }
   356→
   357→          pl3.fallFrom = new Map();
   358→          const res = findMatches(pl3.grid);
   359→          if (res.matched.size > 0) {
   360→            processMatches(pi, res.matched, res.lines, cascade + 1);
   361→          } else {
   362→            pl3.phase = 'idle';
   363→            pl3.cascade = 0;
   364→            pl3.comboShow = 0;
   365→            upd();
   366→          }
   367→        }, 480);
   368→      }, 420);
   369→    },
   370→    [sched, upd],
   371→  );
   372→
   373→  const doSwap = useCallback(
   374→    (pi: number, r1: number, c1: number, r2: number, c2: number) => {
   375→      const pl = pref.current[pi];
   376→      pl.phase = 'swapping';
   377→      pl.swapFrom = [r1, c1];
   378→      pl.swapTo = [r2, c2];
   379→      pl.selected = null;
   380→      upd();
   381→
   382→      sched(() => {
   383→        const pl2 = pref.current[pi];
   384→        if (overRef.current) {
   385→          pl2.phase = 'idle';
   386→          pl2.swapFrom = null;
   387→          pl2.swapTo = null;
   388→          upd();
   389→          return;
   390→        }
   391→
   392→        // Perform the swap in data
   393→        const tmp = pl2.grid[r1][c1];
   394→        pl2.grid[r1][c1] = pl2.grid[r2][c2];
   395→        pl2.grid[r2][c2] = tmp;
   396→        pl2.swapFrom = null;
   397→        pl2.swapTo = null;
   398→        upd();
   399→
   400→        const { matched, lines } = findMatches(pl2.grid);
   401→
   402→        if (matched.size === 0) {
   403→          // No match – revert with animation
   404→          pl2.noMatch = true;
   405→          pl2.phase = 'swapping';
   406→          pl2.swapFrom = [r2, c2];
   407→          pl2.swapTo = [r1, c1];
   408→          upd();
   409→
   410→          sched(() => {
   411→            const pl3 = pref.current[pi];
   412→            const tmp2 = pl3.grid[r1][c1];
   413→            pl3.grid[r1][c1] = pl3.grid[r2][c2];
   414→            pl3.grid[r2][c2] = tmp2;
   415→            pl3.swapFrom = null;
   416→            pl3.swapTo = null;
   417→            pl3.phase = 'idle';
   418→            pl3.noMatch = false;
   419→            upd();
   420→          }, 320);
   421→          return;
   422→        }
   423→
   424→        processMatches(pi, matched, lines, 0);
   425→      }, 260);
   426→    },
   427→    [sched, upd, processMatches],
   428→  );
   429→
   430→  /* ---- click handler ---- */
   431→  const handleClick = useCallback(
   432→    (pi: number, r: number, c: number) => {
   433→      if (!activeRef.current || overRef.current) return;
   434→      const pl = pref.current[pi];
   435→      if (pl.phase !== 'idle' && pl.phase !== 'selected') return;
   436→
   437→      if (!pl.selected) {
   438→        pl.selected = [r, c];
   439→        pl.phase = 'selected';
   440→        upd();
   441→        return;
   442→      }
   443→
   444→      const [sr, sc] = pl.selected;
   445→
   446→      // Clicked same cell → deselect
   447→      if (sr === r && sc === c) {
   448→        pl.selected = null;
   449→        pl.phase = 'idle';
   450→        upd();
   451→        return;
   452→      }
   453→
   454→      // Adjacent → swap
   455→      if (Math.abs(sr - r) + Math.abs(sc - c) === 1) {
   456→        doSwap(pi, sr, sc, r, c);
   457→        return;
   458→      }
   459→
   460→      // Non-adjacent → re-select
   461→      pl.selected = [r, c];
   462→      upd();
   463→    },
   464→    [doSwap, upd],
   465→  );
   466→
   467→  /* ---- controls ---- */
   468→  const bumpGen = useCallback(() => {
   469→    clearT();
   470→    genRef.current++;
   471→    const g = genRef.current;
   472→    setGen(g);
   473→  }, [clearT]);
   474→
   475→  const restart = useCallback(() => {
   476→    bumpGen();
   477→    setP([mkPlayer(), mkPlayer()]);
   478→    setTime(dur);
   479→    setOver(false);
   480→    setActive(false);
   481→  }, [dur, bumpGen]);
   482→
   483→  const start = useCallback(() => {
   484→    bumpGen();
   485→    setP([mkPlayer(), mkPlayer()]);
   486→    setTime(dur);
   487→    setOver(false);
   488→    setActive(true);
   489→  }, [dur, bumpGen]);
   490→
   491→  const changeDur = useCallback(
   492→    (d: number) => {
   493→      setDur(d);
   494→      if (!active) setTime(d);
   495→    },
   496→    [active],
   497→  );
   498→
   499→  /* ============================================================
   500→     Render
   501→     ============================================================ */
   502→
   503→  const renderGrid = (pi: number) => {
   504→    const pl = p[pi];
   505→    const borderC = pi === 0 ? 'border-cyan-500/30' : 'border-pink-500/30';
   506→    const interactive = active && !over;
   507→
   508→    return (
   509→      <div className="relative">
   510→        <div
   511→          className={cn(
   512→            'relative bg-slate-900/50 border border-slate-800 rounded-2xl p-2',
   513→            borderC,
   514→            !interactive && 'opacity-60',
   515→          )}
   516→        >
   517→          <div className="grid grid-cols-6 gap-1">
   518→            {pl.grid.map((row, r) =>
   519→              row.map((cell, c) => {
   520→                const isSel =
   521→                  pl.selected !== null &&
   522→                  pl.selected[0] === r &&
   523→                  pl.selected[1] === c;
   524→                const isMat = pl.matchedCells.has(CK(r, c));
   525→                const isSF =
   526→                  pl.swapFrom !== null &&
   527→                  pl.swapFrom[0] === r &&
   528→                  pl.swapFrom[1] === c;
   529→                const isST =
   530→                  pl.swapTo !== null &&
   531→                  pl.swapTo[0] === r &&
   532→                  pl.swapTo[1] === c;
   533→                const foff = pl.fallFrom.get(CK(r, c));
   534→                const shape = SHAPES[cell.type];
   535→
   536→                // Swap animation offsets
   537→                let ax = 0;
   538→                let ay = 0;
   539→                if (pl.phase === 'swapping') {
   540→                  if (isSF && pl.swapTo && pl.swapFrom) {
   541→                    ax = (pl.swapTo[1] - pl.swapFrom[1]) * CELL_PITCH;
   542→                    ay = (pl.swapTo[0] - pl.swapFrom[0]) * CELL_PITCH;
   543→                  } else if (isST && pl.swapFrom && pl.swapTo) {
   544→                    ax = (pl.swapFrom[1] - pl.swapTo[1]) * CELL_PITCH;
   545→                    ay = (pl.swapFrom[0] - pl.swapTo[0]) * CELL_PITCH;
   546→                  }
   547→                }
   548→
   549→                const fallInitial =
   550→                  foff !== undefined
   551→                    ? { y: (foff - r) * CELL_PITCH, opacity: 0.2, scale: 0.7 }
   552→                    : false;
   553→
   554→                return (
   555→                  <motion.div
   556→                    key={`${cell.id}-v${pl.gravVer}`}
   557→                    initial={fallInitial}
   558→                    animate={{
   559→                      scale: isMat ? 0 : isSel ? 1.15 : 1,
   560→                      opacity: isMat ? 0 : 1,
   561→                      x: ax,
   562→                      y: 0,
   563→                    }}
   564→                    transition={
   565→                      pl.phase === 'falling'
   566→                        ? {
   567→                            type: 'spring',
   568→                            bounce: 0.35,
   569→                            stiffness: 260,
   570→                            damping: 18,
   571→                          }
   572→                        : pl.phase === 'swapping'
   573→                          ? { duration: 0.22, ease: 'easeInOut' }
   574→                          : pl.phase === 'removing'
   575→                            ? { duration: 0.35, ease: 'easeIn' }
   576→                            : { duration: 0.15 }
   577→                    }
   578→                    className={cn(
   579→                      'w-10 h-10 sm:w-12 sm:h-12 rounded-lg flex items-center justify-center select-none transition-shadow duration-150',
   580→                      shape.bg,
   581→                      interactive ? 'cursor-pointer' : 'cursor-default',
   582→                      isSel &&
   583→                        'ring-2 ring-white shadow-[0_0_12px_rgba(255,255,255,0.4)] z-10',
   584→                    )}
   585→                    onClick={() => handleClick(pi, r, c)}
   586→                    whileTap={
   587→                      interactive && !isMat
   588→                        ? { scale: 0.88 }
   589→                        : undefined
   590→                    }
   591→                  >
   592→                    <span className="text-white/90 text-[10px] sm:text-xs font-bold leading-none select-none pointer-events-none drop-shadow-[0_1px_2px_rgba(0,0,0,0.5)]">
   593→                      {shape.sym}
   594→                    </span>
   595→                  </motion.div>
   596→                );
   597→              }),
   598→            )}
   599→          </div>
   600→
   601→          <AnimatePresence>
   602→            {pl.popups.map((pop) => (
   603→              <motion.div
   604→                key={pop.id}
   605→                initial={{ opacity: 1, y: 0, scale: 0.6 }}
   606→                animate={{ opacity: 0, y: -44, scale: 1.4 }}
   607→                exit={{ opacity: 0 }}
   608→                transition={{ duration: 1, ease: 'easeOut' }}
   609→                className="absolute pointer-events-none z-30 font-extrabold text-white text-base sm:text-lg"
   610→                style={{
   611→                  left: `calc(0.5rem + ${pop.col} * (100% - 1rem) / 6 + (100% - 1rem) / 12)`,
   612→                  top: `calc(0.5rem + ${pop.row} * (100% - 1rem) / 6 + (100% - 1rem) / 12 - 0.25rem)`,
   613→                  transform: 'translateX(-50%)',
   614→                  textShadow: '0 2px 8px rgba(0,0,0,0.7)',
   615→                }}
