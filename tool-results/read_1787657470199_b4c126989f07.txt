    69→function RaffleProgressRing({ sold, total, size = 80, strokeWidth = 5, color = C_ACCENT }: { sold: number; total: number; size?: number; strokeWidth?: number; color?: string }) {
    70→  const [pct, setPct] = useState(0);
    71→  const safeTotal = Math.max(total, 1);
    72→  useEffect(() => {
    73→    const timer = setTimeout(() => setPct(Math.min(sold / safeTotal, 1)), 500);
    74→    return () => clearTimeout(timer);
    75→  }, [sold, safeTotal]);
    76→  const r = (size - strokeWidth) / 2;
    77→  const circ = 2 * Math.PI * r;
    78→  const offset = circ * (1 - pct);
    79→  return (
    80→    <div className="relative inline-flex items-center justify-center">
    81→      <svg width={size} height={size} className="-rotate-90">
    82→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="rgba(255,255,255,0.05)" strokeWidth={strokeWidth} />
    83→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={strokeWidth} strokeDasharray={circ} strokeDashoffset={offset} strokeLinecap="round" style={{ transition: "stroke-dashoffset 1.5s cubic-bezier(0.22,1,0.36,1)", filter: "drop-shadow(0 0 6px " + color + "40)" }} />
    84→      </svg>
    85→      <div className="absolute text-center">
    86→        <p className="text-xl font-black font-display" style={{ color }}>{Math.round(pct * 100)}%</p>
    87→        <p className="text-[9px] text-muted-foreground font-semibold uppercase">vendidos</p>
    88→      </div>
    89→    </div>
    90→  );
    91→}

   363→                  <motion.button

   406→                          style={{ background: whiteLabelConfig ? whiteLabelConfig.primary_color : "linear-gradient(135deg, " + C_PRIMARY + ", " + C_ACCENT + ")" }}

   409→                          <ShoppingCart className="h-5 w-5" /> Comprar Bilhetes
   410→                        </motion.button>
   411→                      </motion.div>
   412→                    )}
   413→                  </AnimatePresence>

   416→                    <span className="flex items-center gap-1.5"><span className="h-2.5 w-2.5 rounded" style={{ background: "rgba(255,255,255,0.06)" }} /> Disponivel</span>

   418→                    <span className="flex items-center gap-1.5"><span className="h-2.5 w-2.5 rounded" style={{ background: "rgba(255,255,255,0.03)" }} /> Vendido</span>
   419→                  </div>
   420→                </div>
   421→              </motion.div>
   422→            )}
   423→          </div>
   424→        </div>
   425→      </div>
   426→
   427→      <AnimatePresence>
   428→        {checkoutStep > 0 && (
   429→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex items-end md:items-center justify-center" style={{ background: "rgba(0,0,0,0.6)", backdropFilter: "blur(8px)" }} onClick={(e) => { if (e.target === e.currentTarget && checkoutStep < 4) goToStep(0); }}>
   430→            <motion.div initial={{ y: 100, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: 100, opacity: 0 }} transition={{ ...SPRING }} className="rd-checkout-sheet">

   432→                <div className="rd-checkout-header">
   433→                  <div className="w-10 h-1 rounded-full bg-muted-foreground/30 mx-auto mb-4 md:hidden" />

   435→                    <h3 className="font-display text-lg font-bold">Checkout Seguro</h3>

   437→                  </div>
   438→                  <div className="flex items-center gap-2 text-[11px] text-muted-foreground">
   439→                    <ShieldCheck className="h-3.5 w-3.5" style={{ color: C_PRIMARY }} />

   450→                        <div className="flex justify-between text-sm"><span className="text-muted-foreground">Bilhetes</span><span className="text-foreground font-medium">{selectedNumbers.sort((a, b) => a - b).join(", ")}</span></div>
   451→                        <div className="flex justify-between text-sm"><span className="text-muted-foreground">Preco unitario</span><span className="text-foreground">{fmt(raffle.ticket_price)}</span></div>
   452→                        <div className="rd-order-total">
   453→                          <span className="font-semibold">Total</span>
   454→                          <span className="font-display text-2xl font-black" style={{ color: C_PRIMARY }}>{fmt(totalPrice)}</span>
   455→                        </div>
   456→                        <div className="flex items-center gap-2 text-xs" style={{ color: C_GOLD }}>
   457→                          <Star className="h-3.5 w-3.5" />+{selectedNumbers.length * 10} Pontos de Sorte apos pagamento
   458→                        </div>
   459→                      </div>
   460→                      <div className="grid grid-cols-3 gap-2 mb-4 text-center">
   461→                        {[{ icon: ShieldCheck, label: "100% Seguro", color: C_PRIMARY }, { icon: Zap, label: "Bilhetes Instantaneos", color: C_ACCENT }, { icon: Trophy, label: "Sorteios Verificados", color: C_GOLD }].map((item) => (
   462→                          <div key={item.label} className="rd-trust-badge"><item.icon className="h-4 w-4 mx-auto mb-1" style={{ color: item.color }} /><p className="text-[10px] text-muted-foreground">{item.label}</p></div>
   463→                        ))}
   464→                      </div>
   465→                      {!user ? (
   466→                        <button className="rd-buy-btn w-full" onClick={() => navigate("/login")}>Entrar para pagar</button>
   467→                      ) : (
   468→                        <PaymentGatewaySelector raffleId={raffle.id} quantity={selectedNumbers.length} ticketNumbers={selectedNumbers} amount={totalPrice} currency={currency} description={"Raffle: " + raffle.title} onSuccess={onPayPalSuccess} disabled={paid} />
   469→                      )}
   470→                    </motion.div>
   471→                  )}
   472→                  {checkoutStep === 4 && (
   473→                    <motion.div key="success" initial={{ scale: 0.8, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} transition={{ ...SPRING_BOUNCE }} className="text-center py-6">
   474→                      <motion.div animate={{ rotate: [0, 10, -10, 0], scale: [1, 1.2, 1] }} transition={{ duration: 0.6 }} className="inline-flex h-20 w-20 items-center justify-center rounded-full mb-4" style={{ background: C_PRIMARY + "20" }}>
   475→                        <PartyPopper className="h-10 w-10" style={{ color: C_PRIMARY }} />
   476→                      </motion.div>
   477→                      <h3 className="font-display text-2xl font-bold mb-2">Estas dentro!</h3>
   478→                      <p className="text-muted-foreground mb-1">Pagamento confirmado — bilhetes garantidos.</p>
   479→                      <p className="text-sm font-medium" style={{ color: C_GOLD }}>+{selectedNumbers.length * 10} Pontos de Sorte ganhos</p>
   480→                      {[...Array(12)].map((_, i) => (
   481→                        <motion.div key={i} className="absolute w-2 h-2 rounded-full" style={{ left: 20 + Math.random() * 60 + "%", top: 10 + Math.random() * 30 + "%", backgroundColor: i % 3 === 0 ? C_PRIMARY : i % 3 === 1 ? C_ACCENT : C_GOLD }} initial={{ y: 0, opacity: 1, scale: 0 }} animate={{ y: [0, -60, 20], opacity: [0, 1, 0], scale: [0, 1, 0.5] }} transition={{ duration: 1.5, delay: i * 0.08, ease: "easeOut" }} />
   482→                      ))}
   483→                    </motion.div>
   484→                  )}
   485→                </AnimatePresence>
   486→              </div>
   487→              {checkoutStep === 1 && (
   488→                <div className="px-5 pb-5 pt-2" style={{ borderTop: "1px solid rgba(255,255,255,0.05)" }}>
   489→                  <motion.button className="rd-cancel-btn w-full" onClick={() => goToStep(0)} whileHover={{ scale: 1.01 }} whileTap={{ scale: 0.99 }}>Cancelar</motion.button>
   490→                </div>
   491→              )}
   492→            </motion.div>
   493→          </motion.div>
   494→        )}
   495→      </AnimatePresence>
   496→
   497→      <BolaoModal raffleId={raffle.id} raffleTitle={raffle.title} open={bolaoOpen} onClose={() => setBolaoOpen(false)} />
