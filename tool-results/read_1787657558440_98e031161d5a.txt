    27→function timeAgo(dateStr: string): string {
    28→  const diff = Date.now() - new Date(dateStr).getTime();
    29→  const mins = Math.floor(diff / 60000);
    30→  if (mins < 1) return 'agora';
    31→  if (mins < 60) return `${mins}min`;
    32→  const hrs = Math.floor(mins / 60);
    33→  if (hrs < 24) return `${hrs}h`;
    34→  return `${Math.floor(hrs / 24)}d`;
    35→}

    99→// Animated counter hook
   100→function useAnimatedCounter(target: number, duration: number = 2000, startOnMount: boolean = true) {
   101→  const [count, setCount] = useState(0);
   102→
   103→  useEffect(() => {
   104→    if (!startOnMount) return;
   105→    let startTime: number;
   106→    let raf: number;
   107→
   108→    const animate = (timestamp: number) => {
   109→      if (!startTime) startTime = timestamp;
   110→      const progress = Math.min((timestamp - startTime) / duration, 1);
   111→      const eased = 1 - Math.pow(1 - progress, 3);
   112→      setCount(Math.floor(eased * target));
   113→      if (progress < 1) {
   114→        raf = requestAnimationFrame(animate);
   115→      }
   116→    };
   117→
   118→    raf = requestAnimationFrame(animate);
   119→    return () => cancelAnimationFrame(raf);
   120→  }, [target, duration, startOnMount]);
   121→
   122→  return count;
   123→}
