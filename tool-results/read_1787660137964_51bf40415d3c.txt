   334→  const [visible, setVisible] = useState(false);
   335→  const [message, setMessage] = useState("");

   342→  const [positionIndex, setPositionIndex] = useState(0);
   343→  const [miniMascots, setMiniMascots] = useState<number[]>([]);
   344→  const [celebration, setCelebration] = useState<CelebrationData | null>(null);
   345→  const lastPointsRef = useRef<number | null>(null);
   346→  const celebratedMilestonesRef = useRef<Set<number>>(new Set());
   347→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   348→  const miniTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   349→  const appearCountRef = useRef(0);
   350→  const chatEndRef = useRef<HTMLDivElement>(null);
   351→  const inputRef = useRef<HTMLInputElement>(null);
   352→  const modeRef = useRef<MascotMode>(mode);
   353→  modeRef.current = mode;
   354→
   355→  const userName = profile?.display_name || (lang === "en" ? "friend" : "amigo/a");
   356→  const { context, mood } = useMemo(() => getRouteContext(location.pathname), [location.pathname]);
   357→
   358→  // ── Track point milestones ──
   359→  useEffect(() => {
   360→    if (!user) return;
   361→    const checkPoints = async () => {
   362→      const { data } = await supabase
   363→        .from("luck_points")
   364→        .select("points")
   365→        .eq("user_id", user.id);
   366→      if (!data) return;
   367→      const total = data.reduce((sum, r) => sum + r.points, 0);
   368→
   369→      if (lastPointsRef.current !== null) {
   370→        for (const milestone of POINT_MILESTONES) {
   371→          if (
   372→            total >= milestone &&
   373→            lastPointsRef.current < milestone &&
   374→            !celebratedMilestonesRef.current.has(milestone)
   375→          ) {
   376→            celebratedMilestonesRef.current.add(milestone);
   377→            playMilestoneChime();
   378→            setCelebration({
   379→              type: "milestone",
   380→              title: `${milestone} Luck Points! 🌟`,
   381→              subtitle: `Parabéns, ${userName}! Alcançaste um marco incrível!`,
   382→              emoji: milestone >= 1000 ? "💎" : milestone >= 500 ? "🏆" : "⭐",
   383→            });
   384→            break;
   385→          }
   386→        }
   387→      }
   388→      lastPointsRef.current = total;
   389→    };
   390→
   391→    checkPoints();
   392→    const interval = setInterval(checkPoints, 30000);
   393→    return () => clearInterval(interval);
   394→  }, [user, userName]);
   395→
   396→  // ── Listen for raffle wins via notifications ──
   397→  useEffect(() => {
   398→    if (!user) return;
   399→    const channel = supabase
   400→      .channel("mascot-celebrations")
   401→      .on(
   402→        "postgres_changes",
   403→        {
   404→          event: "INSERT",
   405→          schema: "public",
   406→          table: "notifications",
   407→          filter: `user_id=eq.${user.id}`,
   408→        },
   409→        (payload) => {
   410→          const row = payload.new as any;
   411→          if (row.type === "winner" || row.title?.toLowerCase().includes("ganhas")) {
   412→            playVictoryFanfare();
   413→            setCelebration({
   414→              type: "win",
   415→              title: "GANHASTE! 🏆🎊",
   416→              subtitle: row.message || `Parabéns, ${userName}! Foste o grande vencedor!`,
   417→              emoji: "👑",
   418→            });
   419→          }
   420→        }
   421→      )
   422→      .subscribe();
   423→
   424→    return () => { supabase.removeChannel(channel); };
   425→  }, [user, userName]);
   426→
   427→  // Show mini mascots in various positions (rare, max 1, only desktop)
   428→  useEffect(() => {
   429→    const isMobile = typeof window !== "undefined" && window.innerWidth < 768;
   430→    if (isMobile) return; // No floating minis on mobile to avoid clutter
   431→    const spawnMini = () => {
   432→      if (modeRef.current === "chat") return;
   433→      const pos = Math.floor(Math.random() * POSITIONS.length);
   434→      setMiniMascots((prev) => {
   435→        if (prev.length >= 1) return prev;
   436→        if (prev.includes(pos)) return prev;
   437→        return [...prev, pos];
   438→      });
   439→      setTimeout(() => {
   440→        setMiniMascots((prev) => prev.filter((p) => p !== pos));
   441→      }, 4000);
   442→    };
   443→
   444→    miniTimerRef.current = setInterval(spawnMini, 90000 + Math.random() * 60000);
   445→    const initial = setTimeout(spawnMini, 30000);
   446→    return () => {
   447→      if (miniTimerRef.current) clearInterval(miniTimerRef.current);
   448→      clearTimeout(initial);
   449→    };
   450→  }, [location.pathname]);
   451→
   452→  const fetchMessage = useCallback(async (ctx: string, currentMood: MascotMood) => {
   453→    try {
   454→      setLoading(true);
   455→      const { data, error } = await supabase.functions.invoke("mascot-message", {
   456→        body: { userName, context: ctx, mood: currentMood, lang, userId: user?.id },
   457→      });
   458→      if (error || !data?.message) throw new Error("No message");
   459→      setMessage(data.message);
   460→    } catch {
   461→      const routePool = ROUTE_MESSAGES[location.pathname];
   462→      const pool = routePool && routePool.length > 0 ? routePool : FALLBACK_MESSAGES[currentMood];
   463→      setMessage(pool[Math.floor(Math.random() * pool.length)](userName));
   464→    } finally {
   465→      setLoading(false);
   466→    }
   467→  }, [userName, lang, user?.id, location.pathname]);
   468→
   469→  const showMascot = useCallback(async () => {
   470→    if (dismissed || modeRef.current === "chat") return;
   471→    // Randomize position
   472→    setPositionIndex(Math.floor(Math.random() * POSITIONS.length));
   473→    await fetchMessage(context, mood);
   474→    setVisible(true);
   475→    appearCountRef.current += 1;
   476→    playPopSound();
   477→
   478→    timerRef.current = setTimeout(() => {
   479→      if (modeRef.current !== "chat") setVisible(false);
   480→      const baseDelay = Math.min(90000 + appearCountRef.current * 30000, 240000);
   481→      timerRef.current = setTimeout(showMascot, baseDelay + Math.random() * 30000);
   482→    }, 10000);
   483→  }, [dismissed, fetchMessage, context, mood]);
   484→
   485→  useEffect(() => {
   486→    if (dismissed) return;
   487→    if (mode === "chat") return;
   488→    if (timerRef.current) clearTimeout(timerRef.current);
   489→    const delay = appearCountRef.current === 0 ? 12000 : 8000;
   490→    timerRef.current = setTimeout(showMascot, delay);
   491→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   492→  }, [location.pathname, showMascot, dismissed, mode]);
   493→
   494→  useEffect(() => {
   495→    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
   496→  }, [chatMessages]);
   497→
   498→  const openChat = () => {
   499→    if (timerRef.current) clearTimeout(timerRef.current);
   500→    setMode("chat");
   501→    setVisible(true);
   502→    setMiniMascots([]);
   503→    playPopSound();
   504→    if (chatMessages.length === 0) {
   505→      setChatMessages([{
   506→        role: "assistant",
   507→        content: lang === "en"
   508→          ? `Hey ${userName}! 😊 I'm Bateu, your assistant. Ask me anything about the platform!`
   509→          : `Olá ${userName}! 😊 Sou o Bateu, o teu assistente. Pergunta-me qualquer coisa sobre a plataforma!`,
   510→      }]);
   511→    }
   512→    setTimeout(() => inputRef.current?.focus(), 300);
   513→  };
   514→
   515→  const sendMessage = async (text: string) => {
   516→    if (!text.trim() || chatLoading) return;
   517→    const userMsg: ChatMsg = { role: "user", content: text.trim() };
   518→    const updatedMessages = [...chatMessages, userMsg];
   519→    setChatMessages(updatedMessages);
   520→    setChatInput("");
   521→    setChatLoading(true);
   522→    playSendSound();
   523→
   524→    try {
   525→      const { data, error } = await supabase.functions.invoke("mascot-chat", {
   526→        body: {
   527→          messages: updatedMessages.map(m => ({ role: m.role, content: m.content })),
   528→          userName,
   529→          context,
   530→          lang,
   531→          userId: user?.id,
   532→        },
   533→      });
   534→      if (error || !data?.message) throw new Error("No response");
   535→      setChatMessages(prev => [...prev, { role: "assistant", content: data.message }]);
   536→    } catch {
   537→      setChatMessages(prev => [...prev, { role: "assistant", content: lang === "en" ? "Oops, couldn't reply! 😅 Try again." : "Ops, não consegui responder! 😅 Tenta de novo." }]);
   538→    } finally {
   539→      setChatLoading(false);
   540→    }
   541→  };
   542→
   543→  const handleDismiss = () => {
   544→    if (mode === "chat") return;
   545→    openChat();
   546→  };
   547→
   548→  const handleClose = () => {
   549→    setVisible(false);
   550→    setMode("bubble");
   551→    setDismissed(true);
   552→    setMiniMascots([]);
   553→    playDismissSound();
   554→    if (timerRef.current) clearTimeout(timerRef.current);
   555→  };
   556→
   557→  const minimizeChat = () => {
   558→    setMode("bubble");
   559→    setVisible(false);
   560→    playDismissSound();
   561→    if (timerRef.current) clearTimeout(timerRef.current);
   562→    timerRef.current = setTimeout(showMascot, 60000);
   563→  };
   564→
   565→  const currentImage = MOOD_IMAGES[mood];
   566→  const currentAnimation = MOOD_ANIMATIONS[mood];
   567→  const currentPos = POSITIONS[positionIndex % POSITIONS.length];
   568→
   569→  return (
   570→    <>
   571→      <AnimatePresence>
   572→        {celebration && (
   573→          <CelebrationOverlay data={celebration} onClose={() => setCelebration(null)} />
   574→        )}
   575→      </AnimatePresence>
   576→
   577→      <AnimatePresence>
   578→        {miniMascots.map((pos) => (
   579→          <FloatingMiniMascot key={pos} mood={mood} onClick={openChat} position={pos} />
   580→        ))}
   581→      </AnimatePresence>
   582→
   583→      {!visible && !dismissed && (
   584→        <motion.button
   585→          onClick={openChat}
   586→          className="fixed bottom-20 right-3 sm:bottom-24 sm:right-4 z-50 rounded-full shadow-lg hover:shadow-xl transition-shadow"
   587→          initial={{ scale: 0 }}
   588→          animate={{ scale: 1 }}
   589→          whileHover={{ scale: 1.1 }}
   590→          whileTap={{ scale: 0.9 }}
   591→        >
   592→          <img src={currentImage} alt="Bateu" className="h-11 w-11 sm:h-14 sm:w-14 drop-shadow-md" width={56} height={56} />
   593→          <motion.div
   594→            className="absolute -top-1 -right-1 h-3 w-3 sm:h-4 sm:w-4 rounded-full bg-primary"
   595→            animate={{ scale: [1, 1.3, 1] }}
   596→            transition={{ repeat: Infinity, duration: 2 }}
   597→          />
   598→        </motion.button>
   599→      )}
   600→
   601→      <AnimatePresence>
   602→        {visible && mode === "chat" ? (
   603→          /* ── Chat mode ── */
   604→          <motion.div
   605→            initial={{ scale: 0.8, opacity: 0, y: 20 }}
   606→            animate={{ scale: 1, opacity: 1, y: 0 }}
   607→            exit={{ scale: 0.8, opacity: 0, y: 20 }}
   608→            transition={{ type: "spring", damping: 15, stiffness: 200 }}
   609→            className="fixed bottom-24 right-4 z-50 w-[320px] sm:w-[360px] rounded-2xl border border-border bg-card shadow-2xl overflow-hidden flex flex-col"
   610→            style={{ maxHeight: "70vh" }}
   611→          >
   612→            <div className="flex items-center gap-2 p-3 border-b border-border bg-primary/5">
   613→              <motion.img
   614→                src={currentImage}
   615→                alt="Bateu"
   616→                className="h-10 w-10"
   617→                width={40}
   618→                height={40}
   619→                animate={{ rotate: [0, 5, -5, 0] }}
   620→                transition={{ repeat: Infinity, duration: 4, ease: "easeInOut" }}
   621→              />
   622→              <div className="flex-1 min-w-0">
   623→                <p className="text-sm font-bold text-foreground">Bateu Assistente</p>
   624→                <p className="text-[10px] text-muted-foreground">
   625→                  {mood === "happy" && "😊 Feliz"}
   626→                  {mood === "thinking" && "🤔 Pensativo"}
   627→                  {mood === "excited" && "🤩 Animado"}
   628→                  {mood === "winner" && "🏆 Festivo"}
   629→                  {" · "}Online
   630→                </p>
   631→              </div>
   632→              <button onClick={minimizeChat} className="p-1 rounded hover:bg-secondary transition-colors" title="Minimizar">
   633→                <ChevronDown className="h-4 w-4 text-muted-foreground" />
   634→              </button>
   635→              <button onClick={handleClose} className="p-1 rounded hover:bg-destructive/20 transition-colors" title="Fechar">
   636→                <X className="h-4 w-4 text-muted-foreground" />
   637→              </button>
   638→            </div>
   639→
   640→            <div className="flex-1 overflow-y-auto p-3 space-y-2 min-h-[200px] max-h-[400px]">
   641→              {chatMessages.map((msg, i) => (
   642→                <motion.div
   643→                  key={i}
   644→                  initial={{ opacity: 0, y: 8, scale: 0.95 }}
   645→                  animate={{ opacity: 1, y: 0, scale: 1 }}
   646→                  className={`flex ${msg.role === "user" ? "justify-end" : "justify-start"}`}
   647→                >
   648→                  {msg.role === "assistant" && (
   649→                    <motion.img
   650→                      src={currentImage}
   651→                      alt=""
   652→                      className="h-6 w-6 mr-1 mt-1 shrink-0"
   653→                      width={24}
   654→                      height={24}
   655→                      animate={{ y: [0, -2, 0] }}
   656→                      transition={{ repeat: Infinity, duration: 2 }}
   657→                    />
   658→                  )}
   659→                  <div className={`max-w-[80%] rounded-2xl px-3 py-2 text-xs leading-relaxed ${
   660→                    msg.role === "user"
   661→                      ? "bg-primary text-primary-foreground rounded-br-md"
   662→                      : "bg-secondary text-foreground rounded-bl-md"
   663→                  }`}>
   664→                    {msg.content}
   665→                  </div>
   666→                </motion.div>
   667→              ))}
   668→              {chatLoading && (
   669→                <div className="flex justify-start">
   670→                  <motion.img src={currentImage} alt="" className="h-6 w-6 mr-1 mt-1" width={24} height={24}
   671→                    animate={{ rotate: [0, 10, -10, 0] }} transition={{ repeat: Infinity, duration: 0.8 }} />
   672→                  <div className="bg-secondary rounded-2xl rounded-bl-md px-3 py-2">
   673→                    <motion.div className="flex gap-1" animate={{ opacity: [0.4, 1, 0.4] }} transition={{ repeat: Infinity, duration: 1.2 }}>
   674→                      <span className="h-1.5 w-1.5 rounded-full bg-muted-foreground" />
   675→                      <span className="h-1.5 w-1.5 rounded-full bg-muted-foreground" />
   676→                      <span className="h-1.5 w-1.5 rounded-full bg-muted-foreground" />
   677→                    </motion.div>
   678→                  </div>
   679→                </div>
   680→              )}
   681→              <div ref={chatEndRef} />
   682→            </div>
   683→
   684→            {chatMessages.length <= 1 && (
   685→              <div className="px-3 pb-2 flex flex-wrap gap-1">
   686→                {QUICK_QUESTIONS.map((q) => (
   687→                  <button
   688→                    key={q}
   689→                    onClick={() => sendMessage(q)}
   690→                    className="text-[10px] px-2 py-1 rounded-full border border-border bg-secondary hover:bg-accent/10 text-foreground transition-colors"
   691→                  >
   692→                    {q}
   693→                  </button>
   694→                ))}
   695→              </div>
   696→            )}
   697→
   698→            <form
   699→              onSubmit={(e) => { e.preventDefault(); sendMessage(chatInput); }}
   700→              className="flex items-center gap-2 p-2 border-t border-border"
   701→            >
   702→              <input
   703→                ref={inputRef}
   704→                value={chatInput}
   705→                onChange={(e) => setChatInput(e.target.value)}
   706→                placeholder={lang === "en" ? "Ask Bateu..." : "Pergunta ao Bateu..."}
   707→                className="flex-1 bg-secondary rounded-xl px-3 py-2 text-xs text-foreground placeholder:text-muted-foreground outline-none focus:ring-1 focus:ring-primary/30"
   708→                disabled={chatLoading}
   709→              />
   710→              <button
   711→                type="submit"
   712→                disabled={!chatInput.trim() || chatLoading}
   713→                className="flex h-8 w-8 items-center justify-center rounded-xl bg-primary text-primary-foreground disabled:opacity-40 hover:opacity-90 transition-opacity"
   714→              >
   715→                <Send className="h-3.5 w-3.5" />
   716→              </button>
   717→            </form>
   718→          </motion.div>
   719→        ) : visible && mode === "bubble" ? (
   720→          /* ── Bubble mode - appears at random positions ── */
   721→          <motion.div
   722→            initial={{ scale: 0, opacity: 0, y: 50 }}
   723→            animate={{ scale: 1, opacity: 1, y: 0 }}
   724→            exit={{ scale: 0, opacity: 0, y: 50 }}
   725→            transition={{ type: "spring", damping: 12, stiffness: 200 }}
   726→            className={`fixed z-50 flex items-end gap-2 ${currentPos.className}`}
   727→            style={{ flexDirection: currentPos.align === "left" ? "row-reverse" : "row" }}
   728→          >
   729→            <motion.div
   730→              initial={{ opacity: 0, x: currentPos.align === "left" ? -20 : 20 }}
   731→              animate={{ opacity: 1, x: 0 }}
   732→              transition={{ delay: 0.3, duration: 0.4 }}
   733→              className="relative max-w-[240px] rounded-2xl border border-border bg-card p-3 shadow-lg"
   734→            >
   735→              <button
   736→                onClick={handleClose}
   737→                className="absolute -top-2 -right-2 flex h-5 w-5 items-center justify-center rounded-full bg-destructive text-destructive-foreground text-[10px] hover:opacity-80"
   738→              >
   739→                <X className="h-3 w-3" />
   740→              </button>
   741→              <div className="mb-1 flex items-center justify-between">
   742→                <span className="text-[10px] font-medium text-muted-foreground">
   743→                  {mood === "happy" && "😊 Feliz"}
   744→                  {mood === "thinking" && "🤔 Pensativo"}
   745→                  {mood === "excited" && "🤩 Animado"}
   746→                  {mood === "winner" && "🏆 Festivo"}
   747→                </span>
   748→                <button
   749→                  onClick={openChat}
   750→                  className="text-[10px] text-primary hover:underline flex items-center gap-0.5"
   751→                >
   752→                  <MessageCircle className="h-3 w-3" /> Chat
   753→                </button>
   754→              </div>
   755→              {loading ? (
   756→                <div className="flex items-center gap-2 py-1">
   757→                  <Sparkles className="h-3.5 w-3.5 text-primary animate-pulse" />
   758→                  <span className="text-xs text-muted-foreground">A pensar...</span>
   759→                </div>
   760→              ) : (
   761→                <p className="text-xs leading-relaxed text-foreground">{message}</p>
   762→              )}
   763→            </motion.div>
   764→
   765→            <motion.div
   766→              onClick={handleDismiss}
   767→              className="cursor-pointer"
   768→              whileHover={{ scale: 1.1 }}
   769→              whileTap={{ scale: 0.9 }}
   770→              animate={currentAnimation}
   771→              transition={{
   772→                y: { repeat: Infinity, duration: mood === "excited" ? 1.5 : 2, ease: "easeInOut" },
   773→                rotate: { repeat: Infinity, duration: mood === "excited" ? 2 : 3, ease: "easeInOut" },
   774→                scale: { repeat: Infinity, duration: mood === "excited" ? 1.5 : 2, ease: "easeInOut" },
   775→              }}
   776→            >
   777→              <img src={currentImage} alt={`Bateu - ${mood}`} className="h-14 w-14 sm:h-20 sm:w-20 drop-shadow-lg" width={80} height={80} />
   778→            </motion.div>
   779→          </motion.div>
   780→        ) : null}
   781→      </AnimatePresence>
   782→    </>
   783→  );
   784→}
