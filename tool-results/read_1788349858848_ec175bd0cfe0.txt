   106→  const [correctAnswer, setCorrectAnswer] = useState("A");

   144→    } catch (error) {

   218→      console.error("Error creating game:", error);

   220→    } finally {
   221→      setSaving(false);
   222→    }

   236→        .from("millionaire_games")

   262→      toast.error("Erro ao atualizar jogo");

   326→      const { error } = await supabase
   327→        .from("millionaire_games")

   347→    setGameDescription("");

   349→    setTimePerQuestion(30);
   350→    setPrimaryColor("#fbbf24");
   351→    setBackgroundColor("#0a0e17");

   360→    setQuestionText("");
   361→    setOptionA("");
   362→    setOptionB("");

   405→            <div className="grid gap-4">
   406→              {games.map((game) => (
   407→                <Card
   408→                  key={game.id}
   409→                  className="cursor-pointer hover:shadow-lg transition-shadow"
   410→                  onClick={() => selectGameForEdit(game)}
   411→                >
   412→                  <CardHeader>
   413→                    <div className="flex items-start justify-between">
   414→                      <div>
   415→                        <CardTitle>{game.name}</CardTitle>

   417→                          {game.description}

   424→                      </div>
   425→                    </div>
