  5127→function format(template: string, vars?: Record<string, string>) {
  5128→  if (!vars) return template;
  5129→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? `{${k}}`);
  5130→}

  5132→const SUPPORTED: Lang[] = ["en", "pt", "pt-BR", "es", "fr", "hi"];
  5133→
  5134→export function LanguageProvider({ children }: { children: ReactNode }) {
  5135→  // English is the universal default. Only honor an explicit non-English pick
  5136→  // made via the in-app switcher (tracked with `bateu_lang_explicit`). Legacy
  5137→  // PT defaults from previous releases are reset to English automatically.
  5138→  const [lang, setLangState] = useState<Lang>(() => {
  5139→    if (typeof window === "undefined") return "pt";
  5140→    const explicit = localStorage.getItem("bateu_lang_explicit") === "1";
  5141→    const stored = localStorage.getItem("bateu_lang") as Lang | null;
  5142→    if (explicit && stored && SUPPORTED.includes(stored)) return stored;
  5143→    return resolveLangFromCountry(getStoredCountry());
  5144→  });
  5145→
  5146→  useEffect(() => {
  5147→    localStorage.setItem("bateu_lang", lang);
  5148→    document.documentElement.lang = lang;
  5149→  }, [lang]);
  5150→
  5151→  const setLang = (l: Lang, options?: { explicit?: boolean }) => {
  5152→    if (!SUPPORTED.includes(l)) return;
  5153→    setLangState(l);
  5154→    if (typeof window !== "undefined" && options?.explicit !== false) {
  5155→      localStorage.setItem("bateu_lang_explicit", "1");
  5156→    }
  5157→  };
  5158→
  5159→  const t = (key: string, vars?: Record<string, string>) => {
  5160→    // For PT/PT-BR: selected → pt → en → raw key
  5161→    // For other langs (hi, es, fr): selected → en → raw key (skip PT)
  5162→    const en = translations.en!;
  5163→    const dict = translations[lang] ?? en;
