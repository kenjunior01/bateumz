   176→function PrizeWheelWrapper() {
   177→  const { gameId } = useParams<{ gameId: string }>();
   178→  // Import DEFAULT_WHEEL_PRIZES dynamically to avoid eager loading
   179→  const [prizes, setPrizes] = useState<unknown[]>([]);
   180→  useEffect(() => {
   181→    import("./components/livegames/PrizeWheel.tsx").then((m) => setPrizes(m.DEFAULT_WHEEL_PRIZES));
   182→  }, []);
   183→  if (prizes.length === 0) return <MinimalPageLoader />;
   184→  return (
   185→    <Suspense fallback={<MinimalPageLoader />}>
   186→      <PrizeWheel prizes={prizes as any[]} gameId={gameId} />
   187→    </Suspense>
   188→  );
   189→}

   380→class AppErrorBoundary extends Component<{children: ReactNode}, {hasError: boolean; error: Error | null}> {
   381→  constructor(props: {children: ReactNode}) { super(props); this.state = {hasError: false, error: null}; }
   382→  static getDerivedStateFromError(error: Error) { return {hasError: true, error}; }
   383→  componentDidCatch(error: Error, info: ErrorInfo) { console.error('AppErrorBoundary:', error, info); }
   384→  render() {
   385→    if (this.state.hasError) {
   386→      return (
   387→      <div className="min-h-screen flex items-center justify-center p-6 overflow-hidden" style={{ background: "hsl(var(--background))" }}>
   388→        <motion.div
   389→          initial={{ opacity: 0, scale: 0.9, y: 20 }}
   390→          animate={{ opacity: 1, scale: 1, y: 0 }}
   391→          transition={{ type: "spring", stiffness: 260, damping: 20 }}
   392→          className="text-center space-y-5 max-w-md relative z-10"
   393→        >
   394→          <motion.div
   395→            className="text-7xl"
   396→            animate={{ y: [0, -10, 0], rotate: [0, 5, -5, 0] }}
   397→            transition={{ duration: 3, repeat: Infinity, ease: "easeInOut" }}
   398→          >
   399→            ⚠️
   400→          </motion.div>
   401→          <h2 className="text-2xl font-display font-bold">Something went wrong</h2>
   402→          <p className="text-muted-foreground text-sm">An unexpected error occurred. Try reloading the page.</p>
   403→          <div className="flex gap-3 justify-center">
   404→            <Button onClick={() => this.setState({ hasError: false, error: null })} variant="outline" className="gap-2 rounded-full px-6">
   405→              Try again
   406→            </Button>
   407→            <Button onClick={() => window.location.reload()} className="gap-2 rounded-full px-6">
   408→              Reload page
   409→          </Button>
   410→          </div>
   411→        </motion.div>
   412→      </div>
   413→      );
   414→    }
   415→    return this.props.children;
   416→  }
   417→}
