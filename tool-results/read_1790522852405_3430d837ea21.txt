    53→
    54→export default function ContestDetail() {
    55→  const { id } = useParams<{ id: string }>();
    56→  const location = useLocation();
    57→  const { user } = useAuth();
    58→  const { toast } = useToast();
    59→  const [contest, setContest] = useState<Contest | null>(null);
    60→  const [submissions, setSubmissions] = useState<Submission[]>([]);
    61→  const [myVotes, setMyVotes] = useState<Set<string>>(new Set());
    62→  const [loading, setLoading] = useState(true);
    63→  const [submitting, setSubmitting] = useState(false);
    64→  const [showForm, setShowForm] = useState(false);
    65→  const [selectedMedia, setSelectedMedia] = useState<{ url: string; type: "photo" | "video"; submissionId?: string } | null>(null);
    66→  const [votingId, setVotingId] = useState<string | null>(null);
    67→
    68→  const [name, setName] = useState("");
    69→  const [desc, setDesc] = useState("");
    70→  const [photoFile, setPhotoFile] = useState<File | null>(null);
    71→  const [videoFile, setVideoFile] = useState<File | null>(null);
    72→  const [videoLink, setVideoLink] = useState("");
    73→  const [uploadMode, setUploadMode] = useState<"file" | "link">("link");
    74→
    75→  const loadData = useCallback(async () => {
    76→    if (!id) return;
    77→    const [{ data: c }, { data: subs }] = await Promise.all([
    78→      supabase.from("contests").select("*").eq("id", id).single(),
    79→      supabase.from("contest_submissions").select("*").eq("contest_id", id).eq("status", "approved").order("votes_count", { ascending: false }),
    80→    ]);
    81→    setContest(c);
    82→    setSubmissions(subs || []);
    83→
    84→    if (user) {
    85→      const { data: votes } = await supabase
    86→        .from("contest_votes")
    87→        .select("submission_id")
    88→        .eq("user_id", user.id);
    89→      setMyVotes(new Set((votes || []).map((v: any) => v.submission_id)));
    90→    }
    91→    setLoading(false);
    92→  }, [id, user]);
    93→
    94→  useEffect(() => { loadData(); }, [loadData]);
    95→
    96→  const handleSubmit = async () => {
    97→    if (!user || !contest || !name.trim()) return;
    98→    setSubmitting(true);
    99→    try {
   100→      let photo_url: string | null = null;
   101→      let video_url: string | null = null;
   102→
   103→      if (photoFile) {
   104→        const path = `${user.id}/${Date.now()}-${photoFile.name}`;
   105→        const { error } = await supabase.storage.from("contest-media").upload(path, photoFile);
   106→        if (!error) {
   107→          const { data: urlData } = supabase.storage.from("contest-media").getPublicUrl(path);
   108→          photo_url = urlData.publicUrl;
   109→        }
   110→      }
   111→
   112→      if (uploadMode === "file" && videoFile) {
   113→        const path = `${user.id}/${Date.now()}-${videoFile.name}`;
   114→        const { error } = await supabase.storage.from("contest-media").upload(path, videoFile);
   115→        if (!error) {
   116→          const { data: urlData } = supabase.storage.from("contest-media").getPublicUrl(path);
   117→          video_url = urlData.publicUrl;
   118→        }
   119→      } else if (uploadMode === "link" && videoLink.trim()) {
   120→        video_url = videoLink.trim();
   121→      }
   122→
   123→      const { error } = await supabase.from("contest_submissions").insert({
   124→        contest_id: contest.id,
   125→        user_id: user.id,
   126→        participant_name: name.trim(),
   127→        description: desc.trim() || null,
   128→        photo_url,
   129→        video_url,
   130→        status: "approved",
   131→      });
   132→
   133→      if (error) throw error;
   134→
   135→      const { data: admins } = await supabase.from("user_roles").select("user_id").eq("role", "admin");
   136→      if (admins) {
   137→        const notifications = admins.map((a: any) => ({
   138→          user_id: a.user_id,
   139→          title: "Nova participação em concurso",
   140→          message: `${name} submeteu uma participação no concurso "${contest.title}"`,
   141→          type: "info",
   142→        }));
   143→        await supabase.from("notifications").insert(notifications);
   144→      }
   145→
   146→      toast({ title: "🎉 Participação enviada!", description: "A sua submissão foi registada com sucesso." });
   147→      fireSideCannons();
   148→      setShowForm(false);
   149→      setName("");
   150→      setDesc("");
   151→      setPhotoFile(null);
   152→      setVideoFile(null);
   153→      setVideoLink("");
   154→      loadData();
   155→    } catch {
   156→      toast({ title: "Erro", description: "Não foi possível enviar a participação.", variant: "destructive" });
   157→    }
   158→    setSubmitting(false);
   159→  };
   160→
   161→  const handleVote = async (submissionId: string) => {
   162→    if (!user) {
   163→      toast({ title: "Inicie sessão", description: "Precisa de estar autenticado para votar.", variant: "destructive" });
   164→      return;
   165→    }
   166→    setVotingId(submissionId);
   167→    const hasVoted = myVotes.has(submissionId);
   168→    if (hasVoted) {
   169→      await supabase.from("contest_votes").delete().eq("submission_id", submissionId).eq("user_id", user.id);
   170→      setMyVotes((prev) => { const n = new Set(prev); n.delete(submissionId); return n; });
   171→    } else {
   172→      await supabase.from("contest_votes").insert({ submission_id: submissionId, user_id: user.id });
   173→      setMyVotes((prev) => new Set(prev).add(submissionId));
   174→      fireConfetti({ intensity: "low" });
   175→    }
   176→    await loadData();
   177→    setVotingId(null);
   178→  };
   179→
   180→  const isVideoLink = (url: string) => {
   181→    try {
   182→      const u = new URL(url);
   183→      return u.hostname.includes("youtube") || u.hostname.includes("youtu.be") ||
   184→        u.hostname.includes("instagram") || u.hostname.includes("tiktok");
   185→    } catch { return false; }
   186→  };
   187→
   188→  const timeLeft = (date: string | null) => {
   189→    if (!date) return null;
   190→    const diff = new Date(date).getTime() - Date.now();
   191→    if (diff <= 0) return "Encerrado";
   192→    const days = Math.floor(diff / 86400000);
   193→    if (days > 0) return `${days}d restantes`;
   194→    const hours = Math.floor(diff / 3600000);
   195→    return `${hours}h restantes`;
   196→  };
   197→
   198→  useSEO({
   199→    title: contest?.title ? `${contest.title} — Concurso` : 'Concurso',
   200→    description: contest
   201→      ? `Concurso ${contest.title}${contest.prize_description ? ` — Prémio: ${contest.prize_description}` : ''}${contest.description ? `. ${contest.description}` : ''}`.trim().slice(0, 160)
   202→      : 'Descubra e participe em concursos na Bateu.',
   203→    canonicalPath: location.pathname,
   204→  });
   205→
   206→  if (loading) {
   207→    return (
   208→      <div className="min-h-screen bg-background">
   209→        <Navbar />
   210→        <div className="flex justify-center items-center pt-32">
   211→          <motion.div animate={{ rotate: 360 }} transition={{ repeat: Infinity, duration: 1, ease: "linear" }} className="h-8 w-8 rounded-full border-2 border-primary border-t-transparent" />
   212→        </div>
   213→      </div>
   214→    );
   215→  }
   216→
   217→  if (!contest) {
   218→    return (
   219→      <div className="min-h-screen bg-background">
   220→        <Navbar />
   221→        <div className="text-center pt-32">
   222→          <motion.div initial={{ scale: 0 }} animate={{ scale: 1 }}>
   223→            <Trophy className="h-16 w-16 text-muted-foreground/30 mx-auto mb-4" />
   224→          </motion.div>
   225→          <p className="text-muted-foreground">Concurso não encontrado.</p>
   226→        </div>
   227→      </div>
   228→    );
   229→  }
   230→
   231→  const isOpen = contest.status === "active";
   232→  const isVoting = contest.status === "voting";
   233→  const sorted = [...submissions].sort((a, b) =>
   234→    contest.evaluation_type === "views" ? b.views_count - a.views_count : b.votes_count - a.votes_count
   235→  );
   236→  const winners = sorted.filter((s) => s.is_winner);
   237→  const topScore = sorted.length > 0 ? (contest.evaluation_type === "views" ? sorted[0].views_count : sorted[0].votes_count) : 1;
   238→
   239→  return (
   240→    <div className="min-h-screen bg-background">
   241→      <Navbar />
   242→      <div className="container mx-auto px-4 pt-24 pb-16">
   243→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="mb-8">
   244→          <Link to="/concursos" className="text-sm text-muted-foreground hover:text-primary mb-4 inline-flex items-center gap-1 transition-colors">
   245→            <ArrowLeft className="h-4 w-4" /> Voltar aos concursos
   246→          </Link>
   247→
   248→          {contest.image_url && (
   249→            <motion.div initial={{ opacity: 0, scale: 0.98 }} animate={{ opacity: 1, scale: 1 }} transition={{ delay: 0.1 }} className="aspect-[3/1] rounded-2xl overflow-hidden mb-6 relative">
   250→              <img src={contest.image_url} alt={contest.title} className="w-full h-full object-cover" />
   251→              <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-transparent" />
   252→              <div className="absolute bottom-4 left-4 right-4">
   253→                <Badge className={isOpen ? "bg-primary text-primary-foreground" : isVoting ? "bg-accent text-accent-foreground" : "bg-muted"}>
   254→                  {isOpen ? "🔥 Aberto" : isVoting ? "🗳️ Em Votação" : "✅ Encerrado"}
   255→                </Badge>
   256→              </div>
   257→            </motion.div>
   258→          )}
   259→
   260→          <div className="flex items-start justify-between flex-wrap gap-4">
   261→            <motion.div initial={{ opacity: 0, x: -20 }} animate={{ opacity: 1, x: 0 }} transition={{ delay: 0.2 }}>
   262→              <h1 className="text-3xl md:text-4xl font-display font-bold text-foreground">{contest.title}</h1>
   263→              {contest.description && <p className="text-muted-foreground mt-2 max-w-2xl">{contest.description}</p>}
   264→              {contest.end_date && (isOpen || isVoting) && (
   265→                <div className="mt-4">
   266→                  <p className="text-xs uppercase tracking-wider text-muted-foreground mb-2 font-semibold">⏰ Termina em</p>
   267→                  <ContestCountdown endDate={contest.end_date} />
   268→                </div>
   269→              )}
   270→              <div className="mt-4">
   271→                <ShareButtons data={{ title: contest.title, text: `${contest.title} — participa e ganha!`, url: typeof window !== "undefined" ? window.location.href : "" }} />
   272→              </div>
   273→            </motion.div>
   274→            <motion.div initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} transition={{ delay: 0.3 }} className="flex items-center gap-3">
   275→              {!contest.image_url && (
   276→                <Badge className={isOpen ? "bg-primary text-primary-foreground" : isVoting ? "bg-accent text-accent-foreground" : "bg-muted"}>
   277→                  {isOpen ? "🔥 Aberto" : isVoting ? "🗳️ Em Votação" : "✅ Encerrado"}
   278→                </Badge>
   279→              )}
   280→              {isOpen && user && (
   281→                <Dialog open={showForm} onOpenChange={setShowForm}>
   282→                  <DialogTrigger asChild>
   283→                    <Button size="lg" className="gap-2 shadow-[0_0_15px_hsl(var(--primary)/0.3)] hover:shadow-[0_0_25px_hsl(var(--primary)/0.5)]"><Send className="h-4 w-4" /> Participar</Button>
   284→                  </DialogTrigger>
   285→                  <DialogContent className="max-w-lg">
   286→                    <DialogHeader><DialogTitle className="flex items-center gap-2"><Trophy className="h-5 w-5 text-primary" /> Submeter Participação</DialogTitle></DialogHeader>
   287→                    <div className="space-y-4">
   288→                      <div>
   289→                        <Label>Nome do participante *</Label>
   290→                        <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="O seu nome" />
   291→                      </div>
   292→                      <div>
   293→                        <Label>Descrição da sua criação</Label>
   294→                        <Textarea value={desc} onChange={(e) => setDesc(e.target.value)} placeholder="Descreva a sua receita, criação..." rows={3} />
   295→                      </div>
   296→                      <div>
   297→                        <Label className="flex items-center gap-2"><ImageIcon className="h-4 w-4" /> Foto</Label>
   298→                        <Input type="file" accept="image/*" onChange={(e) => setPhotoFile(e.target.files?.[0] || null)} />
   299→                      </div>
   300→
   301→                      <div>
   302→                        <Label className="flex items-center gap-2 mb-2"><Video className="h-4 w-4" /> Vídeo</Label>
   303→                        <div className="flex gap-2 mb-2">
   304→                          <Button type="button" variant={uploadMode === "link" ? "default" : "outline"} size="sm" onClick={() => setUploadMode("link")} className="gap-1 text-xs">
   305→                            <Link2 className="h-3 w-3" /> Link (Instagram/TikTok/YouTube)
   306→                          </Button>
   307→                          <Button type="button" variant={uploadMode === "file" ? "default" : "outline"} size="sm" onClick={() => setUploadMode("file")} className="gap-1 text-xs">
   308→                            <Video className="h-3 w-3" /> Upload ficheiro
   309→                          </Button>
   310→                        </div>
   311→                        {uploadMode === "link" ? (
   312→                          <Input value={videoLink} onChange={(e) => setVideoLink(e.target.value)} placeholder="https://instagram.com/reel/... ou https://youtube.com/watch?v=..." />
   313→                        ) : (
   314→                          <Input type="file" accept="video/*" onChange={(e) => setVideoFile(e.target.files?.[0] || null)} />
   315→                        )}
   316→                        <p className="text-[10px] text-muted-foreground mt-1">
   317→                          💡 Use links de redes sociais para economizar espaço — o vídeo será reproduzido diretamente na plataforma
   318→                        </p>
   319→                      </div>
   320→
   321→                      <Button onClick={handleSubmit} disabled={submitting || !name.trim()} className="w-full gap-2">
   322→                        {submitting ? "A enviar..." : <><Send className="h-4 w-4" /> Enviar Participação</>}
   323→                      </Button>
   324→                    </div>
   325→                  </DialogContent>
   326→                </Dialog>
   327→              )}
   328→              {!user && isOpen && (
   329→                <Link to="/login"><Button variant="outline" size="lg">Entrar para Participar</Button></Link>
   330→              )}
   331→            </motion.div>
   332→          </div>
   333→
   334→          {(contest as any).contest_mode === "multi" && Array.isArray((contest as any).phases) && (contest as any).phases.length > 0 && (
   335→            <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.35 }} className="mt-6 p-4 rounded-xl glass">
   336→              <p className="text-xs uppercase tracking-wider text-muted-foreground mb-3 font-semibold flex items-center gap-1">🏆 Fases do Concurso</p>
   337→              <div className="flex items-center gap-1 overflow-x-auto pb-2">
   338→                {((contest as any).phases as { name: string; durationDays: number; type: string }[]).map((phase, i) => {
   339→                  const isCurrent = i === ((contest as any).current_phase || 0);
   340→                  const isPast = i < ((contest as any).current_phase || 0);
   341→                  return (
   342→                    <div key={i} className="flex items-center gap-1 shrink-0">
   343→                      <motion.div
   344→                        initial={{ scale: 0 }}
   345→                        animate={{ scale: 1 }}
   346→                        transition={{ delay: 0.4 + i * 0.1, type: "spring" }}
   347→                        className={`px-3 py-2 rounded-lg text-center border transition-all ${
   348→                          isCurrent ? "border-primary bg-primary/10 ring-2 ring-primary/30" : isPast ? "border-muted bg-muted/50 opacity-60" : "border-border/50 bg-background/50"
   349→                        }`}
   350→                      >
   351→                        <p className={`text-xs font-bold ${isCurrent ? "text-primary" : "text-foreground"}`}>{phase.name}</p>
   352→                        <p className="text-[10px] text-muted-foreground">{phase.durationDays}d</p>
   353→                      </motion.div>
   354→                      {i < (contest as any).phases.length - 1 && (
   355→                        <div className={`w-6 h-0.5 ${isPast ? "bg-primary" : "bg-border"}`} />
   356→                      )}
   357→                    </div>
   358→                  );
   359→                })}
   360→              </div>
   361→            </motion.div>
   362→          )}
   363→
   364→          {(contest as any).sponsor_name && (
   365→            <motion.div initial={{ opacity: 0, scale: 0.95 }} animate={{ opacity: 1, scale: 1 }} transition={{ delay: 0.4 }} className="mt-4 inline-flex items-center gap-2 px-3 py-1.5 rounded-full border border-accent/30 bg-accent/5">
   366→              {(contest as any).sponsor_logo_url && <img src={(contest as any).sponsor_logo_url} alt="" className="h-5 w-5 rounded-full object-cover" />}
   367→              <span className="text-xs text-muted-foreground">Patrocinado por</span>
   368→              <span className="text-xs font-semibold text-foreground">{(contest as any).sponsor_name}</span>
   369→            </motion.div>
   370→          )}
   371→
   372→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.4 }} className="flex flex-wrap gap-4 mt-6 p-4 rounded-xl glass">
   373→            {contest.prize_description && (
   374→              <div className="flex items-center gap-2 text-sm">
   375→                <div className="h-8 w-8 rounded-full bg-primary/10 flex items-center justify-center"><Trophy className="h-4 w-4 text-primary" /></div>
   376→                <div><span className="text-muted-foreground text-xs">Prémio</span><p className="font-semibold text-foreground">{contest.prize_description}</p></div>
   377→              </div>
   378→            )}
   379→            {contest.end_date && (
   380→              <div className="flex items-center gap-2 text-sm">
   381→                <div className="h-8 w-8 rounded-full bg-accent/10 flex items-center justify-center"><Clock className="h-4 w-4 text-accent" /></div>
   382→                <div><span className="text-muted-foreground text-xs">Prazo</span><p className="font-semibold text-foreground">{timeLeft(contest.end_date) || new Date(contest.end_date).toLocaleDateString("pt-MZ")}</p></div>
   383→              </div>
   384→            )}
   385→            <div className="flex items-center gap-2 text-sm">
   386→              <div className="h-8 w-8 rounded-full bg-secondary flex items-center justify-center"><Users className="h-4 w-4 text-foreground" /></div>
   387→              <div><span className="text-muted-foreground text-xs">Participações</span><p className="font-semibold text-foreground">{sorted.length}</p></div>
   388→            </div>
   389→            <div className="flex items-center gap-2 text-sm">
   390→              <div className="h-8 w-8 rounded-full bg-secondary flex items-center justify-center">
   391→                {contest.evaluation_type === "views" ? <Eye className="h-4 w-4 text-foreground" /> : <ThumbsUp className="h-4 w-4 text-foreground" />}
   392→              </div>
   393→              <div><span className="text-muted-foreground text-xs">Avaliação</span><p className="font-semibold text-foreground">{contest.evaluation_type === "views" ? "Visualizações" : "Votos"}</p></div>
   394→            </div>
   395→            {(contest as any).entry_fee > 0 && (
   396→              <div className="flex items-center gap-2 text-sm">
   397→                <div className="h-8 w-8 rounded-full bg-accent/10 flex items-center justify-center"><span className="text-xs font-bold text-accent">$</span></div>
   398→                <div><span className="text-muted-foreground text-xs">Inscrição</span><p className="font-semibold text-foreground">{(contest as any).entry_fee} MZN</p></div>
   399→              </div>
   400→            )}
   401→          </motion.div>
   402→        </motion.div>
   403→
   404→        {sorted.length > 0 && (isOpen || isVoting) && (
   405→          <LiveLeaderboard contestId={contest.id} evaluationType={contest.evaluation_type} />
   406→        )}
   407→
   408→        <AnimatePresence>
   409→          {winners.length > 0 && (
   410→            <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="mb-10">
   411→              <h2 className="text-2xl font-bold mb-4 flex items-center gap-2">
   412→                <motion.div animate={{ rotate: [0, -10, 10, 0] }} transition={{ repeat: Infinity, duration: 2 }}>
